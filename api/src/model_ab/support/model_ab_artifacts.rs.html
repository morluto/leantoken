127pub fn validate_orientation_capsule(
128    capsule: &OrientationCapsule,
129    tokenizer: leantoken::tokens::Tokenizer,
130) -> Result<(), String> {
131    if capsule.entries.is_empty() || capsule.entries.len() > ORIENTATION_CAPSULE_MAX_PATHS {
132        return Err("orientation capsule must contain exactly one bounded path".to_owned());
133    }
134    for entry in &capsule.entries {
135        if !is_relative_capsule_path(&entry.path)
136            || entry.matched_terms.is_empty()
137            || entry.matched_terms.len() > ORIENTATION_CAPSULE_MAX_TERMS
138            || entry.definitions.len() > ORIENTATION_CAPSULE_MAX_DEFINITIONS
139            || entry
140                .matched_terms
141                .iter()
142                .chain(&entry.definitions)
143                .any(|value| value.trim().is_empty())
144        {
145            return Err("orientation capsule entry exceeds its structural bounds".to_owned());
146        }
147    }
148    let serialized = serde_json::to_string(&capsule.entries)
149        .map_err(|error| format!("orientation capsule serialization failed: {error}"))?;
150    let exact_tokens = tokenizer.count(&serialized);
151    if exact_tokens != capsule.capsule_tokens || exact_tokens > ORIENTATION_CAPSULE_MAX_TOKENS {
152        return Err(format!(
153            "orientation capsule token count is {}, expected {}, with maximum {}",
154            capsule.capsule_tokens, exact_tokens, ORIENTATION_CAPSULE_MAX_TOKENS
155        ));
156    }
157    Ok(())
158}

173fn is_relative_capsule_path(path: &str) -> bool {
174    !path.is_empty()
175        && !path.starts_with('/')
176        && !path.starts_with('\\')
177        && !path.contains('\\')
178        && path
179            .split('/')
180            .all(|component| !component.is_empty() && component != "." && component != "..")
181}

183pub fn is_bounded_prewalk_todo_event(event: &Value) -> bool {
184    if event["type"].as_str() != Some("item.completed")
185        || event.pointer("/item/type").and_then(Value::as_str) != Some("agent_message")
186    {
187        return false;
188    }
189    let Some(text) = event.pointer("/item/text").and_then(Value::as_str) else {
190        return false;
191    };
192    let Ok(response) = serde_json::from_str::<Value>(text) else {
193        return false;
194    };
195    let Some(response) = response.as_object() else {
196        return false;
197    };
198    if response.len() != 2
199        || response
200            .get("summary")
201            .and_then(Value::as_str)
202            .is_none_or(|summary| summary.trim().is_empty())
203    {
204        return false;
205    }
206    let Some(todo) = response.get("todo").and_then(Value::as_array) else {
207        return false;
208    };
209    !todo.is_empty()
210        && todo.len() <= 8
211        && todo.iter().all(|item| {
212            item.as_object().is_some_and(|item| {
213                item.len() == 2
214                    && item
215                        .get("step")
216                        .and_then(Value::as_str)
217                        .is_some_and(|step| !step.trim().is_empty())
218                    && matches!(
219                        item.get("status").and_then(Value::as_str),
220                        Some("pending" | "in_progress" | "completed")
221                    )
222            })
223        })
224}
