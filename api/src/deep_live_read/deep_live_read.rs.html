87async fn run(args: &Args) -> AnyResult<Report> {
88    validate(args)?;
89    let root = tempfile::tempdir()?;
90    let (content, lines) = synthetic_source(args.file_bytes);
91    fs::write(root.path().join("near_limit.rs"), &content)?;
92    let config = Config::discover(root.path(), Some(root.path().join("index.sqlite")))?;
93    let services = Services::open(config)?;
94    services.refresh(leantoken::IndexingMode::Reconcile).await?;
95
96    let shallow = (1, args.range_lines.min(lines));
97    let deep_start = lines
98        .saturating_sub(args.range_lines)
99        .saturating_add(1)
100        .max(1);
101    let deep = (deep_start, lines);
102    services
103        .read_worktree(request(shallow, args.max_tokens))
104        .await?;
105    services
106        .read_worktree(request(deep, args.max_tokens))
107        .await?;
108
109    let (shallow_range, shallow_response) =
110        measure(&services, shallow, args.iterations, args.max_tokens).await?;
111    let (deep_range, deep_response) =
112        measure(&services, deep, args.iterations, args.max_tokens).await?;
113
114    Ok(Report {
115        schema_version: 1,
116        release_build: !cfg!(debug_assertions),
117        fixture: Fixture {
118            file_bytes: content.len(),
119            lines,
120            iterations: args.iterations,
121            range_lines: args.range_lines,
122            max_tokens: args.max_tokens,
123            deep_start_line: deep_start,
124        },
125        algorithm: Algorithm {
126            name: "single_forward_hash_and_range_stream",
127            complete_read_file_streams: 1,
128            truncated_read_file_streams: 2,
129        },
130        shallow_range,
131        deep_range,
132        response_parity: ResponseParity {
133            shallow_lines: shallow_response.returned_end_line
134                - shallow_response.returned_start_line
135                + 1,
136            deep_lines: deep_response.returned_end_line - deep_response.returned_start_line + 1,
137            shallow_index_stale: shallow_response.index_stale,
138            deep_index_stale: deep_response.index_stale,
139            shallow_truncated: shallow_response.truncated,
140            deep_truncated: deep_response.truncated,
141        },
142        limitations: vec![
143            "The synthetic file is warm in the operating-system page cache after indexing and warmup.",
144            "Elapsed times are diagnostics; correctness and stream-count invariants are not inferred from latency thresholds.",
145            "A complete read uses one forward stream. A token-truncated read retains a second full-hash verification before issuing its continuation cursor.",
146        ],
147    })
148}

150fn validate(args: &Args) -> AnyResult<()> {
151    if args.iterations == 0 || args.range_lines == 0 || args.max_tokens == 0 {
152        return Err("iterations, range-lines, and max-tokens must be positive".into());
153    }
154    if args.file_bytes < 4_096 || args.file_bytes > DiscoveryLimits::DEFAULT_MAX_FILE_BYTES as usize
155    {
156        return Err("file-bytes must be between 4096 and the default file limit".into());
157    }
158    Ok(())
159}

161fn synthetic_source(target_bytes: usize) -> (String, usize) {
162    let mut content = String::with_capacity(target_bytes);
163    let mut line = 1usize;
164    while content.len() < target_bytes {
165        let row = format!("pub const VALUE_{line:06}: usize = {line}; // deep read padding\n");
166        if content.len().saturating_add(row.len()) > target_bytes {
167            break;
168        }
169        content.push_str(&row);
170        line = line.saturating_add(1);
171    }
172    (content, line.saturating_sub(1).max(1))
173}

175async fn measure(
176    services: &Services,
177    range: (usize, usize),
178    iterations: usize,
179    max_tokens: usize,
180) -> AnyResult<(TimingStats, leantoken::ReadResponse)> {
181    let mut durations = Vec::with_capacity(iterations);
182    let mut response = None;
183    for _ in 0..iterations {
184        let started = Instant::now();
185        let current = services.read_worktree(request(range, max_tokens)).await?;
186        durations.push(started.elapsed());
187        black_box(&current);
188        response = Some(current);
189    }
190    Ok((
191        TimingStats::from_durations(durations),
192        response.expect("iterations are positive"),
193    ))
194}
195
196fn request(range: (usize, usize), max_tokens: usize) -> WorktreeReadRequest {
197    WorktreeReadRequest {
198        path: PathBuf::from("near_limit.rs")
199            .to_string_lossy()
200            .into_owned(),
201        start_line: Some(range.0),
202        end_line: Some(range.1),
203        symbol: None,
204        heading: None,
205        heading_occurrence: None,
206        continuation_cursor: None,
207        max_tokens: Some(max_tokens),
208        expected_hash: None,
209        delta: false,
210        delta_base_artifact_id: None,
211        receipt_id: None,
212        policy: leantoken::model::ReadPolicy::default(),
213    }
214}
215
216impl TimingStats {
217    fn from_durations(durations: Vec<Duration>) -> Self {
218        let mut micros = durations
219            .iter()
220            .map(|duration| duration.as_secs_f64() * 1_000_000.0)
221            .collect::<Vec<_>>();
222        micros.sort_by(f64::total_cmp);
223        let mean_us = micros.iter().sum::<f64>() / micros.len() as f64;
224        Self {
225            samples: micros.len(),
226            p50_us: percentile(&micros, 50),
227            p95_us: percentile(&micros, 95),
228            mean_us,
229        }
230    }
231}
232
233fn percentile(values: &[f64], percentile: usize) -> f64 {
234    let rank = (percentile * values.len()).div_ceil(100);
235    values[rank.saturating_sub(1).min(values.len() - 1)]
236}
