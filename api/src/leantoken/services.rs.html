99fn encode_savings_snapshot(state: &TokenSavingsSnapshotState) -> Result<String> {
100    let payload = serde_json::to_vec(state)?;
101    let snapshot = format!(
102        "lts1.{}.{}",
103        URL_SAFE_NO_PAD.encode(&payload),
104        savings_snapshot_checksum(&payload)
105    );
106    if snapshot.len() > MAX_SAVINGS_SNAPSHOT_BYTES {
107        return Err(Error::RequestLimitExceeded {
108            field: "snapshot",
109            requested: snapshot.len(),
110            limit: MAX_SAVINGS_SNAPSHOT_BYTES,
111        });
112    }
113    Ok(snapshot)
114}

116fn decode_savings_snapshot(snapshot: &str) -> Result<TokenSavingsSnapshotState> {
117    if snapshot.len() > MAX_SAVINGS_SNAPSHOT_BYTES {
118        return Err(Error::RequestLimitExceeded {
119            field: "snapshot",
120            requested: snapshot.len(),
121            limit: MAX_SAVINGS_SNAPSHOT_BYTES,
122        });
123    }
124    let mut parts = snapshot.split('.');
125    if parts.next() != Some("lts1") {
126        return Err(snapshot_invalid());
127    }
128    let payload = parts
129        .next()
130        .and_then(|value| URL_SAFE_NO_PAD.decode(value).ok())
131        .ok_or_else(snapshot_invalid)?;
132    let checksum = parts.next().ok_or_else(snapshot_invalid)?;
133    if parts.next().is_some() || checksum != savings_snapshot_checksum(&payload) {
134        return Err(snapshot_invalid());
135    }
136    let state: TokenSavingsSnapshotState =
137        serde_json::from_slice(&payload).map_err(|_| snapshot_invalid())?;
138    if state.version != SAVINGS_SNAPSHOT_VERSION {
139        return Err(snapshot_invalid());
140    }
141    Ok(state)
142}

224fn savings_snapshot_state(
225    repository_id: String,
226    tokenizer: String,
227    records: &HashMap<String, TokenSavingsRecord>,
228    failures: &[ServiceFailureRecord],
229) -> TokenSavingsSnapshotState {
230    let mut records = records
231        .iter()
232        .map(|(operation, record)| (operation.clone(), savings_record_values(record)))
233        .collect::<Vec<_>>();
234    records.sort_by(|left, right| left.0.cmp(&right.0));
235    let mut failures = failures
236        .iter()
237        .map(|record| {
238            (
239                record.operation.clone(),
240                record.error_category.clone(),
241                record.failed_requests,
242            )
243        })
244        .collect::<Vec<_>>();
245    failures.sort_by(|left, right| (&left.0, &left.1).cmp(&(&right.0, &right.1)));
246    TokenSavingsSnapshotState {
247        version: SAVINGS_SNAPSHOT_VERSION,
248        repository_id,
249        tokenizer,
250        records,
251        failures,
252    }

255fn subtract_savings_records(
256    current: &HashMap<String, TokenSavingsRecord>,
257    base: Vec<(String, [u64; 19])>,
258) -> Result<HashMap<String, TokenSavingsRecord>> {
259    let mut delta = current.clone();
260    for (operation, base_record) in base {
261        let base_record = savings_record_from_values(base_record);
262        let current_record = current.get(&operation).cloned().unwrap_or_default();
263        let difference =

265        delta.insert(operation, difference);
266    }
267    Ok(delta)

270fn subtract_service_failures(
271    current: &[ServiceFailureRecord],
272    base: Vec<(String, String, u64)>,
273) -> Result<Vec<ServiceFailureRecord>> {
274    let current = current
275        .iter()
276        .map(|record| {
277            (
278                (record.operation.clone(), record.error_category.clone()),
279                record.failed_requests,
280            )
281        })
282        .collect::<HashMap<_, _>>();
283    let mut delta = current.clone();
284    for (operation, error_category, failed_requests) in base {
285        let key = (operation, error_category);
286        let current_count = current.get(&key).copied().unwrap_or(0);
287        delta.insert(
288            key,
289            current_count
290                .checked_sub(failed_requests)
291                .ok_or_else(snapshot_invalid)?,
292        );
293    }
294    let mut delta = delta
295        .into_iter()
296        .filter(|(_, failed_requests)| *failed_requests != 0)
297        .map(
298            |((operation, error_category), failed_requests)| ServiceFailureRecord {
299                operation,
300                error_category,
301                failed_requests,
302            },
303        )
304        .collect::<Vec<_>>();
305    delta.sort_by(|left, right| {
306        (&left.operation, &left.error_category).cmp(&(&right.operation, &right.error_category))
307    });
308    Ok(delta)
309}

465    pub fn open(config: Config) -> Result<Self> {
466        config.validate()?;
467        let coordination = IndexCoordination::for_database(&config.database_path);
468        let cancellation = CancellationToken::new();
469        let cache_lease = coordination.acquire_cache_lease(&cancellation)?;

471        Self::open_once(&config, None, cache_lease)

474    /// Open services under exclusive cache initialization ownership, retrying
475    /// transient SQLite contention until the caller cancels.
476    pub fn open_cancellable(config: Config, cancellation: &CancellationToken) -> Result<Self> {
477        config.validate()?;
478        let coordination = IndexCoordination::for_database(&config.database_path);
479        let cache_lease = coordination.acquire_cache_lease(cancellation)?;
480        let _initialization = coordination.acquire_initialization(cancellation)?;
481        let mut delay = STARTUP_RETRY_INITIAL_DELAY;
482        let mut attempt = 0u32;
483
484        loop {
485            validation::check_cancelled(cancellation)?;
486            match Self::open_once(&config, Some(STARTUP_BUSY_TIMEOUT), cache_lease.clone()) {
487                Ok(services) => return Ok(services),
488                Err(error) if is_database_contention(&error) => {
489                    attempt = attempt.saturating_add(1);
490                    if attempt == 1 || attempt.is_multiple_of(20) {
491                        tracing::warn!(
492                            attempt,
493                            retry_delay_ms = delay.as_millis(),
494                            database = %config.database_path.display(),
495                            %error,
496                            "cache initialization is waiting for SQLite contention"
497                        );
498                    }
499                    wait_cancellable(cancellation, delay)?;
500                    delay = delay.saturating_mul(2).min(STARTUP_RETRY_MAX_DELAY);
501                }
502                Err(error) => return Err(error),
503            }
504        }
505    }

567    fn finalize_response<T: RetrievalResponse>(&self, response: &mut T) -> Result<()> {
568        let source_tokens = {
569            let meta = response.meta_mut();
570            meta.protocol_tokens = 0;
571            meta.path_and_metadata_tokens = 0;
572            meta.total_response_tokens = 0;
573            meta.payload_tokens = 0;
574            meta.source_tokens
575        };
576        const MAX_ACCOUNTING_PASSES: usize = 32;
577        for _ in 0..MAX_ACCOUNTING_PASSES {
578            let accounting =
579                response_token_accounting(&*response, source_tokens, &self.config.tokenizer)?;
580            let meta = response.meta_mut();
581            if meta.protocol_tokens == accounting.protocol_tokens
582                && meta.path_and_metadata_tokens == accounting.path_and_metadata_tokens
583                && meta.total_response_tokens == accounting.total_response_tokens
584                && meta.payload_tokens == accounting.total_response_tokens
585            {
586                return Ok(());
587            }
588            meta.protocol_tokens = accounting.protocol_tokens;
589            meta.path_and_metadata_tokens = accounting.path_and_metadata_tokens;
590            meta.total_response_tokens = accounting.total_response_tokens;
591            meta.payload_tokens = accounting.total_response_tokens;
592        }
593        Err(Error::InternalFailure(
594            "serialized response accounting did not reach a fixed point".into(),
595        ))
596    }

651    ) -> Result<usize>
652    where
653        T: RetrievalResponse + Clone,
654    {
655        let mut sized = response.clone();
656        {
657            let meta = sized.meta_mut();
658            meta.receipt_id = Some(
659                meta.receipt_id
660                    .clone()
661                    .unwrap_or_else(|| "rffffffffffffffff".into()),
662            );
663            meta.receipt_suppressed_exact = returned_items;
664            meta.receipt_suppressed_overlap = returned_items;
665            meta.receipt_near_duplicates = returned_items;
666        }
667        self.finalized_response_tokens(&sized)
668    }

790    /// Wait until the first committed generation is no longer being published.
791    pub(crate) async fn wait_for_initial_index_cancellable(
792        &self,
793        cancellation: CancellationToken,
794    ) -> Result<()> {
795        let mut idle_deadline = None;
796        loop {
797            validation::check_cancelled(&cancellation)?;
798
799            let changed = self.reconciliation_changed.notified();
800            tokio::pin!(changed);
801            changed.as_mut().enable();
802            if self.active_reconciliations.load(Ordering::Acquire) > 0 {
803                idle_deadline = None;
804                tokio::select! {
805                    _ = cancellation.cancelled() => return Err(Error::Cancelled),
806                    _ = &mut changed => {}
807                }
808                continue;
809            }
810
811            let this = self.clone();
812            let probe = tokio::task::spawn_blocking(move || {
813                let Some(operation) = this.coordination.try_acquire_operation()? else {
814                    return Ok(None);
815                };
816                let generation = this.storage.repository_generation();
817                operation.release()?;
818                generation.map(Some)
819            });
820            let generation = tokio::select! {
821                _ = cancellation.cancelled() => return Err(Error::Cancelled),
822                _ = &mut changed => {
823                    idle_deadline = None;
824                    continue;
825                },
826                result = probe => result??,
827            };
828            if generation.is_some_and(|generation| generation > 0) {
829                return Ok(());
830            }
831            let delay = if generation.is_none() {
832                idle_deadline = None;
833                INITIAL_INDEX_PROBE_INTERVAL
834            } else {
835                let deadline = idle_deadline
836                    .get_or_insert_with(|| tokio::time::Instant::now() + INITIAL_INDEX_IDLE_GRACE);
837                let remaining = deadline.saturating_duration_since(tokio::time::Instant::now());
838                if remaining.is_zero() {
839                    return Err(Error::IndexNotReady);
840                }
841                remaining.min(INITIAL_INDEX_PROBE_INTERVAL)
842            };
843
844            tokio::select! {
845                _ = cancellation.cancelled() => return Err(Error::Cancelled),
846                _ = &mut changed => {
847                    idle_deadline = None;
848                },
849                _ = tokio::time::sleep(delay) => {}
850            }
851        }
852    }

1218    /// Assemble a response against one WAL snapshot (DEFERRED read transaction).
1219    /// Concurrent writers cannot mix generations inside a single assembly. If
1220    /// opening the snapshot fails transiently or the index is still empty,
1221    /// returns a typed retryable error rather than a partial response.
1222    fn consistent_inner<T>(
1223        &self,
1224        allow_empty: bool,
1225        operation: impl Fn(&ReadSession, u64) -> Result<T>,
1226    ) -> Result<T> {
1227        for attempt in 0..3 {
1228            let snapshot = self.storage.begin_read().and_then(|session| {
1229                let generation = session.repository_generation()?;
1230                Ok((session, generation))
1231            });
1232            let (session, generation) = match snapshot {
1233                Ok(snapshot) => snapshot,
1234                Err(error) if is_database_contention(&error) => {
1235                    if attempt + 1 < 3 {
1236                        thread::sleep(CANCELLATION_POLL_INTERVAL);
1237                    }
1238                    continue;
1239                }
1240                Err(error) => return Err(error),
1241            };
1242            if generation == 0 && !allow_empty {
1243                return Err(Error::IndexNotReady);
1244            }
1245            // Do not retry operation errors: after the first read, this session
1246            // is pinned and concurrent publication cannot have caused them.
1247            return operation(&session, generation);
1248        }
1249        Err(Error::RetryableConflict(RetryableOperation::Retrieval))
1250    }
