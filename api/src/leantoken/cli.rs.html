26fn parse_positive_usize(value: &str) -> std::result::Result<usize, String> {
27    let value = value
28        .parse::<usize>()
29        .map_err(|_| "value must be a positive integer".to_owned())?;
30    if value == 0 {
31        return Err("value must be a positive integer".to_owned());
32    }
33    Ok(value)
34}

40fn parse_cache_list_limit(value: &str) -> std::result::Result<usize, String> {
41    let value = parse_positive_usize(value)?;
42    if value > MAX_CACHE_LIST_LIMIT {
43        return Err(format!("must not exceed {MAX_CACHE_LIST_LIMIT}"));
44    }
45    Ok(value)
46}

121fn hide_advanced_repository_options(command: Command) -> Command {
122    command
123        .mut_args(|argument| {
124            if COMMAND_SCOPE_OPTIONS
125                .iter()
126                .any(|option| option.advanced && option.id == argument.get_id().as_str())
127            {
128                argument.hide(true)
129            } else {
130                argument
131            }
132        })
133        .mut_subcommands(|subcommand| subcommand.defer(hide_advanced_repository_options))
134}

136fn keep_advanced_repository_options_in_root_help(command: Command) -> Command {
137    command.mut_subcommands(|subcommand| subcommand.defer(hide_advanced_repository_options))
138}

250    /// Reject repository-scoped global options for repository-free commands.
251    ///
252    /// Clap propagates global arguments to every subcommand. Checking the
253    /// original argument tokens preserves before/after-subcommand placement
254    /// for repository commands while distinguishing an explicit default value
255    /// such as `--root .` from an omitted option.
256    pub fn validate_option_scope(
257        &self,
258        arguments: &[OsString],
259    ) -> std::result::Result<(), clap::Error> {
260        if !matches!(
261            self.command,
262            Commands::Setup(_)
263                | Commands::Remove(_)
264                | Commands::Cache(_)
265                | Commands::Update(_)
266                | Commands::Upgrade(_)
267        ) {
268            return Ok(());
269        }
270        let supplied = arguments.iter().skip(1).find_map(|argument| {
271            let argument = argument.as_os_str().as_encoded_bytes();
272            COMMAND_SCOPE_OPTIONS
273                .iter()
274                .map(|option| option.long)
275                .find(|option| {
276                    argument == option.as_bytes()
277                        || argument
278                            .strip_prefix(option.as_bytes())
279                            .is_some_and(|suffix| suffix.starts_with(b"="))
280                })
281        });
282        let Some(option) = supplied else {
283            return Ok(());
284        };
285        let command = match self.command {
286            Commands::Setup(_) => "setup",
287            Commands::Remove(_) => "remove",
288            Commands::Cache(_) => "cache",
289            Commands::Update(_) => "update",
290            Commands::Upgrade(_) => "upgrade",
291            _ => unreachable!("repository-free commands checked above"),
292        };
293        Err(clap::Error::raw(
294            clap::error::ErrorKind::ArgumentConflict,
295            format!("repository option {option} cannot be used with `{command}`"),
296        ))
297    }

299    /// Resolve global options into a [`Config`].
300    ///
301    /// # Errors
302    ///
303    /// Returns an error when the repository root cannot be canonicalized or is
304    /// an unsafe broad root without the explicit override.
305    pub fn config(&self) -> Result<Config> {
306        let mut config = Config::discover_with_broad_root(
307            &self.root,
308            self.database.clone(),
309            self.allow_broad_root,
310        )?;
311        if let Some(value) = self.max_walk_entries {
312            config.max_walk_entries = value.get();
313        }
314        if let Some(value) = self.max_files {
315            config.max_files = value.get();
316        }
317        if let Some(value) = self.max_total_source_bytes {
318            config.max_total_source_bytes = value.get();
319        }
320        if let Some(value) = self.max_depth {
321            config.max_depth = value.get();
322        }
323        if let Some(value) = self.max_file_bytes {
324            config.max_file_bytes = value.get();
325        }
326        if let Some(value) = self.max_prepare_batch_files {
327            config.max_prepare_batch_files = value.get();
328        }
329        if let Some(value) = self.max_prepare_batch_bytes {
330            config.max_prepare_batch_bytes = value.get();
331        }
332        if let Some(value) = self.max_index_workers {
333            config.max_index_workers = value.get();
334        }
335        config.include_generated = self.include_generated;
336        config.tokenizer = self.tokenizer;
337        config.discovery_limits().validate()?;
338        Ok(config)
339    }

355    /// Convert the parsed CLI into an application request.
356    pub fn app_request(self) -> AppRequest {
357        match self.command {
358            Commands::Index { rebuild } => AppRequest::Index { rebuild },
359            Commands::Status => AppRequest::Status,
360            Commands::Savings(args) => {
361                args.snapshot
362                    .map_or(AppRequest::Savings, |snapshot| AppRequest::SavingsDelta {
363                        snapshot,
364                    })
365            }
366            Commands::Files(args) => {
367                let max_response_tokens = args.max_response_tokens;
368                let request: FilesRequest = args.into();
369                max_response_tokens.map_or(AppRequest::Files(request.clone()), |limit| {
370                    AppRequest::FilesWithOptions {
371                        request,
372                        max_response_tokens: limit,
373                    }
374                })
375            }
376            Commands::Search(args) => {
377                let max_response_tokens = args.max_response_tokens;
378                let request: SearchRequest = args.into();
379                max_response_tokens.map_or(AppRequest::Search(request.clone()), |limit| {
380                    AppRequest::SearchWithOptions {
381                        request,
382                        max_response_tokens: limit,
383                    }
384                })
385            }
386            Commands::Outline(args) => {
387                let max_response_tokens = args.max_response_tokens;
388                let request: OutlineRequest = args.into();
389                max_response_tokens.map_or(AppRequest::Outline(request.clone()), |limit| {
390                    AppRequest::OutlineWithOptions {
391                        request,
392                        max_response_tokens: limit,
393                    }
394                })
395            }
396            Commands::Read(args) => {
397                let max_response_tokens = args.max_response_tokens;
398                let request: ReadRequest = args.into();
399                max_response_tokens.map_or(AppRequest::Read(request.clone()), |limit| {
400                    AppRequest::ReadWithOptions {
401                        request,
402                        max_response_tokens: limit,
403                    }
404                })
405            }
406            Commands::History(args) => {
407                let max_response_tokens = args.max_response_tokens;
408                let request: HistoryRequest = args.into();
409                max_response_tokens.map_or(AppRequest::History(request.clone()), |limit| {
410                    AppRequest::HistoryWithOptions {
411                        request,
412                        max_response_tokens: limit,
413                    }
414                })
415            }
416            Commands::Json(args) => {
417                let max_response_tokens = args.max_response_tokens;
418                let request: JsonRequest = args.into();
419                max_response_tokens.map_or(AppRequest::Json(request.clone()), |limit| {
420                    AppRequest::JsonWithOptions {
421                        request,
422                        max_response_tokens: limit,
423                    }
424                })
425            }
426            Commands::Context(args) => {
427                let workflow = args.workflow.into();
428                let handoff = args.handoff_request();
429                let workflow_evidence = args.workflow_evidence();
430                let max_response_tokens = args.max_response_tokens;
431                AppRequest::Context {
432                    request: args.into(),
433                    workflow,
434                    workflow_evidence,
435                    handoff: handoff.map(Box::new),
436                    max_response_tokens,
437                }
438            }
439            Commands::Doctor => AppRequest::Doctor,
440            Commands::Mcp(args) => AppRequest::Mcp {
441                result_mode: args.result_mode,
442            },
443            Commands::Setup(args) => AppRequest::Setup(args.into()),
444            Commands::Remove(args) => AppRequest::Remove(args.into()),
445            Commands::Cache(args) => match args.command {
446                CacheCommand::List(args) => AppRequest::CacheListV2(args.into()),
447                CacheCommand::Prune(args) => AppRequest::CachePruneV2(args.into()),
448            },
449            Commands::Update(args) | Commands::Upgrade(args) => AppRequest::Upgrade {
450                check: args.check,
451                yes: args.yes,
452            },
453        }
454    }
