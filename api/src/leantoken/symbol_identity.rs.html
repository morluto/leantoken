3use std::collections::{BTreeSet, HashMap};
4
5/// Maximum number of index literals generated for one Unicode simple-fold query.
6///
7/// Most queries have one representation. The bound prevents repeated characters
8/// such as `s`/long-s or `k`/Kelvin-sign from creating an exponential FTS query.
9pub(crate) const MAX_CASE_FOLD_LITERAL_VARIANTS: usize = 32;
10
11/// Complete, bounded index literals for one Unicode simple-case-folded value.
12#[derive(Debug, Clone, PartialEq, Eq)]
13pub(crate) struct CaseFoldLiteralVariants {
14    pub(crate) values: Vec<String>,
15    pub(crate) expanded: bool,
16}
17
18/// Expand a literal into the scalar alternatives accepted by Rust's regex engine.
19///
20/// SQLite's bundled FTS and `lower()` behavior cannot be used as the correctness
21/// oracle for Unicode case-insensitive matching. Returning `None` means the
22/// complete Cartesian product exceeded the hard bound and the caller must use a
23/// bounded full-scan verifier instead of an incomplete index query.
24pub(crate) fn case_fold_literal_variants(value: &str) -> Option<CaseFoldLiteralVariants> {
25    let unique = value.chars().collect::<BTreeSet<_>>();
26    let mut alternatives = HashMap::with_capacity(unique.len());
27    for character in unique {
28        let mut class =
29            regex_syntax::hir::ClassUnicode::new([regex_syntax::hir::ClassUnicodeRange::new(
30                character, character,
31            )]);
32        class.try_case_fold_simple().ok()?;
33        let mut choices = BTreeSet::new();
34        for range in class.iter() {
35            for codepoint in u32::from(range.start())..=u32::from(range.end()) {
36                choices.insert(ascii_fold(char::from_u32(codepoint)?));
37            }
38        }
39        alternatives.insert(character, choices.into_iter().collect::<Vec<_>>());
40    }
41
42    let mut values = vec![String::new()];
43    for character in value.chars() {
44        let choices = alternatives
45            .get(&character)
46            .expect("query character alternatives were prepared");
47        if values.len().checked_mul(choices.len())? > MAX_CASE_FOLD_LITERAL_VARIANTS {
48            return None;
49        }
50        let mut expanded = Vec::with_capacity(values.len() * choices.len());
51        for prefix in &values {
52            for choice in choices {
53                let mut variant = String::with_capacity(prefix.len() + choice.len_utf8());
54                variant.push_str(prefix);
55                variant.push(*choice);
56                expanded.push(variant);
57            }
58        }
59        values = expanded;
60    }
61    values.sort();
62    values.dedup();
63    let ascii_canonical = value.chars().map(ascii_fold).collect::<String>();
64    let expanded = values.len() != 1 || values.first() != Some(&ascii_canonical);
65    Some(CaseFoldLiteralVariants { values, expanded })
66}

68/// Build one FTS expression that admits every complete case-folded literal.
69pub(crate) fn case_fold_fts_query(variants: &CaseFoldLiteralVariants) -> String {
70    variants
71        .values
72        .iter()
73        .map(|value| format!("\"{}\"", value.replace('"', "\"\"")))
74        .collect::<Vec<_>>()
75        .join(" OR ")
76}
77
78fn ascii_fold(character: char) -> char {
79    if character.is_ascii() {
80        character.to_ascii_lowercase()
81    } else {
82        character
83    }
84}

86/// Whether a literal regex match covers the complete candidate identity.
87pub(crate) fn literal_match_is_entire(matcher: &regex::Regex, candidate: &str) -> bool {
88    matcher
89        .find(candidate)
90        .is_some_and(|matched| matched.start() == 0 && matched.end() == candidate.len())
91}

93/// Match a bare or qualified symbol identity with a compiled literal matcher.
94pub(crate) fn symbol_identity_matches_case_fold(
95    matcher: &regex::Regex,
96    name: &str,
97    parent: Option<&str>,
98) -> bool {
99    literal_match_is_entire(matcher, name)
100        || parent.is_some_and(|parent| {
101            let mut qualified = String::with_capacity(parent.len() + name.len() + 1);
102            qualified.push_str(parent);
103            qualified.push('.');
104            qualified.push_str(name);
105            literal_match_is_entire(matcher, &qualified)
106        })
107}

109/// Search a bare name or, for a qualified request, its canonical identity.
110pub(crate) fn symbol_identity_contains_case_fold(
111    requested: &str,
112    matcher: &regex::Regex,
113    name: &str,
114    parent: Option<&str>,
115) -> bool {
116    matcher.is_match(name)
117        || split_qualified_symbol(requested).is_some()
118            && parent.is_some_and(|parent| {
119                let mut qualified = String::with_capacity(parent.len() + name.len() + 1);
120                qualified.push_str(parent);
121                qualified.push('.');
122                qualified.push_str(name);
123                matcher.is_match(&qualified)
124            })
125}

127/// Result of resolving one symbol identity within an already bounded scope.
128pub(crate) enum SymbolResolution<T> {
129    NotFound,
130    Unique(T),
131    Ambiguous,
132}
133
134/// Return the qualified owner/name parts when `requested` uses `parent.name`.
135pub(crate) fn split_qualified_symbol(requested: &str) -> Option<(&str, &str)> {
136    let (parent, name) = requested.rsplit_once('.')?;
137    (!parent.is_empty() && !name.is_empty()).then_some((parent, name))
138}
139
140/// Match either a bare parsed name or its canonical `parent.name` identity.
141pub(crate) fn symbol_identity_matches(requested: &str, name: &str, parent: Option<&str>) -> bool {
142    requested == name
143        || parent.is_some_and(|parent| {
144            requested
145                .strip_prefix(parent)
146                .and_then(|suffix| suffix.strip_prefix('.'))
147                == Some(name)
148        })
149}
150
151/// Resolve zero, one, or multiple matches without silently choosing the first.
152pub(crate) fn resolve_symbol_matches<T>(
153    mut matches: impl Iterator<Item = T>,
154) -> SymbolResolution<T> {
155    let Some(first) = matches.next() else {
156        return SymbolResolution::NotFound;
157    };
158    if matches.next().is_some() {
159        SymbolResolution::Ambiguous
160    } else {
161        SymbolResolution::Unique(first)
162    }
163}

166mod tests {

169    #[test]
170    fn canonical_symbol_identity_requires_an_exact_owner_boundary() {
171        assert!(symbol_identity_matches("run", "run", Some("Service")));
172        assert!(symbol_identity_matches(
173            "Service.run",
174            "run",
175            Some("Service")
176        ));
177        assert!(!symbol_identity_matches(
178            "Other.run",
179            "run",
180            Some("Service")
181        ));
182        assert!(!symbol_identity_matches(
183            "MyService.run",
184            "run",
185            Some("Service")
186        ));
187        assert_eq!(
188            split_qualified_symbol("module.Service.run"),
189            Some(("module.Service", "run"))
190        );
191        assert_eq!(split_qualified_symbol("run"), None);
192    }
193
194    #[test]
195    fn case_fold_literals_expand_only_the_unicode_equivalence_classes() {
196        let ordinary = case_fold_literal_variants("Alpha").expect("ordinary variants");
197        assert_eq!(ordinary.values, ["alpha"]);
198        assert!(!ordinary.expanded);
199
200        let kelvin = case_fold_literal_variants("k").expect("Kelvin variants");
201        assert_eq!(kelvin.values, ["k", "K"]);
202        assert!(kelvin.expanded);
203
204        let georgian = case_fold_literal_variants("აbc").expect("Georgian variants");
205        assert_eq!(georgian.values, ["აbc", "Აbc"]);
206        assert!(georgian.expanded);
207
208        let uncased = case_fold_literal_variants("🦀").expect("uncased variants");
209        assert_eq!(uncased.values, ["🦀"]);
210        assert!(!uncased.expanded);
211
212        assert!(case_fold_literal_variants(&"s".repeat(6)).is_none());
213    }
