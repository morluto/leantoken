4#[serde(rename_all = "snake_case")]
5/// Search candidate source.
6pub enum SearchMode {
7    /// Combine structural and lexical candidates.
8    #[default]
9    Auto,
10    /// Match a literal substring.
11    Text,
12    /// Verify a Rust regular expression over indexed chunks.
13    Regex,
14    /// Search identifier tokens and structural names.
15    Identifier,
16    /// Search definitions only.
17    Symbol,
18    /// Search syntactic references only.
19    Reference,
20}
21
22impl SearchMode {
23    /// Modes that can produce an exhaustive occurrence result.
24    pub(crate) const EXHAUSTIVE_MODES: [Self; 2] = [Self::Text, Self::Regex];
25
26    pub(crate) fn supports_all_occurrences(self) -> bool {
27        Self::EXHAUSTIVE_MODES.contains(&self)
28    }
29
30    pub(crate) const fn wire_name(self) -> &'static str {
31        match self {
32            Self::Auto => "auto",
33            Self::Text => "text",
34            Self::Regex => "regex",
35            Self::Identifier => "identifier",
36            Self::Symbol => "symbol",
37            Self::Reference => "reference",
38        }
39    }
40}

44/// Explicit lifecycle action for one exhaustive-query coverage receipt.

49    Reuse {

51        #[schemars(length(max = 128))]
52        receipt_id: String,
53    },
54}

57/// Input for `leantoken.search`.
58pub struct SearchRequest {
59    /// Text, identifier, symbol, or regular expression to find.
60    pub query: String,
61    /// Candidate source to search.
62    #[serde(default)]
63    pub mode: SearchMode,
64    /// Include only matching repository paths.
65    #[serde(default)]
66    pub include_paths: Vec<String>,
67    /// Exclude matching repository paths.
68    #[serde(default)]
69    pub exclude_paths: Vec<String>,
70    /// Boost matching repository paths without filtering other results.
71    #[serde(default)]
72    pub focus_paths: Vec<String>,
73    /// Maximum hits to return.
74    #[serde(default)]
75    pub max_results: Option<usize>,
76    /// Maximum source tokens across returned excerpts.
77    #[serde(default)]
78    pub max_tokens: Option<usize>,
79    /// Lines included before and after each match.
80    #[serde(default)]
81    pub context_lines: Option<usize>,
82    /// Preserve query case when matching.
83    #[serde(default)]
84    pub case_sensitive: bool,
85    /// Return every text or regex occurrence instead of one hit per indexed chunk.
86    #[serde(default)]
87    pub all_occurrences: bool,
88    /// Prefer a structural definition when lexical and structural channels find the same definition.
89    #[serde(default)]
90    pub prefer_structural: bool,
91    /// Server-managed receipt whose previously returned evidence should be suppressed.
92    #[serde(default)]
93    pub receipt_id: Option<String>,
94    /// Explicit record or reuse action for an exhaustive text or regex query.
95    #[serde(default)]
96    pub query_receipt: Option<QueryReceiptAction>,
97    /// Cursor returned by an earlier response from the same generation.
98    #[serde(default)]
99    pub cursor: Option<String>,
100}

383#[non_exhaustive]
384#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
385#[serde(rename_all = "snake_case")]
386/// Privacy-safe reason a regex request retained the bounded full-scan path.
387pub enum RegexPlanFallbackReason {
388    /// The evaluation explicitly forced the full-scan oracle.
389    PlanningDisabled,
390    /// Unicode case folding cannot be represented by SQLite's ASCII folding.
391    CaseInsensitiveUnicode,
392    /// The HIR parser unexpectedly rejected a regex accepted by the matcher.
393    HirParseFailed,
394    /// Recursive analysis crossed its bounded HIR-node limit.
395    PlanNodeLimit,
396    /// Candidate translation crossed its bounded term-count limit.
397    PlanTermLimit,
398    /// Candidate translation crossed its bounded aggregate term-byte limit.
399    PlanTermBytesLimit,
400    /// Prefix and suffix literal sequences were infinite or not indexable.
401    LiteralSequenceUnavailable,
402}
