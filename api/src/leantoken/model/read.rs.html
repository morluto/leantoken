3#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
4/// Input for `leantoken.read`.
5pub struct ReadRequest {
6    /// Repository-relative file path.
7    pub path: String,
8    /// First one-based line; defaults to the start of the file.
9    #[serde(default)]
10    pub start_line: Option<usize>,
11    /// Last one-based line; defaults to the end of the file.
12    #[serde(default)]
13    pub end_line: Option<usize>,
14    /// Indexed symbol to read; cannot be combined with line fields.
15    #[serde(default)]
16    pub symbol: Option<String>,
17    /// Indexed Markdown or LaTeX section title or outline signature to read.
18    #[serde(default)]
19    pub heading: Option<String>,
20    /// One-based occurrence of a duplicate document heading; defaults to 1.
21    #[serde(default)]
22    pub heading_occurrence: Option<usize>,
23    /// Opaque cursor returned by a truncated read; cannot be combined with a new target.
24    #[serde(default)]
25    pub continuation_cursor: Option<String>,
26    /// Maximum source tokens to return.
27    #[serde(default)]
28    pub max_tokens: Option<usize>,
29    /// Hash from the same prior range; matching content returns `not_modified`.
30    #[serde(default)]
31    pub expected_hash: Option<String>,
32    /// Record a bounded base and prefer a cheaper changed follow-up. Without
33    /// `expected_hash`, select the latest compatible base for this exact target.
34    #[serde(default)]
35    pub delta: bool,
36    /// Server-managed receipt whose previously returned evidence should be suppressed.
37    #[serde(default)]
38    pub receipt_id: Option<String>,
39}

41#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
42pub struct ReadResponse {
43    pub path: String,
44    pub status: ReadStatus,
45    /// First line in the complete resolved target.
46    #[serde(default)]
47    pub target_start_line: usize,
48    /// Last line in the complete resolved target.
49    #[serde(default)]
50    pub target_end_line: usize,
51    /// First line represented by this response page.
52    #[serde(default)]
53    pub returned_start_line: usize,
54    /// Last line represented by this response page.
55    #[serde(default)]
56    pub returned_end_line: usize,
57    /// Compatibility alias for `returned_start_line`.
58    pub start_line: usize,
59    /// Compatibility alias for `returned_end_line`.
60    pub end_line: usize,
61    /// Whether source remains after this response page.
62    #[serde(default)]
63    pub truncated: bool,
64    /// First line represented by the next response page.
65    #[serde(default, skip_serializing_if = "Option::is_none")]
66    pub next_start_line: Option<usize>,
67    /// Opaque continuation bound to this repository generation and live file content.
68    #[serde(default, skip_serializing_if = "Option::is_none")]
69    pub continuation_cursor: Option<String>,
70    /// Whether an explicit or automatically selected base matched this response page.
71    #[serde(default)]
72    pub not_modified: bool,
73    #[serde(skip_serializing_if = "Option::is_none")]
74    pub content: Option<String>,
75    /// Unified diff from the requested base hash to `content_hash`.
76    #[serde(default, skip_serializing_if = "Option::is_none")]
77    pub delta: Option<String>,
78    /// Bounded delta decision and accounting metadata.
79    #[serde(default, skip_serializing_if = "Option::is_none")]
80    pub delta_receipt: Option<ReadDeltaReceipt>,
81    pub content_hash: String,
82    #[serde(skip_serializing_if = "Option::is_none")]
83    pub indexed_hash: Option<String>,
84    pub index_stale: bool,
85    pub meta: ResponseMeta,
86}

88#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema, PartialEq, Eq)]
89#[serde(rename_all = "snake_case")]
90pub enum ReadStatus {
91    Content,
92    /// The response contains only part of the resolved target.
93    Truncated,
94    NotModified,
95    /// A complete unified diff is returned instead of full current content.
96    Delta,
97    /// A server-managed evidence receipt already contained the exact current content.
98    ReceiptSuppressed,
99}

101/// Result selected by an opt-in read delta request.
102#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema, PartialEq, Eq)]
103#[serde(rename_all = "snake_case")]
104pub enum ReadDeltaOutcome {
105    /// Full current content was returned.
106    Full,
107    /// A complete unified diff was returned.
108    Delta,
109    /// The requested or automatically selected base already identifies current content.
110    NotModified,
111    /// A general evidence receipt already contained the exact current content.
112    ReceiptSuppressed,
113}
114
115/// Why an opt-in read delta attempt returned full content.
116#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema, PartialEq, Eq)]
117#[serde(rename_all = "snake_case")]
118pub enum ReadDeltaFallback {
119    /// No bounded base matched the exact target and optional requested hash.
120    BaseUnavailable,
121    /// The resolved target or returned coordinates changed.
122    TargetChanged,
123    /// The current response is truncated and cannot be a complete delta target.
124    CurrentTruncated,
125    /// The current page exceeds the per-entry delta-state bound.
126    ContentTooLarge,
127    /// The complete delta response was not smaller than a full-content response.
128    DeltaNotSmaller,
129}

131/// Provenance and token accounting for one opt-in read delta decision.
132#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema)]
133pub struct ReadDeltaReceipt {
134    /// Stable hash of the repository and caller-selected target.
135    pub target_key: String,
136    /// Requested or automatically selected prior content hash.
137    #[serde(default, skip_serializing_if = "Option::is_none")]
138    pub base_hash: Option<String>,
139    /// Hash of the complete current response page.
140    pub head_hash: String,
141    /// Repository generation observed when the bounded base was captured.
142    #[serde(default, skip_serializing_if = "Option::is_none")]
143    pub base_generation: Option<u64>,
144    /// Repository generation used to resolve the current target.
145    pub head_generation: u64,
146    /// Selected response form.
147    pub outcome: ReadDeltaOutcome,
148    /// Tokens required by full current content.
149    pub full_tokens: usize,
150    /// Tokens in the returned delta, or zero for `not_modified`.
151    #[serde(default, skip_serializing_if = "Option::is_none")]
152    pub delta_tokens: Option<usize>,
153    /// Full-content tokens avoided by the selected response.
154    pub avoided_tokens: usize,
155    /// Explicit reason full content was retained after a delta attempt.
156    #[serde(default, skip_serializing_if = "Option::is_none")]
157    pub fallback_reason: Option<ReadDeltaFallback>,
158}
