3/// Typed read target selected by the caller. The flat `ReadRequest` option
4/// fields remain the wire-compatible input; this enum provides a typed
5/// projection for programmatic callers and internal resolution.
6#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
7#[serde(tag = "kind", rename_all = "snake_case")]
8pub enum ReadTarget {
9    /// Read one indexed symbol definition.
10    Symbol { identity: SymbolIdentity },
11    /// Read one indexed Markdown or LaTeX section by exact title or outline signature.
12    Heading {
13        name: String,
14        #[serde(default = "default_heading_occurrence")]
15        occurrence: usize,
16    },
17    /// Read one inclusive one-based line range.
18    Lines { start: usize, end: usize },
19    /// Continue a truncated read without losing a partial final line.
20    Continuation { cursor: String },
21}
22
23fn default_heading_occurrence() -> usize {
24    1
25}

27/// I/O and verification policy for a live read.
28///
29/// `Bounded` (default) stops reading after the requested page is satisfied,
30/// reports `index_state: unknown`, and emits a metadata-bound continuation
31/// cursor. `Full` hashes the complete live file, reports current/stale with
32/// live and indexed hashes, and is required for delta requests.
33#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
34#[serde(rename_all = "snake_case")]
35pub enum ReadPolicy {
36    /// Stop after the requested page; no full-file hash or index staleness.
37    #[default]
38    Bounded,
39    /// Hash the complete live file and report index verification metadata.
40    Full,
41}
42
43impl ReadPolicy {
44    pub(crate) const fn is_full(self) -> bool {
45        matches!(self, Self::Full)
46    }
47}
48
49impl std::fmt::Display for ReadPolicy {
50    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
51        match self {
52            Self::Bounded => write!(f, "bounded"),
53            Self::Full => write!(f, "full"),
54        }
55    }
56}

58/// Index verification state reported by a read response.
59///
60/// `Unknown` is reported by bounded reads that stop before EOF. `Current` and
61/// `Stale` are reported by full reads that hash the complete live file and
62/// compare it to the indexed snapshot.
63#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
64#[serde(rename_all = "snake_case")]
65pub enum ReadIndexState {
66    /// The live file hash matches the indexed snapshot.
67    Current,
68    /// The live file hash differs from the indexed snapshot.
69    Stale,
70    /// The read stopped before EOF and could not verify index freshness.
71    #[default]
72    Unknown,
73}

263/// Bounded guidance for avoiding repeated undersized continuation reads.
264#[derive(Debug, Clone, Serialize, Deserialize, JsonSchema, PartialEq, Eq)]

276    /// Fewest additional calls possible under the configured source-token ceiling.
277    pub minimum_remaining_pages: usize,
278}

387    pub avoided_tokens: usize,

392    #[serde(default, skip_serializing_if = "Option::is_none")]
393    pub fallback_reason: Option<ReadDeltaFallback>,
