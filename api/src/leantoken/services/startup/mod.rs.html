3const STARTUP_BUSY_TIMEOUT: Duration = Duration::from_millis(250);
4const STARTUP_RETRY_INITIAL_DELAY: Duration = Duration::from_millis(25);
5const STARTUP_RETRY_MAX_DELAY: Duration = Duration::from_millis(500);
6pub(super) const CANCELLATION_POLL_INTERVAL: Duration = Duration::from_millis(25);
7pub(super) const INITIAL_INDEX_IDLE_GRACE: Duration = Duration::from_secs(1);
8pub(super) const INITIAL_INDEX_PROBE_INTERVAL: Duration = Duration::from_millis(100);

10impl Services {
11    pub fn open(config: Config) -> Result<Self> {
12        config.validate()?;
13        let coordination = IndexCoordination::for_database(&config.database_path);
14        let cancellation = CancellationToken::new();
15        let cache_lease = coordination.acquire_cache_lease(&cancellation)?;
16        let _initialization = coordination.acquire_initialization(&cancellation)?;
17        Self::open_once(&config, None, cache_lease)
18    }
19
20    /// Open services under exclusive cache initialization ownership, retrying
21    /// transient SQLite contention until the caller cancels.
22    pub fn open_cancellable(config: Config, cancellation: &CancellationToken) -> Result<Self> {
23        config.validate()?;
24        let coordination = IndexCoordination::for_database(&config.database_path);
25        let cache_lease = coordination.acquire_cache_lease(cancellation)?;
26        let _initialization = coordination.acquire_initialization(cancellation)?;
27        let mut delay = STARTUP_RETRY_INITIAL_DELAY;
28        let mut attempt = 0u32;
29
30        loop {
31            validation::check_cancelled(cancellation)?;
32            match Self::open_once(&config, Some(STARTUP_BUSY_TIMEOUT), cache_lease.clone()) {
33                Ok(services) => return Ok(services),
34                Err(error) if is_database_contention(&error) => {
35                    attempt = attempt.saturating_add(1);
36                    if attempt == 1 || attempt.is_multiple_of(20) {
37                        tracing::warn!(
38                            attempt,
39                            retry_delay_ms = delay.as_millis(),
40                            database = %config.database_path.display(),
41                            %error,
42                            "cache initialization is waiting for SQLite contention"
43                        );
44                    }
45                    wait_cancellable(cancellation, delay)?;
46                    delay = delay.saturating_mul(2).min(STARTUP_RETRY_MAX_DELAY);
47                }
48                Err(error) => return Err(error),
49            }
50        }
51    }
52
53    fn open_once(
54        config: &Config,
55        startup_timeout: Option<Duration>,
56        cache_lease: CacheLease,
57    ) -> Result<Self> {
58        let open_storage = || match startup_timeout {
59            Some(timeout) => Storage::open_for_repository_scoped_with_startup_timeout(
60                &config.database_path,
61                &config.root,
62                config.index_scope().full_digest(),
63                timeout,
64            ),
65            None => Storage::open_for_repository_scoped(
66                &config.database_path,
67                &config.root,
68                config.index_scope().full_digest(),
69            ),
70        };
71        let storage = match open_storage() {
72            Ok(storage) => storage,
73            Err(error) if config.database_is_managed_cache && is_database_corruption(&error) => {
74                tracing::warn!(database = %config.database_path.display(), "rebuilding corrupt managed index");
75                remove_database_artifacts(&config.database_path)?;
76                open_storage()?
77            }
78            Err(error) => return Err(error),
79        };
80        Self::from_parts(Arc::new(config.clone()), storage, cache_lease)
81    }
82
83    fn from_parts(config: Arc<Config>, storage: Storage, cache_lease: CacheLease) -> Result<Self> {
84        let tokenizer = config.tokenizer;
85        let indexer = Indexer::new(Arc::clone(&config), storage.clone())?;
86        let repository_root = indexer.repository_root();
87        let coordination = IndexCoordination::for_database(&config.database_path);
88        let active_reconciliations = Arc::new(AtomicUsize::new(0));
89        let reconciliation_changed = Arc::new(tokio::sync::Notify::new());
90        let reconciliation = reconciliation::ReconciliationCoordinator::new(
91            indexer.clone(),
92            coordination.clone(),
93            Arc::clone(&active_reconciliations),
94            Arc::clone(&reconciliation_changed),
95        );
96        let observer = observer::ServiceObserver::new(storage.clone(), tokenizer);
97        Ok(Self {
98            config,
99            storage,
100            indexer,
101            repository_root,
102            coordination,
103            _cache_lease: cache_lease,
104            active_reconciliations,
105            reconciliation_changed,
106            read_deltas: Arc::new(read_delta::ReadDeltaRegistry::default()),
107            blocking_executor: executor::BlockingExecutor::default(),
108            response_accountant: accounting::ResponseAccountant::new(tokenizer),
109            observer,
110            reconciliation,
111        })
112    }
113}

115fn is_database_corruption(error: &Error) -> bool {
116    matches!(
117        sqlite_error_code(error),
118        Some(rusqlite::ErrorCode::DatabaseCorrupt | rusqlite::ErrorCode::NotADatabase)
119    )
120}

122fn remove_database_artifacts(database: &std::path::Path) -> Result<()> {
123    for suffix in ["", "-wal", "-shm"] {
124        let mut path = database.as_os_str().to_os_string();
125        path.push(suffix);
126        match fs::remove_file(std::path::PathBuf::from(path)) {
127            Ok(()) => {}
128            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
129            Err(error) => return Err(error.into()),
130        }
131    }
132    Ok(())
133}

135fn wait_cancellable(cancellation: &CancellationToken, duration: Duration) -> Result<()> {
136    let deadline = Instant::now() + duration;
137    loop {
138        validation::check_cancelled(cancellation)?;
139        let remaining = deadline.saturating_duration_since(Instant::now());
140        if remaining.is_zero() {
141            return Ok(());
142        }
143        thread::sleep(remaining.min(CANCELLATION_POLL_INTERVAL));
144    }
