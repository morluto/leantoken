21fn validate_history_request(request: &HistoryRequest) -> Result<()> {
22    match &request.operation {
23        HistoryOperation::ReadSymbol {
24            path,
25            symbol,
26            revision,
27        }
28        | HistoryOperation::SymbolLog {
29            path,
30            symbol,
31            revision: Some(revision),
32        } => {
33            validate_input(path, "path", MAX_PATH_BYTES)?;
34            validate_input(symbol, "symbol", MAX_PATTERN_BYTES)?;
35            validate_input(revision, "revision", MAX_PATTERN_BYTES)?;
36        }
37        HistoryOperation::SymbolLog {
38            path,
39            symbol,
40            revision: None,
41        } => {
42            validate_input(path, "path", MAX_PATH_BYTES)?;
43            validate_input(symbol, "symbol", MAX_PATTERN_BYTES)?;
44        }
45        HistoryOperation::DiffSymbol {
46            path,
47            symbol,
48            base_revision,
49            head_revision,
50        } => {
51            validate_input(path, "path", MAX_PATH_BYTES)?;
52            validate_input(symbol, "symbol", MAX_PATTERN_BYTES)?;
53            validate_input(base_revision, "base revision", MAX_PATTERN_BYTES)?;
54            validate_input(head_revision, "head revision", MAX_PATTERN_BYTES)?;
55        }
56    }
57    if let Some(max_results) = request.max_results {
58        validate_positive_request_limit("max_results", max_results, MAX_HISTORY_RESULTS)?;
59    }
60    Ok(())
61}

63fn normalize_operation(operation: HistoryOperation) -> Result<HistoryOperation> {
64    Ok(match operation {
65        HistoryOperation::ReadSymbol {
66            path,
67            symbol,
68            revision,
69        } => HistoryOperation::ReadSymbol {
70            path: normalize_relative(&path)?,
71            symbol,
72            revision,
73        },
74        HistoryOperation::DiffSymbol {
75            path,
76            symbol,
77            base_revision,
78            head_revision,
79        } => HistoryOperation::DiffSymbol {
80            path: normalize_relative(&path)?,
81            symbol,
82            base_revision,
83            head_revision,
84        },
85        HistoryOperation::SymbolLog {
86            path,
87            symbol,
88            revision,
89        } => HistoryOperation::SymbolLog {
90            path: normalize_relative(&path)?,
91            symbol,
92            revision,
93        },
94    })
95}

97impl Services {
98    /// Retrieve symbol-aware evidence from immutable Git revisions.
99    pub async fn history(&self, request: HistoryRequest) -> Result<HistoryResponse> {
100        self.history_cancellable(request, CancellationToken::new())
101            .await
102    }
103
104    /// Retrieve symbol-aware Git evidence with cooperative cancellation.
105    pub async fn history_cancellable(
106        &self,
107        request: HistoryRequest,
108        cancellation: CancellationToken,
109    ) -> Result<HistoryResponse> {
110        validate_history_request(&request)?;
111        let this = self.clone();
112        tokio::task::spawn_blocking(move || this.history_sync(request, &cancellation)).await?
113    }
114
115    fn history_sync(
116        &self,
117        request: HistoryRequest,
118        cancellation: &CancellationToken,
119    ) -> Result<HistoryResponse> {
120        check_cancelled(cancellation)?;
121        validate_history_request(&request)?;
122        let max_results = request.max_results.unwrap_or(DEFAULT_HISTORY_RESULTS);
123        let max_tokens = self.token_limit(request.max_tokens, DEFAULT_HISTORY_TOKENS)?;
124        let operation = normalize_operation(request.operation)?;
125        let generation = self.consistent(|_, generation| Ok(generation))?;
126        let mut response = match operation {
127            HistoryOperation::ReadSymbol {
128                path,
129                symbol,
130                revision,
131            } => {
132                let mut resolved = self.historical_symbol(&path, &symbol, &revision)?;
133                let (content, emitted_tokens) = self
134                    .config
135                    .tokenizer
136                    .truncate(&resolved.content, max_tokens);
137                let truncated = content.len() < resolved.content.len();
138                resolved.symbol.returned_end_line =
139                    returned_end_line(resolved.symbol.target_start_line, content);
140                resolved.symbol.truncated = truncated;
141                resolved.symbol.content = Some(content.to_owned());
142                HistoryResponse {
143                    kind: "read_symbol".into(),
144                    symbol: Some(resolved.symbol),
145                    before: None,
146                    after: None,
147                    diff: None,
148                    diff_truncated: false,
149                    commits: Vec::new(),
150                    result_complete: !truncated,
151                    meta: self.meta(generation, emitted_tokens, None),
152                }
153            }
154            HistoryOperation::DiffSymbol {
155                path,
156                symbol,
157                base_revision,
158                head_revision,
159            } => {
160                check_cancelled(cancellation)?;
161                let mut before = self.historical_symbol(&path, &symbol, &base_revision)?;
162                check_cancelled(cancellation)?;
163                let mut after = self.historical_symbol(&path, &symbol, &head_revision)?;
164                let full_diff = TextDiff::from_lines(&before.content, &after.content)
165                    .unified_diff()
166                    .context_radius(3)
167                    .header(
168                        &format!("{}:{}", before.symbol.revision, path),
169                        &format!("{}:{}", after.symbol.revision, path),
170                    )
171                    .to_string();
172                let total_diff_tokens = self.config.tokenizer.count(&full_diff);
173                let (diff, emitted_tokens) = self.config.tokenizer.truncate(&full_diff, max_tokens);
174                let diff_truncated = emitted_tokens < total_diff_tokens;
175                before.symbol.content = None;
176                before.symbol.returned_end_line = 0;
177                after.symbol.content = None;
178                after.symbol.returned_end_line = 0;
179                HistoryResponse {
180                    kind: "diff_symbol".into(),
181                    symbol: None,
182                    before: Some(before.symbol),
183                    after: Some(after.symbol),
184                    diff: Some(diff.to_owned()),
185                    diff_truncated,
186                    commits: Vec::new(),
187                    result_complete: !diff_truncated,
188                    meta: self.meta(generation, emitted_tokens, None),
189                }
190            }
191            HistoryOperation::SymbolLog {
192                path,
193                symbol,
194                revision,
195            } => {
196                let revision = revision.as_deref().unwrap_or("HEAD");
197                let resolved = self.historical_symbol(&path, &symbol, revision)?;
198                check_cancelled(cancellation)?;
199                let mut commits = git_line_history(
200                    &self.config.root,
201                    revision,
202                    &path,
203                    resolved.symbol.target_start_line,
204                    resolved.symbol.target_end_line,
205                    max_results.saturating_add(1),
206                )?;
207                let result_complete = commits.len() <= max_results;
208                commits.truncate(max_results);
209                HistoryResponse {
210                    kind: "symbol_log".into(),
211                    symbol: Some(HistoricalSymbol {
212                        content: None,
213                        returned_end_line: 0,
214                        ..resolved.symbol
215                    }),
216                    before: None,
217                    after: None,
218                    diff: None,
219                    diff_truncated: false,
220                    commits: commits
221                        .into_iter()
222                        .map(|commit| SymbolHistoryCommit {
223                            commit: commit.commit,
224                            authored_at: commit.authored_at,
225                            subject: commit.subject,
226                        })
227                        .collect(),
228                    result_complete,
229                    meta: self.meta(generation, 0, None),
230                }
231            }
232        };
233        self.finalize_response(&mut response)?;
234        Ok(response)
235    }

237    fn historical_symbol(
238        &self,
239        path: &str,
240        symbol_name: &str,
241        revision: &str,
242    ) -> Result<ResolvedHistoricalSymbol> {
243        let blob = git_blob_at_revision(
244            &self.config.root,
245            revision,
246            path,
247            usize::try_from(self.config.max_file_bytes).unwrap_or(usize::MAX),
248        )?;
249        let parsed = parser::parse(path, &blob.content)?;
250        let symbol = parsed
251            .symbols
252            .into_iter()
253            .find(|symbol| symbol.name == symbol_name)
254            .ok_or_else(|| Error::SymbolNotFound {
255                path: format!("{path}@{}", blob.revision),
256                symbol: symbol_name.to_owned(),
257            })?;
258        let content = blob
259            .content
260            .get(symbol.start_byte..symbol.end_byte)
261            .ok_or_else(|| Error::InternalFailure("invalid historical symbol range".into()))?
262            .to_owned();
263        let content_hash = crate::text::hash(&content);
264        Ok(ResolvedHistoricalSymbol {
265            symbol: HistoricalSymbol {
266                revision: blob.revision,
267                path: path.to_owned(),
268                name: symbol.name,
269                kind: symbol.kind,
270                parent: symbol.parent,
271                target_start_line: symbol.start_line,
272                target_end_line: symbol.end_line,
273                returned_end_line: symbol.end_line,
274                truncated: false,
275                content: None,
276                content_hash,
277            },
278            content,
279        })
280    }
281}

283fn returned_end_line(start_line: usize, content: &str) -> usize {
284    start_line.saturating_add(content.lines().count().saturating_sub(1))
