32fn char_boundaries(value: &str) -> Vec<usize> {
33    let mut boundaries = value
34        .char_indices()
35        .map(|(index, _)| index)
36        .collect::<Vec<_>>();
37    boundaries.push(value.len());
38    boundaries
39}

41fn symbol_diff_buffer(content: &str) -> Cow<'_, str> {
42    if content.is_empty() || content.ends_with('\n') {
43        Cow::Borrowed(content)
44    } else {
45        Cow::Owned(format!("{content}\n"))
46    }
47}

49fn validate_history_request(request: &HistoryRequest) -> Result<()> {
50    match &request.operation {
51        HistoryOperation::ReadSymbol {
52            path,
53            symbol,
54            revision,
55        }
56        | HistoryOperation::SymbolLog {
57            path,
58            symbol,
59            revision: Some(revision),
60        } => {
61            validate_input(path, "path", MAX_PATH_BYTES)?;
62            validate_input(symbol, "symbol", MAX_PATTERN_BYTES)?;
63            validate_input(revision, "revision", MAX_PATTERN_BYTES)?;
64        }
65        HistoryOperation::SymbolLog {
66            path,
67            symbol,
68            revision: None,
69        } => {
70            validate_input(path, "path", MAX_PATH_BYTES)?;
71            validate_input(symbol, "symbol", MAX_PATTERN_BYTES)?;
72        }
73        HistoryOperation::DiffSymbol {
74            path,
75            symbol,
76            base_revision,
77            head_revision,
78        } => {
79            validate_input(path, "path", MAX_PATH_BYTES)?;
80            validate_input(symbol, "symbol", MAX_PATTERN_BYTES)?;
81            validate_input(base_revision, "base revision", MAX_PATTERN_BYTES)?;
82            validate_input(head_revision, "head revision", MAX_PATTERN_BYTES)?;
83        }
84    }
85    if let Some(max_results) = request.max_results {
86        validate_positive_request_limit("max_results", max_results, MAX_HISTORY_RESULTS)?;
87    }
88    Ok(())
89}

91fn normalize_operation(operation: HistoryOperation) -> Result<HistoryOperation> {
92    Ok(match operation {
93        HistoryOperation::ReadSymbol {
94            path,
95            symbol,
96            revision,
97        } => HistoryOperation::ReadSymbol {
98            path: normalize_relative(&path)?,
99            symbol,
100            revision,
101        },
102        HistoryOperation::DiffSymbol {
103            path,
104            symbol,
105            base_revision,
106            head_revision,
107        } => HistoryOperation::DiffSymbol {
108            path: normalize_relative(&path)?,
109            symbol,
110            base_revision,
111            head_revision,
112        },
113        HistoryOperation::SymbolLog {
114            path,
115            symbol,
116            revision,
117        } => HistoryOperation::SymbolLog {
118            path: normalize_relative(&path)?,
119            symbol,
120            revision,
121        },
122    })
123}

152    /// Retrieve symbol-aware Git evidence under response controls and cancellation.
153    pub async fn history_cancellable_with_options(
154        &self,
155        request: HistoryRequest,
156        options: ServiceCallOptions,
157        cancellation: CancellationToken,
158    ) -> Result<HistoryResponse> {
159        let operation = TokenAccountingOperation::History;
160        self.observe_service_result(operation, self.validate_call_options(options))?;
161        self.observe_service_result(operation, validate_history_request(&request))?;
162        let this = self.clone();
163        let result = self
164            .blocking_executor
165            .run(cancellation, move |cancellation| {
166                this.history_sync(request, options, cancellation)
167            })
168            .await;
169        self.observe_service_result(operation, result)
170    }

172    fn history_sync(
173        &self,
174        request: HistoryRequest,
175        options: ServiceCallOptions,
176        cancellation: &CancellationToken,
177    ) -> Result<HistoryResponse> {
178        check_cancelled(cancellation)?;
179        validate_history_request(&request)?;
180        let max_results = request.max_results.unwrap_or(DEFAULT_HISTORY_RESULTS);
181        let max_tokens = self.token_limit(request.max_tokens, DEFAULT_HISTORY_TOKENS)?;
182        let operation = normalize_operation(request.operation)?;
183        let generation = self.consistent(|_, generation| Ok(generation))?;
184        let mut response = match operation {
185            HistoryOperation::ReadSymbol {
186                path,
187                symbol,
188                revision,
189            } => {
190                let mut resolved = self.historical_symbol(&path, &symbol, &revision)?;
191                let (content, emitted_tokens) = self
192                    .config
193                    .tokenizer
194                    .truncate(&resolved.content, max_tokens);
195                let truncated = content.len() < resolved.content.len();
196                resolved.symbol.returned_end_line =
197                    returned_end_line(resolved.symbol.target_start_line, content);
198                resolved.symbol.truncated = truncated;
199                resolved.symbol.content = Some(content.to_owned());
200                HistoryResponse {
201                    kind: "read_symbol".into(),
202                    symbol: Some(resolved.symbol),
203                    before: None,
204                    after: None,
205                    diff: None,
206                    diff_truncated: false,
207                    semantic_change: None,
208                    commits: Vec::new(),
209                    result_complete: !truncated,
210                    meta: self.meta(generation, emitted_tokens, None),
211                }
212            }
213            HistoryOperation::DiffSymbol {
214                path,
215                symbol,
216                base_revision,
217                head_revision,
218            } => {
219                let revisions =
220                    git_diff_identity(&self.config.root, &base_revision, Some(&head_revision))?;
221                check_cancelled(cancellation)?;
222                let before =
223                    self.historical_symbol_optional(&path, &symbol, &revisions.base_revision)?;
224                check_cancelled(cancellation)?;
225                let after =
226                    self.historical_symbol_optional(&path, &symbol, &revisions.head_revision)?;
227                if before.is_none() && after.is_none() {
228                    return Err(Error::SymbolNotFound {
229                        path: format!(
230                            "{path}@{}..{}",
231                            revisions.base_revision, revisions.head_revision
232                        ),
233                        symbol,
234                    });
235                }
236                let before_content = before
237                    .as_ref()
238                    .map_or("", |resolved| resolved.content.as_str());
239                let after_content = after
240                    .as_ref()
241                    .map_or("", |resolved| resolved.content.as_str());
242                let before_diff = symbol_diff_buffer(before_content);
243                let after_diff = symbol_diff_buffer(after_content);
244                let full_diff = TextDiff::from_lines(&before_diff, &after_diff)
245                    .unified_diff()
246                    .context_radius(3)
247                    .header(
248                        &format!("{}:{path}", revisions.base_revision),
249                        &format!("{}:{path}", revisions.head_revision),
250                    )
251                    .to_string();
252                let total_diff_tokens = self.config.tokenizer.count(&full_diff);
253                let (diff, emitted_tokens) = self.config.tokenizer.truncate(&full_diff, max_tokens);
254                let diff_truncated = emitted_tokens < total_diff_tokens;
255                let semantic_change = match (&before, &after) {
256                    (Some(before), Some(after)) => classify_historical_symbol_change(
257                        &path,
258                        &before.symbol,
259                        before.signature.as_deref(),
260                        &before.content,
261                        &after.symbol,
262                        after.signature.as_deref(),
263                        &after.content,
264                    ),
265                    (None, Some(after)) => Some(classify_historical_symbol_added(
266                        &path,
267                        &after.symbol,
268                        after.signature.as_deref(),
269                    )),
270                    (Some(before), None) => Some(classify_historical_symbol_removed(
271                        &path,
272                        &before.symbol,
273                        before.signature.as_deref(),
274                    )),
275                    (None, None) => unreachable!("both absent endpoints returned above"),
276                };
277                let before = before.map(|mut resolved| {
278                    resolved.symbol.content = None;
279                    resolved.symbol.returned_end_line = 0;
280                    resolved.symbol
281                });
282                let after = after.map(|mut resolved| {
283                    resolved.symbol.content = None;
284                    resolved.symbol.returned_end_line = 0;
285                    resolved.symbol
286                });
287                HistoryResponse {
288                    kind: "diff_symbol".into(),
289                    symbol: None,
290                    before,
291                    after,
292                    diff: Some(diff.to_owned()),
293                    diff_truncated,
294                    semantic_change,
295                    commits: Vec::new(),
296                    result_complete: !diff_truncated,
297                    meta: self.meta(generation, emitted_tokens, None),
298                }
299            }
300            HistoryOperation::SymbolLog {
301                path,
302                symbol,
303                revision,
304            } => {
305                let revision = revision.as_deref().unwrap_or("HEAD");
306                let resolved = self.historical_symbol(&path, &symbol, revision)?;
307                check_cancelled(cancellation)?;
308                let mut commits = git_line_history(
309                    &self.config.root,
310                    revision,
311                    &path,
312                    resolved.symbol.target_start_line,
313                    resolved.symbol.target_end_line,
314                    max_results.saturating_add(1),
315                )?;
316                let result_complete = commits.len() <= max_results;
317                commits.truncate(max_results);
318                HistoryResponse {
319                    kind: "symbol_log".into(),
320                    symbol: Some(HistoricalSymbol {
321                        content: None,
322                        returned_end_line: 0,
323                        ..resolved.symbol
324                    }),
325                    before: None,
326                    after: None,
327                    diff: None,
328                    diff_truncated: false,
329                    semantic_change: None,
330                    commits: commits
331                        .into_iter()
332                        .map(|commit| SymbolHistoryCommit {
333                            commit: commit.commit,
334                            authored_at: commit.authored_at,
335                            subject: commit.subject,
336                        })
337                        .collect(),
338                    result_complete,
339                    meta: self.meta(generation, 0, None),
340                }
341            }
342        };
343        self.fit_history_response(&mut response, options)?;
344        self.finalize_bounded_response(&mut response, options)?;
345        self.record_token_savings(TokenAccountingOperation::History, None, &response.meta);
346        Ok(response)
347    }

349    fn fit_history_response(
350        &self,
351        response: &mut HistoryResponse,
352        options: ServiceCallOptions,
353    ) -> Result<()> {
354        if self.response_fits(response, options)? {
355            return Ok(());
356        }
357
358        if let Some(content) = response
359            .symbol
360            .as_ref()
361            .and_then(|symbol| symbol.content.as_ref())
362            .cloned()
363        {
364            let boundaries = char_boundaries(&content);
365            if let Some(candidate) =
366                self.fit_history_text_prefix(response, &content, &boundaries, options, false)?
367            {
368                *response = candidate;
369                return Ok(());
370            }
371        } else if let Some(diff) = response.diff.clone() {
372            let boundaries = char_boundaries(&diff);
373            if let Some(candidate) =
374                self.fit_history_text_prefix(response, &diff, &boundaries, options, true)?
375            {
376                *response = candidate;
377                return Ok(());
378            }
379        } else if !response.commits.is_empty() {
380            let original = response.clone();
381            let max_response_tokens = options
382                .max_response_tokens()
383                .expect("fitting only runs with a response limit");
384            let budget = ResponseBudget::new(&self.config.tokenizer, max_response_tokens);
385            let keep = budget.largest_fitting_prefix(original.commits.len(), |keep| {
386                let mut candidate = original.clone();
387                candidate.commits.truncate(keep);
388                candidate.result_complete = false;
389                self.finalized_response_tokens(&candidate)
390            })?;
391            if let Some(keep) = keep.filter(|keep| *keep > 0) {
392                response.commits.truncate(keep);
393                response.result_complete = false;
394                return Ok(());
395            }
396        }
397
398        let requested = self.finalized_response_tokens(response)?;
399        Err(Error::RequestLimitExceeded {
400            field: "max_response_tokens",
401            requested,
402            limit: options
403                .max_response_tokens()
404                .expect("fitting only runs with a response limit"),
405        })
406    }

408    fn fit_history_text_prefix(
409        &self,
410        response: &HistoryResponse,
411        text: &str,
412        boundaries: &[usize],
413        options: ServiceCallOptions,
414        is_diff: bool,
415    ) -> Result<Option<HistoryResponse>> {
416        let max_response_tokens = options
417            .max_response_tokens()
418            .expect("fitting only runs with a response limit");
419        let budget = ResponseBudget::new(&self.config.tokenizer, max_response_tokens);
420        let keep = budget.largest_fitting_prefix(boundaries.len().saturating_sub(1), |keep| {
421            let prefix = &text[..boundaries[keep]];
422            let mut candidate = response.clone();
423            let source_tokens = self.config.tokenizer.count(prefix);
424            candidate.meta.source_tokens = source_tokens;
425            candidate.meta.emitted_tokens = source_tokens;
426            candidate.result_complete = false;
427            if is_diff {
428                candidate.diff = Some(prefix.to_owned());
429                candidate.diff_truncated = true;
430            } else if let Some(symbol) = candidate.symbol.as_mut() {
431                symbol.content = Some(prefix.to_owned());
432                symbol.returned_end_line = returned_end_line(symbol.target_start_line, prefix);
433                symbol.truncated = true;
434            }
435            self.finalized_response_tokens(&candidate)
436        })?;
437        let Some(keep) = keep.filter(|keep| *keep > 0) else {
438            return Ok(None);
439        };
440        let prefix = &text[..boundaries[keep]];
441        let mut candidate = response.clone();
442        let source_tokens = self.config.tokenizer.count(prefix);
443        candidate.meta.source_tokens = source_tokens;
444        candidate.meta.emitted_tokens = source_tokens;
445        candidate.result_complete = false;
446        if is_diff {
447            candidate.diff = Some(prefix.to_owned());
448            candidate.diff_truncated = true;
449        } else if let Some(symbol) = candidate.symbol.as_mut() {
450            symbol.content = Some(prefix.to_owned());
451            symbol.returned_end_line = returned_end_line(symbol.target_start_line, prefix);
452            symbol.truncated = true;
453        }
454        Ok(Some(candidate))
455    }

457    fn historical_symbol(
458        &self,
459        path: &str,
460        symbol_name: &str,
461        revision: &str,
462    ) -> Result<ResolvedHistoricalSymbol> {
463        let blob = git_blob_at_revision(
464            &self.config.root,
465            revision,
466            path,
467            usize::try_from(self.config.max_file_bytes).unwrap_or(usize::MAX),
468        )?;
469        let resolved_revision = blob.revision.clone();
470        self.resolve_historical_symbol(path, symbol_name, blob)?
471            .ok_or_else(|| Error::SymbolNotFound {
472                path: format!("{path}@{resolved_revision}"),
473                symbol: symbol_name.to_owned(),
474            })
475    }

477    fn historical_symbol_optional(
478        &self,
479        path: &str,
480        symbol_name: &str,
481        revision: &str,
482    ) -> Result<Option<ResolvedHistoricalSymbol>> {
483        let blob = match git_blob_at_revision(
484            &self.config.root,
485            revision,
486            path,
487            usize::try_from(self.config.max_file_bytes).unwrap_or(usize::MAX),
488        ) {
489            Ok(blob) => blob,
490            Err(Error::InvalidInput {
491                field: "path",
492                reason: "file does not exist at revision",
493            }) => return Ok(None),
494            Err(error) => return Err(error),
495        };
496        self.resolve_historical_symbol(path, symbol_name, blob)
497    }

499    fn resolve_historical_symbol(
500        &self,
501        path: &str,
502        symbol_name: &str,
503        blob: GitBlob,
504    ) -> Result<Option<ResolvedHistoricalSymbol>> {
505        let parsed = parser::parse(path, &blob.content)?;
506        let mut symbols = parsed.symbols;
507        let symbol_index = symbols
508            .iter()
509            .position(|symbol| symbol.name == symbol_name)
510            .or_else(|| {
511                symbols.iter().position(|symbol| {
512                    symbol.parent.as_deref().is_some_and(|parent| {
513                        symbol_name
514                            .strip_prefix(parent)
515                            .and_then(|suffix| suffix.strip_prefix('.'))
516                            == Some(symbol.name.as_str())
517                    })
518                })
519            });
520        let Some(symbol_index) = symbol_index else {
521            return Ok(None);
522        };
523        let symbol = symbols.remove(symbol_index);
524        let content = blob
525            .content
526            .get(symbol.start_byte..symbol.end_byte)
527            .ok_or_else(|| Error::InternalFailure("invalid historical symbol range".into()))?
528            .to_owned();
529        let content_hash = crate::text::hash(&content);
530        Ok(Some(ResolvedHistoricalSymbol {
531            symbol: HistoricalSymbol {
532                revision: blob.revision,
533                path: path.to_owned(),
534                name: symbol.name,
535                kind: symbol.kind,
536                parent: symbol.parent,
537                target_start_line: symbol.start_line,
538                target_end_line: symbol.end_line,
539                returned_end_line: symbol.end_line,
540                truncated: false,
541                content: None,
542                content_hash,
543            },
544            signature: symbol.signature,
545            content,
546        }))
547    }
