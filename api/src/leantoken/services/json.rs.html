59fn invalid_json_selector(stage: &'static str, error: jmespath::JmespathError) -> Error {
60    Error::InvalidJsonSelector {
61        stage,
62        offset: error.offset,
63        line: error.line.saturating_add(1),
64        column: error.column.saturating_add(1),
65        reason: error.reason.to_string(),
66    }

69fn validate_selector(selector: &JsonSelector) -> Result<()> {
70    match selector {
71        JsonSelector::Pointer { pointer } => {
72            validate_input(pointer, "JSON Pointer", MAX_PATTERN_BYTES)?;
73            if !pointer.is_empty() && !pointer.starts_with('/') {
74                return Err(Error::InvalidInput {
75                    field: "JSON Pointer",
76                    reason: "must be empty or start with a slash",
77                });
78            }
79        }
80        JsonSelector::Jmespath { expression } => {
81            validate_input(expression, "JMESPath expression", MAX_PATTERN_BYTES)?;
82            if expression.trim().is_empty() {
83                return Err(Error::InvalidInput {
84                    field: "JMESPath expression",
85                    reason: "must not be empty",
86                });
87            }
88            jmespath::compile(expression)
89                .map_err(|error| invalid_json_selector("compile", error))?;
90        }
91    }
92    Ok(())
93}

95fn is_fingerprint(value: &str) -> bool {

99fn decode_json_cursor(cursor: &str) -> Result<JsonCursor> {
100    if cursor.len() > MAX_JSON_CURSOR_BYTES {
101        return Err(Error::StaleCursor);
102    }
103    let mut fields = cursor.split(':');
104    let version = fields.next();
105    let source_hash = fields.next();
106    let query_hash = fields.next();
107    let offset = fields.next();
108    if version != Some("j1") || fields.next().is_some() {
109        return Err(Error::StaleCursor);
110    }
111    let (Some(source_hash), Some(query_hash), Some(offset)) = (source_hash, query_hash, offset)
112    else {
113        return Err(Error::StaleCursor);
114    };
115    if !is_fingerprint(source_hash) || !is_fingerprint(query_hash) {
116        return Err(Error::StaleCursor);
117    }
118    let offset = offset.parse::<usize>().map_err(|_| Error::StaleCursor)?;
119    if offset == 0 {
120        return Err(Error::StaleCursor);
121    }
122    Ok(JsonCursor {
123        source_hash: source_hash.to_owned(),
124        query_hash: query_hash.to_owned(),
125        offset,
126    })
127}

129fn json_query_hash(operation: &JsonOperation) -> Result<String> {
130    let serialized = serde_json::to_string(operation)
131        .map_err(|error| Error::InternalFailure(error.to_string()))?;
132    Ok(crate::text::hash(&serialized))
133}

139fn validate_json_request(request: &JsonRequest) -> Result<()> {
140    match &request.operation {
141        JsonOperation::Query { path, selector, .. }
142        | JsonOperation::NumericSummary { path, selector } => {
143            validate_input(path, "path", MAX_PATH_BYTES)?;
144            if let Some(selector) = selector {
145                validate_selector(selector)?;
146            }
147        }
148        JsonOperation::DiffFields {
149            base_path,
150            head_path,
151            selectors,
152            ..
153        } => {
154            validate_input(base_path, "base path", MAX_PATH_BYTES)?;
155            validate_input(head_path, "head path", MAX_PATH_BYTES)?;
156            validate_positive_request_limit("selectors", selectors.len(), MAX_JSON_SELECTORS)?;
157            for selector in selectors {
158                validate_selector(selector)?;
159            }
160        }
161    }
162    if let Some(max_items) = request.max_items {
163        validate_positive_request_limit("max_items", max_items, MAX_JSON_ITEMS)?;
164    }
165    if let Some(array_sample_size) = request.array_sample_size {
166        validate_request_limit(
167            "array_sample_size",
168            array_sample_size,
169            MAX_ARRAY_SAMPLE_SIZE,
170        )?;
171    }
172    if let Some(cursor) = request.cursor.as_deref() {
173        decode_json_cursor(cursor)?;
174        if !matches!(
175            &request.operation,
176            JsonOperation::Query {
177                projection: JsonProjection::Keys,
178                ..
179            }
180        ) {
181            return Err(Error::InvalidInput {
182                field: "cursor",
183                reason: "is supported only for query operations with the keys projection",
184            });
185        }
186    }
187    Ok(())
188}

212    fn json_sync(
213        &self,
214        request: JsonRequest,
215        cancellation: &CancellationToken,
216    ) -> Result<JsonResponse> {
217        check_cancelled(cancellation)?;
218        validate_json_request(&request)?;
219        let max_tokens = self.token_limit(request.max_tokens, DEFAULT_JSON_TOKENS)?;
220        let max_items = request.max_items.unwrap_or(DEFAULT_JSON_ITEMS);
221        let array_sample_size = request
222            .array_sample_size
223            .unwrap_or(DEFAULT_ARRAY_SAMPLE_SIZE);
224        let cursor = request
225            .cursor
226            .as_deref()
227            .map(decode_json_cursor)
228            .transpose()?;
229        let query_hash = json_query_hash(&request.operation)?;
230        let generation = self.storage.repository_generation()?;
231        let mut projected_tokens = 0usize;
232        let baseline_source_tokens;
233        let mut response = match request.operation {
234            JsonOperation::Query {
235                path,
236                selector,
237                projection,
238            } => {
239                let loaded = self.load_json(&path)?;
240                baseline_source_tokens = loaded.source_tokens;
241                let selected = select_json(&loaded.value, selector.as_ref())?;
242                let value = selected.value.ok_or(Error::InvalidInput {
243                    field: "selector",
244                    reason: "did not match a JSON value",
245                })?;
246                if projection == JsonProjection::Keys {
247                    let page = project_key_page(
248                        self,
249                        &value,
250                        max_items,
251                        max_tokens,
252                        cursor.as_ref(),
253                        &loaded.source.content_hash,
254                        &query_hash,
255                    )?;
256                    projected_tokens = page.projected_tokens;
257                    JsonResponse {
258                        kind: "query".into(),
259                        value: Some(page.value),
260                        numeric_summary: None,
261                        differences: Vec::new(),
262                        sources: vec![loaded.source],
263                        result_complete: page.remaining_items == 0,
264                        total_items: Some(page.total_items),
265                        returned_items: Some(page.returned_items),
266                        remaining_items: Some(page.remaining_items),
267                        incomplete_reason: page.incomplete_reason,
268                        meta: self.meta(generation, 0, page.next_cursor),
269                    }
270                } else {
271                    let total_items = projection_item_count(&value, projection, array_sample_size);
272                    let mut state = ProjectionState {
273                        remaining: max_items,
274                        array_sample_size,
275                        complete: true,
276                    };
277                    let value = project_json(&value, projection, &mut state)?;
278                    projected_tokens = json_tokens(self, &value)?;
279                    let returned_items = total_items.min(max_items);
280                    let remaining_items = total_items.saturating_sub(returned_items);
281                    JsonResponse {
282                        kind: "query".into(),
283                        value: Some(value),
284                        numeric_summary: None,
285                        differences: Vec::new(),
286                        sources: vec![loaded.source],
287                        result_complete: state.complete,
288                        total_items: (!state.complete).then_some(total_items),
289                        returned_items: (!state.complete).then_some(returned_items),
290                        remaining_items: (!state.complete).then_some(remaining_items),
291                        incomplete_reason: (!state.complete)
292                            .then_some(JsonIncompleteReason::MaxItems),
293                        meta: self.meta(generation, 0, None),
294                    }
295                }
296            }
297            JsonOperation::NumericSummary { path, selector } => {
298                let loaded = self.load_json(&path)?;
299                baseline_source_tokens = loaded.source_tokens;
300                let selected = select_json(&loaded.value, selector.as_ref())?;
301                let value = selected.value.ok_or(Error::InvalidInput {
302                    field: "selector",
303                    reason: "did not match a JSON value",
304                })?;
305                JsonResponse {
306                    kind: "numeric_summary".into(),
307                    value: None,
308                    numeric_summary: Some(numeric_summary(&value)),
309                    differences: Vec::new(),
310                    sources: vec![loaded.source],
311                    result_complete: true,
312                    total_items: None,
313                    returned_items: None,
314                    remaining_items: None,
315                    incomplete_reason: None,
316                    meta: self.meta(generation, 0, None),
317                }
318            }
319            JsonOperation::DiffFields {
320                base_path,
321                head_path,
322                selectors,
323                projection,
324            } => {
325                let before = self.load_json(&base_path)?;
326                check_cancelled(cancellation)?;
327                let after = self.load_json(&head_path)?;
328                baseline_source_tokens = before.source_tokens.saturating_add(after.source_tokens);
329                let mut state = ProjectionState {
330                    remaining: max_items,
331                    array_sample_size,
332                    complete: true,
333                };
334                let mut differences = Vec::with_capacity(selectors.len());
335                let mut total_items = 0usize;
336                for selector in selectors {
337                    check_cancelled(cancellation)?;
338                    let before_selected = select_json(&before.value, Some(&selector))?;
339                    let after_selected = select_json(&after.value, Some(&selector))?;
340                    let changed = before_selected.present != after_selected.present
341                        || before_selected.value != after_selected.value;
342                    total_items = total_items.saturating_add(
343                        before_selected
344                            .value
345                            .as_ref()
346                            .map(|value| {
347                                projection_item_count(value, projection, array_sample_size)
348                            })
349                            .unwrap_or_default(),
350                    );
351                    total_items = total_items.saturating_add(
352                        after_selected
353                            .value
354                            .as_ref()
355                            .map(|value| {
356                                projection_item_count(value, projection, array_sample_size)
357                            })
358                            .unwrap_or_default(),
359                    );
360                    let before_value = before_selected
361                        .value
362                        .as_ref()
363                        .map(|value| project_json(value, projection, &mut state))
364                        .transpose()?;
365                    let after_value = after_selected
366                        .value
367                        .as_ref()
368                        .map(|value| project_json(value, projection, &mut state))
369                        .transpose()?;
370                    if let Some(value) = &before_value {
371                        projected_tokens =
372                            projected_tokens.saturating_add(json_tokens(self, value)?);
373                    }
374                    if let Some(value) = &after_value {
375                        projected_tokens =
376                            projected_tokens.saturating_add(json_tokens(self, value)?);
377                    }
378                    differences.push(JsonFieldDiff {
379                        selector,
380                        before_present: before_selected.present,
381                        before: before_value,
382                        after_present: after_selected.present,
383                        after: after_value,
384                        changed,
385                    });
386                }
387                let returned_items = total_items.min(max_items);
388                let remaining_items = total_items.saturating_sub(returned_items);
389                JsonResponse {
390                    kind: "diff_fields".into(),
391                    value: None,
392                    numeric_summary: None,
393                    differences,
394                    sources: vec![before.source, after.source],
395                    result_complete: state.complete,
396                    total_items: (!state.complete).then_some(total_items),
397                    returned_items: (!state.complete).then_some(returned_items),
398                    remaining_items: (!state.complete).then_some(remaining_items),
399                    incomplete_reason: (!state.complete).then_some(JsonIncompleteReason::MaxItems),
400                    meta: self.meta(generation, 0, None),
401                }
402            }
403        };
404        if projected_tokens > max_tokens {
405            return Err(Error::RequestLimitExceeded {
406                field: "projected JSON tokens",
407                requested: projected_tokens,
408                limit: max_tokens,
409            });
410        }
411        response.meta.source_tokens = projected_tokens;
412        response.meta.emitted_tokens = projected_tokens;
413        self.finalize_response(&mut response)?;
414        self.record_token_savings(
415            TokenAccountingOperation::Json,
416            Some(baseline_source_tokens),
417            &response.meta,
418        );
419        Ok(response)
420    }

422    fn load_json(&self, path: &str) -> Result<LoadedJson> {
423        let path = normalize_relative(path)?;
424        let mut file = open_live_file(self, &path)?;
425        let max_bytes = usize::try_from(self.config.max_file_bytes).unwrap_or(usize::MAX);
426        let metadata_bytes = usize::try_from(file.metadata()?.len()).unwrap_or(usize::MAX);
427        if metadata_bytes > max_bytes {
428            return Err(Error::RequestLimitExceeded {
429                field: "JSON file bytes",
430                requested: metadata_bytes,
431                limit: max_bytes,
432            });
433        }
434        let mut bytes = Vec::with_capacity(metadata_bytes.min(max_bytes));
435        file.by_ref()
436            .take(
437                u64::try_from(max_bytes)
438                    .unwrap_or(u64::MAX)
439                    .saturating_add(1),
440            )
441            .read_to_end(&mut bytes)?;
442        if bytes.len() > max_bytes {
443            return Err(Error::RequestLimitExceeded {
444                field: "JSON file bytes",
445                requested: bytes.len(),
446                limit: max_bytes,
447            });
448        }
449        let content = std::str::from_utf8(&bytes).map_err(|_| Error::InvalidInput {
450            field: "path",
451            reason: "JSON file is not valid UTF-8",
452        })?;
453        let value = serde_json::from_str(content).map_err(|error| {
454            let syntax_category = match error.classify() {
455                serde_json::error::Category::Io => "io",
456                serde_json::error::Category::Syntax => "syntax",
457                serde_json::error::Category::Data => "data",
458                serde_json::error::Category::Eof => "eof",
459            };
460            Error::InvalidJson {
461                syntax_category,
462                byte_offset: json_error_byte_offset(content, error.line(), error.column()),
463                line: error.line(),
464                column: error.column(),
465                reason: error.to_string(),
466            }
467        })?;
468        Ok(LoadedJson {
469            source: JsonSource {
470                path,
471                content_hash: crate::text::hash(content),
472                bytes: bytes.len(),
473            },
474            value,
475            source_tokens: self.config.tokenizer.count(content),
476        })
477    }
478}

480fn select_json(value: &Value, selector: Option<&JsonSelector>) -> Result<SelectedJson> {
481    match selector {
482        None => Ok(SelectedJson {
483            present: true,
484            value: Some(value.clone()),
485        }),
486        Some(JsonSelector::Pointer { pointer }) => {
487            let selected = value.pointer(pointer).cloned();
488            Ok(SelectedJson {
489                present: selected.is_some(),
490                value: selected,
491            })
492        }
493        Some(JsonSelector::Jmespath { expression }) => {
494            let expression = jmespath::compile(expression)
495                .map_err(|error| invalid_json_selector("compile", error))?;
496            let selected = expression
497                .search(value)
498                .map_err(|error| invalid_json_selector("evaluate", error))?;
499            let selected = serde_json::to_value(selected.as_ref())
500                .map_err(|error| Error::InternalFailure(error.to_string()))?;
501            Ok(SelectedJson {
502                present: true,
503                value: Some(selected),
504            })
505        }
506    }
507}

509fn json_error_byte_offset(content: &str, line: usize, column: usize) -> usize {
510    if line == 0 {
511        return 0;
512    }
513    let mut line_start = 0usize;
514    for (index, segment) in content.split_inclusive('\n').enumerate() {
515        if index.saturating_add(1) == line {
516            return line_start
517                .saturating_add(column.saturating_sub(1).min(segment.len()))
518                .min(content.len());
519        }
520        line_start = line_start.saturating_add(segment.len());
521    }
522    content.len()
523}

525fn collect_all_keys(value: &Value, pointer: &str, keys: &mut BTreeMap<String, &'static str>) {
526    if !keys.contains_key(pointer) {
527        keys.insert(pointer.to_owned(), json_type(value));
528    }
529    match value {
530        Value::Object(values) => {
531            for (key, value) in values {
532                let pointer = format!("{pointer}/{}", escape_pointer(key));
533                collect_all_keys(value, &pointer, keys);
534            }
535        }
536        Value::Array(values) => {
537            let pointer = format!("{pointer}/*");
538            for value in values {
539                collect_all_keys(value, &pointer, keys);
540            }
541        }
542        _ => {}
543    }
544}

546fn key_entries(value: &Value) -> Vec<Value> {
547    let mut keys = BTreeMap::new();
548    collect_all_keys(value, "", &mut keys);
549    keys.into_iter()
550        .map(|(pointer, value_type)| json!({"pointer": pointer, "type": value_type}))
551        .collect()
552}

554fn key_prefix_tokens(services: &Services, entries: &[Value], length: usize) -> Result<usize> {
555    json_tokens(services, &Value::Array(entries[..length].to_vec()))
556}
557
558fn largest_key_prefix_within_tokens(
559    services: &Services,
560    entries: &[Value],
561    max_tokens: usize,
562) -> Result<(usize, usize)> {
563    if entries.is_empty() {
564        return Ok((0, json_tokens(services, &Value::Array(Vec::new()))?));
565    }
566    let full_tokens = key_prefix_tokens(services, entries, entries.len())?;
567    if full_tokens <= max_tokens {
568        return Ok((entries.len(), full_tokens));
569    }
570
571    let mut lower = 0usize;
572    let mut upper = entries.len();
573    while lower < upper {
574        let middle = lower.saturating_add(upper).saturating_add(1) / 2;
575        if key_prefix_tokens(services, entries, middle)? <= max_tokens {
576            lower = middle;
577        } else {
578            upper = middle.saturating_sub(1);
579        }
580    }
581    let tokens = key_prefix_tokens(services, entries, lower)?;
582    Ok((lower, tokens))
583}

585fn project_key_page(
586    services: &Services,
587    value: &Value,
588    max_items: usize,
589    max_tokens: usize,
590    cursor: Option<&JsonCursor>,
591    source_hash: &str,
592    query_hash: &str,
593) -> Result<KeyProjectionPage> {
594    let entries = key_entries(value);
595    let total_items = entries.len();
596    let offset = match cursor {
597        Some(cursor) if cursor.source_hash == source_hash && cursor.query_hash == query_hash => {
598            cursor.offset
599        }
600        Some(_) => return Err(Error::StaleCursor),
601        None => 0,
602    };
603    if offset > total_items || (offset == total_items && offset != 0) {
604        return Err(Error::StaleCursor);
605    }
606
607    let page_end = offset.saturating_add(max_items).min(total_items);
608    let candidates = &entries[offset..page_end];
609    let (returned_items, projected_tokens) =
610        largest_key_prefix_within_tokens(services, candidates, max_tokens)?;
611    if returned_items == 0 && !candidates.is_empty() {
612        return Err(Error::RequestLimitExceeded {
613            field: "one projected JSON item tokens",
614            requested: key_prefix_tokens(services, candidates, 1)?,
615            limit: max_tokens,
616        });
617    }
618
619    let consumed = offset.saturating_add(returned_items);
620    let remaining_items = total_items.saturating_sub(consumed);
621    let incomplete_reason = (remaining_items > 0).then_some(if returned_items < candidates.len() {
622        JsonIncompleteReason::MaxTokens
623    } else {
624        JsonIncompleteReason::MaxItems
625    });
626    let next_cursor =
627        (remaining_items > 0).then(|| make_json_cursor(source_hash, query_hash, consumed));
628    Ok(KeyProjectionPage {
629        value: Value::Array(candidates[..returned_items].to_vec()),
630        total_items,
631        returned_items,
632        remaining_items,
633        incomplete_reason,
634        next_cursor,
635        projected_tokens,
636    })
637}

639fn collapsed_item_count(value: &Value, array_sample_size: usize) -> usize {
640    match value {
641        Value::Array(values) => 1usize.saturating_add(
642            values
643                .iter()
644                .take(array_sample_size)
645                .map(|value| collapsed_item_count(value, array_sample_size))
646                .fold(0usize, usize::saturating_add),
647        ),
648        Value::Object(values) => 1usize.saturating_add(
649            values
650                .values()
651                .map(|value| collapsed_item_count(value, array_sample_size))
652                .fold(0usize, usize::saturating_add),
653        ),
654        _ => 1,
655    }
656}

658fn projection_item_count(
659    value: &Value,
660    projection: JsonProjection,
661    array_sample_size: usize,
662) -> usize {
663    match projection {
664        JsonProjection::Value | JsonProjection::Schema => count_nodes(value),
665        JsonProjection::Collapsed => collapsed_item_count(value, array_sample_size),
666        JsonProjection::Keys => key_entries(value).len(),
667    }

670fn project_json(
671    value: &Value,
672    projection: JsonProjection,
673    state: &mut ProjectionState,
674) -> Result<Value> {
675    match projection {
676        JsonProjection::Value => {
677            let items = count_nodes(value);
678            if items > state.remaining {
679                return Err(Error::RequestLimitExceeded {
680                    field: "selected JSON items",
681                    requested: items,
682                    limit: state.remaining,
683                });
684            }
685            state.remaining -= items;
686            Ok(value.clone())
687        }
688        JsonProjection::Collapsed => Ok(collapse_json(value, state)),
689        JsonProjection::Keys => {
690            let mut keys = BTreeMap::new();
691            collect_keys(value, "", &mut keys, state);
692            Ok(Value::Array(
693                keys.into_iter()
694                    .map(|(pointer, value_type)| json!({"pointer": pointer, "type": value_type}))
695                    .collect(),
696            ))
697        }
698        JsonProjection::Schema => Ok(infer_schema(value, state)),
699    }
700}
701
702fn count_nodes(value: &Value) -> usize {
703    match value {
704        Value::Array(values) => 1usize.saturating_add(
705            values
706                .iter()
707                .map(count_nodes)
708                .fold(0usize, usize::saturating_add),
709        ),
710        Value::Object(values) => 1usize.saturating_add(
711            values
712                .values()
713                .map(count_nodes)
714                .fold(0usize, usize::saturating_add),
715        ),
716        _ => 1,
717    }
718}
719
720fn take_item(state: &mut ProjectionState) -> bool {
721    if state.remaining == 0 {
722        state.complete = false;
723        return false;
724    }
725    state.remaining -= 1;
726    true
727}
728
729fn collapse_json(value: &Value, state: &mut ProjectionState) -> Value {
730    if !take_item(state) {
731        return Value::Null;
732    }
733    match value {
734        Value::Array(values) => {
735            let mut sample = Vec::new();
736            for value in values.iter().take(state.array_sample_size) {
737                if state.remaining == 0 {
738                    state.complete = false;
739                    break;
740                }
741                sample.push(collapse_json(value, state));
742            }
743            let omitted = values.len().saturating_sub(sample.len());
744            json!({
745                "$array": {
746                    "count": values.len(),
747                    "sample": sample,
748                    "omitted": omitted,
749                }
750            })
751        }
752        Value::Object(values) => {
753            let mut projected = Map::new();
754            for (key, value) in values {
755                if state.remaining == 0 {
756                    state.complete = false;
757                    break;
758                }
759                projected.insert(key.clone(), collapse_json(value, state));
760            }
761            Value::Object(projected)
762        }
763        _ => value.clone(),
764    }
765}
766
767fn collect_keys(
768    value: &Value,
769    pointer: &str,
770    keys: &mut BTreeMap<String, &'static str>,
771    state: &mut ProjectionState,
772) {
773    if !keys.contains_key(pointer) {
774        if !take_item(state) {
775            return;
776        }
777        keys.insert(pointer.to_owned(), json_type(value));
778    }
779    match value {
780        Value::Object(values) => {
781            for (key, value) in values {
782                let pointer = format!("{pointer}/{}", escape_pointer(key));
783                collect_keys(value, &pointer, keys, state);
784            }
785        }
786        Value::Array(values) => {
787            let pointer = format!("{pointer}/*");
788            for value in values {
789                collect_keys(value, &pointer, keys, state);
790            }
791        }
792        _ => {}
793    }
794}
795
796fn escape_pointer(value: &str) -> String {
797    value.replace('~', "~0").replace('/', "~1")
798}
799
800fn infer_schema(value: &Value, state: &mut ProjectionState) -> Value {
801    if !take_item(state) {
802        return json!({"type": "unknown"});
803    }
804    match value {
805        Value::Object(values) => {
806            let mut properties = Map::new();
807            for (key, value) in values {
808                if state.remaining == 0 {
809                    state.complete = false;
810                    break;
811                }
812                properties.insert(key.clone(), infer_schema(value, state));
813            }
814            json!({"type": "object", "properties": properties})
815        }
816        Value::Array(values) => {
817            let mut variants = BTreeMap::new();
818            for value in values {
819                if state.remaining == 0 {
820                    state.complete = false;
821                    break;
822                }
823                let schema = infer_schema(value, state);
824                let key = serde_json::to_string(&schema).unwrap_or_default();
825                variants.entry(key).or_insert(schema);
826            }
827            let items = match variants.len() {
828                0 => json!({}),
829                1 => variants.into_values().next().unwrap_or_else(|| json!({})),
830                _ => json!({"anyOf": variants.into_values().collect::<Vec<_>>() }),
831            };
832            json!({"type": "array", "count": values.len(), "items": items})
833        }
834        _ => json!({"type": json_type(value)}),
835    }
836}
837
838fn json_type(value: &Value) -> &'static str {
839    match value {
840        Value::Null => "null",
841        Value::Bool(_) => "boolean",
842        Value::Number(_) => "number",
843        Value::String(_) => "string",
844        Value::Array(_) => "array",
845        Value::Object(_) => "object",
846    }
847}

849fn numeric_summary(value: &Value) -> JsonNumericSummary {
850    let mut values = Vec::new();
851    let mut non_numeric_count = 0usize;
852    collect_numbers(value, &mut values, &mut non_numeric_count);
853    values.sort_by(f64::total_cmp);
854    let count = values.len();
855    let median = match count {
856        0 => None,
857        count if count % 2 == 1 => Some(values[count / 2]),
858        count => Some((values[count / 2 - 1] + values[count / 2]) / 2.0),
859    };
860    let p95 = (count > 0).then(|| {
861        let rank = (count.saturating_mul(95).saturating_add(99)) / 100;
862        values[rank.saturating_sub(1).min(count - 1)]
863    });
864    JsonNumericSummary {
865        count,
866        non_numeric_count,
867        min: values.first().copied(),
868        median,
869        p95,
870        max: values.last().copied(),
871    }
872}
873
874fn collect_numbers(value: &Value, values: &mut Vec<f64>, non_numeric_count: &mut usize) {
875    match value {
876        Value::Number(value) => {
877            if let Some(value) = value.as_f64() {
878                values.push(value);
879            } else {
880                *non_numeric_count = non_numeric_count.saturating_add(1);
881            }
882        }
883        Value::Array(items) => {
884            for item in items {
885                collect_numbers(item, values, non_numeric_count);
886            }
887        }
888        Value::Object(items) => {
889            for item in items.values() {
890                collect_numbers(item, values, non_numeric_count);
891            }
892        }
893        _ => *non_numeric_count = non_numeric_count.saturating_add(1),
894    }
895}

897fn json_tokens(services: &Services, value: &Value) -> Result<usize> {
898    let serialized =
899        serde_json::to_string(value).map_err(|error| Error::InternalFailure(error.to_string()))?;
900    Ok(services.config.tokenizer.count(&serialized))
901}
