62fn invalid_json_selector(stage: &'static str, error: jmespath::JmespathError) -> Error {
63    Error::InvalidJsonSelector {
64        stage,
65        offset: error.offset,
66        line: error.line.saturating_add(1),
67        column: error.column.saturating_add(1),
68        reason: error.reason.to_string(),
69    }

72fn validate_selector(selector: &JsonSelector) -> Result<()> {
73    match selector {
74        JsonSelector::Pointer { pointer } => {
75            validate_input(pointer, "JSON Pointer", MAX_PATTERN_BYTES)?;
76            if !pointer.is_empty() && !pointer.starts_with('/') {
77                return Err(Error::InvalidInput {
78                    field: "JSON Pointer",
79                    reason: "must be empty or start with a slash",
80                });
81            }
82        }
83        JsonSelector::Jmespath { expression } => {
84            validate_input(expression, "JMESPath expression", MAX_PATTERN_BYTES)?;
85            if expression.trim().is_empty() {
86                return Err(Error::InvalidInput {
87                    field: "JMESPath expression",
88                    reason: "must not be empty",
89                });
90            }
91            jmespath::compile(expression)
92                .map_err(|error| invalid_json_selector("compile", error))?;
93        }
94    }
95    Ok(())
96}

98fn is_fingerprint(value: &str) -> bool {
99    value.len() == CONTENT_FINGERPRINT_HEX_LEN && value.bytes().all(|byte| byte.is_ascii_hexdigit())
100}

102fn decode_json_cursor(cursor: &str) -> Result<JsonCursor> {
103    if cursor.len() > MAX_JSON_CURSOR_BYTES {
104        return Err(Error::StaleCursor);
105    }
106    let mut fields = cursor.split(':');
107    let version = fields.next();
108    let source_hash = fields.next();
109    let query_hash = fields.next();
110    let offset = fields.next();
111    if version != Some("j1") || fields.next().is_some() {
112        return Err(Error::StaleCursor);
113    }
114    let (Some(source_hash), Some(query_hash), Some(offset)) = (source_hash, query_hash, offset)
115    else {
116        return Err(Error::StaleCursor);
117    };
118    if !is_fingerprint(source_hash) || !is_fingerprint(query_hash) {
119        return Err(Error::StaleCursor);
120    }
121    let offset = offset.parse::<usize>().map_err(|_| Error::StaleCursor)?;
122    if offset == 0 {
123        return Err(Error::StaleCursor);
124    }
125    Ok(JsonCursor {
126        source_hash: source_hash.to_owned(),
127        query_hash: query_hash.to_owned(),
128        offset,
129    })
130}

132fn json_query_hash(operation: &JsonOperation) -> Result<String> {
133    let serialized = serde_json::to_string(operation)
134        .map_err(|error| Error::InternalFailure(error.to_string()))?;
135    Ok(crate::text::hash(&serialized))
136}

142fn validate_json_request(request: &JsonRequest) -> Result<()> {
143    match &request.operation {
144        JsonOperation::Query { path, selector, .. }
145        | JsonOperation::NumericSummary { path, selector } => {
146            validate_input(path, "path", MAX_PATH_BYTES)?;
147            if let Some(selector) = selector {
148                validate_selector(selector)?;
149            }
150        }
151        JsonOperation::DiffFields {
152            base_path,
153            head_path,
154            selectors,
155            ..
156        } => {
157            validate_input(base_path, "base path", MAX_PATH_BYTES)?;
158            validate_input(head_path, "head path", MAX_PATH_BYTES)?;
159            validate_positive_request_limit("selectors", selectors.len(), MAX_JSON_SELECTORS)?;
160            for selector in selectors {
161                validate_selector(selector)?;
162            }
163        }
164    }
165    if let Some(max_items) = request.max_items {
166        validate_positive_request_limit("max_items", max_items, MAX_JSON_ITEMS)?;
167    }
168    if let Some(array_sample_size) = request.array_sample_size {
169        validate_request_limit(
170            "array_sample_size",
171            array_sample_size,
172            MAX_ARRAY_SAMPLE_SIZE,
173        )?;
174    }
175    if let Some(cursor) = request.cursor.as_deref() {
176        decode_json_cursor(cursor)?;
177        if !matches!(
178            &request.operation,
179            JsonOperation::Query {
180                projection: JsonProjection::Keys,
181                ..
182            }
183        ) {
184            return Err(Error::InvalidInput {
185                field: "cursor",
186                reason: "is supported only for query operations with the keys projection",
187            });
188        }
189    }
190    Ok(())
191}

237    fn json_sync(
238        &self,
239        request: JsonRequest,
240        options: ServiceCallOptions,
241        cancellation: &CancellationToken,
242    ) -> Result<JsonResponse> {
243        check_cancelled(cancellation)?;
244        validate_json_request(&request)?;
245        let max_tokens = self.token_limit(request.max_tokens, DEFAULT_JSON_TOKENS)?;
246        let max_items = request.max_items.unwrap_or(DEFAULT_JSON_ITEMS);
247        let array_sample_size = request
248            .array_sample_size
249            .unwrap_or(DEFAULT_ARRAY_SAMPLE_SIZE);
250        let cursor = request
251            .cursor
252            .as_deref()
253            .map(decode_json_cursor)
254            .transpose()?;
255        let query_hash = json_query_hash(&request.operation)?;
256        let generation = self.storage.repository_generation()?;
257        let mut projected_tokens = 0usize;
258        let baseline_source_tokens;
259        let mut key_page_context = None;
260        let mut response = match request.operation {
261            JsonOperation::Query {
262                path,
263                selector,
264                projection,
265            } => {
266                let loaded = self.load_json(&path)?;
267                baseline_source_tokens = loaded.source_tokens;
268                let selected = select_json(&loaded.value, selector.as_ref())?;
269                let value = selected.value.ok_or(Error::InvalidInput {
270                    field: "selector",
271                    reason: "did not match a JSON value",
272                })?;
273                if projection == JsonProjection::Keys {
274                    let offset = cursor.as_ref().map_or(0, |cursor| cursor.offset);
275                    let page = project_key_page(
276                        self,
277                        &value,
278                        max_items,
279                        max_tokens,
280                        cursor.as_ref(),
281                        &loaded.source.content_hash,
282                        &query_hash,
283                    )?;
284                    key_page_context = Some((
285                        loaded.source.content_hash.clone(),
286                        query_hash.clone(),
287                        offset,
288                    ));
289                    projected_tokens = page.projected_tokens;
290                    JsonResponse {
291                        kind: "query".into(),
292                        value: Some(page.value),
293                        numeric_summary: None,
294                        differences: Vec::new(),
295                        sources: vec![loaded.source],
296                        result_complete: page.remaining_items == 0,
297                        total_items: Some(page.total_items),
298                        returned_items: Some(page.returned_items),
299                        remaining_items: Some(page.remaining_items),
300                        incomplete_reason: page.incomplete_reason,
301                        meta: self.meta(generation, 0, page.next_cursor),
302                    }
303                } else {
304                    let total_items = projection_item_count(&value, projection, array_sample_size);
305                    let mut state = ProjectionState {
306                        remaining: max_items,
307                        array_sample_size,
308                        complete: true,
309                    };
310                    let value = project_json(&value, projection, &mut state)?;
311                    projected_tokens = json_tokens(self, &value)?;
312                    let returned_items = total_items.min(max_items);
313                    let remaining_items = total_items.saturating_sub(returned_items);
314                    JsonResponse {
315                        kind: "query".into(),
316                        value: Some(value),
317                        numeric_summary: None,
318                        differences: Vec::new(),
319                        sources: vec![loaded.source],
320                        result_complete: state.complete,
321                        total_items: (!state.complete).then_some(total_items),
322                        returned_items: (!state.complete).then_some(returned_items),
323                        remaining_items: (!state.complete).then_some(remaining_items),
324                        incomplete_reason: (!state.complete)
325                            .then_some(JsonIncompleteReason::MaxItems),
326                        meta: self.meta(generation, 0, None),
327                    }
328                }
329            }
330            JsonOperation::NumericSummary { path, selector } => {
331                let loaded = self.load_json(&path)?;
332                baseline_source_tokens = loaded.source_tokens;
333                let selected = select_json(&loaded.value, selector.as_ref())?;
334                let value = selected.value.ok_or(Error::InvalidInput {
335                    field: "selector",
336                    reason: "did not match a JSON value",
337                })?;
338                JsonResponse {
339                    kind: "numeric_summary".into(),
340                    value: None,
341                    numeric_summary: Some(numeric_summary(&value)),
342                    differences: Vec::new(),
343                    sources: vec![loaded.source],
344                    result_complete: true,
345                    total_items: None,
346                    returned_items: None,
347                    remaining_items: None,
348                    incomplete_reason: None,
349                    meta: self.meta(generation, 0, None),
350                }
351            }
352            JsonOperation::DiffFields {
353                base_path,
354                head_path,
355                selectors,
356                projection,
357            } => {
358                let before = self.load_json(&base_path)?;
359                check_cancelled(cancellation)?;
360                let after = self.load_json(&head_path)?;
361                baseline_source_tokens = before.source_tokens.saturating_add(after.source_tokens);
362                let mut state = ProjectionState {
363                    remaining: max_items,
364                    array_sample_size,
365                    complete: true,
366                };
367                let mut differences = Vec::with_capacity(selectors.len());
368                let mut total_items = 0usize;
369                for selector in selectors {
370                    check_cancelled(cancellation)?;
371                    let before_selected = select_json(&before.value, Some(&selector))?;
372                    let after_selected = select_json(&after.value, Some(&selector))?;
373                    let changed = before_selected.present != after_selected.present
374                        || before_selected.value != after_selected.value;
375                    total_items = total_items.saturating_add(
376                        before_selected
377                            .value
378                            .as_ref()
379                            .map(|value| {
380                                projection_item_count(value, projection, array_sample_size)
381                            })
382                            .unwrap_or_default(),
383                    );
384                    total_items = total_items.saturating_add(
385                        after_selected
386                            .value
387                            .as_ref()
388                            .map(|value| {
389                                projection_item_count(value, projection, array_sample_size)
390                            })
391                            .unwrap_or_default(),
392                    );
393                    let before_value = before_selected
394                        .value
395                        .as_ref()
396                        .map(|value| project_json(value, projection, &mut state))
397                        .transpose()?;
398                    let after_value = after_selected
399                        .value
400                        .as_ref()
401                        .map(|value| project_json(value, projection, &mut state))
402                        .transpose()?;
403                    if let Some(value) = &before_value {
404                        projected_tokens =
405                            projected_tokens.saturating_add(json_tokens(self, value)?);
406                    }
407                    if let Some(value) = &after_value {
408                        projected_tokens =
409                            projected_tokens.saturating_add(json_tokens(self, value)?);
410                    }
411                    differences.push(JsonFieldDiff {
412                        selector,
413                        before_present: before_selected.present,
414                        before: before_value,
415                        after_present: after_selected.present,
416                        after: after_value,
417                        changed,
418                    });
419                }
420                let returned_items = total_items.min(max_items);
421                let remaining_items = total_items.saturating_sub(returned_items);
422                JsonResponse {
423                    kind: "diff_fields".into(),
424                    value: None,
425                    numeric_summary: None,
426                    differences,
427                    sources: vec![before.source, after.source],
428                    result_complete: state.complete,
429                    total_items: (!state.complete).then_some(total_items),
430                    returned_items: (!state.complete).then_some(returned_items),
431                    remaining_items: (!state.complete).then_some(remaining_items),
432                    incomplete_reason: (!state.complete).then_some(JsonIncompleteReason::MaxItems),
433                    meta: self.meta(generation, 0, None),
434                }
435            }
436        };
437        if projected_tokens > max_tokens {
438            return Err(Error::RequestLimitExceeded {
439                field: "projected JSON tokens",
440                requested: projected_tokens,
441                limit: max_tokens,
442            });
443        }
444        response.meta.source_tokens = projected_tokens;
445        response.meta.emitted_tokens = projected_tokens;
446        self.fit_json_response(&mut response, key_page_context.as_ref(), options)?;
447        self.finalize_bounded_response(&mut response, options)?;
448        self.record_token_savings(
449            TokenAccountingOperation::Json,
450            Some(baseline_source_tokens),
451            &response.meta,
452        );
453        Ok(response)
454    }

456    fn fit_json_response(
457        &self,
458        response: &mut JsonResponse,
459        key_page_context: Option<&(String, String, usize)>,
460        options: ServiceCallOptions,
461    ) -> Result<()> {
462        if self.response_fits(response, options)? {
463            return Ok(());
464        }
465
466        if let (Some((source_hash, query_hash, offset)), Some(Value::Array(entries))) =
467            (key_page_context, response.value.as_ref())
468        {
469            let original = response.clone();
470            let max_response_tokens = options
471                .max_response_tokens()
472                .expect("fitting only runs with a response limit");
473            let budget = ResponseBudget::new(&self.config.tokenizer, max_response_tokens);
474            let keep = budget.largest_fitting_prefix(entries.len(), |keep| {
475                let mut candidate = original.clone();
476                let mut value = entries.clone();
477                value.truncate(keep);
478                candidate.value = Some(Value::Array(value));
479                let consumed = offset.saturating_add(keep);
480                let total = candidate.total_items.unwrap_or(consumed);
481                let remaining = total.saturating_sub(consumed);
482                candidate.returned_items = Some(keep);
483                candidate.remaining_items = Some(remaining);
484                candidate.result_complete = remaining == 0;
485                candidate.incomplete_reason =
486                    (remaining > 0).then_some(JsonIncompleteReason::MaxTokens);
487                candidate.meta.next_cursor =
488                    (remaining > 0).then(|| make_json_cursor(source_hash, query_hash, consumed));
489                let source_tokens = json_tokens(
490                    self,
491                    candidate
492                        .value
493                        .as_ref()
494                        .expect("keys projection keeps a value"),
495                )?;
496                candidate.meta.source_tokens = source_tokens;
497                candidate.meta.emitted_tokens = source_tokens;
498                self.finalized_response_tokens(&candidate)
499            })?;
500            if let Some(keep) = keep.filter(|keep| *keep > 0) {
501                let mut value = entries.clone();
502                value.truncate(keep);
503                response.value = Some(Value::Array(value));
504                let consumed = offset.saturating_add(keep);
505                let total = response.total_items.unwrap_or(consumed);
506                let remaining = total.saturating_sub(consumed);
507                response.returned_items = Some(keep);
508                response.remaining_items = Some(remaining);
509                response.result_complete = remaining == 0;
510                response.incomplete_reason =
511                    (remaining > 0).then_some(JsonIncompleteReason::MaxTokens);
512                response.meta.next_cursor =
513                    (remaining > 0).then(|| make_json_cursor(source_hash, query_hash, consumed));
514                let source_tokens = json_tokens(
515                    self,
516                    response
517                        .value
518                        .as_ref()
519                        .expect("keys projection keeps a value"),
520                )?;
521                response.meta.source_tokens = source_tokens;
522                response.meta.emitted_tokens = source_tokens;
523                return Ok(());
524            }
525        }
526
527        let requested = self.finalized_response_tokens(response)?;
528        Err(Error::RequestLimitExceeded {
529            field: "max_response_tokens",
530            requested,
531            limit: options
532                .max_response_tokens()
533                .expect("fitting only runs with a response limit"),
534        })
535    }

537    fn load_json(&self, path: &str) -> Result<LoadedJson> {
538        let path = normalize_relative(path)?;
539        let mut file = open_live_file(self, &path)?;
540        let max_bytes = usize::try_from(self.config.max_file_bytes).unwrap_or(usize::MAX);
541        let metadata_bytes = usize::try_from(file.metadata()?.len()).unwrap_or(usize::MAX);
542        if metadata_bytes > max_bytes {
543            return Err(Error::RequestLimitExceeded {
544                field: "JSON file bytes",
545                requested: metadata_bytes,
546                limit: max_bytes,
547            });
548        }
549        let mut bytes = Vec::with_capacity(metadata_bytes.min(max_bytes));
550        file.by_ref()
551            .take(
552                u64::try_from(max_bytes)
553                    .unwrap_or(u64::MAX)
554                    .saturating_add(1),
555            )
556            .read_to_end(&mut bytes)?;
557        if bytes.len() > max_bytes {
558            return Err(Error::RequestLimitExceeded {
559                field: "JSON file bytes",
560                requested: bytes.len(),
561                limit: max_bytes,
562            });
563        }
564        let content = std::str::from_utf8(&bytes).map_err(|_| Error::InvalidInput {
565            field: "path",
566            reason: "JSON file is not valid UTF-8",
567        })?;
568        let value = serde_json::from_str(content).map_err(|error| {
569            let syntax_category = match error.classify() {
570                serde_json::error::Category::Io => "io",
571                serde_json::error::Category::Syntax => "syntax",
572                serde_json::error::Category::Data => "data",
573                serde_json::error::Category::Eof => "eof",
574            };
575            Error::InvalidJson {
576                syntax_category,
577                byte_offset: json_error_byte_offset(content, error.line(), error.column()),
578                line: error.line(),
579                column: error.column(),
580                reason: error.to_string(),
581            }
582        })?;
583        Ok(LoadedJson {
584            source: JsonSource {
585                path,
586                content_hash: crate::text::hash(content),
587                bytes: bytes.len(),
588            },
589            value,
590            source_tokens: self.config.tokenizer.count(content),
591        })
592    }
593}

595fn select_json(value: &Value, selector: Option<&JsonSelector>) -> Result<SelectedJson> {
596    match selector {
597        None => Ok(SelectedJson {
598            present: true,
599            value: Some(value.clone()),
600        }),
601        Some(JsonSelector::Pointer { pointer }) => {
602            let selected = value.pointer(pointer).cloned();
603            Ok(SelectedJson {
604                present: selected.is_some(),
605                value: selected,
606            })
607        }
608        Some(JsonSelector::Jmespath { expression }) => {
609            let expression = jmespath::compile(expression)
610                .map_err(|error| invalid_json_selector("compile", error))?;
611            let selected = expression
612                .search(value)
613                .map_err(|error| invalid_json_selector("evaluate", error))?;
614            let selected = serde_json::to_value(selected.as_ref())
615                .map_err(|error| Error::InternalFailure(error.to_string()))?;
616            Ok(SelectedJson {
617                present: true,
618                value: Some(selected),
619            })
620        }
621    }
622}

624fn json_error_byte_offset(content: &str, line: usize, column: usize) -> usize {
625    if line == 0 {
626        return 0;
627    }
628    let mut line_start = 0usize;
629    for (index, segment) in content.split_inclusive('\n').enumerate() {
630        if index.saturating_add(1) == line {
631            return line_start
632                .saturating_add(column.saturating_sub(1).min(segment.len()))
633                .min(content.len());
634        }
635        line_start = line_start.saturating_add(segment.len());
636    }
637    content.len()
638}

640fn collect_all_keys(value: &Value, pointer: &str, keys: &mut BTreeMap<String, &'static str>) {
641    if !keys.contains_key(pointer) {
642        keys.insert(pointer.to_owned(), json_type(value));
643    }
644    match value {
645        Value::Object(values) => {
646            for (key, value) in values {
647                let pointer = format!("{pointer}/{}", escape_pointer(key));
648                collect_all_keys(value, &pointer, keys);
649            }
650        }
651        Value::Array(values) => {
652            let pointer = format!("{pointer}/*");
653            for value in values {
654                collect_all_keys(value, &pointer, keys);
655            }
656        }
657        _ => {}
658    }
659}

661fn key_entries(value: &Value) -> Vec<Value> {
662    let mut keys = BTreeMap::new();
663    collect_all_keys(value, "", &mut keys);
664    keys.into_iter()
665        .map(|(pointer, value_type)| json!({"pointer": pointer, "type": value_type}))
666        .collect()
667}

670    json_tokens(services, &Value::Array(entries[..length].to_vec()))
671}
672
673fn largest_key_prefix_within_tokens(
674    services: &Services,
675    entries: &[Value],
676    max_tokens: usize,
677) -> Result<(usize, usize)> {
678    if entries.is_empty() {
679        return Ok((0, json_tokens(services, &Value::Array(Vec::new()))?));
680    }
681    let full_tokens = key_prefix_tokens(services, entries, entries.len())?;
682    if full_tokens <= max_tokens {
683        return Ok((entries.len(), full_tokens));
684    }
685
686    let mut lower = 0usize;
687    let mut upper = entries.len();
688    while lower < upper {
689        let middle = lower.saturating_add(upper).saturating_add(1) / 2;
690        if key_prefix_tokens(services, entries, middle)? <= max_tokens {
691            lower = middle;
692        } else {
693            upper = middle.saturating_sub(1);
694        }
695    }
696    let tokens = key_prefix_tokens(services, entries, lower)?;
697    Ok((lower, tokens))
698}

700fn project_key_page(
701    services: &Services,
702    value: &Value,
703    max_items: usize,
704    max_tokens: usize,
705    cursor: Option<&JsonCursor>,
706    source_hash: &str,
707    query_hash: &str,
708) -> Result<KeyProjectionPage> {
709    let entries = key_entries(value);
710    let total_items = entries.len();
711    let offset = match cursor {
712        Some(cursor) if cursor.source_hash == source_hash && cursor.query_hash == query_hash => {
713            cursor.offset
714        }
715        Some(_) => return Err(Error::StaleCursor),
716        None => 0,
717    };
718    if offset > total_items || (offset == total_items && offset != 0) {
719        return Err(Error::StaleCursor);
720    }
721
722    let page_end = offset.saturating_add(max_items).min(total_items);
723    let candidates = &entries[offset..page_end];
724    let (returned_items, projected_tokens) =
725        largest_key_prefix_within_tokens(services, candidates, max_tokens)?;
726    if returned_items == 0 && !candidates.is_empty() {
727        return Err(Error::RequestLimitExceeded {
728            field: "one projected JSON item tokens",
729            requested: key_prefix_tokens(services, candidates, 1)?,
730            limit: max_tokens,
731        });
732    }
733
734    let consumed = offset.saturating_add(returned_items);
735    let remaining_items = total_items.saturating_sub(consumed);
736    let incomplete_reason = (remaining_items > 0).then_some(if returned_items < candidates.len() {
737        JsonIncompleteReason::MaxTokens
738    } else {
739        JsonIncompleteReason::MaxItems
740    });
741    let next_cursor =
742        (remaining_items > 0).then(|| make_json_cursor(source_hash, query_hash, consumed));
743    Ok(KeyProjectionPage {
744        value: Value::Array(candidates[..returned_items].to_vec()),
745        total_items,
746        returned_items,
747        remaining_items,
748        incomplete_reason,
749        next_cursor,
750        projected_tokens,
751    })
752}

754fn collapsed_item_count(value: &Value, array_sample_size: usize) -> usize {
755    match value {
756        Value::Array(values) => 1usize.saturating_add(
757            values
758                .iter()
759                .take(array_sample_size)
760                .map(|value| collapsed_item_count(value, array_sample_size))
761                .fold(0usize, usize::saturating_add),
762        ),
763        Value::Object(values) => 1usize.saturating_add(
764            values
765                .values()
766                .map(|value| collapsed_item_count(value, array_sample_size))
767                .fold(0usize, usize::saturating_add),
768        ),
769        _ => 1,
770    }
771}

773fn projection_item_count(
774    value: &Value,
775    projection: JsonProjection,
776    array_sample_size: usize,
777) -> usize {
778    match projection {
779        JsonProjection::Value | JsonProjection::Schema => count_nodes(value),
780        JsonProjection::Collapsed => collapsed_item_count(value, array_sample_size),
781        JsonProjection::Keys => key_entries(value).len(),
782    }
783}
784
785fn project_json(
786    value: &Value,
787    projection: JsonProjection,
788    state: &mut ProjectionState,
789) -> Result<Value> {
790    match projection {
791        JsonProjection::Value => {
792            let items = count_nodes(value);
793            if items > state.remaining {
794                return Err(Error::RequestLimitExceeded {
795                    field: "selected JSON items",
796                    requested: items,
797                    limit: state.remaining,
798                });
799            }
800            state.remaining -= items;
801            Ok(value.clone())
802        }
803        JsonProjection::Collapsed => Ok(collapse_json(value, state)),
804        JsonProjection::Keys => {
805            let mut keys = BTreeMap::new();
806            collect_keys(value, "", &mut keys, state);
807            Ok(Value::Array(
808                keys.into_iter()
809                    .map(|(pointer, value_type)| json!({"pointer": pointer, "type": value_type}))
810                    .collect(),
811            ))
812        }
813        JsonProjection::Schema => Ok(infer_schema(value, state)),
814    }
815}
816
817fn count_nodes(value: &Value) -> usize {
818    match value {
819        Value::Array(values) => 1usize.saturating_add(
820            values
821                .iter()
822                .map(count_nodes)
823                .fold(0usize, usize::saturating_add),
824        ),
825        Value::Object(values) => 1usize.saturating_add(
826            values
827                .values()
828                .map(count_nodes)
829                .fold(0usize, usize::saturating_add),
830        ),
831        _ => 1,
832    }
833}
834
835fn take_item(state: &mut ProjectionState) -> bool {
836    if state.remaining == 0 {
837        state.complete = false;
838        return false;
839    }
840    state.remaining -= 1;
841    true
842}
843
844fn collapse_json(value: &Value, state: &mut ProjectionState) -> Value {
845    if !take_item(state) {
846        return Value::Null;
847    }
848    match value {
849        Value::Array(values) => {
850            let mut sample = Vec::new();
851            for value in values.iter().take(state.array_sample_size) {
852                if state.remaining == 0 {
853                    state.complete = false;
854                    break;
855                }
856                sample.push(collapse_json(value, state));
857            }
858            let omitted = values.len().saturating_sub(sample.len());
859            json!({
860                "$array": {
861                    "count": values.len(),
862                    "sample": sample,
863                    "omitted": omitted,
864                }
865            })
866        }
867        Value::Object(values) => {
868            let mut projected = Map::new();
869            for (key, value) in values {
870                if state.remaining == 0 {
871                    state.complete = false;
872                    break;
873                }
874                projected.insert(key.clone(), collapse_json(value, state));
875            }
876            Value::Object(projected)
877        }
878        _ => value.clone(),
879    }
880}
881
882fn collect_keys(
883    value: &Value,
884    pointer: &str,
885    keys: &mut BTreeMap<String, &'static str>,
886    state: &mut ProjectionState,
887) {
888    if !keys.contains_key(pointer) {
889        if !take_item(state) {
890            return;
891        }
892        keys.insert(pointer.to_owned(), json_type(value));
893    }
894    match value {
895        Value::Object(values) => {
896            for (key, value) in values {
897                let pointer = format!("{pointer}/{}", escape_pointer(key));
898                collect_keys(value, &pointer, keys, state);
899            }
900        }
901        Value::Array(values) => {
902            let pointer = format!("{pointer}/*");
903            for value in values {
904                collect_keys(value, &pointer, keys, state);
905            }
906        }
907        _ => {}
908    }
909}
910
911fn escape_pointer(value: &str) -> String {
912    value.replace('~', "~0").replace('/', "~1")
913}
914
915fn infer_schema(value: &Value, state: &mut ProjectionState) -> Value {
916    if !take_item(state) {
917        return json!({"type": "unknown"});
918    }
919    match value {
920        Value::Object(values) => {
921            let mut properties = Map::new();
922            for (key, value) in values {
923                if state.remaining == 0 {
924                    state.complete = false;
925                    break;
926                }
927                properties.insert(key.clone(), infer_schema(value, state));
928            }
929            json!({"type": "object", "properties": properties})
930        }
931        Value::Array(values) => {
932            let mut variants = BTreeMap::new();
933            for value in values {
934                if state.remaining == 0 {
935                    state.complete = false;
936                    break;
937                }
938                let schema = infer_schema(value, state);
939                let key = serde_json::to_string(&schema).unwrap_or_default();
940                variants.entry(key).or_insert(schema);
941            }
942            let items = match variants.len() {
943                0 => json!({}),
944                1 => variants.into_values().next().unwrap_or_else(|| json!({})),
945                _ => json!({"anyOf": variants.into_values().collect::<Vec<_>>() }),
946            };
947            json!({"type": "array", "count": values.len(), "items": items})
948        }
949        _ => json!({"type": json_type(value)}),
950    }
951}
952
953fn json_type(value: &Value) -> &'static str {
954    match value {
955        Value::Null => "null",
956        Value::Bool(_) => "boolean",
957        Value::Number(_) => "number",
958        Value::String(_) => "string",
959        Value::Array(_) => "array",
960        Value::Object(_) => "object",
961    }
962}

964fn numeric_summary(value: &Value) -> JsonNumericSummary {
965    let mut values = Vec::new();
966    let mut non_numeric_count = 0usize;
967    collect_numbers(value, &mut values, &mut non_numeric_count);
968    values.sort_by(f64::total_cmp);
969    let count = values.len();
970    let median = match count {
971        0 => None,
972        count if count % 2 == 1 => Some(values[count / 2]),
973        count => Some((values[count / 2 - 1] + values[count / 2]) / 2.0),
974    };
975    let p95 = (count > 0).then(|| {
976        let rank = (count.saturating_mul(95).saturating_add(99)) / 100;
977        values[rank.saturating_sub(1).min(count - 1)]
978    });
979    JsonNumericSummary {
980        count,
981        non_numeric_count,
982        min: values.first().copied(),
983        median,
984        p95,
985        max: values.last().copied(),
986    }
987}
988
989fn collect_numbers(value: &Value, values: &mut Vec<f64>, non_numeric_count: &mut usize) {
990    match value {
991        Value::Number(value) => {
992            if let Some(value) = value.as_f64() {
993                values.push(value);
994            } else {
995                *non_numeric_count = non_numeric_count.saturating_add(1);
996            }
997        }
998        Value::Array(items) => {
999            for item in items {
1000                collect_numbers(item, values, non_numeric_count);
1001            }
1002        }
1003        Value::Object(items) => {
1004            for item in items.values() {
1005                collect_numbers(item, values, non_numeric_count);
1006            }
1007        }
1008        _ => *non_numeric_count = non_numeric_count.saturating_add(1),
1009    }
1010}

1012fn json_tokens(services: &Services, value: &Value) -> Result<usize> {
1013    let serialized =
1014        serde_json::to_string(value).map_err(|error| Error::InternalFailure(error.to_string()))?;
1015    Ok(services.config.tokenizer.count(&serialized))
1016}
