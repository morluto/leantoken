19fn outline_request_class(response: &OutlineResponse) -> TokenSavingsRequestClass {
20    let empty_latex_outline = response.total_symbols == 0
21        && response
22            .files
23            .iter()
24            .any(|file| file.language.as_deref() == Some("latex"));
25    if !response.parse_complete || empty_latex_outline {
26        TokenSavingsRequestClass::Unsupported
27    } else if !response.result_complete {
28        TokenSavingsRequestClass::Incomplete
29    } else {
30        TokenSavingsRequestClass::Useful
31    }
32}

64fn decode_outline_cursor(cursor: &str) -> Result<(u64, usize, String)> {
65    let fields = cursor.split(':').collect::<Vec<_>>();
66    let [generation, kind, offset, query_hash] = fields.as_slice() else {
67        return Err(Error::StaleCursor);
68    };
69    if *kind != "outline" || query_hash.len() != 16 || !query_hash.bytes().all(is_lower_hex) {
70        return Err(Error::StaleCursor);
71    }
72    Ok((
73        generation.parse().map_err(|_| Error::StaleCursor)?,
74        offset.parse().map_err(|_| Error::StaleCursor)?,
75        (*query_hash).into(),
76    ))
77}

79fn outline_query_hash(request: &OutlineRequest, projection: Option<&str>) -> String {
80    fn update_field(hasher: &mut blake3::Hasher, value: &str) {
81        hasher.update(&(value.len() as u64).to_le_bytes());
82        hasher.update(value.as_bytes());
83    }
84
85    let mut hasher = blake3::Hasher::new();
86    hasher.update(&(request.paths.len() as u64).to_le_bytes());
87    for path in &request.paths {
88        update_field(&mut hasher, path);
89    }
90    for value in [&request.symbol_name, &request.symbol_kind] {
91        match value {
92            Some(value) => {
93                hasher.update(&[1]);
94                update_field(&mut hasher, value);
95            }
96            None => {
97                hasher.update(&[0]);
98            }
99        }
100    }
101    if let Some(projection) = projection {
102        hasher.update(&[2]);
103        update_field(&mut hasher, projection);
104    }
105    hasher.finalize().to_hex()[..16].to_string()
106}

108fn parse_outline_cursor(
109    cursor: Option<&str>,
110    generation: u64,
111    request: &OutlineRequest,
112    projection: Option<&str>,
113) -> Result<usize> {
114    let Some(cursor) = cursor else {
115        return Ok(0);
116    };
117    let (cursor_generation, offset, query_hash) = decode_outline_cursor(cursor)?;
118    if cursor_generation != generation || query_hash != outline_query_hash(request, projection) {
119        return Err(Error::StaleCursor);
120    }
121    Ok(offset)
122}

124fn make_outline_cursor(
125    generation: u64,
126    offset: usize,
127    request: &OutlineRequest,
128    projection: Option<&str>,
129) -> String {
130    format!(
131        "{generation}:outline:{offset}:{}",
132        outline_query_hash(request, projection)
133    )
134}

136fn validate_outline_input(request: &OutlineRequest) -> Result<()> {
137    if request.paths.is_empty() {
138        return Err(Error::InvalidInput {
139            field: "paths",
140            reason: "must contain at least one path",
141        });
142    }
143    if request.paths.len() > MAX_INPUT_ITEMS {
144        return Err(Error::LimitExceeded);
145    }
146    for path in &request.paths {
147        validate_input(path, "path", MAX_PATH_BYTES)?;
148        validate_relative(path)?;
149    }
150    validate_optional_input(
151        request.symbol_name.as_deref(),
152        "symbol name",
153        MAX_PATTERN_BYTES,
154    )?;
155    validate_optional_input(
156        request.symbol_kind.as_deref(),
157        "symbol kind",
158        MAX_PATTERN_BYTES,
159    )?;
160    validate_optional_input(request.cursor.as_deref(), "cursor", 256)?;
161    if let Some(cursor) = request.cursor.as_deref() {
162        decode_outline_cursor(cursor)?;
163    }
164    Ok(())
165}

227    async fn outline_execute(
228        &self,
229        request: OutlineRequest,
230        execution: RetrievalExecution,
231    ) -> Result<OutlineResponse> {
232        let operation = TokenAccountingOperation::Outline;
233        let RetrievalExecution {
234            consistency,
235            options,
236            cancellation,
237        } = execution;
238        self.observe_service_result(operation, self.validate_call_options(options))?;
239        if let Some(consistency) = consistency {
240            self.observe_service_result(operation, validate_outline_input(&request))?;
241            self.observe_service_result(operation, self.result_limit(request.max_results))?;
242            self.observe_service_result(
243                operation,
244                self.token_limit(request.max_tokens, self.config.default_read_tokens),
245            )?;
246            let consistency_result = self
247                .apply_consistency_with_initial_deadline(
248                    consistency,
249                    cancellation.clone(),
250                    options.initial_reconciliation_deadline(),
251                )
252                .await;
253            self.observe_service_result(operation, consistency_result)?;
254        }
255        let this = self.clone();
256        let result = self
257            .blocking_executor
258            .run(cancellation, move |cancellation| {
259                this.outline_sync(request, options, true, true, cancellation)
260            })
261            .await;
262        self.observe_service_result(operation, result)
263    }

303        &self,
304        request: OutlineRequest,
305        execution: RetrievalExecution,
306    ) -> Result<OutlineSignaturesResponse> {
307        let operation = TokenAccountingOperation::Outline;
308        let RetrievalExecution {
309            consistency,
310            options,
311            cancellation,
312        } = execution;
313        self.observe_service_result(operation, self.validate_call_options(options))?;
314        if let Some(consistency) = consistency {
315            self.observe_service_result(operation, validate_outline_input(&request))?;
316            self.observe_service_result(operation, self.result_limit(request.max_results))?;
317            self.observe_service_result(
318                operation,
319                self.token_limit(request.max_tokens, self.config.default_read_tokens),
320            )?;
321            let consistency_result = self
322                .apply_consistency_with_initial_deadline(
323                    consistency,
324                    cancellation.clone(),
325                    options.initial_reconciliation_deadline(),
326                )
327                .await;
328            self.observe_service_result(operation, consistency_result)?;
329        }
330        let this = self.clone();
331        let result = self
332            .blocking_executor
333            .run(cancellation, move |cancellation| {
334                let response = this.outline_sync(
335                    request,
336                    ServiceCallOptions::new(),
337                    false,
338                    false,
339                    cancellation,
340                )?;
341                let mut files = Vec::with_capacity(response.files.len());
342                for file in response.files {
343                    let signatures = file
344                        .symbols
345                        .into_iter()
346                        .map(|symbol| OutlineSignature {
347                            name: symbol.name,
348                            kind: symbol.kind,
349                            parent: symbol.parent,
350                            signature: symbol.signature,
351                            start_line: symbol.start_line,
352                            end_line: symbol.end_line,
353                        })
354                        .collect::<Vec<_>>();
355                    let serialized = serde_json::to_string(&signatures)
356                        .map_err(|error| Error::InternalFailure(error.to_string()))?;
357                    files.push(OutlineSignaturesFile {
358                        path: file.path,
359                        content_hash: hash(&serialized),
360                        language: file.language,
361                        parse_complete: file.parse_complete,
362                        structurally_complete: file.structurally_complete,
363                        signatures,
364                    });
365                }
366                let mut compact = OutlineSignaturesResponse {
367                    files,
368                    path_results: response.path_results,
369                    parse_complete: response.parse_complete,
370                    result_complete: response.result_complete,
371                    total_symbols: response.total_symbols,
372                    returned_symbols: response.returned_symbols,
373                    truncated_by_max_results: response.truncated_by_max_results,
374                    truncated_by_max_tokens: response.truncated_by_max_tokens,
375                    symbol_counts_by_kind: response.symbol_counts_by_kind,
376                    meta: response.meta,
377                };
378                this.finalize_bounded_response(&mut compact, options)?;
379                this.record_token_savings_classified(
380                    TokenAccountingOperation::Outline,
381                    None,
382                    &compact.meta,
383                    outline_signatures_request_class(&compact),
384                );
385                Ok(compact)
386            })
387            .await;
388        self.observe_service_result(operation, result)
389    }

391    fn outline_sync(
392        &self,
393        mut request: OutlineRequest,
394        options: ServiceCallOptions,
395        include_imports: bool,
396        record_savings: bool,
397        cancellation: &CancellationToken,
398    ) -> Result<OutlineResponse> {
399        check_cancelled(cancellation)?;
400        validate_outline_input(&request)?;
401        request.paths = request
402            .paths
403            .iter()
404            .map(|path| normalize_relative(path))
405            .collect::<Result<Vec<_>>>()?;
406        let limit = self.result_limit(request.max_results)?;
407        let token_limit = self.token_limit(request.max_tokens, self.config.default_read_tokens)?;
408        let (mut response, baseline_source_tokens) = self.consistent(|session, generation| {
409            let cursor_projection = (!include_imports).then_some("signatures");
410            let offset = parse_outline_cursor(
411                request.cursor.as_deref(),
412                generation,
413                &request,
414                cursor_projection,
415            )?;
416            let mut total_symbols = 0usize;
417            let mut total_imports = 0usize;
418            let mut symbol_counts_by_kind = BTreeMap::new();
419            let mut parse_complete = true;
420            let mut all_paths_indexed = true;
421            let mut files = Vec::with_capacity(request.paths.len());
422            let mut file_totals = Vec::with_capacity(request.paths.len());
423            let mut path_results = Vec::with_capacity(request.paths.len());
424            for (request_index, path) in request.paths.iter().enumerate() {
425                check_cancelled(cancellation)?;
426                let Some(file) = session.find_file(path)? else {
427                    parse_complete = false;
428                    all_paths_indexed = false;
429                    path_results.push(OutlinePathResult {
430                        request_index,
431                        path: path.clone(),
432                        status: OutlinePathStatus::NotIndexed,
433                    });
434                    continue;
435                };
436                path_results.push(OutlinePathResult {
437                    request_index,
438                    path: path.clone(),
439                    status: OutlinePathStatus::Indexed,
440                });
441                let kind_counts = session.symbol_counts_for_file_filtered(
442                    file.id,
443                    request.symbol_name.as_deref(),
444                    request.symbol_kind.as_deref(),
445                )?;
446                let file_symbol_total = kind_counts.iter().map(|(_, count)| *count).sum::<usize>();
447                for (kind, count) in kind_counts {
448                    *symbol_counts_by_kind.entry(kind).or_insert(0usize) += count;
449                }
450                let file_import_total = if include_imports {
451                    session.count_imports_for_file(file.id)?
452                } else {
453                    0
454                };
455                total_symbols = total_symbols.saturating_add(file_symbol_total);
456                total_imports = total_imports.saturating_add(file_import_total);
457                parse_complete &= file.structurally_complete;
458                file_totals.push((file.id, file_symbol_total, file_import_total));
459                files.push(OutlineFile {
460                    path: file.path,
461                    language: file.language,
462                    parse_complete: file.structurally_complete,
463                    structurally_complete: file.structurally_complete,
464                    symbols: Vec::new(),
465                    imports: Vec::new(),
466                });
467            }
468
469            let total_entries = total_symbols.saturating_add(total_imports);
470            if offset > total_entries {
471                return Err(Error::StaleCursor);
472            }
473
474            let mut remaining = limit;
475            let mut emitted_tokens = 0usize;
476            let mut returned_symbols = 0usize;
477            let mut returned_imports = 0usize;
478            let mut consumed = offset;
479            let mut prefix = 0usize;
480            let mut truncated_by_max_tokens = false;
481            'files: for (file_index, (file_id, file_symbol_total, file_import_total)) in
482                file_totals.iter().copied().enumerate()
483            {
484                check_cancelled(cancellation)?;
485                let file_total = file_symbol_total.saturating_add(file_import_total);
486                let file_end = prefix.saturating_add(file_total);
487                if offset >= file_end {
488                    prefix = file_end;
489                    continue;
490                }
491                let local_offset = offset.saturating_sub(prefix);
492                let mut symbol_offset = local_offset.min(file_symbol_total);
493                while symbol_offset < file_symbol_total {
494                    if remaining == 0 {
495                        break 'files;
496                    }
497                    let batch_limit = file_symbol_total.saturating_sub(symbol_offset).min(100);
498                    let symbols = session.get_symbols_for_file_filtered_page(
499                        file_id,
500                        request.symbol_name.as_deref(),
501                        request.symbol_kind.as_deref(),
502                        batch_limit,
503                        symbol_offset,
504                    )?;
505                    if symbols.is_empty() {
506                        return Err(Error::StaleCursor);
507                    }
508                    for symbol in symbols {
509                        if remaining == 0 {
510                            break 'files;
511                        }
512                        consumed = consumed.saturating_add(1);
513                        symbol_offset = symbol_offset.saturating_add(1);
514                        let symbol = storage_symbol(symbol);
515                        let cost = symbol
516                            .signature
517                            .as_deref()
518                            .map_or(1, |value| self.config.tokenizer.count(value));
519                        if emitted_tokens.saturating_add(cost) > token_limit {
520                            truncated_by_max_tokens = true;
521                            continue;
522                        }
523                        emitted_tokens = emitted_tokens.saturating_add(cost);
524                        remaining -= 1;
525                        returned_symbols = returned_symbols.saturating_add(1);
526                        files[file_index].symbols.push(symbol);
527                    }
528                }
529
530                let mut import_offset = local_offset.saturating_sub(file_symbol_total);
531                while import_offset < file_import_total {
532                    if remaining == 0 {
533                        break 'files;
534                    }
535                    let batch_limit = file_import_total.saturating_sub(import_offset).min(100);
536                    let imports =
537                        session.get_imports_for_file_page(file_id, batch_limit, import_offset)?;
538                    if imports.is_empty() {
539                        return Err(Error::StaleCursor);
540                    }
541                    for import in imports {
542                        if remaining == 0 {
543                            break 'files;
544                        }
545                        consumed = consumed.saturating_add(1);
546                        import_offset = import_offset.saturating_add(1);
547                        let import = Import {
548                            raw_target: import.raw_target,
549                            resolved_path: import.resolved_path,
550                            line: import.line,
551                        };
552                        let cost = self.config.tokenizer.count(&import.raw_target)
553                            + import
554                                .resolved_path
555                                .as_deref()
556                                .map_or(0, |value| self.config.tokenizer.count(value));
557                        if emitted_tokens.saturating_add(cost) > token_limit {
558                            truncated_by_max_tokens = true;
559                            continue;
560                        }
561                        emitted_tokens = emitted_tokens.saturating_add(cost);
562                        remaining -= 1;
563                        returned_imports = returned_imports.saturating_add(1);
564                        files[file_index].imports.push(import);
565                    }
566                }
567                prefix = file_end;
568            }
569
570            let truncated_by_max_results = remaining == 0 && consumed < total_entries;
571            let next_cursor = truncated_by_max_results
572                .then(|| make_outline_cursor(generation, consumed, &request, cursor_projection));
573            let result_complete = offset == 0
574                && returned_symbols == total_symbols
575                && returned_imports == total_imports;
576            let paths = files
577                .iter()
578                .map(|file| file.path.clone())
579                .collect::<Vec<_>>();
580            let baseline_source_tokens =
581                session.whole_file_source_tokens(&paths, self.config.tokenizer.name())?;
582            Ok((
583                OutlineResponse {
584                    files,
585                    path_results,
586                    parse_complete,
587                    result_complete: result_complete && all_paths_indexed,
588                    total_symbols,
589                    returned_symbols,
590                    total_imports,
591                    returned_imports,
592                    truncated_by_max_results,
593                    truncated_by_max_tokens,
594                    symbol_counts_by_kind,
595                    meta: self.meta(generation, emitted_tokens, next_cursor),
596                },
597                baseline_source_tokens,
598            ))
599        })?;
600        let returned_entries = response
601            .returned_symbols
602            .saturating_add(response.returned_imports);
603        if !self.response_fits_with_receipt_reserve(&response, returned_entries, options)? {
604            return Err(self.response_budget_error_with_receipt_reserve(
605                &response,
606                returned_entries,
607                options
608                    .max_response_tokens()
609                    .expect("fitting only runs with a response limit"),
610            )?);
611        }
612        let receipt_candidates = response
613            .files
614            .iter()
615            .flat_map(|file| {
616                let symbol_evidence = file.symbols.iter().map(|symbol| {
617                    let content = symbol.signature.as_deref().unwrap_or(&symbol.name);
618                    ReceiptEvidence::new(
619                        file.path.clone(),
620                        symbol.start_line,
621                        symbol.end_line,
622                        hash(content),
623                        Some(content),
624                    )
625                });
626                let import_evidence = file.imports.iter().map(|import| {
627                    ReceiptEvidence::new(
628                        file.path.clone(),
629                        import.line,
630                        import.line,
631                        hash(&import.raw_target),
632                        Some(&import.raw_target),
633                    )
634                });
635                symbol_evidence.chain(import_evidence)
636            })
637            .collect::<Vec<_>>();
638        let receipt = self.evaluate_receipt(
639            request.receipt_id.as_deref(),
640            response.meta.repository_generation,
641            &receipt_candidates,
642        )?;
643        let mut decision_index = 0usize;
644        for file in &mut response.files {
645            file.symbols.retain(|_| {
646                let keep = matches!(
647                    receipt.decisions[decision_index],
648                    ReceiptDecision::Return | ReceiptDecision::ReturnNearDuplicate
649                );
650                decision_index += 1;
651                keep
652            });
653            file.imports.retain(|_| {
654                let keep = matches!(
655                    receipt.decisions[decision_index],
656                    ReceiptDecision::Return | ReceiptDecision::ReturnNearDuplicate
657                );
658                decision_index += 1;
659                keep
660            });
661        }
662        response.returned_symbols = response.files.iter().map(|file| file.symbols.len()).sum();
663        response.returned_imports = response.files.iter().map(|file| file.imports.len()).sum();
664        response.result_complete = response.result_complete
665            && response.returned_symbols == response.total_symbols
666            && response.returned_imports == response.total_imports;
667        let symbol_tokens = response
668            .files
669            .iter()
670            .flat_map(|file| &file.symbols)
671            .map(|symbol| {
672                symbol
673                    .signature
674                    .as_deref()
675                    .map_or(1, |signature| self.config.tokenizer.count(signature))
676            })
677            .sum::<usize>();
678        let import_tokens = response
679            .files
680            .iter()
681            .flat_map(|file| &file.imports)
682            .map(|import| {
683                self.config.tokenizer.count(&import.raw_target)
684                    + import
685                        .resolved_path
686                        .as_deref()
687                        .map_or(0, |path| self.config.tokenizer.count(path))
688            })
689            .sum::<usize>();
690        response.meta.source_tokens = symbol_tokens.saturating_add(import_tokens);
691        response.meta.emitted_tokens = response.meta.source_tokens;
692        receipt.apply_meta(&mut response.meta);
693        self.finalize_bounded_response(&mut response, options)?;
694        if record_savings {
695            self.record_token_savings_classified(
696                TokenAccountingOperation::Outline,
697                baseline_source_tokens,
698                &response.meta,
699                outline_request_class(&response),
700            );
701        }
702        Ok(response)
703    }
704}
