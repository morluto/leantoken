32fn validate_read_input(request: &ReadRequest) -> Result<()> {
33    validate_input(&request.path, "path", MAX_PATH_BYTES)?;
34    if request.symbol.as_deref().is_some_and(str::is_empty) {
35        return Err(Error::InvalidInput {
36            field: "symbol",
37            reason: "must not be empty",
38        });
39    }
40    validate_optional_input(request.symbol.as_deref(), "symbol", MAX_PATTERN_BYTES)?;
41    if request
42        .heading
43        .as_deref()
44        .is_some_and(|heading| heading.trim().is_empty())
45    {
46        return Err(Error::InvalidInput {
47            field: "heading",
48            reason: "must not be empty",
49        });
50    }
51    validate_optional_input(request.heading.as_deref(), "heading", MAX_PATTERN_BYTES)?;
52    if request.heading_occurrence == Some(0) {
53        return Err(Error::InvalidInput {
54            field: "heading occurrence",
55            reason: "must be one-based",
56        });
57    }
58    if request.heading_occurrence.is_some() && request.heading.is_none() {
59        return Err(Error::InvalidInput {
60            field: "heading occurrence",
61            reason: "requires a heading target",
62        });
63    }
64    validate_optional_input(request.expected_hash.as_deref(), "expected hash", 128)?;
65    validate_optional_input(
66        request.continuation_cursor.as_deref(),
67        "continuation cursor",
68        256,
69    )?;
70    if let Some(cursor) = request.continuation_cursor.as_deref() {
71        decode_read_cursor(cursor)?;
72    }
73    validate_relative(&request.path)?;
74    let has_line_target = request.start_line.is_some() || request.end_line.is_some();
75    if request.symbol.is_some() && has_line_target {
76        return Err(Error::InvalidInput {
77            field: "read target",
78            reason: "must use either a symbol or line range, not both",
79        });
80    }
81    if request.heading.is_some() && (request.symbol.is_some() || has_line_target) {
82        return Err(Error::InvalidInput {
83            field: "read target",
84            reason: "must use either a heading, symbol, or line range",
85        });
86    }
87    if request.continuation_cursor.is_some()
88        && (request.symbol.is_some() || request.heading.is_some() || has_line_target)
89    {
90        return Err(Error::InvalidInput {
91            field: "read target",
92            reason: "must use either a continuation cursor or a new target, not both",
93        });
94    }
95    if request.delta && request.continuation_cursor.is_some() {
96        return Err(Error::InvalidInput {
97            field: "delta",
98            reason: "is supported only for a new line, symbol, or heading target",
99        });
100    }
101    if request.symbol.is_none()
102        && request.heading.is_none()
103        && request.continuation_cursor.is_none()
104    {
105        let start_line = request.start_line.unwrap_or(1);
106        if start_line == 0
107            || request
108                .end_line
109                .is_some_and(|end_line| end_line < start_line)
110        {
111            return Err(invalid_line_range());
112        }
113    }
114    Ok(())
115}

177    async fn read_execute(
178        &self,
179        request: ReadRequest,
180        execution: RetrievalExecution,
181    ) -> Result<ReadResponse> {
182        let operation = TokenAccountingOperation::Read;
183        let RetrievalExecution {
184            consistency,
185            options,
186            cancellation,
187        } = execution;
188        self.observe_service_result(operation, self.validate_call_options(options))?;
189        if let Some(consistency) = consistency {
190            self.observe_service_result(operation, validate_read_input(&request))?;
191            self.observe_service_result(
192                operation,
193                self.token_limit(request.max_tokens, self.config.default_read_tokens),
194            )?;
195            let consistency_result = self
196                .apply_consistency(consistency, cancellation.clone())
197                .await;
198            self.observe_service_result(operation, consistency_result)?;
199        }
200        let this = self.clone();
201        let result = self
202            .blocking_executor
203            .run(cancellation, move |cancellation| {
204                this.read_sync(request, options, cancellation)
205            })
206            .await;
207        self.observe_service_result(operation, result)
208    }

210    fn read_sync(
211        &self,
212        mut request: ReadRequest,
213        options: ServiceCallOptions,
214        cancellation: &CancellationToken,
215    ) -> Result<ReadResponse> {
216        check_cancelled(cancellation)?;
217        validate_read_input(&request)?;
218        request.path = normalize_relative(&request.path)?;
219        let max_tokens = self.token_limit(request.max_tokens, self.config.default_read_tokens)?;
220        let materialized = self.consistent(|session, generation| {
221            check_cancelled(cancellation)?;
222            self.read_at_generation_with_options(session, &request, generation, max_tokens, options)
223        })?;
224        let mut response = materialized.response;
225        let direct_response = response.clone();
226        if request.delta {
227            let evaluation = self.read_deltas.evaluate(ReadDeltaInput {
228                repository_id: &response.meta.repository_id,
229                storage: &self.storage,
230                request: &request,
231                response: &response,
232                current_content: &materialized.current_content,
233                full_tokens: materialized.current_tokens,
234                tokenizer: self.config.tokenizer,
235            })?;
236            if evaluation.receipt.outcome == ReadDeltaOutcome::NotModified {
237                response.status = ReadStatus::NotModified;
238                response.not_modified = true;
239                response.content = None;
240                response.delta = None;
241                response.meta.source_tokens = 0;
242                response.meta.emitted_tokens = 0;
243            } else if let Some(delta) = evaluation.delta {
244                let emitted_tokens = evaluation
245                    .receipt
246                    .delta_tokens
247                    .expect("delta evaluation reports its token count");
248                response.status = ReadStatus::Delta;
249                response.content = None;
250                response.delta = Some(delta);
251                response.meta.source_tokens = emitted_tokens;
252                response.meta.emitted_tokens = emitted_tokens;
253            }
254            response.delta_receipt = Some(evaluation.receipt);
255            prefer_full_if_delta_payload_not_smaller(
256                &mut response,
257                &materialized.current_content,
258                materialized.current_tokens,
259                self.config.tokenizer,
260            )?;
261        }
262        let mut returned_items = usize::from(!response.not_modified);
263        if let Some(limit) = options.max_response_tokens() {
264            let mut reserved =
265                self.finalized_response_tokens_with_receipt_reserve(&response, returned_items)?;
266            if reserved > limit && request.delta {
267                response = direct_response;
268                returned_items = usize::from(!response.not_modified);
269                reserved =
270                    self.finalized_response_tokens_with_receipt_reserve(&response, returned_items)?;
271            }
272            if reserved > limit {
273                return Err(Error::RequestLimitExceeded {
274                    field: "max_response_tokens",
275                    requested: reserved,
276                    limit,
277                });
278            }
279        }
280        let receipt_candidates = if response.not_modified {
281            Vec::new()
282        } else {
283            vec![ReceiptEvidence::new(
284                response.path.clone(),
285                response.returned_start_line,
286                response.returned_end_line,
287                response.content_hash.clone(),
288                Some(&materialized.current_content),
289            )]
290        };
291        let receipt = self.evaluate_read_receipt(
292            request.receipt_id.as_deref(),
293            response.meta.repository_generation,
294            &receipt_candidates,
295        )?;
296        if receipt
297            .decisions
298            .first()
299            .is_some_and(|decision| *decision == ReceiptDecision::SuppressExact)
300        {
301            response.content = None;
302            response.delta = None;
303            response.status = ReadStatus::ReceiptSuppressed;
304            response.not_modified = false;
305            response.meta.source_tokens = 0;
306            response.meta.emitted_tokens = 0;
307            if let Some(delta_receipt) = response.delta_receipt.as_mut() {
308                delta_receipt.outcome = ReadDeltaOutcome::ReceiptSuppressed;
309                delta_receipt.delta_tokens = Some(0);
310                delta_receipt.avoided_tokens = delta_receipt.full_tokens;
311                delta_receipt.fallback_reason = None;
312            }
313        }
314        receipt.apply_meta(&mut response.meta);
315        self.finalize_bounded_response(&mut response, options)?;
316        let expected_hash_not_modified = request.expected_hash.is_some() && response.not_modified;
317        self.record_token_savings_with_expected_hash(
318            TokenAccountingOperation::Read,
319            Some(materialized.baseline_source_tokens),
320            &response.meta,
321            if response.not_modified {
322                TokenSavingsRequestClass::HashSuppressed
323            } else if matches!(
324                &response.status,
325                ReadStatus::Truncated | ReadStatus::ReceiptSuppressed
326            ) {
327                TokenSavingsRequestClass::Incomplete
328            } else {
329                TokenSavingsRequestClass::Useful
330            },
331            expected_hash_not_modified,
332            if expected_hash_not_modified {
333                materialized.baseline_source_tokens
334            } else {
335                0
336            },
337        );
338        Ok(response)
339    }

341    fn read_at_generation_with_options(
342        &self,
343        session: &ReadSession,
344        request: &ReadRequest,
345        generation: u64,
346        max_tokens: usize,
347        options: ServiceCallOptions,
348    ) -> Result<MaterializedRead> {
349        let materialized = self.read_at_generation(session, request, generation, max_tokens)?;
350        let returned_items = usize::from(!materialized.response.not_modified);
351        if self.response_fits_with_receipt_reserve(
352            &materialized.response,
353            returned_items,
354            options,
355        )? {
356            return Ok(materialized);
357        }
358
359        let max_response_tokens = options
360            .max_response_tokens()
361            .expect("fitting only runs with a response limit");
362        let budget = ResponseBudget::new(&self.config.tokenizer, max_response_tokens);
363        let keep = budget.largest_fitting_prefix(max_tokens, |candidate_limit| {
364            let candidate_limit = candidate_limit.max(1);
365            let candidate =
366                self.read_at_generation(session, request, generation, candidate_limit)?;
367            let returned_items = usize::from(!candidate.response.not_modified);
368            self.finalized_response_tokens_with_receipt_reserve(&candidate.response, returned_items)
369        })?;
370        if let Some(candidate_limit) = keep.filter(|keep| *keep > 0) {
371            return self.read_at_generation(session, request, generation, candidate_limit);
372        }
373
374        let minimum = self.read_at_generation(session, request, generation, 1)?;
375        let requested = self.finalized_response_tokens_with_receipt_reserve(
376            &minimum.response,
377            usize::from(!minimum.response.not_modified),
378        )?;
379        Err(Error::RequestLimitExceeded {
380            field: "max_response_tokens",
381            requested,
382            limit: max_response_tokens,
383        })
384    }

386    fn read_at_generation(
387        &self,
388        session: &ReadSession,
389        request: &ReadRequest,
390        generation: u64,
391        max_tokens: usize,
392    ) -> Result<MaterializedRead> {
393        let indexed = session
394            .find_file(&request.path)?
395            .ok_or_else(|| Error::NotIndexed(request.path.clone()))?;
396        let target = resolve_read_target(session, indexed.id, request, generation)?;
397
398        // Hash the complete live file and extract the bounded target during
399        // the same stream. Truncated responses retain one verification pass
400        // before issuing a continuation cursor.
401        let file = open_live_file(self, &request.path)?;
402        let observation = observe_live_range(
403            &file,
404            target.target_start_line,
405            target.target_end_line,
406            target.page_start_byte,
407            max_tokens,
408            self.config.tokenizer,
409        )?;
410        let snapshot = observation.snapshot;
411        let range = observation.range;
412        if target
413            .expected_full_hash
414            .as_deref()
415            .is_some_and(|expected| expected != snapshot.content_hash)
416        {
417            return Err(Error::StaleCursor);
418        }
419        let target_end_line = target
420            .target_end_line
421            .unwrap_or(snapshot.end_line)
422            .min(snapshot.end_line);
423        if target.target_start_line > target_end_line || target.page_start_line > target_end_line {
424            return Err(invalid_line_range());
425        }
426        if range.page_start_line != target.page_start_line {
427            return Err(Error::StaleCursor);
428        }
429        let baseline_source_tokens = self.config.tokenizer.count(&range.content);
430        let (content, emitted_tokens) = self.config.tokenizer.truncate(&range.content, max_tokens);
431        let returned_start_line = range.page_start_line;
432        let returned_end_line = returned_end_line(returned_start_line, content);
433        let next_byte = target.page_start_byte.saturating_add(content.len());
434        let truncated = next_byte < range.target_bytes;
435        let next_start_line = truncated.then(|| {
436            if content.ends_with('\n') {
437                returned_end_line.saturating_add(1)
438            } else {
439                returned_end_line
440            }
441        });
442        if truncated {
443            let after_read = stream_snapshot(&file)?;
444            if after_read.content_hash != snapshot.content_hash
445                || after_read.end_line != snapshot.end_line
446            {
447                return Err(Error::RetryableConflict(
448                    crate::error::RetryableOperation::Retrieval,
449                ));
450            }
451        }
452        let continuation_cursor = next_start_line.map(|next_start_line| {
453            ReadCursor {
454                generation,
455                target_start_line: target.target_start_line,
456                target_end_line,
457                next_start_line,
458                next_byte,
459                full_hash: snapshot.content_hash.clone(),
460                path_hash: read_path_hash(&request.path),
461            }
462            .encode()
463        });
464        let content_hash = hash(content);
465        let index_stale = indexed.content_hash != snapshot.content_hash;
466        let indexed_hash = Some(indexed.content_hash);
467        let not_modified = request.expected_hash.as_deref() == Some(content_hash.as_str());
468        let status = if truncated {
469            ReadStatus::Truncated
470        } else if not_modified {
471            ReadStatus::NotModified
472        } else {
473            ReadStatus::Content
474        };
475
476        Ok(MaterializedRead {
477            response: ReadResponse {
478                path: request.path.clone(),
479                status,
480                target_start_line: target.target_start_line,
481                target_end_line,
482                returned_start_line,
483                returned_end_line,
484                start_line: returned_start_line,
485                end_line: returned_end_line,
486                truncated,
487                next_start_line,
488                continuation_cursor,
489                not_modified,
490                content: (!not_modified).then(|| content.to_string()),
491                delta: None,
492                delta_receipt: None,
493                content_hash,
494                indexed_hash,
495                index_stale,
496                meta: self.meta(
497                    generation,
498                    if not_modified { 0 } else { emitted_tokens },
499                    None,
500                ),
501            },
502            baseline_source_tokens,
503            current_content: content.to_owned(),
504            current_tokens: emitted_tokens,
505        })
506    }

509fn prefer_full_if_delta_payload_not_smaller(
510    response: &mut ReadResponse,
511    current_content: &str,
512    current_tokens: usize,
513    tokenizer: crate::tokens::Tokenizer,
514) -> Result<()> {
515    if response.status != ReadStatus::Delta {
516        return Ok(());
517    }
518    let delta_tokens = finalized_serialized_read_tokens(response, tokenizer)?;
519    let mut full = response.clone();
520    full.status = ReadStatus::Content;
521    full.content = Some(current_content.to_owned());
522    full.delta = None;
523    full.meta.source_tokens = current_tokens;
524    full.meta.emitted_tokens = current_tokens;
525    if let Some(receipt) = full.delta_receipt.as_mut() {
526        receipt.outcome = ReadDeltaOutcome::Full;
527        receipt.delta_tokens = None;
528        receipt.avoided_tokens = 0;
529        receipt.fallback_reason = Some(ReadDeltaFallback::DeltaNotSmaller);
530    }
531    if delta_tokens >= finalized_serialized_read_tokens(&full, tokenizer)? {
532        *response = full;
533    }
534    Ok(())
535}
536
537fn finalized_serialized_read_tokens(
538    response: &ReadResponse,
539    tokenizer: crate::tokens::Tokenizer,
540) -> Result<usize> {
541    let mut finalized = response.clone();
542    finalized.meta.protocol_tokens = 0;
543    finalized.meta.path_and_metadata_tokens = 0;
544    finalized.meta.total_response_tokens = 0;
545    finalized.meta.payload_tokens = 0;
546    let accounting = crate::tokens::response_token_accounting(
547        &finalized,
548        finalized.meta.source_tokens,
549        &tokenizer,
550    )?;
551    finalized.meta.protocol_tokens = accounting.protocol_tokens;
552    finalized.meta.path_and_metadata_tokens = accounting.path_and_metadata_tokens;
553    finalized.meta.total_response_tokens = accounting.total_response_tokens;
554    finalized.meta.payload_tokens = accounting.total_response_tokens;
555    Ok(tokenizer.count(&serde_json::to_string(&finalized)?))
556}
