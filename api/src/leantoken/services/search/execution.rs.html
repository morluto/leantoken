46    pub(super) fn prepare_search(
47        &self,
48        request: &SearchRequest,
49        kind: &SearchKind,
50    ) -> Result<PreparedSearch> {
51        let regex = kind
52            .is_regex()
53            .then(|| compile_regex(request))
54            .transpose()?;
55        let literal_regex = if kind.is_regex() {
56            None
57        } else {
58            compile_literal_regex(&request.query, request.case_sensitive)?
59        };
60        let occurrence_literal_regex = kind
61            .is_exhaustive_text()
62            .then(|| compile_occurrence_literal_regex(&request.query, request.case_sensitive))
63            .transpose()?;
64        Ok(PreparedSearch {
65            regex,
66            literal_regex,
67            occurrence_literal_regex,
68            limit: self.result_limit(request.max_results)?,
69            token_limit: self.token_limit(request.max_tokens, self.config.default_read_tokens)?,
70            context_lines: self.context_line_limit(request.context_lines)?,
71        })
72    }

74    pub(super) fn search_snapshot(
75        &self,
76        snapshot: SearchSnapshot<'_>,
77        query: SearchQuery<'_>,
78        output_shape: SearchOutputShape,
79        execution: SearchExecutionOptions,
80        scan: SearchScan,
81    ) -> Result<SearchSnapshotResult> {
82        let SearchSnapshot {
83            session,
84            generation,
85            cancellation,
86        } = snapshot;
87        let SearchQuery { request, prepared } = query;
88        if let Some(PreparedQueryReceipt::Reuse {
89            receipt_id,
90            predicate,
91        }) = request.kind.query_receipt()
92        {
93            return self.reuse_query_receipt(
94                session,
95                generation,
96                receipt_id,
97                predicate,
98                cancellation,
99            );
100        }
101        let offset = parse_cursor(request.cursor.as_deref(), generation)?;
102        let mut hits = self.collect_structural_search_hits(
103            session,
104            request,
105            prepared,
106            prepared.limit,
107            prepared.context_lines,
108            cancellation,
109        )?;
110        let snapshot = SearchSnapshot {
111            session,
112            generation,
113            cancellation,
114        };
115        let query = SearchQuery { request, prepared };
116        let lexical = self.collect_lexical_search_hits(snapshot, query, scan)?;
117        hits.extend(lexical.hits);
118        let hits = order_search_hits(hits, request)?;
119        let page = OrderedSearchPage { hits, offset };
120        let (response, baseline_source_tokens, query_receipt) =
121            self.build_search_page(snapshot, query, output_shape, execution, page)?;
122        Ok(SearchSnapshotResult {
123            response,
124            baseline_source_tokens,
125            phases: lexical.phases,
126            primitive_keys: lexical.primitive_keys,
127            query_receipt,
128        })
129    }

131    pub(super) fn reuse_query_receipt(
132        &self,
133        session: &IndexReadSnapshot,
134        generation: u64,
135        receipt_id: &str,
136        requested_predicate: &ExactQueryPredicate,
137        cancellation: &CancellationToken,
138    ) -> Result<SearchSnapshotResult> {
139        check_cancelled(cancellation)?;
140        let stored = session.load_query_receipt(receipt_id)?;
141        let Some(scope_relation) = requested_predicate.scope_relation_to(&stored.predicate) else {
142            return Err(Error::QueryReceiptMismatch);
143        };
144        if scope_relation == QueryReceiptScopeRelation::Subset && stored.match_count != 0 {
145            return Err(Error::QueryReceiptMismatch);
146        }
147        let current_meta = session.meta()?;
148        if current_meta.config_hash != stored.config_hash {
149            return Err(Error::StaleQueryReceipt {
150                receipt_generation: stored.repository_generation,
151                repository_generation: generation,
152            });
153        }
154        let reused_across_generation = stored.repository_generation != generation;
155        if reused_across_generation {
156            let recorded_filter = PathFilter::new(
157                stored.predicate.include_paths(),
158                stored.predicate.exclude_paths(),
159            )?;
160            let current_partition = session.exact_query_partition(
161                |path| recorded_filter.allows(path),
162                || check_cancelled(cancellation),
163            )?;
164            if current_partition != stored.partition {
165                return Err(Error::StaleQueryReceipt {
166                    receipt_generation: stored.repository_generation,
167                    repository_generation: generation,
168                });
169            }
170        }
171        let requested_predicate_blake3 = requested_predicate.digest()?;
172        let outcome = QueryReceiptOutcome {
173            status: QueryReceiptStatus::AlreadyCovered,
174            receipt_id: Some(stored.receipt_id),
175            complete: true,
176            match_count: stored.match_count,
177            requested_predicate_blake3,
178            covered_predicate_blake3: stored.predicate_blake3,
179            result_blake3: Some(stored.result_blake3),
180            receipt_generation: stored.repository_generation,
181            reused_across_generation,
182            scope_relation,
183        };
184        Ok(SearchSnapshotResult {
185            response: SearchResponse {
186                hits: Vec::new(),
187                coverage: SearchCoverage {
188                    text_matches: SearchCoverageCount {
189                        total: stored.match_count,
190                        returned: 0,
191                        truncated: stored.match_count,
192                    },
193                    ..SearchCoverage::default()
194                },
195                occurrences_returned: 0,
196                occurrences_total: Some(stored.match_count),
197                meta: self.meta(generation, 0, None),
198            },
199            baseline_source_tokens: None,
200            phases: SearchPhaseCounters::default(),
201            primitive_keys: Vec::new(),
202            query_receipt: QueryReceiptExecution::Outcome(outcome),
203        })
204    }

206    pub(super) fn collect_structural_search_hits(
207        &self,
208        session: &IndexReadSnapshot,
209        request: &SearchInput,
210        prepared: &PreparedSearch,
211        limit: usize,
212        context_lines: usize,
213        cancellation: &CancellationToken,
214    ) -> Result<Vec<CandidateSearchHit>> {
215        let mut hits = Vec::new();
216        if matches!(
217            request.kind.mode(),
218            SearchMode::Auto | SearchMode::Identifier | SearchMode::Symbol
219        ) {
220            let max_candidates = limit.saturating_mul(4);
221            let unicode_full_scan = !request.case_sensitive
222                && crate::symbol_identity::case_fold_literal_variants(&request.query).is_none();
223            let symbol_hits = if unicode_full_scan {
224                filter_materialized_hits(
225                    request,
226                    max_candidates,
227                    cancellation,
228                    session.search_symbols_page(&request.query, false, MAX_FILTER_SCAN_ROWS, 0)?,
229                    |hit: &SymbolHit| &hit.path,
230                )?
231            } else {
232                collect_filtered_hits(
233                    request,
234                    max_candidates,
235                    cancellation,
236                    |offset, page_limit| {
237                        session.search_symbols_page(
238                            &request.query,
239                            request.case_sensitive,
240                            page_limit,
241                            offset,
242                        )
243                    },
244                    |hit: &SymbolHit| &hit.path,
245                )?
246            };
247            let excerpt_requests = symbol_hits
248                .iter()
249                .map(|hit| StoredExcerptRequest {
250                    file_id: hit.symbol.file_id,
251                    desired_start_line: hit.symbol.start_line.saturating_sub(context_lines).max(1),
252                    desired_end_line: hit.symbol.end_line.saturating_add(context_lines),
253                    required_start_line: hit.symbol.start_line,
254                    required_end_line: hit.symbol.start_line,
255                    max_lines: 30,
256                })
257                .collect::<Vec<_>>();
258            for (hit, excerpt) in symbol_hits
259                .into_iter()
260                .zip(self.stored_excerpts(session, &excerpt_requests)?)
261            {
262                if let Some(excerpt) = excerpt {
263                    let definition = DefinitionIdentity {
264                        path: hit.path.clone(),
265                        start_line: hit.symbol.start_line,
266                        end_line: hit.symbol.end_line,
267                    };
268                    hits.push(CandidateSearchHit {
269                        hit: self.symbol_search_hit(
270                            hit,
271                            &request.query,
272                            request.case_sensitive,
273                            prepared.literal_regex.as_ref(),
274                            excerpt,
275                        ),
276                        definition: Some(definition),
277                    });
278                }
279            }
280        }
281        if matches!(
282            request.kind.mode(),
283            SearchMode::Auto | SearchMode::Identifier | SearchMode::Reference
284        ) {
285            let max_candidates = limit.saturating_mul(4);
286            let unicode_full_scan = !request.case_sensitive
287                && crate::symbol_identity::case_fold_literal_variants(&request.query).is_none();
288            let reference_hits = if unicode_full_scan {
289                filter_materialized_hits(
290                    request,
291                    max_candidates,
292                    cancellation,
293                    session.search_references_page(
294                        &request.query,
295                        false,
296                        MAX_FILTER_SCAN_ROWS,
297                        0,
298                    )?,
299                    |hit: &ReferenceHit| &hit.path,
300                )?
301            } else {
302                collect_filtered_hits(
303                    request,
304                    max_candidates,
305                    cancellation,
306                    |offset, page_limit| {
307                        session.search_references_page(
308                            &request.query,
309                            request.case_sensitive,
310                            page_limit,
311                            offset,
312                        )
313                    },
314                    |hit: &ReferenceHit| &hit.path,
315                )?
316            };
317            let excerpt_requests = reference_hits
318                .iter()
319                .map(|hit| StoredExcerptRequest {
320                    file_id: hit.reference.file_id,
321                    desired_start_line: hit
322                        .reference
323                        .start_line
324                        .saturating_sub(context_lines)
325                        .max(1),
326                    desired_end_line: hit.reference.end_line.saturating_add(context_lines),
327                    required_start_line: hit.reference.start_line,
328                    required_end_line: hit.reference.end_line,
329                    max_lines: 12,
330                })
331                .collect::<Vec<_>>();
332            for (hit, excerpt) in reference_hits
333                .into_iter()
334                .zip(self.stored_excerpts(session, &excerpt_requests)?)
335            {
336                if let Some(excerpt) = excerpt {
337                    hits.push(CandidateSearchHit {
338                        hit: self.reference_search_hit(
339                            hit,
340                            &request.query,
341                            request.case_sensitive,
342                            prepared.literal_regex.as_ref(),
343                            excerpt,
344                        ),
345                        definition: None,
346                    });
347                }
348            }
349        }
350        Ok(hits)
351    }

353    pub(super) fn collect_lexical_search_hits(
354        &self,
355        snapshot: SearchSnapshot<'_>,
356        query: SearchQuery<'_>,
357        scan: SearchScan,
358    ) -> Result<LexicalSearchBatch> {
359        let SearchSnapshot {
360            session,
361            generation,
362            cancellation,
363        } = snapshot;
364        let SearchQuery { request, prepared } = query;
365        let SearchScan {
366            regex_planning,
367            diagnostics,
368        } = scan;
369        let mut phases = SearchPhaseCounters::default();
370        let mut primitive_keys = Vec::new();
371        let lexical = match request.kind.mode() {
372            SearchMode::Regex => {
373                let scan = self.regex_hits(
374                    session,
375                    request,
376                    prepared
377                        .regex
378                        .as_ref()
379                        .expect("regex mode compiles a pattern"),
380                    (!request.kind.is_exhaustive()).then_some(prepared.limit.saturating_mul(20)),
381                    cancellation,
382                    regex_planning,
383                )?;
384                phases = scan.phases;
385                let primitive_kind = match phases.regex_planning.strategy() {
386                    RegexCandidateStrategy::Trigram => "regex_trigram_candidates",
387                    RegexCandidateStrategy::FullScan => "regex_full_scan",
388                };
389                if diagnostics == SearchDiagnostics::Collect {
390                    primitive_keys.push(retrieval_primitive_key(
391                        generation,
392                        primitive_kind,
393                        &format!(
394                            "case_sensitive:{}:include:{:?}:exclude:{:?}:query:{}",
395                            request.case_sensitive,
396                            request.include_paths,
397                            request.exclude_paths,
398                            request.query
399                        ),
400                    ));
401                }
402                scan.hits
403            }
404            SearchMode::Text if request.kind.is_exhaustive() => {
405                let scan = self.regex_hits(
406                    session,
407                    request,
408                    prepared
409                        .occurrence_literal_regex
410                        .as_ref()
411                        .expect("exhaustive text mode compiles a literal pattern"),
412                    None,
413                    cancellation,
414                    regex_planning,
415                )?;
416                phases = scan.phases;
417                scan.hits
418            }
419            SearchMode::Text | SearchMode::Auto | SearchMode::Identifier
420                if request.query.chars().count() < 3 =>
421            {
422                // Word FTS cannot match substrings shorter than three
423                // characters; reuse the literal regex lexical path so
424                // Identifier stays aligned with Text/Auto short queries.
425                let short_literal_regex =
426                    compile_occurrence_literal_regex(&request.query, request.case_sensitive)?;
427                let scan = self.regex_hits(
428                    session,
429                    request,
430                    &short_literal_regex,
431                    Some(prepared.limit.saturating_mul(20)),
432                    cancellation,
433                    RegexPlanning::Disabled,
434                )?;
435                phases = scan.phases;
436                scan.hits
437            }
438            SearchMode::Text | SearchMode::Auto | SearchMode::Identifier
439                if !request.case_sensitive
440                    && crate::symbol_identity::case_fold_literal_variants(&request.query)
441                        .is_none() =>
442            {
443                let scan = self.regex_hits(
444                    session,
445                    request,
446                    prepared
447                        .literal_regex
448                        .as_ref()
449                        .expect("case-insensitive literal search compiles a matcher"),
450                    Some(prepared.limit.saturating_mul(20)),
451                    cancellation,
452                    RegexPlanning::Disabled,
453                )?;
454                phases = scan.phases;
455                scan.hits
456            }
457            SearchMode::Text | SearchMode::Auto | SearchMode::Identifier => {
458                let folded = (!request.case_sensitive)
459                    .then(|| crate::symbol_identity::case_fold_literal_variants(&request.query))
460                    .flatten()
461                    .filter(|variants| variants.expanded);
462                let indexed_query = folded
463                    .as_ref()
464                    .map(crate::symbol_identity::case_fold_fts_query)
465                    .unwrap_or_else(|| fts_quote(&request.query));
466                let fetch_page = |offset, page_limit| {
467                    if matches!(request.kind.mode(), SearchMode::Identifier) {
468                        session.search_word_page(&indexed_query, page_limit, offset)
469                    } else if folded.is_some() {
470                        session.search_trigram_expression_page(&indexed_query, page_limit, offset)
471                    } else {
472                        session.search_trigram_page(&request.query, page_limit, offset)
473                    }
474                };
475                collect_filtered_hits(
476                    request,
477                    prepared.limit.saturating_mul(8),
478                    cancellation,
479                    fetch_page,
480                    |hit: &ChunkHit| &hit.path,
481                )?
482            }
483            SearchMode::Symbol | SearchMode::Reference => Vec::new(),
484        };
485        let hits =
486            self.hydrate_lexical_search_hits(session, request, prepared, cancellation, lexical)?;
487        Ok(LexicalSearchBatch {
488            hits,
489            phases,
490            primitive_keys,
491        })
492    }

494    pub(super) fn hydrate_lexical_search_hits(
495        &self,
496        session: &IndexReadSnapshot,
497        request: &SearchInput,
498        prepared: &PreparedSearch,
499        cancellation: &CancellationToken,
500        lexical: Vec<ChunkHit>,
501    ) -> Result<Vec<CandidateSearchHit>> {
502        let mut lexical_hits = Vec::new();
503        for hit in lexical {
504            check_cancelled(cancellation)?;
505            let chunk_hits = if request.kind.is_exhaustive() {
506                chunk_search_hits(
507                    &hit,
508                    &request.query,
509                    request.case_sensitive,
510                    prepared.context_lines,
511                    prepared
512                        .regex
513                        .as_ref()
514                        .or(prepared.occurrence_literal_regex.as_ref())
515                        .or(prepared.literal_regex.as_ref()),
516                    request.kind.lexical_match_kind(),
517                    OccurrenceMaterializationLimit {
518                        existing_hits: lexical_hits.len(),
519                        max_hits: MAX_EXHAUSTIVE_OCCURRENCES,
520                    },
521                )?
522            } else {
523                chunk_search_hit(
524                    &hit,
525                    &request.query,
526                    request.case_sensitive,
527                    prepared.context_lines,
528                    prepared.regex.as_ref().or(prepared.literal_regex.as_ref()),
529                    request.kind.lexical_match_kind(),
530                )?
531                .into_iter()
532                .collect()
533            };
534            for search_hit in chunk_hits {
535                if request.kind.is_exhaustive() && lexical_hits.len() == MAX_EXHAUSTIVE_OCCURRENCES
536                {
537                    return Err(Error::RetrievalLimitExceeded {
538                        kind: RetrievalLimitKind::ExhaustiveOccurrences,
539                        observed: lexical_hits.len().saturating_add(1),
540                        limit: MAX_EXHAUSTIVE_OCCURRENCES,
541                    });
542                }
543                let matched_line = search_hit
544                    .occurrence
545                    .as_ref()
546                    .map(|occurrence| occurrence.start_line)
547                    .or_else(|| {
548                        matching_line(
549                            &hit,
550                            &request.query,
551                            request.case_sensitive,
552                            prepared.regex.as_ref().or(prepared.literal_regex.as_ref()),
553                        )
554                    })
555                    .unwrap_or(search_hit.start_line);
556                lexical_hits.push((hit.file_id, search_hit, matched_line));
557            }
558        }
559        let lexical_locations = lexical_hits
560            .iter()
561            .map(|(file_id, _, matched_line)| (*file_id, *matched_line))
562            .collect::<Vec<_>>();
563        let mut enclosing = Vec::with_capacity(lexical_locations.len());
564        for locations in lexical_locations.chunks(512) {
565            check_cancelled(cancellation)?;
566            enclosing.extend(session.find_enclosing_symbols_batch(locations)?);
567        }
568        Ok(lexical_hits
569            .into_iter()
570            .zip(enclosing)
571            .map(|((_, mut hit, _), symbol)| {
572                let definition = symbol.map(|symbol| {
573                    hit.enclosing_symbol = Some(symbol.name);
574                    DefinitionIdentity {
575                        path: hit.path.clone(),
576                        start_line: symbol.start_line,
577                        end_line: symbol.end_line,
578                    }
579                });
580                CandidateSearchHit { hit, definition }
581            })
582            .collect())
583    }

585    pub(super) fn build_search_page(
586        &self,
587        snapshot: SearchSnapshot<'_>,
588        query: SearchQuery<'_>,
589        output_shape: SearchOutputShape,
590        execution: SearchExecutionOptions,
591        page: OrderedSearchPage,
592    ) -> Result<(SearchResponse, Option<usize>, QueryReceiptExecution)> {
593        let SearchSnapshot {
594            session,
595            generation,
596            cancellation,
597        } = snapshot;
598        let SearchQuery { request, prepared } = query;
599        let OrderedSearchPage { hits, offset } = page;
600        let total_candidates = hits.len();
601        let (mut selected, consumed, _) = select_search_page(
602            &hits,
603            offset,
604            prepared.limit,
605            prepared.token_limit,
606            output_shape,
607            &self.config.tokenizer,
608            cancellation,
609        )?;
610        let has_more = offset.saturating_add(consumed) < total_candidates;
611        self.ensure_search_page_fits(
612            &mut selected,
613            SearchResponseShape {
614                all: &hits,
615                request,
616                generation,
617                total_candidates,
618                offset,
619                consumed,
620                has_more,
621            },
622            execution.response_options,
623        )?;
624        let receipt_candidates = selected
625            .iter()
626            .map(|candidate| {
627                ReceiptEvidence::new(
628                    candidate.hit.path.clone(),
629                    candidate.hit.start_line,
630                    candidate.hit.end_line,
631                    candidate.hit.content_hash.clone(),
632                    Some(&candidate.hit.excerpt),
633                )
634            })
635            .collect::<Vec<_>>();
636        let receipt = self.evaluate_receipt(
637            matches!(
638                output_shape,
639                SearchOutputShape::Full | SearchOutputShape::Compact
640            )
641            .then_some(request.receipt_id.as_deref())
642            .flatten(),
643            generation,
644            &receipt_candidates,
645        )?;
646        selected = selected
647            .into_iter()
648            .zip(&receipt.decisions)
649            .filter_map(|(candidate, decision)| {
650                matches!(
651                    decision,
652                    ReceiptDecision::Return | ReceiptDecision::ReturnNearDuplicate
653                )
654                .then_some(candidate)
655            })
656            .collect();
657        let emitted_tokens =
658            selected_search_source_tokens(&selected, output_shape, &self.config.tokenizer);
659        let paths = selected
660            .iter()
661            .map(|candidate| candidate.hit.path.clone())
662            .collect::<HashSet<_>>()
663            .into_iter()
664            .collect::<Vec<_>>();
665        let occurrences_returned = selected.len();
666        let coverage = search_coverage(&hits, &selected, request.kind.mode());
667        let selected = selected
668            .into_iter()
669            .map(|candidate| candidate.hit)
670            .collect();
671        let baseline_source_tokens =
672            session.whole_file_source_tokens(&paths, self.config.tokenizer.name())?;
673        let mut response = SearchResponse {
674            hits: selected,
675            coverage,
676            occurrences_returned,
677            occurrences_total: request.kind.is_exhaustive().then_some(total_candidates),
678            meta: self.meta(
679                generation,
680                emitted_tokens,
681                has_more.then(|| make_cursor(generation, offset + consumed)),
682            ),
683        };
684        receipt.apply_meta(&mut response.meta);
685        let query_receipt =
686            if let Some(PreparedQueryReceipt::Record(predicate)) = request.kind.query_receipt() {
687                let predicate_blake3 = predicate.digest()?;
688                if !has_more && occurrences_returned == total_candidates {
689                    let path_filter =
690                        PathFilter::new(predicate.include_paths(), predicate.exclude_paths())?;
691                    let partition = session.exact_query_partition(
692                        |path| path_filter.allows(path),
693                        || check_cancelled(cancellation),
694                    )?;
695                    let occurrences = hits
696                        .iter()
697                        .map(|candidate| {
698                            candidate
699                                .hit
700                                .occurrence
701                                .as_ref()
702                                .map(|occurrence| (candidate.hit.path.as_str(), occurrence))
703                                .ok_or_else(|| {
704                                    Error::OperationFailure(
705                                        "exhaustive query result omitted exact coordinates".into(),
706                                    )
707                                })
708                        })
709                        .collect::<Result<Vec<_>>>()?;
710                    QueryReceiptExecution::Pending(QueryReceiptRecord {
711                        repository_generation: generation,
712                        config_hash: session.meta()?.config_hash,
713                        predicate: predicate.clone(),
714                        predicate_blake3,
715                        partition,
716                        match_count: total_candidates,
717                        result_blake3: exhaustive_result_digest(occurrences),
718                    })
719                } else {
720                    QueryReceiptExecution::Outcome(QueryReceiptOutcome {
721                        status: QueryReceiptStatus::NotRecordedIncompleteResponse,
722                        receipt_id: None,
723                        complete: false,
724                        match_count: total_candidates,
725                        requested_predicate_blake3: predicate_blake3.clone(),
726                        covered_predicate_blake3: predicate_blake3,
727                        result_blake3: None,
728                        receipt_generation: generation,
729                        reused_across_generation: false,
730                        scope_relation: QueryReceiptScopeRelation::Exact,
731                    })
732                }
733            } else {
734                QueryReceiptExecution::None
735            };
736        Ok((response, baseline_source_tokens, query_receipt))
737    }
738}

740pub(super) fn order_search_hits(
741    mut hits: Vec<CandidateSearchHit>,
742    request: &SearchInput,
743) -> Result<Vec<CandidateSearchHit>> {
744    apply_focus(&mut hits, &request.focus_paths)?;
745    hits.sort_by(|left, right| {
746        right
747            .hit
748            .score
749            .total_cmp(&left.hit.score)
750            .then_with(|| left.hit.path.cmp(&right.hit.path))
751            .then_with(|| left.hit.start_line.cmp(&right.hit.start_line))
752            .then_with(|| {
753                left.hit
754                    .occurrence
755                    .as_ref()
756                    .map(|occurrence| occurrence.start_byte)
757                    .cmp(
758                        &right
759                            .hit
760                            .occurrence
761                            .as_ref()
762                            .map(|occurrence| occurrence.start_byte),
763                    )
764            })
765    });
766    let preference = request.kind.definition_preference();
767    hits = deduplicate_exact_hits(hits, preference);
768    if matches!(
769        request.kind.mode(),
770        SearchMode::Auto | SearchMode::Identifier
771    ) {
772        hits = deduplicate_definition_channels(hits, preference);
773    }
774    normalize_search_scores(&mut hits);
775    Ok(hits)
776}
