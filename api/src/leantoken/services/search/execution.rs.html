32    pub(super) fn parse_search_request(
33        &self,
34        request: SearchRequest,
35        output_shape: SearchOutputShape,
36    ) -> Result<ParsedSearchRequest> {
37        validate_search_input(&request)?;
38        let cursor = ContinuationCursor::parse_optional(request.cursor.as_deref())?;
39        let patterns = SearchPatterns::parse(&request)?;
40        let kind = parse_search_kind(&request, output_shape)?;
41        let prepared = self.prepare_search(&request, &kind)?;
42        let request = SearchInput::from_request(request, kind, patterns, cursor);
43        let stream_id = search_stream_id(self, &request, &prepared, output_shape);
44        Ok(ParsedSearchRequest {
45            request,
46            prepared,
47            output_shape,
48            stream_id,
49        })
50    }

52    pub(super) fn prepare_search(
53        &self,
54        request: &SearchRequest,
55        kind: &SearchKind,
56    ) -> Result<PreparedSearch> {
57        let regex = kind
58            .is_regex()
59            .then(|| compile_regex(request))
60            .transpose()?;
61        let literal_regex = if kind.is_regex() {
62            None
63        } else {
64            compile_literal_regex(&request.query, request.case_sensitive)?
65        };
66        let occurrence_literal_regex = kind
67            .is_exhaustive_text()
68            .then(|| compile_occurrence_literal_regex(&request.query, request.case_sensitive))
69            .transpose()?;
70        Ok(PreparedSearch {
71            regex,
72            literal_regex,
73            occurrence_literal_regex,
74            limit: self.result_limit(request.max_results)?,
75            token_limit: self.token_limit(request.max_tokens, self.config.default_read_tokens)?,
76            context_lines: self.context_line_limit(request.context_lines)?,
77        })
78    }

80    pub(super) fn search_snapshot(
81        &self,
82        snapshot: SearchSnapshot<'_>,
83        query: SearchQuery<'_>,
84        output_shape: SearchOutputShape,
85        execution: SearchExecutionOptions,
86        scan: SearchScan,
87    ) -> Result<SearchSnapshotResult> {
88        let SearchSnapshot {
89            session,
90            generation,
91            cancellation,
92        } = snapshot;
93        let SearchQuery {
94            request,
95            prepared,
96            stream_id,
97        } = query;
98        if let Some(PreparedQueryReceipt::Reuse {
99            receipt_id,
100            predicate,
101        }) = request.kind.query_receipt()
102        {
103            return self.reuse_query_receipt(
104                session,
105                generation,
106                receipt_id,
107                predicate,
108                cancellation,
109            );
110        }
111        let offset = request
112            .cursor
113            .map(|cursor| cursor.position_for(CursorKind::Search, generation, stream_id))
114            .transpose()?
115            .unwrap_or(0);
116        let mut hits = self.collect_structural_search_hits(
117            session,
118            request,
119            prepared,
120            prepared.limit,
121            prepared.context_lines,
122            cancellation,
123        )?;
124        let snapshot = SearchSnapshot {
125            session,
126            generation,
127            cancellation,
128        };
129        let query = SearchQuery {
130            request,
131            prepared,
132            stream_id,
133        };
134        let lexical = self.collect_lexical_search_hits(snapshot, query, scan)?;
135        hits.extend(lexical.hits);
136        let hits = order_search_hits(hits, request)?;
137        let page = OrderedSearchPage { hits, offset };
138        let (response, baseline_source_tokens, query_receipt) =
139            self.build_search_page(snapshot, query, output_shape, execution, page)?;
140        Ok(SearchSnapshotResult {
141            response,
142            baseline_source_tokens,
143            phases: lexical.phases,
144            primitive_keys: lexical.primitive_keys,
145            query_receipt,
146        })
147    }

149    pub(super) fn reuse_query_receipt(
150        &self,
151        session: &IndexReadSnapshot,
152        generation: u64,
153        receipt_id: &str,
154        requested_predicate: &ExactQueryPredicate,
155        cancellation: &CancellationToken,
156    ) -> Result<SearchSnapshotResult> {
157        check_cancelled(cancellation)?;
158        let stored = session.load_query_receipt(receipt_id)?;
159        let Some(scope_relation) = requested_predicate.scope_relation_to(&stored.predicate) else {
160            return Err(Error::QueryReceiptMismatch);
161        };
162        if scope_relation == QueryReceiptScopeRelation::Subset && stored.match_count != 0 {
163            return Err(Error::QueryReceiptMismatch);
164        }
165        let current_meta = session.meta()?;
166        if current_meta.config_hash != stored.config_hash {
167            return Err(Error::StaleQueryReceipt {
168                receipt_generation: stored.repository_generation,
169                repository_generation: generation,
170            });
171        }
172        let reused_across_generation = stored.repository_generation != generation;
173        if reused_across_generation {
174            let recorded_filter = PathFilter::new(
175                stored.predicate.include_paths(),
176                stored.predicate.exclude_paths(),
177            )?;
178            let current_partition = session.exact_query_partition(
179                |path| recorded_filter.allows(path),
180                || check_cancelled(cancellation),
181            )?;
182            if current_partition != stored.partition {
183                return Err(Error::StaleQueryReceipt {
184                    receipt_generation: stored.repository_generation,
185                    repository_generation: generation,
186                });
187            }
188        }
189        // Best-effort touch: a concurrent quota eviction may have removed
190        // this receipt after the pinned snapshot validated it. The reuse
191        // was already validated above, so losing the touch race must not
192        // turn a valid already_covered into an error.
193        if let Err(Error::UnknownQueryReceipt(_)) = self.storage.touch_query_receipt(receipt_id) {
194            // Receipt was evicted concurrently; reuse result is still valid.
195        }
196        let requested_predicate_blake3 = requested_predicate.digest()?;
197        let outcome = QueryReceiptOutcome {
198            status: QueryReceiptStatus::AlreadyCovered,
199            receipt_id: Some(stored.receipt_id),
200            complete: true,
201            match_count: stored.match_count,
202            requested_predicate_blake3,
203            covered_predicate_blake3: stored.predicate_blake3,
204            result_blake3: Some(stored.result_blake3),
205            receipt_generation: stored.repository_generation,
206            reused_across_generation,
207            scope_relation,
208        };
209        Ok(SearchSnapshotResult {
210            response: SearchResponse {
211                hits: Vec::new(),
212                coverage: SearchCoverage {
213                    text_matches: SearchCoverageCount {
214                        total: stored.match_count,
215                        returned: 0,
216                        truncated: stored.match_count,
217                    },
218                    ..SearchCoverage::default()
219                },
220                occurrences_returned: 0,
221                occurrences_total: Some(stored.match_count),
222                meta: self.meta(generation, 0, None),
223            },
224            baseline_source_tokens: None,
225            phases: SearchPhaseCounters::default(),
226            primitive_keys: Vec::new(),
227            query_receipt: QueryReceiptExecution::Outcome(outcome),
228        })
229    }

231    pub(super) fn collect_structural_search_hits(
232        &self,
233        session: &IndexReadSnapshot,
234        request: &SearchInput,
235        prepared: &PreparedSearch,
236        limit: usize,
237        context_lines: usize,
238        cancellation: &CancellationToken,
239    ) -> Result<Vec<CandidateSearchHit>> {
240        let mut hits = Vec::new();
241        if matches!(
242            request.kind.mode(),
243            SearchMode::Auto | SearchMode::Identifier | SearchMode::Symbol
244        ) {
245            let max_candidates = limit.saturating_mul(4);
246            let unicode_full_scan = !request.case_sensitive
247                && crate::symbol_identity::case_fold_literal_variants(&request.query).is_none();
248            let symbol_hits = if unicode_full_scan {
249                filter_materialized_hits(
250                    request,
251                    max_candidates,
252                    cancellation,
253                    session.search_symbols_page(&request.query, false, MAX_FILTER_SCAN_ROWS, 0)?,
254                    |hit: &SymbolHit| &hit.path,
255                )?
256            } else {
257                collect_filtered_hits(
258                    request,
259                    max_candidates,
260                    cancellation,
261                    |offset, page_limit| {
262                        session.search_symbols_page(
263                            &request.query,
264                            request.case_sensitive,
265                            page_limit,
266                            offset,
267                        )
268                    },
269                    |hit: &SymbolHit| &hit.path,
270                )?
271            };
272            let excerpt_requests = symbol_hits
273                .iter()
274                .map(|hit| StoredExcerptRequest {
275                    file_id: hit.symbol.file_id,
276                    desired_start_line: hit.symbol.start_line.saturating_sub(context_lines).max(1),
277                    desired_end_line: hit.symbol.end_line.saturating_add(context_lines),
278                    required_start_line: hit.symbol.start_line,
279                    required_end_line: hit.symbol.start_line,
280                    max_lines: 30,
281                })
282                .collect::<Vec<_>>();
283            for (hit, excerpt) in symbol_hits
284                .into_iter()
285                .zip(self.stored_excerpts(session, &excerpt_requests)?)
286            {
287                if let Some(excerpt) = excerpt {
288                    let definition = DefinitionIdentity {
289                        path: hit.path.clone(),
290                        start_line: hit.symbol.start_line,
291                        end_line: hit.symbol.end_line,
292                    };
293                    hits.push(CandidateSearchHit {
294                        hit: self.symbol_search_hit(
295                            hit,
296                            &request.query,
297                            request.case_sensitive,
298                            prepared.literal_regex.as_ref(),
299                            excerpt,
300                        ),
301                        definition: Some(definition),
302                    });
303                }
304            }
305        }
306        if matches!(
307            request.kind.mode(),
308            SearchMode::Auto | SearchMode::Identifier | SearchMode::Reference
309        ) {
310            let max_candidates = limit.saturating_mul(4);
311            let unicode_full_scan = !request.case_sensitive
312                && crate::symbol_identity::case_fold_literal_variants(&request.query).is_none();
313            let reference_hits = if unicode_full_scan {
314                filter_materialized_hits(
315                    request,
316                    max_candidates,
317                    cancellation,
318                    session.search_references_page(
319                        &request.query,
320                        false,
321                        MAX_FILTER_SCAN_ROWS,
322                        0,
323                    )?,
324                    |hit: &ReferenceHit| &hit.path,
325                )?
326            } else {
327                collect_filtered_hits(
328                    request,
329                    max_candidates,
330                    cancellation,
331                    |offset, page_limit| {
332                        session.search_references_page(
333                            &request.query,
334                            request.case_sensitive,
335                            page_limit,
336                            offset,
337                        )
338                    },
339                    |hit: &ReferenceHit| &hit.path,
340                )?
341            };
342            let excerpt_requests = reference_hits
343                .iter()
344                .map(|hit| StoredExcerptRequest {
345                    file_id: hit.reference.file_id,
346                    desired_start_line: hit
347                        .reference
348                        .start_line
349                        .saturating_sub(context_lines)
350                        .max(1),
351                    desired_end_line: hit.reference.end_line.saturating_add(context_lines),
352                    required_start_line: hit.reference.start_line,
353                    required_end_line: hit.reference.end_line,
354                    max_lines: 12,
355                })
356                .collect::<Vec<_>>();
357            for (hit, excerpt) in reference_hits
358                .into_iter()
359                .zip(self.stored_excerpts(session, &excerpt_requests)?)
360            {
361                if let Some(excerpt) = excerpt {
362                    hits.push(CandidateSearchHit {
363                        hit: self.reference_search_hit(
364                            hit,
365                            &request.query,
366                            request.case_sensitive,
367                            prepared.literal_regex.as_ref(),
368                            excerpt,
369                        ),
370                        definition: None,
371                    });
372                }
373            }
374        }
375        Ok(hits)
376    }

378    pub(super) fn collect_lexical_search_hits(
379        &self,
380        snapshot: SearchSnapshot<'_>,
381        query: SearchQuery<'_>,
382        scan: SearchScan,
383    ) -> Result<LexicalSearchBatch> {
384        let SearchSnapshot {
385            session,
386            generation,
387            cancellation,
388        } = snapshot;
389        let SearchQuery {
390            request,
391            prepared,
392            stream_id: _,
393        } = query;
394        let SearchScan {
395            regex_planning,
396            diagnostics,
397        } = scan;
398        let mut phases = SearchPhaseCounters::default();
399        let mut primitive_keys = Vec::new();
400        let lexical = match request.kind.mode() {
401            SearchMode::Regex => {
402                let scan = self.regex_hits(
403                    session,
404                    request,
405                    prepared
406                        .regex
407                        .as_ref()
408                        .expect("regex mode compiles a pattern"),
409                    (!request.kind.is_exhaustive()).then_some(prepared.limit.saturating_mul(20)),
410                    cancellation,
411                    regex_planning,
412                )?;
413                phases = scan.phases;
414                let primitive_kind = match phases.regex_planning.strategy() {
415                    RegexCandidateStrategy::Trigram => "regex_trigram_candidates",
416                    RegexCandidateStrategy::FullScan => "regex_full_scan",
417                };
418                if diagnostics == SearchDiagnostics::Collect {
419                    primitive_keys.push(retrieval_primitive_key(
420                        generation,
421                        primitive_kind,
422                        &format!(
423                            "case_sensitive:{}:include:{:?}:exclude:{:?}:query:{}",
424                            request.case_sensitive,
425                            request.include_paths,
426                            request.exclude_paths,
427                            request.query
428                        ),
429                    ));
430                }
431                scan.hits
432            }
433            SearchMode::Text if request.kind.is_exhaustive() => {
434                let scan = self.regex_hits(
435                    session,
436                    request,
437                    prepared
438                        .occurrence_literal_regex
439                        .as_ref()
440                        .expect("exhaustive text mode compiles a literal pattern"),
441                    None,
442                    cancellation,
443                    regex_planning,
444                )?;
445                phases = scan.phases;
446                scan.hits
447            }
448            SearchMode::Text | SearchMode::Auto | SearchMode::Identifier
449                if request.query.chars().count() < 3 =>
450            {
451                // Word FTS cannot match substrings shorter than three
452                // characters; reuse the literal regex lexical path so
453                // Identifier stays aligned with Text/Auto short queries.
454                let short_literal_regex =
455                    compile_occurrence_literal_regex(&request.query, request.case_sensitive)?;
456                let scan = self.regex_hits(
457                    session,
458                    request,
459                    &short_literal_regex,
460                    Some(prepared.limit.saturating_mul(20)),
461                    cancellation,
462                    RegexPlanning::Disabled,
463                )?;
464                phases = scan.phases;
465                scan.hits
466            }
467            SearchMode::Text | SearchMode::Auto | SearchMode::Identifier
468                if !request.case_sensitive
469                    && crate::symbol_identity::case_fold_literal_variants(&request.query)
470                        .is_none() =>
471            {
472                let scan = self.regex_hits(
473                    session,
474                    request,
475                    prepared
476                        .literal_regex
477                        .as_ref()
478                        .expect("case-insensitive literal search compiles a matcher"),
479                    Some(prepared.limit.saturating_mul(20)),
480                    cancellation,
481                    RegexPlanning::Disabled,
482                )?;
483                phases = scan.phases;
484                scan.hits
485            }
486            SearchMode::Text | SearchMode::Auto | SearchMode::Identifier => {
487                let folded = (!request.case_sensitive)
488                    .then(|| crate::symbol_identity::case_fold_literal_variants(&request.query))
489                    .flatten()
490                    .filter(|variants| variants.expanded);
491                let indexed_query = folded
492                    .as_ref()
493                    .map(crate::symbol_identity::case_fold_fts_query)
494                    .unwrap_or_else(|| fts_quote(&request.query));
495                let fetch_page = |offset, page_limit| {
496                    if matches!(request.kind.mode(), SearchMode::Identifier) {
497                        session.search_word_page(&indexed_query, page_limit, offset)
498                    } else if folded.is_some() {
499                        session.search_trigram_expression_page(&indexed_query, page_limit, offset)
500                    } else {
501                        session.search_trigram_page(&request.query, page_limit, offset)
502                    }
503                };
504                collect_filtered_hits(
505                    request,
506                    prepared.limit.saturating_mul(8),
507                    cancellation,
508                    fetch_page,
509                    |hit: &ChunkHit| &hit.path,
510                )?
511            }
512            SearchMode::Symbol | SearchMode::Reference => Vec::new(),
513        };
514        let hits =
515            self.hydrate_lexical_search_hits(session, request, prepared, cancellation, lexical)?;
516        Ok(LexicalSearchBatch {
517            hits,
518            phases,
519            primitive_keys,
520        })
521    }

523    pub(super) fn hydrate_lexical_search_hits(
524        &self,
525        session: &IndexReadSnapshot,
526        request: &SearchInput,
527        prepared: &PreparedSearch,
528        cancellation: &CancellationToken,
529        lexical: Vec<ChunkHit>,
530    ) -> Result<Vec<CandidateSearchHit>> {
531        let mut lexical_hits = Vec::new();
532        for hit in lexical {
533            check_cancelled(cancellation)?;
534            let chunk_hits = if request.kind.is_exhaustive() {
535                chunk_search_hits(
536                    &hit,
537                    &request.query,
538                    request.case_sensitive,
539                    prepared.context_lines,
540                    prepared
541                        .regex
542                        .as_ref()
543                        .or(prepared.occurrence_literal_regex.as_ref())
544                        .or(prepared.literal_regex.as_ref()),
545                    request.kind.lexical_match_kind(),
546                    OccurrenceMaterializationLimit {
547                        existing_hits: lexical_hits.len(),
548                        max_hits: MAX_EXHAUSTIVE_OCCURRENCES,
549                    },
550                )?
551            } else {
552                chunk_search_hit(
553                    &hit,
554                    &request.query,
555                    request.case_sensitive,
556                    prepared.context_lines,
557                    prepared.regex.as_ref().or(prepared.literal_regex.as_ref()),
558                    request.kind.lexical_match_kind(),
559                )?
560                .into_iter()
561                .collect()
562            };
563            for search_hit in chunk_hits {
564                if request.kind.is_exhaustive() && lexical_hits.len() == MAX_EXHAUSTIVE_OCCURRENCES
565                {
566                    return Err(Error::RetrievalLimitExceeded {
567                        kind: RetrievalLimitKind::ExhaustiveOccurrences,
568                        observed: lexical_hits.len().saturating_add(1),
569                        limit: MAX_EXHAUSTIVE_OCCURRENCES,
570                    });
571                }
572                let matched_line = search_hit
573                    .occurrence
574                    .as_ref()
575                    .map(|occurrence| occurrence.start_line)
576                    .or_else(|| {
577                        matching_line(
578                            &hit,
579                            &request.query,
580                            request.case_sensitive,
581                            prepared.regex.as_ref().or(prepared.literal_regex.as_ref()),
582                        )
583                    })
584                    .unwrap_or(search_hit.start_line);
585                lexical_hits.push((hit.file_id, search_hit, matched_line));
586            }
587        }
588        let lexical_locations = lexical_hits
589            .iter()
590            .map(|(file_id, _, matched_line)| (*file_id, *matched_line))
591            .collect::<Vec<_>>();
592        let mut enclosing = Vec::with_capacity(lexical_locations.len());
593        for locations in lexical_locations.chunks(512) {
594            check_cancelled(cancellation)?;
595            enclosing.extend(session.find_enclosing_symbols_batch(locations)?);
596        }
597        Ok(lexical_hits
598            .into_iter()
599            .zip(enclosing)
600            .map(|((_, mut hit, _), symbol)| {
601                let definition = symbol.map(|symbol| {
602                    hit.enclosing_symbol = Some(symbol.name);
603                    DefinitionIdentity {
604                        path: hit.path.clone(),
605                        start_line: symbol.start_line,
606                        end_line: symbol.end_line,
607                    }
608                });
609                CandidateSearchHit { hit, definition }
610            })
611            .collect())
612    }

614    pub(super) fn build_search_page(
615        &self,
616        snapshot: SearchSnapshot<'_>,
617        query: SearchQuery<'_>,
618        output_shape: SearchOutputShape,
619        execution: SearchExecutionOptions,
620        page: OrderedSearchPage,
621    ) -> Result<(SearchResponse, Option<usize>, QueryReceiptExecution)> {
622        let SearchSnapshot {
623            session,
624            generation,
625            cancellation,
626        } = snapshot;
627        let SearchQuery {
628            request,
629            prepared,
630            stream_id,
631        } = query;
632        let OrderedSearchPage { hits, offset } = page;
633        let total_candidates = hits.len();
634        let (mut selected, consumed, _) = select_search_page(
635            &hits,
636            offset,
637            prepared.limit,
638            prepared.token_limit,
639            output_shape,
640            &self.config.tokenizer,
641            cancellation,
642        )?;
643        let has_more = offset.saturating_add(consumed) < total_candidates;
644        self.ensure_search_page_fits(
645            &mut selected,
646            SearchResponseShape {
647                all: &hits,
648                request,
649                stream_id,
650                generation,
651                total_candidates,
652                offset,
653                consumed,
654                has_more,
655            },
656            execution.response_options,
657        )?;
658        let receipt_candidates = selected
659            .iter()
660            .map(|candidate| {
661                ReceiptEvidence::new(
662                    candidate.hit.path.clone(),
663                    candidate.hit.start_line,
664                    candidate.hit.end_line,
665                    candidate.hit.content_hash.clone(),
666                    Some(&candidate.hit.excerpt),
667                )
668            })
669            .collect::<Vec<_>>();
670        let receipt = self.evaluate_receipt(
671            matches!(
672                output_shape,
673                SearchOutputShape::Full | SearchOutputShape::Compact | SearchOutputShape::Grouped
674            )
675            .then_some(request.receipt_id.as_deref())
676            .flatten(),
677            generation,
678            &receipt_candidates,
679        )?;
680        selected = selected
681            .into_iter()
682            .zip(&receipt.decisions)
683            .filter_map(|(candidate, decision)| {
684                matches!(
685                    decision,
686                    ReceiptDecision::Return | ReceiptDecision::ReturnNearDuplicate
687                )
688                .then_some(candidate)
689            })
690            .collect();
691        let emitted_tokens =
692            selected_search_source_tokens(&selected, output_shape, &self.config.tokenizer);
693        let paths = selected
694            .iter()
695            .map(|candidate| candidate.hit.path.clone())
696            .collect::<HashSet<_>>()
697            .into_iter()
698            .collect::<Vec<_>>();
699        let occurrences_returned = selected.len();
700        let coverage = search_coverage(&hits, &selected, request.kind.mode());
701        let selected = selected
702            .into_iter()
703            .map(|candidate| candidate.hit)
704            .collect();
705        let baseline_source_tokens =
706            session.whole_file_source_tokens(&paths, self.config.tokenizer.name())?;
707        let next_cursor = has_more
708            .then(|| {
709                ContinuationCursor::at(CursorKind::Search, generation, stream_id, offset + consumed)
710                    .map(ContinuationCursor::encode)
711            })
712            .transpose()?;
713        let mut response = SearchResponse {
714            hits: selected,
715            coverage,
716            occurrences_returned,
717            occurrences_total: request.kind.is_exhaustive().then_some(total_candidates),
718            meta: self.meta(generation, emitted_tokens, next_cursor),
719        };
720        receipt.apply_meta(&mut response.meta);
721        let query_receipt =
722            if let Some(PreparedQueryReceipt::Record(predicate)) = request.kind.query_receipt() {
723                let predicate_blake3 = predicate.digest()?;
724                if !has_more && occurrences_returned == total_candidates {
725                    let path_filter =
726                        PathFilter::new(predicate.include_paths(), predicate.exclude_paths())?;
727                    let partition = session.exact_query_partition(
728                        |path| path_filter.allows(path),
729                        || check_cancelled(cancellation),
730                    )?;
731                    let occurrences = hits
732                        .iter()
733                        .map(|candidate| {
734                            candidate
735                                .hit
736                                .occurrence
737                                .as_ref()
738                                .map(|occurrence| (candidate.hit.path.as_str(), occurrence))
739                                .ok_or_else(|| {
740                                    Error::OperationFailure(
741                                        "exhaustive query result omitted exact coordinates".into(),
742                                    )
743                                })
744                        })
745                        .collect::<Result<Vec<_>>>()?;
746                    QueryReceiptExecution::Pending(QueryReceiptRecord {
747                        repository_generation: generation,
748                        config_hash: session.meta()?.config_hash,
749                        predicate: predicate.clone(),
750                        predicate_blake3,
751                        partition,
752                        match_count: total_candidates,
753                        result_blake3: exhaustive_result_digest(occurrences),
754                    })
755                } else {
756                    QueryReceiptExecution::Outcome(QueryReceiptOutcome {
757                        status: QueryReceiptStatus::NotRecordedIncompleteResponse,
758                        receipt_id: None,
759                        complete: false,
760                        match_count: total_candidates,
761                        requested_predicate_blake3: predicate_blake3.clone(),
762                        covered_predicate_blake3: predicate_blake3,
763                        result_blake3: None,
764                        receipt_generation: generation,
765                        reused_across_generation: false,
766                        scope_relation: QueryReceiptScopeRelation::Exact,
767                    })
768                }
769            } else {
770                QueryReceiptExecution::None
771            };
772        Ok((response, baseline_source_tokens, query_receipt))
773    }
774}

776pub(super) fn order_search_hits(
777    mut hits: Vec<CandidateSearchHit>,
778    request: &SearchInput,
779) -> Result<Vec<CandidateSearchHit>> {
780    apply_focus(&mut hits, &request.focus_paths)?;
781    hits.sort_by(|left, right| {
782        right
783            .hit
784            .score
785            .total_cmp(&left.hit.score)
786            .then_with(|| left.hit.path.cmp(&right.hit.path))
787            .then_with(|| left.hit.start_line.cmp(&right.hit.start_line))
788            .then_with(|| {
789                left.hit
790                    .occurrence
791                    .as_ref()
792                    .map(|occurrence| occurrence.start_byte)
793                    .cmp(
794                        &right
795                            .hit
796                            .occurrence
797                            .as_ref()
798                            .map(|occurrence| occurrence.start_byte),
799                    )
800            })
801    });
802    let preference = request.kind.definition_preference();
803    hits = deduplicate_exact_hits(hits, preference);
804    if matches!(
805        request.kind.mode(),
806        SearchMode::Auto | SearchMode::Identifier
807    ) {
808        hits = deduplicate_definition_channels(hits, preference);
809    }
810    normalize_search_scores(&mut hits);
811    Ok(hits)
812}
