1pub(super) fn validate_search_input(request: &SearchRequest) -> Result<()> {
2    if request.query.trim().is_empty() {
3        return Err(Error::InvalidInput {
4            field: "search query",
5            reason: "must not be empty",
6        });
7    }
8    validate_input(&request.query, "search query", MAX_QUERY_BYTES)?;
9    validate_glob_patterns(&request.include_paths)?;
10    validate_glob_patterns(&request.exclude_paths)?;
11    validate_glob_patterns(&request.focus_paths)?;
12    validate_cursor(request.cursor.as_deref())?;
13    if request.all_occurrences && !request.mode.supports_all_occurrences() {
14        return Err(incompatible_occurrence_options(
15            request.mode,
16            vec!["all_occurrences=true".into()],
17        ));
18    }
19    if request.prefer_structural
20        && !matches!(request.mode, SearchMode::Auto | SearchMode::Identifier)
21    {
22        return Err(Error::InvalidInput {
23            field: "prefer structural",
24            reason: "requires auto or identifier mode",
25        });
26    }
27    if request.query_receipt.is_some() {
28        if !request.all_occurrences || !matches!(request.mode, SearchMode::Text | SearchMode::Regex)
29        {
30            return Err(Error::InvalidInput {
31                field: "query_receipt",
32                reason: "requires all_occurrences=true with text or regex mode",
33            });
34        }
35        if !request.focus_paths.is_empty() {
36            return Err(Error::InvalidInput {
37                field: "query_receipt",
38                reason: "does not allow focus_paths",
39            });
40        }
41        if request.receipt_id.is_some() {
42            return Err(Error::InvalidInput {
43                field: "query_receipt",
44                reason: "cannot be combined with evidence receipt_id",
45            });
46        }
47        if request.cursor.is_some() {
48            return Err(Error::InvalidInput {
49                field: "query_receipt",
50                reason: "does not allow a cursor",
51            });
52        }
53        if let Some(QueryReceiptAction::Reuse { receipt_id }) = &request.query_receipt {
54            validate_input(
55                receipt_id,
56                "query receipt_id",
57                crate::query_receipt::MAX_QUERY_RECEIPT_ID_BYTES,
58            )?;
59        }
60    }
61    if matches!(request.mode, SearchMode::Regex) {
62        compile_regex(request)?;
63    } else {
64        compile_literal_regex(&request.query, request.case_sensitive)?;
65    }
66    Ok(())
67}

69pub(super) fn validate_occurrence_group_input(request: &SearchRequest) -> Result<()> {
70    validate_search_input(request)?;
71    if !request.all_occurrences {
72        return Err(Error::InvalidInput {
73            field: "occurrence projection",
74            reason: "requires all_occurrences=true",
75        });
76    }
77    Ok(())
