28pub(super) fn merge_search_hits(primary: &mut SearchHit, secondary: SearchHit) {
29    primary.score = primary.score.max(secondary.score);
30    for kind in secondary.match_kinds {
31        if !primary.match_kinds.contains(&kind) {
32            primary.match_kinds.push(kind);
33        }
34    }
35    for reason in secondary.score_reasons {
36        if !primary.score_reasons.contains(&reason) {
37            primary.score_reasons.push(reason);
38        }
39    }
40    if primary.role.is_none() {
41        primary.role = secondary.role;
42    }
43    if primary.symbol.is_none() {
44        primary.symbol = secondary.symbol;
45    }
46    if primary.enclosing_symbol.is_none() {
47        primary.enclosing_symbol = secondary.enclosing_symbol;
48    }
49}

51pub(super) fn deduplicate_definition_channels(
52    hits: Vec<CandidateSearchHit>,
53    prefer_structural: bool,
54) -> Vec<CandidateSearchHit> {
55    let mut deduplicated: Vec<CandidateSearchHit> = Vec::with_capacity(hits.len());
56    for mut candidate in hits {
57        let candidate_structural = hit_has_kind(&candidate.hit, "symbol");
58        let candidate_lexical =
59            hit_has_kind(&candidate.hit, "text") || hit_has_kind(&candidate.hit, "regex");
60        let duplicate = candidate.definition.as_ref().and_then(|identity| {
61            deduplicated.iter().position(|existing| {
62                existing.definition.as_ref() == Some(identity)
63                    && ((candidate_structural
64                        && (hit_has_kind(&existing.hit, "text")
65                            || hit_has_kind(&existing.hit, "regex")))
66                        || (candidate_lexical && hit_has_kind(&existing.hit, "symbol")))
67            })
68        });
69        let Some(index) = duplicate else {
70            deduplicated.push(candidate);
71            continue;
72        };
73        if prefer_structural
74            && candidate_structural
75            && !hit_has_kind(&deduplicated[index].hit, "symbol")
76        {
77            std::mem::swap(&mut candidate, &mut deduplicated[index]);
78        }
79        merge_search_hits(&mut deduplicated[index].hit, candidate.hit);
80    }
81    deduplicated
82}

84pub(super) fn deduplicate_exact_hits(
85    hits: Vec<CandidateSearchHit>,
86    prefer_structural: bool,
87) -> Vec<CandidateSearchHit> {
88    let mut deduplicated: Vec<CandidateSearchHit> = Vec::with_capacity(hits.len());
89    let mut positions = HashMap::new();
90    for mut candidate in hits {
91        let key = (
92            candidate.hit.path.clone(),
93            candidate.hit.start_line,
94            candidate.hit.end_line,
95            candidate.hit.content_hash.clone(),
96            candidate
97                .hit
98                .occurrence
99                .as_ref()
100                .map(|occurrence| (occurrence.start_byte, occurrence.end_byte)),
101        );
102        let Some(&index) = positions.get(&key) else {
103            positions.insert(key, deduplicated.len());
104            deduplicated.push(candidate);
105            continue;
106        };
107        if prefer_structural
108            && hit_has_kind(&candidate.hit, "symbol")
109            && !hit_has_kind(&deduplicated[index].hit, "symbol")
110        {
111            std::mem::swap(&mut candidate, &mut deduplicated[index]);
112        }
113        if deduplicated[index].definition.is_none() {
114            deduplicated[index].definition = candidate.definition.clone();
115        }
116        merge_search_hits(&mut deduplicated[index].hit, candidate.hit);
117    }
118    deduplicated
119}

121pub(super) fn normalize_search_scores(hits: &mut [CandidateSearchHit]) {
122    let max_score = hits
123        .iter()
124        .map(|candidate| candidate.hit.score)
125        .filter(|score| score.is_finite())
126        .fold(0.0_f64, f64::max);
127    for candidate in hits {
128        candidate.hit.normalized_score = if max_score > 0.0 && candidate.hit.score.is_finite() {
129            (candidate.hit.score / max_score).clamp(0.0, 1.0)
130        } else {
131            0.0
132        };
133    }
134}

136pub(super) fn collect_filtered_hits<T>(
137    request: &SearchRequest,
138    max_candidates: usize,
139    cancellation: &CancellationToken,
140    mut fetch_page: impl FnMut(usize, usize) -> Result<Vec<T>>,
141    path: impl Fn(&T) -> &str,
142) -> Result<Vec<T>> {
143    let path_filter = PathFilter::new(&request.include_paths, &request.exclude_paths)?;
144    let mut selected = Vec::new();
145    let mut offset = 0usize;
146    while selected.len() < max_candidates && offset < MAX_FILTER_SCAN_ROWS {
147        check_cancelled(cancellation)?;
148        let page_limit = FILTER_SCAN_PAGE_SIZE.min(MAX_FILTER_SCAN_ROWS - offset);
149        let page = fetch_page(offset, page_limit)?;
150        let page_len = page.len();
151        for hit in page {
152            check_cancelled(cancellation)?;
153            if path_filter.allows(path(&hit)) {
154                selected.push(hit);
155                if selected.len() == max_candidates {
156                    break;
157                }
158            }
159        }
160        offset = offset.saturating_add(page_len);
161        if page_len < page_limit {
162            break;
163        }
164    }
165    Ok(selected)
166}

189pub(super) fn chunk_search_hit(
190    hit: &ChunkHit,
191    query: &str,
192    case_sensitive: bool,
193    context: usize,
194    compiled_matcher: Option<&regex::Regex>,
195    regex_match: bool,
196) -> Result<Option<SearchHit>> {
197    let byte_range = if let Some(regex) = compiled_matcher {
198        regex
199            .find(&hit.content)
200            .map(|matched| (matched.start(), matched.end()))
201    } else if case_sensitive {
202        hit.content
203            .find(query)
204            .map(|start| (start, start + query.len()))
205    } else {
206        regex::RegexBuilder::new(&regex::escape(query))
207            .case_insensitive(true)
208            .build()?
209            .find(&hit.content)
210            .map(|matched| (matched.start(), matched.end()))
211    };
212    let Some((start, end)) = byte_range else {
213        return Ok(None);
214    };
215    let starts = line_starts(&hit.content);
216    Ok(Some(chunk_search_hit_for_range(
217        hit,
218        start,
219        end,
220        context,
221        regex_match,
222        false,
223        &starts,
224    )))
225}

233pub(super) fn chunk_search_hits(
234    hit: &ChunkHit,
235    query: &str,
236    case_sensitive: bool,
237    context: usize,
238    compiled_matcher: Option<&regex::Regex>,
239    regex_match: bool,
240    occurrence_limit: OccurrenceMaterializationLimit,
241) -> Result<Vec<SearchHit>> {
242    let ranges: Box<dyn Iterator<Item = (usize, usize)> + '_> =
243        if let Some(regex) = compiled_matcher {
244            Box::new(
245                regex
246                    .find_iter(&hit.content)
247                    .map(|matched| (matched.start(), matched.end())),
248            )
249        } else if case_sensitive {
250            Box::new(
251                hit.content
252                    .match_indices(query)
253                    .map(|(start, matched)| (start, start + matched.len())),
254            )
255        } else {
256            Box::new(std::iter::empty())
257        };
258    let starts = line_starts(&hit.content);
259    let mut hits = Vec::new();
260    for (start, end) in ranges {
261        if occurrence_limit.existing_hits.saturating_add(hits.len()) >= occurrence_limit.max_hits {
262            return Err(Error::RetrievalLimitExceeded {
263                kind: RetrievalLimitKind::ExhaustiveOccurrences,
264                observed: occurrence_limit
265                    .existing_hits
266                    .saturating_add(hits.len())
267                    .saturating_add(1),
268                limit: occurrence_limit.max_hits,
269            });
270        }
271        hits.push(chunk_search_hit_for_range(
272            hit,
273            start,
274            end,
275            context,
276            regex_match,
277            true,
278            &starts,
279        ));
280    }
281    Ok(hits)
282}

284pub(in crate::services) fn chunk_search_hit_for_range(
285    hit: &ChunkHit,
286    start: usize,
287    end: usize,
288    context: usize,
289    regex_match: bool,
290    include_occurrence: bool,
291    line_starts: &[usize],
292) -> SearchHit {
293    let text_len = hit.content.len();
294    let local_start = byte_to_line(line_starts, text_len, start);
295    let local_end = if end == 0 || end == start {
296        local_start
297    } else {
298        byte_to_line(line_starts, text_len, end.saturating_sub(1))
299    };
300    let available_lines = line_starts.len().max(1);
301    let desired_start = local_start.saturating_sub(context).max(1);
302    let desired_end = local_end.saturating_add(context).min(available_lines);
303    let (excerpt_start, excerpt_end) =
304        anchored_line_window(desired_start, desired_end, local_start, local_end, 20);
305    let excerpt = excerpt(&hit.content, excerpt_start, excerpt_end);
306    SearchHit {
307        path: hit.path.clone(),
308        start_line: hit.start_line + excerpt_start - 1,
309        end_line: hit.start_line + excerpt_end - 1,
310        content_hash: hash(&excerpt),
311        excerpt,
312        match_kind: if regex_match {
313            "regex".into()
314        } else {
315            "text".into()
316        },
317        match_kinds: vec![if regex_match {
318            "regex".into()
319        } else {
320            "text".into()
321        }],
322        role: None,
323        symbol: None,
324        enclosing_symbol: None,
325        occurrence: include_occurrence.then_some(SearchOccurrence {
326            start_line: hit.start_line + local_start - 1,
327            end_line: hit.start_line + local_end - 1,
328            start_column: start.saturating_sub(line_starts[local_start.saturating_sub(1)]),
329            end_column: end.saturating_sub(line_starts[local_end.saturating_sub(1)]),
330            start_byte: hit.start_byte + start,
331            end_byte: hit.start_byte + end,
332        }),
333        score: 3.0 + (-hit.score).max(0.0) * 1_000_000.0,
334        normalized_score: 0.0,
335        score_reasons: vec![if regex_match {
336            "regex match".into()
337        } else {
338            "text match".into()
339        }],
340    }
341}

343pub(super) fn matching_line(
344    hit: &ChunkHit,
345    query: &str,
346    case_sensitive: bool,
347    compiled_regex: Option<&regex::Regex>,
348) -> Option<usize> {
349    if let Some(regex) = compiled_regex {
350        return regex.find(&hit.content).map(|matched| {
351            let (local_start, _) =
352                byte_range_to_line_range(&hit.content, matched.start(), matched.end());
353            hit.start_line + local_start - 1
354        });
355    }
356    if !case_sensitive {
357        return None;
358    }
359    hit.content
360        .lines()
361        .position(|line| line.contains(query))
362        .map(|offset| hit.start_line + offset)
363}

365pub(super) fn apply_focus(hits: &mut [CandidateSearchHit], focus_paths: &[String]) -> Result<()> {
366    let focus_paths = PathMatcher::new(focus_paths)?;
367    for candidate in hits {
368        if focus_paths.is_match(&candidate.hit.path) {
369            candidate.hit.score += 2.0;
370            candidate.hit.score_reasons.push("focus path".into());
371        }
372    }
373    Ok(())
374}
