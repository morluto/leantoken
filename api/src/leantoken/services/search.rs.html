45fn hit_has_kind(hit: &SearchHit, kind: &str) -> bool {
46    hit.match_kind == kind || hit.match_kinds.iter().any(|candidate| candidate == kind)
47}

49fn merge_search_hits(primary: &mut SearchHit, secondary: SearchHit) {
50    primary.score = primary.score.max(secondary.score);
51    for kind in secondary.match_kinds {
52        if !primary.match_kinds.contains(&kind) {
53            primary.match_kinds.push(kind);
54        }
55    }
56    for reason in secondary.score_reasons {
57        if !primary.score_reasons.contains(&reason) {
58            primary.score_reasons.push(reason);
59        }
60    }
61    if primary.role.is_none() {
62        primary.role = secondary.role;
63    }
64    if primary.symbol.is_none() {
65        primary.symbol = secondary.symbol;
66    }
67    if primary.enclosing_symbol.is_none() {
68        primary.enclosing_symbol = secondary.enclosing_symbol;
69    }
70}

72fn deduplicate_definition_channels(
73    hits: Vec<CandidateSearchHit>,
74    prefer_structural: bool,
75) -> Vec<CandidateSearchHit> {
76    let mut deduplicated: Vec<CandidateSearchHit> = Vec::with_capacity(hits.len());
77    for mut candidate in hits {
78        let candidate_structural = hit_has_kind(&candidate.hit, "symbol");
79        let candidate_lexical =
80            hit_has_kind(&candidate.hit, "text") || hit_has_kind(&candidate.hit, "regex");
81        let duplicate = candidate.definition.as_ref().and_then(|identity| {
82            deduplicated.iter().position(|existing| {
83                existing.definition.as_ref() == Some(identity)
84                    && ((candidate_structural
85                        && (hit_has_kind(&existing.hit, "text")
86                            || hit_has_kind(&existing.hit, "regex")))
87                        || (candidate_lexical && hit_has_kind(&existing.hit, "symbol")))
88            })
89        });
90        let Some(index) = duplicate else {
91            deduplicated.push(candidate);
92            continue;
93        };
94        if prefer_structural
95            && candidate_structural
96            && !hit_has_kind(&deduplicated[index].hit, "symbol")
97        {
98            std::mem::swap(&mut candidate, &mut deduplicated[index]);
99        }
100        merge_search_hits(&mut deduplicated[index].hit, candidate.hit);
101    }
102    deduplicated
103}

105fn deduplicate_exact_hits(
106    hits: Vec<CandidateSearchHit>,
107    prefer_structural: bool,
108) -> Vec<CandidateSearchHit> {
109    let mut deduplicated: Vec<CandidateSearchHit> = Vec::with_capacity(hits.len());
110    let mut positions = HashMap::new();
111    for mut candidate in hits {
112        let key = (
113            candidate.hit.path.clone(),
114            candidate.hit.start_line,
115            candidate.hit.end_line,
116            candidate.hit.content_hash.clone(),
117            candidate
118                .hit
119                .occurrence
120                .as_ref()
121                .map(|occurrence| (occurrence.start_byte, occurrence.end_byte)),
122        );
123        let Some(&index) = positions.get(&key) else {
124            positions.insert(key, deduplicated.len());
125            deduplicated.push(candidate);
126            continue;
127        };
128        if prefer_structural
129            && hit_has_kind(&candidate.hit, "symbol")
130            && !hit_has_kind(&deduplicated[index].hit, "symbol")
131        {
132            std::mem::swap(&mut candidate, &mut deduplicated[index]);
133        }
134        if deduplicated[index].definition.is_none() {
135            deduplicated[index].definition = candidate.definition.clone();
136        }
137        merge_search_hits(&mut deduplicated[index].hit, candidate.hit);
138    }
139    deduplicated
140}

142fn normalize_search_scores(hits: &mut [CandidateSearchHit]) {
143    let max_score = hits
144        .iter()
145        .map(|candidate| candidate.hit.score)
146        .filter(|score| score.is_finite())
147        .fold(0.0_f64, f64::max);
148    for candidate in hits {
149        candidate.hit.normalized_score = if max_score > 0.0 && candidate.hit.score.is_finite() {
150            (candidate.hit.score / max_score).clamp(0.0, 1.0)
151        } else {
152            0.0
153        };
154    }
155}

157fn coverage_count(
158    all: &[CandidateSearchHit],
159    returned: &[CandidateSearchHit],
160    matches: impl Fn(&SearchHit) -> bool,
161) -> SearchCoverageCount {
162    let total = all
163        .iter()
164        .filter(|candidate| matches(&candidate.hit))
165        .count();
166    let returned = returned
167        .iter()
168        .filter(|candidate| matches(&candidate.hit))
169        .count();
170    SearchCoverageCount {
171        total,
172        returned,
173        truncated: total.saturating_sub(returned),
174    }
175}

177fn search_coverage(all: &[CandidateSearchHit], returned: &[CandidateSearchHit]) -> SearchCoverage {
178    SearchCoverage {
179        definitions: coverage_count(all, returned, |hit| hit_has_kind(hit, "symbol")),
180        references: coverage_count(all, returned, |hit| hit_has_kind(hit, "reference")),
181        text_matches: coverage_count(all, returned, |hit| {
182            hit_has_kind(hit, "text") || hit_has_kind(hit, "regex")
183        }),
184    }

187fn collect_filtered_hits<T>(
188    request: &SearchRequest,
189    max_candidates: usize,
190    cancellation: &CancellationToken,
191    mut fetch_page: impl FnMut(usize, usize) -> Result<Vec<T>>,
192    path: impl Fn(&T) -> &str,
193) -> Result<Vec<T>> {
194    let path_filter = PathFilter::new(&request.include_paths, &request.exclude_paths)?;
195    let mut selected = Vec::new();
196    let mut offset = 0usize;
197    while selected.len() < max_candidates && offset < MAX_FILTER_SCAN_ROWS {
198        check_cancelled(cancellation)?;
199        let page_limit = FILTER_SCAN_PAGE_SIZE.min(MAX_FILTER_SCAN_ROWS - offset);
200        let page = fetch_page(offset, page_limit)?;
201        let page_len = page.len();
202        for hit in page {
203            check_cancelled(cancellation)?;
204            if path_filter.allows(path(&hit)) {
205                selected.push(hit);
206                if selected.len() == max_candidates {
207                    break;
208                }
209            }
210        }
211        offset = offset.saturating_add(page_len);
212        if page_len < page_limit {
213            break;
214        }
215    }
216    Ok(selected)
217}

219pub(super) fn chunk_search_hit(
220    hit: &ChunkHit,
221    query: &str,
222    case_sensitive: bool,
223    context: usize,
224    compiled_matcher: Option<&regex::Regex>,
225    regex_match: bool,
226) -> Result<Option<SearchHit>> {
227    let byte_range = if let Some(regex) = compiled_matcher {
228        regex
229            .find(&hit.content)
230            .map(|matched| (matched.start(), matched.end()))
231    } else if case_sensitive {
232        hit.content
233            .find(query)
234            .map(|start| (start, start + query.len()))
235    } else {
236        regex::RegexBuilder::new(&regex::escape(query))
237            .case_insensitive(true)
238            .build()?
239            .find(&hit.content)
240            .map(|matched| (matched.start(), matched.end()))
241    };
242    let Some((start, end)) = byte_range else {
243        return Ok(None);
244    };
245    let starts = line_starts(&hit.content);
246    Ok(Some(chunk_search_hit_for_range(
247        hit,
248        start,
249        end,
250        context,
251        regex_match,
252        false,
253        &starts,
254    )))
255}

257fn chunk_search_hits(
258    hit: &ChunkHit,
259    query: &str,
260    case_sensitive: bool,
261    context: usize,
262    compiled_matcher: Option<&regex::Regex>,
263    regex_match: bool,
264    max_hits: usize,
265) -> Result<Vec<SearchHit>> {
266    let ranges: Box<dyn Iterator<Item = (usize, usize)> + '_> =
267        if let Some(regex) = compiled_matcher {
268            Box::new(
269                regex
270                    .find_iter(&hit.content)
271                    .map(|matched| (matched.start(), matched.end())),
272            )
273        } else if case_sensitive {
274            Box::new(
275                hit.content
276                    .match_indices(query)
277                    .map(|(start, matched)| (start, start + matched.len())),
278            )
279        } else {
280            Box::new(std::iter::empty())
281        };
282    let starts = line_starts(&hit.content);
283    let mut hits = Vec::new();
284    for (start, end) in ranges {
285        if hits.len() == max_hits {
286            return Err(Error::LimitExceeded);
287        }
288        hits.push(chunk_search_hit_for_range(
289            hit,
290            start,
291            end,
292            context,
293            regex_match,
294            true,
295            &starts,
296        ));
297    }
298    Ok(hits)
299}

301fn chunk_search_hit_for_range(
302    hit: &ChunkHit,
303    start: usize,
304    end: usize,
305    context: usize,
306    regex_match: bool,
307    include_occurrence: bool,
308    line_starts: &[usize],
309) -> SearchHit {
310    let text_len = hit.content.len();
311    let local_start = byte_to_line(line_starts, text_len, start);
312    let local_end = if end == 0 || end == start {
313        local_start
314    } else {
315        byte_to_line(line_starts, text_len, end.saturating_sub(1))
316    };
317    let available_lines = line_starts.len().max(1);
318    let desired_start = local_start.saturating_sub(context).max(1);
319    let desired_end = local_end.saturating_add(context).min(available_lines);
320    let (excerpt_start, excerpt_end) =
321        anchored_line_window(desired_start, desired_end, local_start, local_end, 20);
322    let excerpt = excerpt(&hit.content, excerpt_start, excerpt_end);
323    SearchHit {
324        path: hit.path.clone(),
325        start_line: hit.start_line + excerpt_start - 1,
326        end_line: hit.start_line + excerpt_end - 1,
327        content_hash: hash(&excerpt),
328        excerpt,
329        match_kind: if regex_match {
330            "regex".into()
331        } else {
332            "text".into()
333        },
334        match_kinds: vec![if regex_match {
335            "regex".into()
336        } else {
337            "text".into()
338        }],
339        role: None,
340        symbol: None,
341        enclosing_symbol: None,
342        occurrence: include_occurrence.then_some(SearchOccurrence {
343            start_line: hit.start_line + local_start - 1,
344            end_line: hit.start_line + local_end - 1,
345            start_byte: hit.start_byte + start,
346            end_byte: hit.start_byte + end,
347        }),
348        score: 3.0 + (-hit.score).max(0.0) * 1_000_000.0,
349        normalized_score: 0.0,
350        score_reasons: vec![if regex_match {
351            "regex match".into()
352        } else {
353            "text match".into()
354        }],
355    }
356}

358pub(super) fn matching_line(
359    hit: &ChunkHit,
360    query: &str,
361    case_sensitive: bool,
362    compiled_regex: Option<&regex::Regex>,
363) -> Option<usize> {
364    if let Some(regex) = compiled_regex {
365        return regex.find(&hit.content).map(|matched| {
366            let (local_start, _) =
367                byte_range_to_line_range(&hit.content, matched.start(), matched.end());
368            hit.start_line + local_start - 1
369        });
370    }
371    if !case_sensitive {
372        return None;
373    }
374    hit.content
375        .lines()
376        .position(|line| line.contains(query))
377        .map(|offset| hit.start_line + offset)
378}

380fn apply_focus(hits: &mut [CandidateSearchHit], focus_paths: &[String]) -> Result<()> {
381    let focus_paths = PathMatcher::new(focus_paths)?;
382    for candidate in hits {
383        if focus_paths.is_match(&candidate.hit.path) {
384            candidate.hit.score += 2.0;
385            candidate.hit.score_reasons.push("focus path".into());
386        }
387    }
388    Ok(())
389}

395fn compile_regex(request: &SearchRequest) -> Result<regex::Regex> {
396    Ok(regex::RegexBuilder::new(&request.query)
397        .case_insensitive(!request.case_sensitive)
398        .size_limit(1 << 20)
399        .dfa_size_limit(1 << 20)
400        .build()?)
401}
402
403fn compile_occurrence_literal_regex(request: &SearchRequest) -> Result<regex::Regex> {
404    Ok(regex::RegexBuilder::new(&regex::escape(&request.query))
405        .case_insensitive(!request.case_sensitive)
406        .size_limit(1 << 20)
407        .dfa_size_limit(1 << 20)
408        .build()?)
409}

411/// Compile a case-insensitive literal matcher for non-regex search modes.
412///
413/// In Auto/Text/Identifier modes the query is a literal string, not a regex
414/// pattern. Compile it once per request so every lexical hit reuses the same
415/// matcher. Case-sensitive search uses `str::find` without compilation.
416pub(super) fn compile_literal_regex(
417    query: &str,
418    case_sensitive: bool,
419) -> Result<Option<regex::Regex>> {
420    if case_sensitive {
421        return Ok(None);
422    }
423    Ok(Some(
424        regex::RegexBuilder::new(&regex::escape(query))
425            .case_insensitive(true)
426            .size_limit(1 << 20)
427            .dfa_size_limit(1 << 20)
428            .build()?,
429    ))
430}

432fn validate_search_input(request: &SearchRequest) -> Result<()> {
433    if request.query.trim().is_empty() {
434        return Err(Error::InvalidInput {
435            field: "search query",
436            reason: "must not be empty",
437        });
438    }
439    validate_input(&request.query, "search query", MAX_QUERY_BYTES)?;
440    validate_glob_patterns(&request.include_paths)?;
441    validate_glob_patterns(&request.exclude_paths)?;
442    validate_glob_patterns(&request.focus_paths)?;
443    validate_cursor(request.cursor.as_deref())?;
444    if request.all_occurrences && !matches!(request.mode, SearchMode::Text | SearchMode::Regex) {
445        return Err(Error::InvalidInput {
446            field: "all occurrences",
447            reason: "requires text or regex mode",
448        });
449    }
450    if request.prefer_structural
451        && !matches!(request.mode, SearchMode::Auto | SearchMode::Identifier)
452    {
453        return Err(Error::InvalidInput {
454            field: "prefer structural",
455            reason: "requires auto or identifier mode",
456        });
457    }
458    if matches!(request.mode, SearchMode::Regex) {
459        compile_regex(request)?;
460    } else {
461        compile_literal_regex(&request.query, request.case_sensitive)?;
462    }
463    Ok(())
464}

489    pub async fn search_cancellable(
490        &self,
491        request: SearchRequest,
492        cancellation: CancellationToken,
493    ) -> Result<SearchResponse> {
494        let this = self.clone();
495        tokio::task::spawn_blocking(move || this.search_sync(request, &cancellation)).await?

497
498    fn search_sync(
499        &self,
500        request: SearchRequest,
501        cancellation: &CancellationToken,
502    ) -> Result<SearchResponse> {
503        check_cancelled(cancellation)?;
504        validate_search_input(&request)?;
505        let regex = matches!(request.mode, SearchMode::Regex)
506            .then(|| compile_regex(&request))
507            .transpose()?;
508        let literal_regex = if !matches!(request.mode, SearchMode::Regex) {
509            compile_literal_regex(&request.query, request.case_sensitive)?
510        } else {
511            None
512        };
513        let occurrence_literal_regex = (request.all_occurrences
514            && matches!(request.mode, SearchMode::Text))
515        .then(|| compile_occurrence_literal_regex(&request))
516        .transpose()?;
517        let limit = self.result_limit(request.max_results)?;
518        let token_limit = self.token_limit(request.max_tokens, self.config.default_read_tokens)?;
519        let context_lines = self.context_line_limit(request.context_lines)?;
520        let (mut response, baseline_source_tokens) = self.consistent(|session, generation| {
521            let offset = parse_cursor(request.cursor.as_deref(), generation)?;
522            let mut hits = Vec::new();
523            if matches!(
524                request.mode,
525                SearchMode::Auto | SearchMode::Identifier | SearchMode::Symbol
526            ) {
527                let symbol_hits = collect_filtered_hits(
528                    &request,
529                    limit.saturating_mul(4),
530                    cancellation,
531                    |offset, page_limit| {
532                        session.search_symbols_page(
533                            &request.query,
534                            request.case_sensitive,
535                            page_limit,
536                            offset,
537                        )
538                    },
539                    |hit: &SymbolHit| &hit.path,
540                )?;
541                let excerpt_requests = symbol_hits
542                    .iter()
543                    .map(|hit| StoredExcerptRequest {
544                        file_id: hit.symbol.file_id,
545                        desired_start_line: hit
546                            .symbol
547                            .start_line
548                            .saturating_sub(context_lines)
549                            .max(1),
550                        desired_end_line: hit.symbol.end_line.saturating_add(context_lines),
551                        required_start_line: hit.symbol.start_line,
552                        required_end_line: hit.symbol.start_line,
553                        max_lines: 30,
554                    })
555                    .collect::<Vec<_>>();
556                for (hit, excerpt) in symbol_hits
557                    .into_iter()
558                    .zip(self.stored_excerpts(session, &excerpt_requests)?)
559                {
560                    if let Some(excerpt) = excerpt {
561                        let definition = DefinitionIdentity {
562                            path: hit.path.clone(),
563                            start_line: hit.symbol.start_line,
564                            end_line: hit.symbol.end_line,
565                        };
566                        hits.push(CandidateSearchHit {
567                            hit: self.symbol_search_hit(hit, &request.query, excerpt),
568                            definition: Some(definition),
569                        });
570                    }
571                }
572            }
573            if matches!(
574                request.mode,
575                SearchMode::Auto | SearchMode::Identifier | SearchMode::Reference
576            ) {
577                let reference_hits = collect_filtered_hits(
578                    &request,
579                    limit.saturating_mul(4),
580                    cancellation,
581                    |offset, page_limit| {
582                        session.search_references_page(
583                            &request.query,
584                            request.case_sensitive,
585                            page_limit,
586                            offset,
587                        )
588                    },
589                    |hit: &ReferenceHit| &hit.path,
590                )?;
591                let excerpt_requests = reference_hits
592                    .iter()
593                    .map(|hit| StoredExcerptRequest {
594                        file_id: hit.reference.file_id,
595                        desired_start_line: hit
596                            .reference
597                            .start_line
598                            .saturating_sub(context_lines)
599                            .max(1),
600                        desired_end_line: hit.reference.end_line.saturating_add(context_lines),
601                        required_start_line: hit.reference.start_line,
602                        required_end_line: hit.reference.end_line,
603                        max_lines: 12,
604                    })
605                    .collect::<Vec<_>>();
606                for (hit, excerpt) in reference_hits
607                    .into_iter()
608                    .zip(self.stored_excerpts(session, &excerpt_requests)?)
609                {
610                    if let Some(excerpt) = excerpt {
611                        hits.push(CandidateSearchHit {
612                            hit: self.reference_search_hit(hit, &request.query, excerpt),
613                            definition: None,
614                        });
615                    }
616                }
617            }
618
619            let lexical = match request.mode {
620                SearchMode::Regex => self.regex_hits(
621                    session,
622                    &request,
623                    regex.as_ref().expect("regex mode compiles a pattern"),
624                    (!request.all_occurrences).then_some(limit.saturating_mul(20)),
625                    cancellation,
626                )?,
627                SearchMode::Text if request.all_occurrences => self.regex_hits(
628                    session,
629                    &request,
630                    occurrence_literal_regex
631                        .as_ref()
632                        .expect("exhaustive text mode compiles a literal pattern"),
633                    None,
634                    cancellation,
635                )?,
636                SearchMode::Text | SearchMode::Auto | SearchMode::Identifier => {
637                    let fetch_page = |offset, page_limit| {
638                        if matches!(request.mode, SearchMode::Identifier)
639                            || request.query.chars().count() < 3
640                        {
641                            session.search_word_page(&fts_quote(&request.query), page_limit, offset)
642                        } else {
643                            session.search_trigram_page(&request.query, page_limit, offset)
644                        }
645                    };
646                    collect_filtered_hits(
647                        &request,
648                        limit.saturating_mul(8),
649                        cancellation,
650                        fetch_page,
651                        |hit: &ChunkHit| &hit.path,
652                    )?
653                }
654                SearchMode::Symbol | SearchMode::Reference => Vec::new(),
655            };
656            let mut lexical_hits = Vec::new();
657            for hit in lexical {
658                check_cancelled(cancellation)?;
659                let chunk_hits = if request.all_occurrences {
660                    chunk_search_hits(
661                        &hit,
662                        &request.query,
663                        request.case_sensitive,
664                        context_lines,
665                        regex
666                            .as_ref()
667                            .or(occurrence_literal_regex.as_ref())
668                            .or(literal_regex.as_ref()),
669                        matches!(request.mode, SearchMode::Regex),
670                        MAX_EXHAUSTIVE_OCCURRENCES.saturating_sub(lexical_hits.len()),
671                    )?
672                } else {
673                    chunk_search_hit(
674                        &hit,
675                        &request.query,
676                        request.case_sensitive,
677                        context_lines,
678                        regex.as_ref().or(literal_regex.as_ref()),
679                        matches!(request.mode, SearchMode::Regex),
680                    )?
681                    .into_iter()
682                    .collect()
683                };
684                for search_hit in chunk_hits {
685                    if request.all_occurrences && lexical_hits.len() == MAX_EXHAUSTIVE_OCCURRENCES {
686                        return Err(Error::LimitExceeded);
687                    }
688                    let matched_line = search_hit
689                        .occurrence
690                        .as_ref()
691                        .map(|occurrence| occurrence.start_line)
692                        .or_else(|| {
693                            matching_line(
694                                &hit,
695                                &request.query,
696                                request.case_sensitive,
697                                regex.as_ref().or(literal_regex.as_ref()),
698                            )
699                        })
700                        .unwrap_or(search_hit.start_line);
701                    lexical_hits.push((hit.file_id, search_hit, matched_line));
702                }
703            }
704            let lexical_locations = lexical_hits
705                .iter()
706                .map(|(file_id, _, matched_line)| (*file_id, *matched_line))
707                .collect::<Vec<_>>();
708            let mut enclosing = Vec::with_capacity(lexical_locations.len());
709            for locations in lexical_locations.chunks(512) {
710                check_cancelled(cancellation)?;
711                enclosing.extend(session.find_enclosing_symbols_batch(locations)?);
712            }
713            for ((_, mut hit, _), symbol) in lexical_hits.into_iter().zip(enclosing) {
714                let definition = symbol.map(|symbol| {
715                    hit.enclosing_symbol = Some(symbol.name);
716                    DefinitionIdentity {
717                        path: hit.path.clone(),
718                        start_line: symbol.start_line,
719                        end_line: symbol.end_line,
720                    }
721                });
722                hits.push(CandidateSearchHit { hit, definition });
723            }
724
725            apply_focus(&mut hits, &request.focus_paths)?;
726            hits.sort_by(|left, right| {
727                right
728                    .hit
729                    .score
730                    .total_cmp(&left.hit.score)
731                    .then_with(|| left.hit.path.cmp(&right.hit.path))
732                    .then_with(|| left.hit.start_line.cmp(&right.hit.start_line))
733                    .then_with(|| {
734                        left.hit
735                            .occurrence
736                            .as_ref()
737                            .map(|occurrence| occurrence.start_byte)
738                            .cmp(
739                                &right
740                                    .hit
741                                    .occurrence
742                                    .as_ref()
743                                    .map(|occurrence| occurrence.start_byte),
744                            )
745                    })
746            });
747            hits = deduplicate_exact_hits(hits, request.prefer_structural);
748            if matches!(request.mode, SearchMode::Auto | SearchMode::Identifier) {
749                hits = deduplicate_definition_channels(hits, request.prefer_structural);
750            }
751            normalize_search_scores(&mut hits);
752
753            let mut emitted_tokens = 0usize;
754            let mut selected = Vec::new();
755            let total_candidates = hits.len();
756            let page = hits.iter().skip(offset).take(limit).cloned();
757            let mut consumed = 0usize;
758            for candidate in page {
759                check_cancelled(cancellation)?;
760                consumed += 1;
761                let count = self.config.tokenizer.count(&candidate.hit.excerpt);
762                if emitted_tokens.saturating_add(count) > token_limit {
763                    continue;
764                }
765                emitted_tokens += count;
766                selected.push(candidate);
767            }
768            let has_more = offset.saturating_add(consumed) < total_candidates;
769            let paths = selected
770                .iter()
771                .map(|candidate| candidate.hit.path.clone())
772                .collect::<HashSet<_>>()
773                .into_iter()
774                .collect::<Vec<_>>();
775            let occurrences_returned = selected.len();
776            let coverage = search_coverage(&hits, &selected);
777            let selected = selected
778                .into_iter()
779                .map(|candidate| candidate.hit)
780                .collect();
781            let baseline_source_tokens =
782                session.whole_file_source_tokens(&paths, self.config.tokenizer.name())?;
783            Ok((
784                SearchResponse {
785                    hits: selected,
786                    coverage,
787                    occurrences_returned,
788                    occurrences_total: request.all_occurrences.then_some(total_candidates),
789                    meta: self.meta(
790                        generation,
791                        emitted_tokens,
792                        has_more.then(|| make_cursor(generation, offset + consumed)),
793                    ),
794                },
795                baseline_source_tokens,
796            ))
797        })?;
798        if let Some(baseline_source_tokens) = baseline_source_tokens {
799            self.record_token_savings(
800                TokenSavingsOperation::Search,
801                baseline_source_tokens,
802                response.meta.emitted_tokens,
803            );
804        }
805        self.finalize_response(&mut response)?;
806        Ok(response)
807    }

862    fn regex_hits(
863        &self,
864        session: &ReadSession,
865        request: &SearchRequest,
866        regex: &regex::Regex,
867        max_candidates: Option<usize>,
868        cancellation: &CancellationToken,
869    ) -> Result<Vec<ChunkHit>> {
870        // Hard caps prevent repository-wide lexical scans from running
871        // unbounded. Exhaustive modes lift only the candidate-chunk cap and
872        // fail explicitly if another cap would make the result incomplete.
873        let max_candidates = max_candidates.map(|limit| limit.min(MAX_REGEX_CANDIDATES));
874        let path_filter = PathFilter::new(&request.include_paths, &request.exclude_paths)?;
875        let mut hits = Vec::new();
876        let mut files_scanned = 0usize;
877        let mut cursor = None;
878        loop {
879            check_cancelled(cancellation)?;
880            let page = session.list_files(FILE_LIST_PAGE_SIZE, cursor)?;
881            if page.is_empty() {
882                break;
883            }
884            cursor = page.last().map(|file| file.id);
885            for file in page {
886                check_cancelled(cancellation)?;
887                if files_scanned == MAX_REGEX_FILES_SCANNED {
888                    return Err(Error::LimitExceeded);
889                }
890                files_scanned += 1;
891                if !path_filter.allows(&file.path) {
892                    continue;
893                }
894                let chunks = session
895                    .get_chunks_for_file(file.id, MAX_REGEX_CHUNKS_PER_FILE.saturating_add(1))?;
896                let chunks_truncated = chunks.len() > MAX_REGEX_CHUNKS_PER_FILE;
897                for chunk in chunks.into_iter().take(MAX_REGEX_CHUNKS_PER_FILE) {
898                    check_cancelled(cancellation)?;
899                    if regex.is_match(&chunk.content) {
900                        if max_candidates.is_some_and(|limit| hits.len() == limit) {
901                            return Err(Error::LimitExceeded);
902                        }
903                        hits.push(ChunkHit {
904                            chunk_id: chunk.id,
905                            file_id: chunk.file_id,
906                            path: file.path.clone(),
907                            content: chunk.content,
908                            start_line: chunk.start_line,
909                            end_line: chunk.end_line,
910                            start_byte: chunk.start_byte,
911                            end_byte: chunk.end_byte,
912                            token_count: chunk.token_count,
913                            generation: file.generation,
914                            score: 0.0,
915                        });
916                    }
917                }
918                if chunks_truncated {
919                    return Err(Error::LimitExceeded);
920                }
921            }
922        }
923        Ok(hits)
924    }
