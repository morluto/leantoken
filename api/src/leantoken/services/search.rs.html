71#[derive(Default)]
72struct RegexPlanBudget {
73    nodes: usize,
74    terms: usize,
75    term_bytes: usize,
76}
77
78impl RegexPlanBudget {
79    fn visit(&mut self) -> std::result::Result<(), ()> {
80        self.nodes = self.nodes.saturating_add(1);
81        (self.nodes <= MAX_REGEX_PLAN_NODES).then_some(()).ok_or(())
82    }
83
84    fn add_term(&mut self, term: &str) -> std::result::Result<(), ()> {
85        self.terms = self.terms.saturating_add(1);
86        self.term_bytes = self.term_bytes.saturating_add(term.len());
87        (self.terms <= MAX_REGEX_PLAN_TERMS && self.term_bytes <= MAX_REGEX_PLAN_TERM_BYTES)
88            .then_some(())
89            .ok_or(())
90    }
91}

93impl RegexCandidateExpr {
94    fn fts_query(&self) -> String {
95        match self {
96            Self::Term(term) => fts_quote(term),
97            Self::All(expressions) => expressions
98                .iter()
99                .map(|expression| format!("({})", expression.fts_query()))
100                .collect::<Vec<_>>()
101                .join(" AND "),
102            Self::Any(expressions) => expressions
103                .iter()
104                .map(|expression| format!("({})", expression.fts_query()))
105                .collect::<Vec<_>>()
106                .join(" OR "),
107        }
108    }
109}

111fn regex_candidate_plan(request: &SearchRequest) -> Option<RegexCandidatePlan> {
112    // SQLite's default trigram tokenizer folds ASCII only. Rust regexes use
113    // Unicode simple case folding, so a case-insensitive ASCII literal can
114    // also match non-ASCII code points (for example, Kelvin sign for `k`).
115    // Falling back avoids false negatives until those semantics can be
116    // represented by the candidate index.
117    if !request.case_sensitive {
118        return None;
119    }
120    let hir = regex_syntax::parse(&request.query).ok()?;
121    let mut budget = RegexPlanBudget::default();
122    let expression = regex_candidate_expr(&hir, &mut budget).ok()??;
123    Some(RegexCandidatePlan {
124        expression,
125        term_count: budget.terms,
126    })
127}
128
129fn regex_candidate_expr(
130    hir: &Hir,
131    budget: &mut RegexPlanBudget,
132) -> std::result::Result<Option<RegexCandidateExpr>, ()> {
133    budget.visit()?;
134    match hir.kind() {
135        HirKind::Literal(literal) => literal_candidate_expr(&literal.0, budget),
136        HirKind::Capture(capture) => regex_candidate_expr(&capture.sub, budget),
137        HirKind::Repetition(repetition) if repetition.min > 0 => {
138            regex_candidate_expr(&repetition.sub, budget)
139        }
140        HirKind::Concat(expressions) => {
141            let mut plans = Vec::new();
142            for expression in expressions {
143                if let Some(plan) = regex_candidate_expr(expression, budget)? {
144                    plans.push(plan);
145                }
146            }
147            Ok(combine_candidate_expr(plans, true))
148        }
149        HirKind::Alternation(expressions) => {
150            let mut plans = Vec::with_capacity(expressions.len());
151            for expression in expressions {
152                let Some(plan) = regex_candidate_expr(expression, budget)? else {
153                    return Ok(None);
154                };
155                plans.push(plan);
156            }
157            Ok(combine_candidate_expr(plans, false))
158        }
159        HirKind::Empty | HirKind::Class(_) | HirKind::Look(_) | HirKind::Repetition(_) => Ok(None),
160    }
161}

163fn literal_candidate_expr(
164    literal: &[u8],
165    budget: &mut RegexPlanBudget,
166) -> std::result::Result<Option<RegexCandidateExpr>, ()> {
167    let mut terms = Vec::new();
168    for bytes in literal.split(|byte| !byte.is_ascii_alphanumeric() && *byte != b'_') {
169        if bytes.len() < 3 {
170            continue;
171        }
172        let term = std::str::from_utf8(bytes).map_err(|_| ())?.to_owned();
173        budget.add_term(&term)?;
174        terms.push(RegexCandidateExpr::Term(term));
175    }
176    Ok(combine_candidate_expr(terms, true))
177}
178
179fn combine_candidate_expr(
180    mut expressions: Vec<RegexCandidateExpr>,
181    all: bool,
182) -> Option<RegexCandidateExpr> {
183    match expressions.len() {
184        0 => None,
185        1 => expressions.pop(),
186        _ if all => Some(RegexCandidateExpr::All(expressions)),
187        _ => Some(RegexCandidateExpr::Any(expressions)),
188    }
189}

214fn hit_has_kind(hit: &SearchHit, kind: &str) -> bool {
215    hit.match_kind == kind || hit.match_kinds.iter().any(|candidate| candidate == kind)
216}

218fn merge_search_hits(primary: &mut SearchHit, secondary: SearchHit) {
219    primary.score = primary.score.max(secondary.score);
220    for kind in secondary.match_kinds {
221        if !primary.match_kinds.contains(&kind) {
222            primary.match_kinds.push(kind);
223        }
224    }
225    for reason in secondary.score_reasons {
226        if !primary.score_reasons.contains(&reason) {
227            primary.score_reasons.push(reason);
228        }
229    }
230    if primary.role.is_none() {
231        primary.role = secondary.role;
232    }
233    if primary.symbol.is_none() {
234        primary.symbol = secondary.symbol;
235    }
236    if primary.enclosing_symbol.is_none() {
237        primary.enclosing_symbol = secondary.enclosing_symbol;
238    }
239}

241fn deduplicate_definition_channels(
242    hits: Vec<CandidateSearchHit>,
243    prefer_structural: bool,
244) -> Vec<CandidateSearchHit> {
245    let mut deduplicated: Vec<CandidateSearchHit> = Vec::with_capacity(hits.len());
246    for mut candidate in hits {
247        let candidate_structural = hit_has_kind(&candidate.hit, "symbol");
248        let candidate_lexical =
249            hit_has_kind(&candidate.hit, "text") || hit_has_kind(&candidate.hit, "regex");
250        let duplicate = candidate.definition.as_ref().and_then(|identity| {
251            deduplicated.iter().position(|existing| {
252                existing.definition.as_ref() == Some(identity)
253                    && ((candidate_structural
254                        && (hit_has_kind(&existing.hit, "text")
255                            || hit_has_kind(&existing.hit, "regex")))
256                        || (candidate_lexical && hit_has_kind(&existing.hit, "symbol")))
257            })
258        });
259        let Some(index) = duplicate else {
260            deduplicated.push(candidate);
261            continue;
262        };
263        if prefer_structural
264            && candidate_structural
265            && !hit_has_kind(&deduplicated[index].hit, "symbol")
266        {
267            std::mem::swap(&mut candidate, &mut deduplicated[index]);
268        }
269        merge_search_hits(&mut deduplicated[index].hit, candidate.hit);
270    }
271    deduplicated
272}

274fn deduplicate_exact_hits(
275    hits: Vec<CandidateSearchHit>,
276    prefer_structural: bool,
277) -> Vec<CandidateSearchHit> {
278    let mut deduplicated: Vec<CandidateSearchHit> = Vec::with_capacity(hits.len());
279    let mut positions = HashMap::new();
280    for mut candidate in hits {
281        let key = (
282            candidate.hit.path.clone(),
283            candidate.hit.start_line,
284            candidate.hit.end_line,
285            candidate.hit.content_hash.clone(),
286            candidate
287                .hit
288                .occurrence
289                .as_ref()
290                .map(|occurrence| (occurrence.start_byte, occurrence.end_byte)),
291        );
292        let Some(&index) = positions.get(&key) else {
293            positions.insert(key, deduplicated.len());
294            deduplicated.push(candidate);
295            continue;
296        };
297        if prefer_structural
298            && hit_has_kind(&candidate.hit, "symbol")
299            && !hit_has_kind(&deduplicated[index].hit, "symbol")
300        {
301            std::mem::swap(&mut candidate, &mut deduplicated[index]);
302        }
303        if deduplicated[index].definition.is_none() {
304            deduplicated[index].definition = candidate.definition.clone();
305        }
306        merge_search_hits(&mut deduplicated[index].hit, candidate.hit);
307    }
308    deduplicated
309}

311fn normalize_search_scores(hits: &mut [CandidateSearchHit]) {
312    let max_score = hits
313        .iter()
314        .map(|candidate| candidate.hit.score)
315        .filter(|score| score.is_finite())
316        .fold(0.0_f64, f64::max);
317    for candidate in hits {
318        candidate.hit.normalized_score = if max_score > 0.0 && candidate.hit.score.is_finite() {
319            (candidate.hit.score / max_score).clamp(0.0, 1.0)
320        } else {
321            0.0
322        };
323    }
324}

378fn group_search_hits(hits: &[SearchHit]) -> Vec<SearchGroup> {
379    let mut groups = Vec::<SearchGroup>::new();
380    let mut positions = HashMap::<String, usize>::new();
381    for hit in hits {
382        let key = grouped_search_key(hit);
383        let index = *positions.entry(key).or_insert_with(|| {
384            let index = groups.len();
385            groups.push(SearchGroup {
386                symbol: hit.symbol.clone().or_else(|| hit.enclosing_symbol.clone()),
387                definition: None,
388                representative: None,
389                references: Vec::new(),
390                text_matches: 0,
391                total_hits: 0,
392            });
393            index
394        });
395        let group = &mut groups[index];
396        group.total_hits = group.total_hits.saturating_add(1);
397        if hit_has_kind(hit, "text") || hit_has_kind(hit, "regex") {
398            group.text_matches = group.text_matches.saturating_add(1);
399        }
400
401        if hit.role == Some(ReferenceRole::Definition) {
402            if group.definition.is_none() {
403                group.definition = Some(grouped_search_evidence(hit));
404                group.representative = None;
405            }
406        } else if group.definition.is_none() && group.representative.is_none() {
407            group.representative = Some(grouped_search_evidence(hit));
408        }
409
410        if hit.role == Some(ReferenceRole::Reference) || hit_has_kind(hit, "reference") {
411            if let Some(reference) = group
412                .references
413                .iter_mut()
414                .find(|reference| reference.path == hit.path)
415            {
416                reference.count = reference.count.saturating_add(1);
417                reference.start_line = reference.start_line.min(hit.start_line);
418                reference.end_line = reference.end_line.max(hit.end_line);
419                if let Some(role) = hit.role
420                    && !reference.roles.contains(&role)
421                {
422                    reference.roles.push(role);
423                }
424            } else {
425                group.references.push(SearchReferenceGroup {
426                    path: hit.path.clone(),
427                    count: 1,
428                    start_line: hit.start_line,
429                    end_line: hit.end_line,
430                    roles: hit.role.into_iter().collect(),
431                });
432            }
433        }
434    }
435    groups
436}

438fn collect_filtered_hits<T>(
439    request: &SearchRequest,
440    max_candidates: usize,
441    cancellation: &CancellationToken,
442    mut fetch_page: impl FnMut(usize, usize) -> Result<Vec<T>>,
443    path: impl Fn(&T) -> &str,
444) -> Result<Vec<T>> {
445    let path_filter = PathFilter::new(&request.include_paths, &request.exclude_paths)?;
446    let mut selected = Vec::new();
447    let mut offset = 0usize;
448    while selected.len() < max_candidates && offset < MAX_FILTER_SCAN_ROWS {
449        check_cancelled(cancellation)?;
450        let page_limit = FILTER_SCAN_PAGE_SIZE.min(MAX_FILTER_SCAN_ROWS - offset);
451        let page = fetch_page(offset, page_limit)?;
452        let page_len = page.len();
453        for hit in page {
454            check_cancelled(cancellation)?;
455            if path_filter.allows(path(&hit)) {
456                selected.push(hit);
457                if selected.len() == max_candidates {
458                    break;
459                }
460            }
461        }
462        offset = offset.saturating_add(page_len);
463        if page_len < page_limit {
464            break;
465        }
466    }
467    Ok(selected)
468}

470pub(super) fn chunk_search_hit(
471    hit: &ChunkHit,
472    query: &str,
473    case_sensitive: bool,
474    context: usize,
475    compiled_matcher: Option<&regex::Regex>,
476    regex_match: bool,
477) -> Result<Option<SearchHit>> {
478    let byte_range = if let Some(regex) = compiled_matcher {
479        regex
480            .find(&hit.content)
481            .map(|matched| (matched.start(), matched.end()))
482    } else if case_sensitive {
483        hit.content
484            .find(query)
485            .map(|start| (start, start + query.len()))
486    } else {
487        regex::RegexBuilder::new(&regex::escape(query))
488            .case_insensitive(true)
489            .build()?
490            .find(&hit.content)
491            .map(|matched| (matched.start(), matched.end()))
492    };
493    let Some((start, end)) = byte_range else {
494        return Ok(None);
495    };
496    let starts = line_starts(&hit.content);
497    Ok(Some(chunk_search_hit_for_range(
498        hit,
499        start,
500        end,
501        context,
502        regex_match,
503        false,
504        &starts,
505    )))
506}

508fn chunk_search_hits(
509    hit: &ChunkHit,
510    query: &str,
511    case_sensitive: bool,
512    context: usize,
513    compiled_matcher: Option<&regex::Regex>,
514    regex_match: bool,
515    max_hits: usize,
516) -> Result<Vec<SearchHit>> {
517    let ranges: Box<dyn Iterator<Item = (usize, usize)> + '_> =
518        if let Some(regex) = compiled_matcher {
519            Box::new(
520                regex
521                    .find_iter(&hit.content)
522                    .map(|matched| (matched.start(), matched.end())),
523            )
524        } else if case_sensitive {
525            Box::new(
526                hit.content
527                    .match_indices(query)
528                    .map(|(start, matched)| (start, start + matched.len())),
529            )
530        } else {
531            Box::new(std::iter::empty())
532        };
533    let starts = line_starts(&hit.content);
534    let mut hits = Vec::new();
535    for (start, end) in ranges {
536        if hits.len() == max_hits {
537            return Err(Error::LimitExceeded);
538        }
539        hits.push(chunk_search_hit_for_range(
540            hit,
541            start,
542            end,
543            context,
544            regex_match,
545            true,
546            &starts,
547        ));
548    }
549    Ok(hits)
550}

552pub(super) fn chunk_search_hit_for_range(
553    hit: &ChunkHit,
554    start: usize,
555    end: usize,
556    context: usize,
557    regex_match: bool,
558    include_occurrence: bool,
559    line_starts: &[usize],
560) -> SearchHit {
561    let text_len = hit.content.len();
562    let local_start = byte_to_line(line_starts, text_len, start);
563    let local_end = if end == 0 || end == start {
564        local_start
565    } else {
566        byte_to_line(line_starts, text_len, end.saturating_sub(1))
567    };
568    let available_lines = line_starts.len().max(1);
569    let desired_start = local_start.saturating_sub(context).max(1);
570    let desired_end = local_end.saturating_add(context).min(available_lines);
571    let (excerpt_start, excerpt_end) =
572        anchored_line_window(desired_start, desired_end, local_start, local_end, 20);
573    let excerpt = excerpt(&hit.content, excerpt_start, excerpt_end);
574    SearchHit {
575        path: hit.path.clone(),
576        start_line: hit.start_line + excerpt_start - 1,
577        end_line: hit.start_line + excerpt_end - 1,
578        content_hash: hash(&excerpt),
579        excerpt,
580        match_kind: if regex_match {
581            "regex".into()
582        } else {
583            "text".into()
584        },
585        match_kinds: vec![if regex_match {
586            "regex".into()
587        } else {
588            "text".into()
589        }],
590        role: None,
591        symbol: None,
592        enclosing_symbol: None,
593        occurrence: include_occurrence.then_some(SearchOccurrence {
594            start_line: hit.start_line + local_start - 1,
595            end_line: hit.start_line + local_end - 1,
596            start_byte: hit.start_byte + start,
597            end_byte: hit.start_byte + end,
598        }),
599        score: 3.0 + (-hit.score).max(0.0) * 1_000_000.0,
600        normalized_score: 0.0,
601        score_reasons: vec![if regex_match {
602            "regex match".into()
603        } else {
604            "text match".into()
605        }],
606    }
607}

609pub(super) fn matching_line(
610    hit: &ChunkHit,
611    query: &str,
612    case_sensitive: bool,
613    compiled_regex: Option<&regex::Regex>,
614) -> Option<usize> {
615    if let Some(regex) = compiled_regex {
616        return regex.find(&hit.content).map(|matched| {
617            let (local_start, _) =
618                byte_range_to_line_range(&hit.content, matched.start(), matched.end());
619            hit.start_line + local_start - 1
620        });
621    }
622    if !case_sensitive {
623        return None;
624    }
625    hit.content
626        .lines()
627        .position(|line| line.contains(query))
628        .map(|offset| hit.start_line + offset)
629}

631fn apply_focus(hits: &mut [CandidateSearchHit], focus_paths: &[String]) -> Result<()> {
632    let focus_paths = PathMatcher::new(focus_paths)?;
633    for candidate in hits {
634        if focus_paths.is_match(&candidate.hit.path) {
635            candidate.hit.score += 2.0;
636            candidate.hit.score_reasons.push("focus path".into());
637        }
638    }
639    Ok(())
640}

662/// Compile a case-insensitive literal matcher for non-regex search modes.
663///
664/// In Auto/Text/Identifier modes the query is a literal string, not a regex
665/// pattern. Compile it once per request so every lexical hit reuses the same
666/// matcher. Case-sensitive search uses `str::find` without compilation.
667pub(super) fn compile_literal_regex(
668    query: &str,
669    case_sensitive: bool,
670) -> Result<Option<regex::Regex>> {
671    if case_sensitive {
672        return Ok(None);
673    }
674    Ok(Some(
675        regex::RegexBuilder::new(&regex::escape(query))
676            .case_insensitive(true)
677            .size_limit(1 << 20)
678            .dfa_size_limit(1 << 20)
679            .build()?,
680    ))
681}
682
683fn validate_search_input(request: &SearchRequest) -> Result<()> {
684    if request.query.trim().is_empty() {
685        return Err(Error::InvalidInput {
686            field: "search query",
687            reason: "must not be empty",
688        });
689    }
690    validate_input(&request.query, "search query", MAX_QUERY_BYTES)?;
691    validate_glob_patterns(&request.include_paths)?;
692    validate_glob_patterns(&request.exclude_paths)?;
693    validate_glob_patterns(&request.focus_paths)?;
694    validate_cursor(request.cursor.as_deref())?;
695    if request.all_occurrences && !matches!(request.mode, SearchMode::Text | SearchMode::Regex) {
696        return Err(Error::InvalidInput {
697            field: "all occurrences",
698            reason: "requires text or regex mode",
699        });
700    }
701    if request.prefer_structural
702        && !matches!(request.mode, SearchMode::Auto | SearchMode::Identifier)
703    {
704        return Err(Error::InvalidInput {
705            field: "prefer structural",
706            reason: "requires auto or identifier mode",
707        });
708    }
709    if matches!(request.mode, SearchMode::Regex) {
710        compile_regex(request)?;
711    } else {
712        compile_literal_regex(&request.query, request.case_sensitive)?;
713    }
714    Ok(())
715}

718    fn ensure_search_page_fits(
719        &self,
720        selected: &mut [CandidateSearchHit],
721        shape: SearchResponseShape<'_>,
722        options: ServiceCallOptions,
723    ) -> Result<()> {
724        let provisional = |selected: &[CandidateSearchHit]| SearchResponse {
725            hits: selected
726                .iter()
727                .map(|candidate| candidate.hit.clone())
728                .collect(),
729            coverage: search_coverage(shape.all, selected),
730            occurrences_returned: selected.len(),
731            occurrences_total: shape
732                .request
733                .all_occurrences
734                .then_some(shape.total_candidates),
735            meta: self.meta(
736                shape.generation,
737                selected
738                    .iter()
739                    .map(|candidate| self.config.tokenizer.count(&candidate.hit.excerpt))
740                    .sum(),
741                shape
742                    .has_more
743                    .then(|| make_cursor(shape.generation, shape.offset + shape.consumed)),
744            ),
745        };
746        let mut sized = provisional(selected);
747        if self.response_fits_with_receipt_reserve(&sized, selected.len(), options)? {
748            return Ok(());
749        }
750        for candidate in selected.iter_mut() {
751            candidate.hit.score_reasons.clear();
752        }
753        sized = provisional(selected);
754        if self.response_fits_with_receipt_reserve(&sized, selected.len(), options)? {
755            return Ok(());
756        }
757        Err(Error::RequestLimitExceeded {
758            field: "max_response_tokens",
759            requested: self.finalized_response_tokens(&sized)?,
760            limit: options
761                .max_response_tokens()
762                .expect("fitting only runs with a response limit"),
763        })
764    }

899    async fn search_grouped_cancellable_with_options(
900        &self,
901        request: SearchRequest,
902        options: ServiceCallOptions,
903        cancellation: CancellationToken,
904    ) -> Result<SearchGroupedResponse> {
905        let operation = TokenAccountingOperation::Search;
906        self.observe_service_result(operation, self.validate_call_options(options))?;
907        let this = self.clone();
908        let result = self
909            .blocking_executor
910            .run(cancellation, move |cancellation| {
911                let response = this
912                    .search_sync(
913                        request,
914                        cancellation,
915                        RegexPlanning::Enabled,
916                        SearchDiagnostics::Omit,
917                        ServiceCallOptions::new(),
918                        false,
919                    )?
920                    .response;
921                let hits_returned = response.hits.len();
922                let groups = group_search_hits(&response.hits);
923                let source_tokens = groups
924                    .iter()
925                    .filter_map(|group| group.definition.as_ref().or(group.representative.as_ref()))
926                    .filter_map(|evidence| evidence.excerpt.as_deref())
927                    .map(|excerpt| this.config.tokenizer.count(excerpt))
928                    .sum();
929                let mut meta = response.meta;
930                meta.source_tokens = source_tokens;
931                meta.emitted_tokens = source_tokens;
932                let mut compact = SearchGroupedResponse {
933                    groups_returned: groups.len(),
934                    groups,
935                    coverage: response.coverage,
936                    hits_returned,
937                    occurrences_total: response.occurrences_total,
938                    meta,
939                };
940                this.finalize_bounded_response(&mut compact, options)?;
941                this.record_token_savings(TokenAccountingOperation::Search, None, &compact.meta);
942                Ok(compact)
943            })
944            .await;
945        self.observe_service_result(operation, result)
946    }

990
991    fn search_sync(
992        &self,
993        request: SearchRequest,
994        cancellation: &CancellationToken,
995        regex_planning: RegexPlanning,
996        diagnostics: SearchDiagnostics,
997        options: ServiceCallOptions,
998        record_savings: bool,
999    ) -> Result<SearchEvaluation> {
1000        check_cancelled(cancellation)?;
1001        validate_search_input(&request)?;
1002        let regex = matches!(request.mode, SearchMode::Regex)
1003            .then(|| compile_regex(&request))
1004            .transpose()?;
1005        let literal_regex = if !matches!(request.mode, SearchMode::Regex) {
1006            compile_literal_regex(&request.query, request.case_sensitive)?
1007        } else {
1008            None
1009        };
1010        let occurrence_literal_regex = (request.all_occurrences
1011            && matches!(request.mode, SearchMode::Text))
1012        .then(|| compile_occurrence_literal_regex(&request))
1013        .transpose()?;
1014        let limit = self.result_limit(request.max_results)?;
1015        let token_limit = self.token_limit(request.max_tokens, self.config.default_read_tokens)?;
1016        let context_lines = self.context_line_limit(request.context_lines)?;
1017        let search_result = self.consistent(|session, generation| {
1018            let offset = parse_cursor(request.cursor.as_deref(), generation)?;
1019            let mut hits = Vec::new();
1020            let mut phases = SearchPhaseCounters::default();
1021            let mut primitive_keys = Vec::new();
1022            if matches!(
1023                request.mode,
1024                SearchMode::Auto | SearchMode::Identifier | SearchMode::Symbol
1025            ) {
1026                let symbol_hits = collect_filtered_hits(
1027                    &request,
1028                    limit.saturating_mul(4),
1029                    cancellation,
1030                    |offset, page_limit| {
1031                        session.search_symbols_page(
1032                            &request.query,
1033                            request.case_sensitive,
1034                            page_limit,
1035                            offset,
1036                        )
1037                    },
1038                    |hit: &SymbolHit| &hit.path,
1039                )?;
1040                let excerpt_requests = symbol_hits
1041                    .iter()
1042                    .map(|hit| StoredExcerptRequest {
1043                        file_id: hit.symbol.file_id,
1044                        desired_start_line: hit
1045                            .symbol
1046                            .start_line
1047                            .saturating_sub(context_lines)
1048                            .max(1),
1049                        desired_end_line: hit.symbol.end_line.saturating_add(context_lines),
1050                        required_start_line: hit.symbol.start_line,
1051                        required_end_line: hit.symbol.start_line,
1052                        max_lines: 30,
1053                    })
1054                    .collect::<Vec<_>>();
1055                for (hit, excerpt) in symbol_hits
1056                    .into_iter()
1057                    .zip(self.stored_excerpts(session, &excerpt_requests)?)
1058                {
1059                    if let Some(excerpt) = excerpt {
1060                        let definition = DefinitionIdentity {
1061                            path: hit.path.clone(),
1062                            start_line: hit.symbol.start_line,
1063                            end_line: hit.symbol.end_line,
1064                        };
1065                        hits.push(CandidateSearchHit {
1066                            hit: self.symbol_search_hit(hit, &request.query, excerpt),
1067                            definition: Some(definition),
1068                        });
1069                    }
1070                }
1071            }
1072            if matches!(
1073                request.mode,
1074                SearchMode::Auto | SearchMode::Identifier | SearchMode::Reference
1075            ) {
1076                let reference_hits = collect_filtered_hits(
1077                    &request,
1078                    limit.saturating_mul(4),
1079                    cancellation,
1080                    |offset, page_limit| {
1081                        session.search_references_page(
1082                            &request.query,
1083                            request.case_sensitive,
1084                            page_limit,
1085                            offset,
1086                        )
1087                    },
1088                    |hit: &ReferenceHit| &hit.path,
1089                )?;
1090                let excerpt_requests = reference_hits
1091                    .iter()
1092                    .map(|hit| StoredExcerptRequest {
1093                        file_id: hit.reference.file_id,
1094                        desired_start_line: hit
1095                            .reference
1096                            .start_line
1097                            .saturating_sub(context_lines)
1098                            .max(1),
1099                        desired_end_line: hit.reference.end_line.saturating_add(context_lines),
1100                        required_start_line: hit.reference.start_line,
1101                        required_end_line: hit.reference.end_line,
1102                        max_lines: 12,
1103                    })
1104                    .collect::<Vec<_>>();
1105                for (hit, excerpt) in reference_hits
1106                    .into_iter()
1107                    .zip(self.stored_excerpts(session, &excerpt_requests)?)
1108                {
1109                    if let Some(excerpt) = excerpt {
1110                        hits.push(CandidateSearchHit {
1111                            hit: self.reference_search_hit(hit, &request.query, excerpt),
1112                            definition: None,
1113                        });
1114                    }
1115                }
1116            }
1117
1118            let lexical = match request.mode {
1119                SearchMode::Regex => {
1120                    let scan = self.regex_hits(
1121                        session,
1122                        &request,
1123                        regex.as_ref().expect("regex mode compiles a pattern"),
1124                        (!request.all_occurrences).then_some(limit.saturating_mul(20)),
1125                        cancellation,
1126                        regex_planning,
1127                    )?;
1128                    phases = scan.phases;
1129                    let primitive_kind = match phases.regex_candidate_strategy {
1130                        RegexCandidateStrategy::Trigram => "regex_trigram_candidates",
1131                        RegexCandidateStrategy::FullScan => "regex_full_scan",
1132                    };
1133                    if diagnostics == SearchDiagnostics::Collect {
1134                        primitive_keys.push(retrieval_primitive_key(
1135                            generation,
1136                            primitive_kind,
1137                            &format!(
1138                                "case_sensitive:{}:include:{:?}:exclude:{:?}:query:{}",
1139                                request.case_sensitive,
1140                                request.include_paths,
1141                                request.exclude_paths,
1142                                request.query
1143                            ),
1144                        ));
1145                    }
1146                    scan.hits
1147                }
1148                SearchMode::Text if request.all_occurrences => {
1149                    let scan = self.regex_hits(
1150                        session,
1151                        &request,
1152                        occurrence_literal_regex
1153                            .as_ref()
1154                            .expect("exhaustive text mode compiles a literal pattern"),
1155                        None,
1156                        cancellation,
1157                        RegexPlanning::Disabled,
1158                    )?;
1159                    phases = scan.phases;
1160                    scan.hits
1161                }
1162                SearchMode::Text | SearchMode::Auto | SearchMode::Identifier => {
1163                    let fetch_page = |offset, page_limit| {
1164                        if matches!(request.mode, SearchMode::Identifier)
1165                            || request.query.chars().count() < 3
1166                        {
1167                            session.search_word_page(&fts_quote(&request.query), page_limit, offset)
1168                        } else {
1169                            session.search_trigram_page(&request.query, page_limit, offset)
1170                        }
1171                    };
1172                    collect_filtered_hits(
1173                        &request,
1174                        limit.saturating_mul(8),
1175                        cancellation,
1176                        fetch_page,
1177                        |hit: &ChunkHit| &hit.path,
1178                    )?
1179                }
1180                SearchMode::Symbol | SearchMode::Reference => Vec::new(),
1181            };
1182            let mut lexical_hits = Vec::new();
1183            for hit in lexical {
1184                check_cancelled(cancellation)?;
1185                let chunk_hits = if request.all_occurrences {
1186                    chunk_search_hits(
1187                        &hit,
1188                        &request.query,
1189                        request.case_sensitive,
1190                        context_lines,
1191                        regex
1192                            .as_ref()
1193                            .or(occurrence_literal_regex.as_ref())
1194                            .or(literal_regex.as_ref()),
1195                        matches!(request.mode, SearchMode::Regex),
1196                        MAX_EXHAUSTIVE_OCCURRENCES.saturating_sub(lexical_hits.len()),
1197                    )?
1198                } else {
1199                    chunk_search_hit(
1200                        &hit,
1201                        &request.query,
1202                        request.case_sensitive,
1203                        context_lines,
1204                        regex.as_ref().or(literal_regex.as_ref()),
1205                        matches!(request.mode, SearchMode::Regex),
1206                    )?
1207                    .into_iter()
1208                    .collect()
1209                };
1210                for search_hit in chunk_hits {
1211                    if request.all_occurrences && lexical_hits.len() == MAX_EXHAUSTIVE_OCCURRENCES {
1212                        return Err(Error::LimitExceeded);
1213                    }
1214                    let matched_line = search_hit
1215                        .occurrence
1216                        .as_ref()
1217                        .map(|occurrence| occurrence.start_line)
1218                        .or_else(|| {
1219                            matching_line(
1220                                &hit,
1221                                &request.query,
1222                                request.case_sensitive,
1223                                regex.as_ref().or(literal_regex.as_ref()),
1224                            )
1225                        })
1226                        .unwrap_or(search_hit.start_line);
1227                    lexical_hits.push((hit.file_id, search_hit, matched_line));
1228                }
1229            }
1230            let lexical_locations = lexical_hits
1231                .iter()
1232                .map(|(file_id, _, matched_line)| (*file_id, *matched_line))
1233                .collect::<Vec<_>>();
1234            let mut enclosing = Vec::with_capacity(lexical_locations.len());
1235            for locations in lexical_locations.chunks(512) {
1236                check_cancelled(cancellation)?;
1237                enclosing.extend(session.find_enclosing_symbols_batch(locations)?);
1238            }
1239            for ((_, mut hit, _), symbol) in lexical_hits.into_iter().zip(enclosing) {
1240                let definition = symbol.map(|symbol| {
1241                    hit.enclosing_symbol = Some(symbol.name);
1242                    DefinitionIdentity {
1243                        path: hit.path.clone(),
1244                        start_line: symbol.start_line,
1245                        end_line: symbol.end_line,
1246                    }
1247                });
1248                hits.push(CandidateSearchHit { hit, definition });
1249            }
1250
1251            apply_focus(&mut hits, &request.focus_paths)?;
1252            hits.sort_by(|left, right| {
1253                right
1254                    .hit
1255                    .score
1256                    .total_cmp(&left.hit.score)
1257                    .then_with(|| left.hit.path.cmp(&right.hit.path))
1258                    .then_with(|| left.hit.start_line.cmp(&right.hit.start_line))
1259                    .then_with(|| {
1260                        left.hit
1261                            .occurrence
1262                            .as_ref()
1263                            .map(|occurrence| occurrence.start_byte)
1264                            .cmp(
1265                                &right
1266                                    .hit
1267                                    .occurrence
1268                                    .as_ref()
1269                                    .map(|occurrence| occurrence.start_byte),
1270                            )
1271                    })
1272            });
1273            hits = deduplicate_exact_hits(hits, request.prefer_structural);
1274            if matches!(request.mode, SearchMode::Auto | SearchMode::Identifier) {
1275                hits = deduplicate_definition_channels(hits, request.prefer_structural);
1276            }
1277            normalize_search_scores(&mut hits);
1278
1279            let mut emitted_tokens = 0usize;
1280            let mut selected = Vec::new();
1281            let total_candidates = hits.len();
1282            let page = hits.iter().skip(offset).take(limit).cloned();
1283            let mut consumed = 0usize;
1284            for candidate in page {
1285                check_cancelled(cancellation)?;
1286                consumed += 1;
1287                let count = self.config.tokenizer.count(&candidate.hit.excerpt);
1288                if emitted_tokens.saturating_add(count) > token_limit {
1289                    continue;
1290                }
1291                emitted_tokens += count;
1292                selected.push(candidate);
1293            }
1294            let has_more = offset.saturating_add(consumed) < total_candidates;
1295            self.ensure_search_page_fits(
1296                &mut selected,
1297                SearchResponseShape {
1298                    all: &hits,
1299                    request: &request,
1300                    generation,
1301                    total_candidates,
1302                    offset,
1303                    consumed,
1304                    has_more,
1305                },
1306                options,
1307            )?;
1308            let receipt_candidates = selected
1309                .iter()
1310                .map(|candidate| {
1311                    ReceiptEvidence::new(
1312                        candidate.hit.path.clone(),
1313                        candidate.hit.start_line,
1314                        candidate.hit.end_line,
1315                        candidate.hit.content_hash.clone(),
1316                        Some(&candidate.hit.excerpt),
1317                    )
1318                })
1319                .collect::<Vec<_>>();
1320            let receipt = self.evaluate_receipt(
1321                request.receipt_id.as_deref(),
1322                generation,
1323                &receipt_candidates,
1324            )?;
1325            selected = selected
1326                .into_iter()
1327                .zip(&receipt.decisions)
1328                .filter_map(|(candidate, decision)| {
1329                    matches!(
1330                        decision,
1331                        ReceiptDecision::Return | ReceiptDecision::ReturnNearDuplicate
1332                    )
1333                    .then_some(candidate)
1334                })
1335                .collect();
1336            emitted_tokens = selected
1337                .iter()
1338                .map(|candidate| self.config.tokenizer.count(&candidate.hit.excerpt))
1339                .sum();
1340            let paths = selected
1341                .iter()
1342                .map(|candidate| candidate.hit.path.clone())
1343                .collect::<HashSet<_>>()
1344                .into_iter()
1345                .collect::<Vec<_>>();
1346            let occurrences_returned = selected.len();
1347            let coverage = search_coverage(&hits, &selected);
1348            let selected = selected
1349                .into_iter()
1350                .map(|candidate| candidate.hit)
1351                .collect();
1352            let baseline_source_tokens =
1353                session.whole_file_source_tokens(&paths, self.config.tokenizer.name())?;
1354            let mut response = SearchResponse {
1355                hits: selected,
1356                coverage,
1357                occurrences_returned,
1358                occurrences_total: request.all_occurrences.then_some(total_candidates),
1359                meta: self.meta(
1360                    generation,
1361                    emitted_tokens,
1362                    has_more.then(|| make_cursor(generation, offset + consumed)),
1363                ),
1364            };
1365            receipt.apply_meta(&mut response.meta);
1366            Ok((response, baseline_source_tokens, phases, primitive_keys))
1367        });
1368        let (mut response, baseline_source_tokens, phases, primitive_keys) = search_result?;
1369        self.finalize_bounded_response(&mut response, options)?;
1370        if record_savings {
1371            self.record_token_savings(
1372                TokenAccountingOperation::Search,
1373                baseline_source_tokens,
1374                &response.meta,
1375            );
1376        }
1377        Ok(SearchEvaluation {
1378            response,
1379            phases,
1380            primitive_keys,
1381        })
1382    }

1437    fn regex_hits(
1438        &self,
1439        session: &ReadSession,
1440        request: &SearchRequest,
1441        regex: &regex::Regex,
1442        max_candidates: Option<usize>,
1443        cancellation: &CancellationToken,
1444        planning: RegexPlanning,
1445    ) -> Result<RegexScan> {
1446        // Hard caps prevent repository-wide lexical scans from running
1447        // unbounded. Exhaustive modes lift only the candidate-chunk cap and
1448        // fail explicitly if another cap would make the result incomplete.
1449        let max_candidates = max_candidates.map(|limit| limit.min(MAX_REGEX_CANDIDATES));
1450        let path_filter = PathFilter::new(&request.include_paths, &request.exclude_paths)?;
1451        let has_path_filters =
1452            !request.include_paths.is_empty() || !request.exclude_paths.is_empty();
1453        let file_count = session.file_count()?;
1454        let plan = (planning == RegexPlanning::Enabled)
1455            .then(|| regex_candidate_plan(request))
1456            .flatten();
1457        if let Some(plan) = plan {
1458            return self.regex_candidate_hits(
1459                session,
1460                regex,
1461                max_candidates,
1462                cancellation,
1463                path_filter,
1464                has_path_filters,
1465                file_count,
1466                plan,
1467            );
1468        }
1469
1470        if file_count > MAX_REGEX_FILES_SCANNED {
1471            return Err(Error::LimitExceeded);
1472        }
1473        let files = session.regex_scan_files(MAX_REGEX_FILES_SCANNED)?;
1474        for (file, chunk_count) in &files {
1475            check_cancelled(cancellation)?;
1476            if path_filter.allows(&file.path) && *chunk_count > MAX_REGEX_CHUNKS_PER_FILE {
1477                return Err(Error::LimitExceeded);
1478            }
1479        }
1480        let mut hits = Vec::new();
1481        let mut phases = SearchPhaseCounters {
1482            regex_candidate_strategy: RegexCandidateStrategy::FullScan,
1483            regex_files_considered: files.len(),
1484            ..SearchPhaseCounters::default()
1485        };
1486        for (file, _) in files {
1487            check_cancelled(cancellation)?;
1488            if !path_filter.allows(&file.path) {
1489                continue;
1490            }
1491            let chunks = session.get_chunks_for_file(file.id, MAX_REGEX_CHUNKS_PER_FILE)?;
1492            for chunk in chunks {
1493                check_cancelled(cancellation)?;
1494                phases.regex_chunks_loaded = phases.regex_chunks_loaded.saturating_add(1);
1495                phases.regex_chunks_verified = phases.regex_chunks_verified.saturating_add(1);
1496                if regex.is_match(&chunk.content) {
1497                    if max_candidates.is_some_and(|limit| hits.len() == limit) {
1498                        return Err(Error::LimitExceeded);
1499                    }
1500                    hits.push(ChunkHit {
1501                        chunk_id: chunk.id,
1502                        file_id: chunk.file_id,
1503                        path: file.path.clone(),
1504                        content: chunk.content,
1505                        start_line: chunk.start_line,
1506                        end_line: chunk.end_line,
1507                        start_byte: chunk.start_byte,
1508                        end_byte: chunk.end_byte,
1509                        token_count: chunk.token_count,
1510                        generation: file.generation,
1511                        score: 0.0,
1512                    });
1513                }
1514            }
1515        }
1516        Ok(RegexScan { hits, phases })
1517    }
1518
1519    #[allow(clippy::too_many_arguments)]
1520    fn regex_candidate_hits(
1521        &self,
1522        session: &ReadSession,
1523        regex: &regex::Regex,
1524        max_candidates: Option<usize>,
1525        cancellation: &CancellationToken,
1526        path_filter: PathFilter,
1527        has_path_filters: bool,
1528        files_considered: usize,
1529        plan: RegexCandidatePlan,
1530    ) -> Result<RegexScan> {
1531        let mut phases = SearchPhaseCounters {
1532            regex_candidate_strategy: RegexCandidateStrategy::Trigram,
1533            regex_plan_terms: plan.term_count,
1534            regex_files_considered: files_considered,
1535            ..SearchPhaseCounters::default()
1536        };
1537        let query = plan.expression.fts_query();
1538        if has_path_filters {
1539            let candidate_ids = session.select_scoped_regex_candidate_ids(
1540                &query,
1541                MAX_SCOPED_REGEX_ROWS_SCANNED,
1542                MAX_REGEX_CANDIDATE_CHUNKS,
1543                |path| path_filter.allows(path),
1544            )?;
1545            phases.regex_candidate_chunks = candidate_ids.len();
1546            let mut hits = Vec::new();
1547            for candidate_batch in candidate_ids.chunks(REGEX_CANDIDATE_PAGE_SIZE) {
1548                check_cancelled(cancellation)?;
1549                for hit in session.regex_candidates_by_ids(candidate_batch)? {
1550                    check_cancelled(cancellation)?;
1551                    phases.regex_chunks_verified = phases.regex_chunks_verified.saturating_add(1);
1552                    if regex.is_match(&hit.content) {
1553                        if max_candidates.is_some_and(|limit| hits.len() == limit) {
1554                            return Err(Error::LimitExceeded);
1555                        }
1556                        hits.push(hit);
1557                    }
1558                }
1559            }
1560            return Ok(RegexScan { hits, phases });
1561        }
1562        let candidate_count = session
1563            .regex_candidate_count_up_to(&query, MAX_REGEX_CANDIDATE_CHUNKS.saturating_add(1))?;
1564        if candidate_count > MAX_REGEX_CANDIDATE_CHUNKS {
1565            return Err(Error::LimitExceeded);
1566        }
1567        phases.regex_candidate_chunks = candidate_count;
1568        let mut hits = Vec::new();
1569        let mut offset = 0usize;
1570        loop {
1571            check_cancelled(cancellation)?;
1572            let page =
1573                session.search_regex_candidates_page(&query, REGEX_CANDIDATE_PAGE_SIZE, offset)?;
1574            let page_len = page.len();
1575            if page_len == 0 {
1576                break;
1577            }
1578            for hit in page {
1579                check_cancelled(cancellation)?;
1580                if !path_filter.allows(&hit.path) {
1581                    continue;
1582                }
1583                phases.regex_chunks_verified = phases.regex_chunks_verified.saturating_add(1);
1584                if regex.is_match(&hit.content) {
1585                    if max_candidates.is_some_and(|limit| hits.len() == limit) {
1586                        return Err(Error::LimitExceeded);
1587                    }
1588                    hits.push(hit);
1589                }
1590            }
1591            offset = offset.saturating_add(page_len);
1592            if page_len < REGEX_CANDIDATE_PAGE_SIZE {
1593                break;
1594            }
1595        }
1596        Ok(RegexScan { hits, phases })
1597    }
1598}
