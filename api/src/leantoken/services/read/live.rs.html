1// Re-tokenize bounded candidate windows instead of guessing a byte/token ratio.
2// The hard cap prevents pathological low-token inputs from growing forever.
3pub(super) const LIVE_READ_TOKEN_CHECK_BYTES: usize = 64 * 1024;
4pub(super) const MAX_LIVE_READ_BYTES: usize = 8 * 1024 * 1024;

6pub(in crate::services) fn open_live_file(services: &Services, path: &str) -> Result<File> {
7    services
8        .repository_root
9        .open(path)
10        .map(cap_std::fs::File::into_std)
11        .map_err(|open_error| {
12            // The capability open is authoritative for access. Canonicalization is
13            // only used after refusal to preserve the public escape classification.
14            match resolve_existing(&services.config.root, path) {
15                Err(Error::PathOutsideRoot(external)) => Error::PathOutsideRoot(external),
16                _ => Error::Io(open_error),
17            }
18        })
19}

21pub(super) fn resolve_read_target(
22    session: &ReadSession,
23    file_id: i64,
24    request: &ReadRequest,
25    generation: u64,
26) -> Result<ResolvedReadTarget> {
27    if let Some(cursor) = request.continuation_cursor.as_deref() {
28        let cursor = parse_read_cursor(cursor, generation, &request.path)?;
29        if cursor.full != matches!(request.policy, ReadPolicy::Full) {
30            return Err(Error::StaleCursor);
31        }
32        return Ok(ResolvedReadTarget {
33            target_start_line: cursor.target_start_line,
34            target_end_line: cursor.target_end_line,
35            page_start_line: cursor.next_start_line,
36            page_start_byte: cursor.next_byte,
37            expected_full_hash: cursor.full_hash,
38            expected_file_size: Some(cursor.file_size),
39            expected_modified_ns: cursor.modified_ns,
40            cursor_full: cursor.full,
41        });
42    }
43
44    let (target_start_line, target_end_line) = if let Some(symbol_name) = &request.symbol {
45        let symbol = match session.find_symbol(file_id, symbol_name)? {
46            crate::symbol_identity::SymbolResolution::Unique(symbol) => symbol,
47            crate::symbol_identity::SymbolResolution::NotFound => {
48                return Err(Error::SymbolNotFound {
49                    path: request.path.clone(),
50                    symbol: symbol_name.clone(),
51                });
52            }
53            crate::symbol_identity::SymbolResolution::Ambiguous => {
54                return Err(Error::AmbiguousSymbol {
55                    path: request.path.clone(),
56                    symbol: symbol_name.clone(),
57                });
58            }
59        };
60        (symbol.start_line, Some(symbol.end_line))
61    } else if let Some(heading_name) = &request.heading {
62        let occurrence = request.heading_occurrence.unwrap_or(1);
63        let heading = session
64            .find_document_heading(file_id, heading_name, occurrence)?
65            .ok_or_else(|| Error::HeadingNotFound {
66                path: request.path.clone(),
67                heading: heading_name.clone(),
68                occurrence,
69            })?;
70        (heading.start_line, Some(heading.end_line))
71    } else {
72        (request.start_line.unwrap_or(1), request.end_line)
73    };
74
75    if target_start_line == 0
76        || target_end_line.is_some_and(|end_line| end_line < target_start_line)
77    {
78        return Err(invalid_line_range());
79    }
80    Ok(ResolvedReadTarget {
81        target_start_line,
82        target_end_line,
83        page_start_line: target_start_line,
84        page_start_byte: 0,
85        expected_full_hash: None,
86        expected_file_size: None,
87        expected_modified_ns: None,
88        cursor_full: matches!(request.policy, ReadPolicy::Full),
89    })
90}

92fn modified_ns(file: &File) -> Result<Option<u128>> {
93    Ok(file
94        .metadata()?
95        .modified()
96        .ok()
97        .and_then(|time| time.duration_since(std::time::UNIX_EPOCH).ok())
98        .map(|duration| duration.as_nanos()))
99}

101/// Stream a file without loading it into memory and capture its live identity.
102pub(super) fn stream_snapshot(file: &File) -> Result<LiveFileSnapshot> {
103    let mut file = file.try_clone()?;
104    let file_size = file.metadata()?.len().try_into().unwrap_or(usize::MAX);
105    let file_modified_ns = modified_ns(&file)?;
106    file.seek(SeekFrom::Start(0))?;
107    let mut reader = BufReader::new(file);
108    let mut hasher = blake3::Hasher::new();
109    let mut buf = [0u8; 65_536];
110    let mut bytes_seen = 0usize;
111    let mut newline_count = 0usize;
112    let mut last_byte_was_newline = false;
113    loop {
114        let n = reader.read(&mut buf)?;
115        if n == 0 {
116            break;
117        }
118        hasher.update(&buf[..n]);
119        bytes_seen = bytes_seen.saturating_add(n);
120        newline_count =
121            newline_count.saturating_add(buf[..n].iter().filter(|byte| **byte == b'\n').count());
122        last_byte_was_newline = buf[n - 1] == b'\n';
123    }
124    let end_line = if bytes_seen == 0 {
125        1
126    } else {
127        newline_count.saturating_add(usize::from(!last_byte_was_newline))
128    };
129    Ok(LiveFileSnapshot {
130        content_hash: Some(
131            hasher.finalize().to_hex()[..crate::text::CONTENT_FINGERPRINT_HEX_LEN].to_string(),
132        ),
133        end_line,
134        bytes_read: bytes_seen,
135        file_size,
136        modified_ns: file_modified_ns,
137    })
138}

140/// Hash the live file and read a resolved range in one forward stream.
141///
142/// When `full` is true the complete file is hashed and `content_hash` is
143/// populated. When `full` is false the stream stops as soon as the target
144/// range is satisfied or the token bound is reached, `content_hash` is `None`,
145/// and `bytes_read` reflects only the bytes consumed before early termination.
146pub(super) fn observe_live_range(
147    file: &File,
148    target_start_line: usize,
149    target_end_line: Option<usize>,
150    page_start_byte: usize,
151    max_tokens: usize,
152    tokenizer: crate::tokens::Tokenizer,
153    full: bool,
154) -> Result<LiveReadObservation> {
155    let mut file = file.try_clone()?;
156    let file_size = file.metadata()?.len().try_into().unwrap_or(usize::MAX);
157    let file_modified_ns = modified_ns(&file)?;
158    file.seek(SeekFrom::Start(0))?;
159    let mut reader = BufReader::new(file);
160    let mut hasher = blake3::Hasher::new();
161    let mut selected = Vec::with_capacity(LIVE_READ_TOKEN_CHECK_BYTES);
162    let mut current_line = 1usize;
163    let mut target_finished = false;
164    let mut token_bound_reached = false;
165    let mut final_target_checked = false;
166    let mut target_bytes = 0usize;
167    let mut page_start_line = target_start_line;
168    let mut next_token_check = LIVE_READ_TOKEN_CHECK_BYTES;
169    let mut utf8_pending = Vec::new();
170    let mut bytes_seen = 0usize;
171    let mut newline_count = 0usize;
172    let mut last_byte_was_newline = false;
173    let requested_end_line = target_end_line.unwrap_or(usize::MAX);
174
175    loop {
176        let buffer = reader.fill_buf()?;
177        if buffer.is_empty() {
178            break;
179        }
180        let mut validation_chunk = Vec::new();
181        let mut consumed = 0usize;
182        if !target_finished {
183            for &byte in buffer {
184                consumed = consumed.saturating_add(1);
185                let in_target =
186                    current_line >= target_start_line && current_line <= requested_end_line;
187                if in_target {
188                    validation_chunk.push(byte);
189                    if target_bytes < page_start_byte {
190                        if byte == b'\n' {
191                            page_start_line = current_line.saturating_add(1);
192                        }
193                    } else if !token_bound_reached {
194                        selected.push(byte);
195                    }
196                    target_bytes = target_bytes.saturating_add(1);
197                }
198                if byte == b'\n' {
199                    if requested_end_line == current_line {
200                        target_finished = true;
201                        break;
202                    }
203                    current_line = current_line.saturating_add(1);
204                }
205            }
206        }
207        // For full reads, always consume the entire buffer so the complete file
208        // is hashed. For bounded reads, consume only what the target scan used.
209        if full && consumed < buffer.len() {
210            consumed = buffer.len();
211        }
212        if consumed == 0 {
213            consumed = buffer.len();
214        }
215        let partial_buffer = consumed < buffer.len();
216        let consumed_buffer = &buffer[..consumed];
217        if full {
218            hasher.update(consumed_buffer);
219        }
220        bytes_seen = bytes_seen.saturating_add(consumed);
221        newline_count = newline_count.saturating_add(
222            consumed_buffer
223                .iter()
224                .filter(|byte| **byte == b'\n')
225                .count(),
226        );
227        last_byte_was_newline = consumed_buffer.last() == Some(&b'\n');
228        let final_chunk = target_finished || consumed == 0 || partial_buffer;
229        reader.consume(consumed);
230        validate_utf8_chunk(&mut utf8_pending, &validation_chunk, final_chunk)?;
231
232        if !token_bound_reached
233            && ((target_finished && !final_target_checked)
234                || selected.len() >= next_token_check
235                || selected.len() >= MAX_LIVE_READ_BYTES)
236        {
237            match std::str::from_utf8(&selected) {
238                Ok(content) if tokenizer.count(content) > max_tokens => {
239                    token_bound_reached = true;
240                    final_target_checked = target_finished;
241                }
242                Ok(_) => {
243                    if selected.len() >= MAX_LIVE_READ_BYTES {
244                        return Err(Error::LimitExceeded);
245                    }
246                    next_token_check = selected.len().saturating_add(LIVE_READ_TOKEN_CHECK_BYTES);
247                    final_target_checked = target_finished;
248                }
249                Err(error) if error.error_len().is_none() => {
250                    if target_finished || selected.len() >= MAX_LIVE_READ_BYTES {
251                        return Err(Error::InvalidInput {
252                            field: "path",
253                            reason: "must identify UTF-8 text",
254                        });
255                    }
256                }
257                Err(_) => {
258                    return Err(Error::InvalidInput {
259                        field: "path",
260                        reason: "must identify UTF-8 text",
261                    });
262                }
263            }
264        }
265        // Bounded reads stop as soon as the target is finished or the token
266        // bound is reached. Full reads continue to EOF to hash the complete
267        // file.
268        if !full && (target_finished || token_bound_reached) {
269            break;
270        }
271    }
272
273    validate_utf8_chunk(&mut utf8_pending, &[], true)?;
274    if !utf8_pending.is_empty() {
275        return Err(Error::InvalidInput {
276            field: "path",
277            reason: "must identify UTF-8 text",
278        });
279    }
280
281    if page_start_byte > target_bytes {
282        return Err(Error::StaleCursor);
283    }
284    let content = String::from_utf8(selected).map_err(|_| Error::InvalidInput {
285        field: "path",
286        reason: "must identify UTF-8 text",
287    })?;
288    let end_line = if bytes_seen == 0 {
289        1
290    } else {
291        newline_count.saturating_add(usize::from(!last_byte_was_newline))
292    };
293    Ok(LiveReadObservation {
294        snapshot: LiveFileSnapshot {
295            content_hash: full.then(|| {
296                hasher.finalize().to_hex()[..crate::text::CONTENT_FINGERPRINT_HEX_LEN].to_string()
297            }),
298            end_line,
299            bytes_read: bytes_seen,
300            file_size,
301            modified_ns: file_modified_ns,
302        },
303        range: LiveReadRange {
304            content,
305            page_start_line,
306            target_bytes,
307        },
308    })
309}

311pub(super) fn validate_utf8_chunk(
312    pending: &mut Vec<u8>,
313    bytes: &[u8],
314    final_chunk: bool,
315) -> Result<()> {
316    pending.extend_from_slice(bytes);
317    match std::str::from_utf8(pending) {
318        Ok(_) => {
319            pending.clear();
320            Ok(())
321        }
322        Err(error) if error.error_len().is_none() && !final_chunk => {
323            let valid_up_to = error.valid_up_to();
324            pending.drain(..valid_up_to);
325            Ok(())
326        }
327        Err(_) => Err(Error::InvalidInput {
328            field: "path",
329            reason: "must identify UTF-8 text",
330        }),
331    }
332}
