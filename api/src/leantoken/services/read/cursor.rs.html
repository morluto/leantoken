7pub(super) fn read_stream_id(services: &Services, path: &str, policy: ReadPolicy) -> StreamId {
8    let mut stream = StreamIdentityBuilder::for_service(services, CursorKind::Read);
9    stream.field_str("path", path);
10    stream.field_str(
11        "policy",
12        match policy {
13            ReadPolicy::Bounded => "bounded",
14            ReadPolicy::Full => "full",
15        },
16    );
17    stream.finish()
18}

20pub(super) fn encode_read_cursor(
21    generation: u64,
22    stream_id: StreamId,
23    state: ReadCursorState,
24) -> Result<String> {
25    validate_read_state(&state)?;
26    let mut payload = Vec::with_capacity(160);
27    payload.push(match state.policy {
28        ReadPolicy::Bounded => 1,
29        ReadPolicy::Full => 2,
30    });
31    push_usize(&mut payload, state.target_start_line)?;
32    push_optional_usize(&mut payload, state.target_end_line)?;
33    push_usize(&mut payload, state.next_start_line)?;
34    push_usize(&mut payload, state.next_byte)?;
35    push_fingerprint(&mut payload, state.full_hash.as_deref())?;
36    push_fingerprint(&mut payload, state.prefix_hash.as_deref())?;
37    push_usize(&mut payload, state.file_size)?;
38    push_optional_u128(&mut payload, state.modified_ns);
39    CursorEnvelope::new(CursorKind::Read, generation, stream_id, payload)
40        .map(CursorEnvelope::encode)
41}

43pub(super) fn decode_read_cursor(cursor: &str) -> Result<ReadCursor> {
44    let envelope = CursorEnvelope::parse(cursor, MAX_READ_CURSOR_BYTES)?;
45    let (kind, generation, stream_id) = envelope.identity();
46    let payload = envelope.payload_for(kind, generation, stream_id)?;
47    let mut payload = PayloadReader::new(payload);
48    let policy = match payload.byte()? {
49        1 => ReadPolicy::Bounded,
50        2 => ReadPolicy::Full,
51        _ => return Err(Error::StaleCursor),
52    };
53    let state = ReadCursorState {
54        target_start_line: payload.usize()?,
55        target_end_line: payload.optional_usize()?,
56        next_start_line: payload.usize()?,
57        next_byte: payload.usize()?,
58        full_hash: payload.fingerprint()?,
59        prefix_hash: payload.fingerprint()?,
60        policy,
61        file_size: payload.usize()?,
62        modified_ns: payload.optional_u128()?,
63    };
64    payload.finish()?;
65    validate_read_state(&state).map_err(|_| Error::StaleCursor)?;
66    Ok(ReadCursor { envelope, state })
67}

80fn validate_read_state(state: &ReadCursorState) -> Result<()> {
81    let fingerprints_match_policy = match state.policy {
82        ReadPolicy::Bounded => state.full_hash.is_none() && state.prefix_hash.is_some(),
83        ReadPolicy::Full => state.full_hash.is_some() && state.prefix_hash.is_none(),
84    };
85    if state.target_start_line == 0
86        || state.next_start_line < state.target_start_line
87        || state.target_end_line.is_some_and(|end_line| {
88            end_line < state.target_start_line || state.next_start_line > end_line
89        })
90        || state.next_byte == 0
91        || !fingerprints_match_policy
92    {
93        return Err(Error::StaleCursor);
94    }
95    Ok(())
96}

98fn push_usize(payload: &mut Vec<u8>, value: usize) -> Result<()> {
99    payload.extend_from_slice(
100        &u64::try_from(value)
101            .map_err(|_| Error::OperationFailure("read cursor value overflow".into()))?
102            .to_le_bytes(),
103    );
104    Ok(())
105}
106
107fn push_optional_usize(payload: &mut Vec<u8>, value: Option<usize>) -> Result<()> {
108    match value {
109        Some(value) => {
110            payload.push(1);
111            push_usize(payload, value)
112        }
113        None => {
114            payload.push(0);
115            Ok(())
116        }
117    }
118}

120fn push_optional_u128(payload: &mut Vec<u8>, value: Option<u128>) {
121    match value {
122        Some(value) => {
123            payload.push(1);
124            payload.extend_from_slice(&value.to_le_bytes());
125        }
126        None => payload.push(0),
127    }
128}
129
130fn push_fingerprint(payload: &mut Vec<u8>, value: Option<&str>) -> Result<()> {
131    match value {
132        Some(value)
133            if value.len() == crate::text::CONTENT_FINGERPRINT_HEX_LEN
134                && value.bytes().all(is_lower_hex) =>
135        {
136            payload.push(1);
137            payload.extend_from_slice(value.as_bytes());
138            Ok(())
139        }
140        Some(_) => Err(Error::OperationFailure(
141            "invalid read cursor fingerprint".into(),
142        )),
143        None => {
144            payload.push(0);
145            Ok(())
146        }
147    }
148}

150struct PayloadReader<'a> {
151    remaining: &'a [u8],
152}
153
154impl<'a> PayloadReader<'a> {
155    const fn new(remaining: &'a [u8]) -> Self {
156        Self { remaining }
157    }
158
159    fn byte(&mut self) -> Result<u8> {
160        self.take::<1>().map(|bytes| bytes[0])
161    }
162
163    fn usize(&mut self) -> Result<usize> {
164        let value = u64::from_le_bytes(self.take::<8>()?);
165        usize::try_from(value).map_err(|_| Error::StaleCursor)
166    }
167
168    fn optional_usize(&mut self) -> Result<Option<usize>> {
169        match self.byte()? {
170            0 => Ok(None),
171            1 => self.usize().map(Some),
172            _ => Err(Error::StaleCursor),
173        }
174    }
175
176    fn optional_u128(&mut self) -> Result<Option<u128>> {
177        match self.byte()? {
178            0 => Ok(None),
179            1 => self.take::<16>().map(u128::from_le_bytes).map(Some),
180            _ => Err(Error::StaleCursor),
181        }
182    }
183
184    fn fingerprint(&mut self) -> Result<Option<String>> {
185        match self.byte()? {
186            0 => Ok(None),
187            1 => {
188                let bytes = self.take_slice(crate::text::CONTENT_FINGERPRINT_HEX_LEN)?;
189                let value = std::str::from_utf8(bytes).map_err(|_| Error::StaleCursor)?;
190                if !value.bytes().all(is_lower_hex) {
191                    return Err(Error::StaleCursor);
192                }
193                Ok(Some(value.to_owned()))
194            }
195            _ => Err(Error::StaleCursor),
196        }
197    }
198
199    fn finish(self) -> Result<()> {
200        if self.remaining.is_empty() {
201            Ok(())
202        } else {
203            Err(Error::StaleCursor)
204        }
205    }
206
207    fn take<const N: usize>(&mut self) -> Result<[u8; N]> {
208        self.take_slice(N)?
209            .try_into()
210            .map_err(|_| Error::StaleCursor)
211    }
212
213    fn take_slice(&mut self, length: usize) -> Result<&'a [u8]> {
214        if self.remaining.len() < length {
215            return Err(Error::StaleCursor);
216        }
217        let (value, remaining) = self.remaining.split_at(length);
218        self.remaining = remaining;
219        Ok(value)
220    }
221}

223pub(super) fn returned_end_line(start_line: usize, content: &str) -> usize {
224    let newline_count = content.bytes().filter(|byte| *byte == b'\n').count();
225    start_line
226        .saturating_add(newline_count)
227        .saturating_sub(usize::from(content.ends_with('\n') && newline_count > 0))
228}
