13const MAX_READ_DELTA_ENTRIES: usize = 128;
14const MAX_READ_DELTA_ENTRY_BYTES: usize = 512 * 1024;
15const MAX_READ_DELTA_TOTAL_BYTES: usize = 8 * 1024 * 1024;
16const READ_DELTA_TTL: Duration = Duration::from_secs(30 * 60);

52impl ReadDeltaRegistry {
53    pub(super) fn evaluate(
54        &self,
55        repository_id: &str,
56        request: &ReadRequest,
57        response: &ReadResponse,
58        current_content: &str,
59        full_tokens: usize,
60        tokenizer: Tokenizer,
61    ) -> Result<ReadDeltaEvaluation> {
62        let target_key = target_key(repository_id, request);
63        let base_hash = request.expected_hash.clone();
64        let mut base_generation = None;
65        let mut delta = None;
66        let mut outcome = ReadDeltaOutcome::Full;
67        let mut delta_tokens = None;
68        let mut avoided_tokens = 0;
69        let mut fallback_reason = None;
70
71        if request.expected_hash.as_deref() == Some(response.content_hash.as_str()) {
72            outcome = ReadDeltaOutcome::NotModified;
73            delta_tokens = Some(0);
74            avoided_tokens = full_tokens;
75        } else if response.truncated {
76            fallback_reason = Some(ReadDeltaFallback::CurrentTruncated);
77        } else if current_content.len() > MAX_READ_DELTA_ENTRY_BYTES {
78            fallback_reason = Some(ReadDeltaFallback::ContentTooLarge);
79        } else if let Some(expected_hash) = request.expected_hash.as_deref() {
80            let base = self.lookup(&target_key, expected_hash)?;
81            if let Some(base) = base {
82                base_generation = Some(base.generation);
83                if target_coordinates_changed(&base, response) {
84                    fallback_reason = Some(ReadDeltaFallback::TargetChanged);
85                } else {
86                    let full_delta = TextDiff::from_lines(base.content.as_str(), current_content)
87                        .unified_diff()
88                        .context_radius(3)
89                        .header(
90                            &format!(
91                                "base/{}:{}-{}",
92                                response.path,
93                                response.returned_start_line,
94                                response.returned_end_line
95                            ),
96                            &format!(
97                                "head/{}:{}-{}",
98                                response.path,
99                                response.returned_start_line,
100                                response.returned_end_line
101                            ),
102                        )
103                        .to_string();
104                    let candidate_tokens = tokenizer.count(&full_delta);
105                    if candidate_tokens < full_tokens {
106                        outcome = ReadDeltaOutcome::Delta;
107                        delta_tokens = Some(candidate_tokens);
108                        avoided_tokens = full_tokens - candidate_tokens;
109                        delta = Some(full_delta);
110                    } else {
111                        fallback_reason = Some(ReadDeltaFallback::DeltaNotSmaller);
112                    }
113                }
114            } else {
115                fallback_reason = Some(ReadDeltaFallback::BaseUnavailable);
116            }
117        }
118
119        if !response.truncated && current_content.len() <= MAX_READ_DELTA_ENTRY_BYTES {
120            self.insert(
121                CacheKey {
122                    target_key: target_key.clone(),
123                    content_hash: response.content_hash.clone(),
124                },
125                CacheEntry {
126                    content: current_content.to_owned(),
127                    generation: response.meta.repository_generation,
128                    target_start_line: response.target_start_line,
129                    target_end_line: response.target_end_line,
130                    returned_start_line: response.returned_start_line,
131                    returned_end_line: response.returned_end_line,
132                    inserted_at: Instant::now(),
133                },
134            )?;
135        }
136
137        Ok(ReadDeltaEvaluation {
138            delta,
139            receipt: ReadDeltaReceipt {
140                target_key,
141                base_hash,
142                head_hash: response.content_hash.clone(),
143                base_generation,
144                head_generation: response.meta.repository_generation,
145                outcome,
146                full_tokens,
147                delta_tokens,
148                avoided_tokens,
149                fallback_reason,
150            },
151        })
152    }

154    fn lookup(&self, target_key: &str, content_hash: &str) -> Result<Option<CacheEntry>> {
155        let mut state = self
156            .state
157            .lock()
158            .map_err(|_| Error::InternalFailure("read delta registry poisoned".into()))?;
159        prune_expired(&mut state, Instant::now());
160        Ok(state
161            .entries
162            .get(&CacheKey {
163                target_key: target_key.to_owned(),
164                content_hash: content_hash.to_owned(),
165            })
166            .cloned())
167    }
168
169    fn insert(&self, key: CacheKey, entry: CacheEntry) -> Result<()> {
170        if entry.content.len() > MAX_READ_DELTA_ENTRY_BYTES {
171            return Ok(());
172        }
173        let mut state = self
174            .state
175            .lock()
176            .map_err(|_| Error::InternalFailure("read delta registry poisoned".into()))?;
177        prune_expired(&mut state, Instant::now());
178        if let Some(previous) = state.entries.remove(&key) {
179            state.total_bytes = state.total_bytes.saturating_sub(previous.content.len());
180            state.insertion_order.retain(|candidate| candidate != &key);
181        }
182        while state.entries.len() >= MAX_READ_DELTA_ENTRIES
183            || state.total_bytes.saturating_add(entry.content.len()) > MAX_READ_DELTA_TOTAL_BYTES
184        {
185            if !evict_oldest(&mut state) {
186                break;
187            }
188        }
189        if state.entries.len() < MAX_READ_DELTA_ENTRIES
190            && state.total_bytes.saturating_add(entry.content.len()) <= MAX_READ_DELTA_TOTAL_BYTES
191        {
192            state.total_bytes += entry.content.len();
193            state.insertion_order.push_back(key.clone());
194            state.entries.insert(key, entry);
195        }
196        Ok(())
197    }
198}

200fn target_key(repository_id: &str, request: &ReadRequest) -> String {
201    let mut hasher = blake3::Hasher::new();
202    for value in [repository_id, request.path.as_str()] {
203        hasher.update(value.as_bytes());
204        hasher.update(&[0]);
205    }
206    if let Some(symbol) = request.symbol.as_deref() {
207        hasher.update(b"symbol\0");
208        hasher.update(symbol.as_bytes());
209    } else if let Some(heading) = request.heading.as_deref() {
210        hasher.update(b"heading\0");
211        hasher.update(heading.as_bytes());
212        hasher.update(&[0]);
213        hasher.update(
214            &u64::try_from(request.heading_occurrence.unwrap_or(1))
215                .unwrap_or(u64::MAX)
216                .to_le_bytes(),
217        );
218    } else {
219        hasher.update(b"lines\0");
220        hasher.update(
221            &u64::try_from(request.start_line.unwrap_or(1))
222                .unwrap_or(u64::MAX)
223                .to_le_bytes(),
224        );
225        hasher.update(
226            &request
227                .end_line
228                .and_then(|line| u64::try_from(line).ok())
229                .unwrap_or(u64::MAX)
230                .to_le_bytes(),
231        );
232    }
233    hasher.finalize().to_hex().to_string()
234}

236fn target_coordinates_changed(base: &CacheEntry, head: &ReadResponse) -> bool {
237    base.target_start_line != head.target_start_line
238        || base.target_end_line != head.target_end_line
239        || base.returned_start_line != head.returned_start_line
240        || base.returned_end_line != head.returned_end_line
241}

243fn prune_expired(state: &mut RegistryState, now: Instant) {
244    while let Some(key) = state.insertion_order.front() {
245        let expired = state
246            .entries
247            .get(key)
248            .is_none_or(|entry| now.duration_since(entry.inserted_at) >= READ_DELTA_TTL);
249        if !expired {
250            break;
251        }
252        evict_oldest(state);
253    }
254}
255
256fn evict_oldest(state: &mut RegistryState) -> bool {
257    let Some(key) = state.insertion_order.pop_front() else {
258        return false;
259    };
260    if let Some(entry) = state.entries.remove(&key) {
261        state.total_bytes = state.total_bytes.saturating_sub(entry.content.len());
262    }
263    true
264}
