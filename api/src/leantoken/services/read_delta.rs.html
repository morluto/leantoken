83    pub(super) fn evaluate(&self, input: ReadDeltaInput<'_>) -> Result<ReadDeltaEvaluation> {
84        let ReadDeltaInput {
85            repository_id,
86            storage,
87            request,
88            response,
89            current_content,
90            full_tokens,
91            tokenizer,
92        } = input;
93        let target_key = target_key(repository_id, request);
94        let mut base_hash = request.expected_hash.clone();
95        let mut base_generation = None;
96        let mut base_source = None;
97        let mut delta = None;
98        let mut outcome = ReadDeltaOutcome::Full;
99        let mut delta_tokens = None;
100        let mut avoided_tokens = 0;
101        let mut fallback_reason = None;
102
103        if request.expected_hash.as_deref() == Some(response.content_hash.as_str()) {
104            outcome = ReadDeltaOutcome::NotModified;
105            delta_tokens = Some(0);
106            avoided_tokens = full_tokens;
107        } else if response.truncated {
108            fallback_reason = Some(ReadDeltaFallback::CurrentTruncated);
109        } else if current_content.len() > MAX_READ_DELTA_ENTRY_BYTES {
110            fallback_reason = Some(ReadDeltaFallback::ContentTooLarge);
111        } else {
112            let base = if let Some(expected_hash) = request.expected_hash.as_deref() {
113                if let Some(entry) = self.lookup(&target_key, expected_hash)? {
114                    Some((
115                        expected_hash.to_owned(),
116                        entry,
117                        ReadDeltaBaseSource::ProcessLocal,
118                    ))
119                } else {
120                    storage
121                        .read_delta_base(&target_key, expected_hash)?
122                        .map(|base| {
123                            (
124                                expected_hash.to_owned(),
125                                CacheEntry::from_persistent(base),
126                                ReadDeltaBaseSource::Persistent,
127                            )
128                        })
129                }
130            } else {
131                let local = self.lookup_latest(&target_key)?;
132                let persistent = storage.latest_read_delta_base(&target_key)?;
133                match (local, persistent) {
134                    (Some((hash, entry)), Some((persistent_hash, persistent))) => {
135                        if entry.generation >= persistent.generation {
136                            Some((hash, entry, ReadDeltaBaseSource::ProcessLocal))
137                        } else {
138                            Some((
139                                persistent_hash,
140                                CacheEntry::from_persistent(persistent),
141                                ReadDeltaBaseSource::Persistent,
142                            ))
143                        }
144                    }
145                    (Some((hash, entry)), None) => {
146                        Some((hash, entry, ReadDeltaBaseSource::ProcessLocal))
147                    }
148                    (None, Some((hash, persistent))) => Some((
149                        hash,
150                        CacheEntry::from_persistent(persistent),
151                        ReadDeltaBaseSource::Persistent,
152                    )),
153                    (None, None) => None,
154                }
155            };
156            if let Some((selected_hash, base, selected_source)) = base {
157                base_hash = Some(selected_hash.clone());
158                base_generation = Some(base.generation);
159                base_source = Some(selected_source);
160                if base_hash.as_deref() == Some(response.content_hash.as_str()) {
161                    outcome = ReadDeltaOutcome::NotModified;
162                    delta_tokens = Some(0);
163                    avoided_tokens = full_tokens;
164                } else if target_coordinates_changed(&base, response) {
165                    fallback_reason = Some(ReadDeltaFallback::TargetChanged);
166                } else {
167                    let full_delta = if let Some(cached) =
168                        self.lookup_delta(&target_key, &selected_hash, &response.content_hash)?
169                    {
170                        cached
171                    } else {
172                        let computed = TextDiff::from_lines(base.content.as_str(), current_content)
173                            .unified_diff()
174                            .context_radius(3)
175                            .header(
176                                &format!(
177                                    "base/{}:{}-{}",
178                                    response.path,
179                                    response.returned_start_line,
180                                    response.returned_end_line
181                                ),
182                                &format!(
183                                    "head/{}:{}-{}",
184                                    response.path,
185                                    response.returned_start_line,
186                                    response.returned_end_line
187                                ),
188                            )
189                            .to_string();
190                        self.insert_delta(
191                            DeltaCacheKey {
192                                target_key: target_key.clone(),
193                                base_hash: selected_hash,
194                                head_hash: response.content_hash.clone(),
195                            },
196                            computed.clone(),
197                        )?;
198                        computed
199                    };
200                    let candidate_tokens = tokenizer.count(&full_delta);
201                    if candidate_tokens < full_tokens {
202                        outcome = ReadDeltaOutcome::Delta;
203                        delta_tokens = Some(candidate_tokens);
204                        avoided_tokens = full_tokens - candidate_tokens;
205                        delta = Some(full_delta);
206                    } else {
207                        fallback_reason = Some(ReadDeltaFallback::DeltaNotSmaller);
208                    }
209                }
210            } else {
211                fallback_reason = Some(ReadDeltaFallback::BaseUnavailable);
212            }
213        }
214
215        let persistent_base = ReadDeltaBase {
216            content: current_content.to_owned(),
217            generation: response.meta.repository_generation,
218            target_start_line: response.target_start_line,
219            target_end_line: response.target_end_line,
220            returned_start_line: response.returned_start_line,
221            returned_end_line: response.returned_end_line,
222        };
223        if !response.truncated && current_content.len() <= MAX_READ_DELTA_ENTRY_BYTES {
224            self.insert(
225                CacheKey {
226                    target_key: target_key.clone(),
227                    content_hash: response.content_hash.clone(),
228                },
229                CacheEntry {
230                    content: current_content.to_owned(),
231                    generation: response.meta.repository_generation,
232                    target_start_line: response.target_start_line,
233                    target_end_line: response.target_end_line,
234                    returned_start_line: response.returned_start_line,
235                    returned_end_line: response.returned_end_line,
236                    inserted_at: Instant::now(),
237                },
238            )?;
239        }
240        let mut persistence_fallback_reason = if response.truncated {
241            Some(ReadDeltaPersistenceFallback::CurrentTruncated)
242        } else if current_content.len() > MAX_READ_DELTA_ENTRY_BYTES {
243            Some(ReadDeltaPersistenceFallback::ContentTooLarge)
244        } else if response.index_stale {
245            Some(ReadDeltaPersistenceFallback::LiveDiffersFromIndex)
246        } else if response.indexed_hash.is_none() {
247            Some(ReadDeltaPersistenceFallback::IndexedHashUnavailable)
248        } else {
249            None
250        };
251        let head_persisted = if persistence_fallback_reason.is_none() {
252            let persisted = storage.persist_read_delta_base(
253                &target_key,
254                &response.content_hash,
255                &persistent_base,
256            )?;
257            if !persisted {
258                persistence_fallback_reason = Some(ReadDeltaPersistenceFallback::StorageCapacity);
259            }
260            persisted
261        } else {
262            false
263        };
264
265        Ok(ReadDeltaEvaluation {
266            delta,
267            receipt: ReadDeltaReceipt {
268                target_key,
269                base_hash,
270                head_hash: response.content_hash.clone(),
271                base_generation,
272                base_source,
273                head_generation: response.meta.repository_generation,
274                outcome,
275                full_tokens,
276                delta_tokens,
277                avoided_tokens,
278                head_persisted,
279                persistence_fallback_reason,
280                fallback_reason,
281            },
282        })
283    }

285    fn lookup(&self, target_key: &str, content_hash: &str) -> Result<Option<CacheEntry>> {
286        let mut state = self
287            .state
288            .lock()
289            .map_err(|_| Error::InternalFailure("read delta registry poisoned".into()))?;
290        prune_expired(&mut state, Instant::now());
291        Ok(state
292            .entries
293            .get(&CacheKey {
294                target_key: target_key.to_owned(),
295                content_hash: content_hash.to_owned(),
296            })
297            .cloned())
298    }

300    fn lookup_latest(&self, target_key: &str) -> Result<Option<(String, CacheEntry)>> {
301        let mut state = self
302            .state
303            .lock()
304            .map_err(|_| Error::InternalFailure("read delta registry poisoned".into()))?;
305        prune_expired(&mut state, Instant::now());
306        Ok(state.insertion_order.iter().rev().find_map(|key| {
307            if key.target_key != target_key {
308                return None;
309            }
310            state
311                .entries
312                .get(key)
313                .cloned()
314                .map(|entry| (key.content_hash.clone(), entry))
315        }))
316    }

318    fn lookup_delta(
319        &self,
320        target_key: &str,
321        base_hash: &str,
322        head_hash: &str,
323    ) -> Result<Option<String>> {
324        let mut state = self
325            .state
326            .lock()
327            .map_err(|_| Error::InternalFailure("read delta registry poisoned".into()))?;
328        prune_expired(&mut state, Instant::now());
329        Ok(state
330            .deltas
331            .get(&DeltaCacheKey {
332                target_key: target_key.to_owned(),
333                base_hash: base_hash.to_owned(),
334                head_hash: head_hash.to_owned(),
335            })
336            .map(|entry| entry.delta.clone()))
337    }

339    fn insert(&self, key: CacheKey, entry: CacheEntry) -> Result<()> {
340        if entry.content.len() > MAX_READ_DELTA_ENTRY_BYTES {
341            return Ok(());
342        }
343        let mut state = self
344            .state
345            .lock()
346            .map_err(|_| Error::InternalFailure("read delta registry poisoned".into()))?;
347        prune_expired(&mut state, Instant::now());
348        if let Some(previous) = state.entries.remove(&key) {
349            state.total_bytes = state.total_bytes.saturating_sub(previous.content.len());
350            state.insertion_order.retain(|candidate| candidate != &key);
351        }
352        while state.entries.len() >= MAX_READ_DELTA_ENTRIES
353            || state.total_bytes.saturating_add(entry.content.len()) > MAX_READ_DELTA_TOTAL_BYTES
354        {
355            if !evict_oldest(&mut state) {
356                break;
357            }
358        }
359        if state.entries.len() < MAX_READ_DELTA_ENTRIES
360            && state.total_bytes.saturating_add(entry.content.len()) <= MAX_READ_DELTA_TOTAL_BYTES
361        {
362            state.total_bytes += entry.content.len();
363            state.insertion_order.push_back(key.clone());
364            state.entries.insert(key, entry);
365        }
366        Ok(())
367    }

369    fn insert_delta(&self, key: DeltaCacheKey, delta: String) -> Result<()> {
370        if delta.len() > MAX_READ_DELTA_ENTRY_BYTES {
371            return Ok(());
372        }
373        let mut state = self
374            .state
375            .lock()
376            .map_err(|_| Error::InternalFailure("read delta registry poisoned".into()))?;
377        prune_expired(&mut state, Instant::now());
378        if let Some(previous) = state.deltas.remove(&key) {
379            state.delta_bytes = state.delta_bytes.saturating_sub(previous.delta.len());
380            state.delta_order.retain(|candidate| candidate != &key);
381        }
382        while state.deltas.len() >= MAX_READ_DELTA_ENTRIES
383            || state.delta_bytes.saturating_add(delta.len()) > MAX_READ_DELTA_TOTAL_BYTES
384        {
385            if !evict_oldest_delta(&mut state) {
386                break;
387            }
388        }
389        if state.deltas.len() < MAX_READ_DELTA_ENTRIES
390            && state.delta_bytes.saturating_add(delta.len()) <= MAX_READ_DELTA_TOTAL_BYTES
391        {
392            state.delta_bytes += delta.len();
393            state.delta_order.push_back(key.clone());
394            state.deltas.insert(
395                key,
396                DeltaCacheEntry {
397                    delta,
398                    inserted_at: Instant::now(),
399                },
400            );
401        }
402        Ok(())
403    }

420fn target_key(repository_id: &str, request: &ReadRequest) -> String {
421    let mut hasher = blake3::Hasher::new();
422    for value in [repository_id, request.path.as_str()] {
423        hasher.update(value.as_bytes());
424        hasher.update(&[0]);
425    }
426    if let Some(symbol) = request.symbol.as_deref() {
427        hasher.update(b"symbol\0");
428        hasher.update(symbol.as_bytes());
429    } else if let Some(heading) = request.heading.as_deref() {
430        hasher.update(b"heading\0");
431        hasher.update(heading.as_bytes());
432        hasher.update(&[0]);
433        hasher.update(
434            &u64::try_from(request.heading_occurrence.unwrap_or(1))
435                .unwrap_or(u64::MAX)
436                .to_le_bytes(),
437        );
438    } else {
439        hasher.update(b"lines\0");
440        hasher.update(
441            &u64::try_from(request.start_line.unwrap_or(1))
442                .unwrap_or(u64::MAX)
443                .to_le_bytes(),
444        );
445        hasher.update(
446            &request
447                .end_line
448                .and_then(|line| u64::try_from(line).ok())
449                .unwrap_or(u64::MAX)
450                .to_le_bytes(),
451        );
452    }
453    hasher.finalize().to_hex().to_string()
454}

456fn target_coordinates_changed(base: &CacheEntry, head: &ReadResponse) -> bool {
457    base.target_start_line != head.target_start_line
458        || base.target_end_line != head.target_end_line
459        || base.returned_start_line != head.returned_start_line
460        || base.returned_end_line != head.returned_end_line
461}

463fn prune_expired(state: &mut RegistryState, now: Instant) {
464    while let Some(key) = state.insertion_order.front() {
465        let expired = state
466            .entries
467            .get(key)
468            .is_none_or(|entry| now.duration_since(entry.inserted_at) >= READ_DELTA_TTL);
469        if !expired {
470            break;
471        }
472        evict_oldest(state);
473    }
474    while let Some(key) = state.delta_order.front() {
475        let expired = state
476            .deltas
477            .get(key)
478            .is_none_or(|entry| now.duration_since(entry.inserted_at) >= READ_DELTA_TTL);
479        if !expired {
480            break;
481        }
482        evict_oldest_delta(state);
483    }
484}
485
486fn evict_oldest(state: &mut RegistryState) -> bool {
487    let Some(key) = state.insertion_order.pop_front() else {
488        return false;
489    };
490    if let Some(entry) = state.entries.remove(&key) {
491        state.total_bytes = state.total_bytes.saturating_sub(entry.content.len());
492    }
493    true
494}
495
496fn evict_oldest_delta(state: &mut RegistryState) -> bool {
497    let Some(key) = state.delta_order.pop_front() else {
498        return false;
499    };
500    if let Some(entry) = state.deltas.remove(&key) {
501        state.delta_bytes = state.delta_bytes.saturating_sub(entry.delta.len());
502    }
503    true
504}

692        let key = DeltaCacheKey {
693            target_key: "target".into(),

695            head_hash: "head".into(),

703            .expect("lookup")
704            .expect("cached delta");
