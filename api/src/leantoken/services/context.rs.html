65fn parse_revision_range(revision: &str) -> Result<Option<(&str, &str)>> {
66    let Some((base, head)) = revision.split_once("..") else {
67        return Ok(None);
68    };
69    if base.trim().is_empty()
70        || head.trim().is_empty()
71        || base.ends_with('.')
72        || head.starts_with('.')
73        || head.contains("..")
74    {
75        return Err(Error::InvalidInput {
76            field: "base revision",
77            reason: "revision range must be BASE..HEAD",
78        });
79    }
80    Ok(Some((base, head)))
81}

116impl ContextPathScorer {
117    fn new(terms: &[String], task: &str) -> Self {
118        let code_token_parts = facets::legacy_code_tokens(task)
119            .into_iter()
120            .filter(|token| {
121                token.contains("::")
122                    || token
123                        .split('.')
124                        .any(|part| part.chars().next().is_some_and(char::is_uppercase))
125            })
126            .map(|code_token| {
127                expand_terms(&code_token)
128                    .into_iter()
129                    .map(|part| part.to_ascii_lowercase())
130                    .filter(|part| part.chars().count() >= 2)
131                    .collect::<HashSet<_>>()
132                    .into_iter()
133                    .collect()
134            })
135            .collect();
136        Self {
137            terms: terms.iter().map(|term| term.to_ascii_lowercase()).collect(),
138            code_token_parts,
139            languages: [
140                task_mentions_language(task, "javascript"),
141                task_mentions_language(task, "typescript"),
142                task_mentions_language(task, "python"),
143                task_mentions_language(task, "rust"),
144                task_mentions_language(task, "go"),
145            ],
146        }
147    }
148
149    fn score(&self, path: &str) -> f64 {
150        let path = path.to_lowercase();
151        let mut score = self
152            .terms
153            .iter()
154            .filter(|term| path.contains(term.as_str()))
155            .count() as f64;
156        for parts in &self.code_token_parts {
157            let matched_parts = parts.iter().filter(|part| path.contains(*part)).count();
158            if matched_parts >= 2 {
159                #[allow(clippy::cast_precision_loss)]
160                {
161                    score += (matched_parts * matched_parts) as f64;
162                }
163            }
164        }
165        for (mentioned, component, extensions) in [
166            (self.languages[0], "/js/", &["js", "jsx", "mjs", "cjs"][..]),
167            (self.languages[1], "/ts/", &["ts", "tsx"][..]),
168            (self.languages[2], "/python/", &["py", "pyw"][..]),
169            (self.languages[3], "/rust/", &["rs"][..]),
170            (self.languages[4], "/go/", &["go"][..]),
171        ] {
172            let extension_matches = path
173                .rsplit_once('.')
174                .is_some_and(|(_, extension)| extensions.contains(&extension));
175            let component = component.trim_matches('/');
176            let component_matches = path
177                .split('/')
178                .any(|path_component| path_component == component);
179            if mentioned && (extension_matches || component_matches) {
180                // An explicit language name in the task is strong repository-scope
181                // evidence. Keep this above an exact-name match in another
182                // language so common names such as `Point` do not dominate.
183                score += 12.0;
184            }
185        }
186        score
187    }
188}

190fn context_path_group(path: &str) -> String {
191    let mut components = path.split('/').filter(|component| !component.is_empty());
192    let first = components.next().unwrap_or("<root>");
193    let second = components.next();
194    if second.is_none() {
195        return "<root>".into();
196    }
197    if matches!(
198        first,
199        "src" | "lib" | "app" | "apps" | "crates" | "packages"
200    ) && let Some(second) = second
201    {
202        format!("{first}/{second}")
203    } else {
204        first.to_owned()
205    }
206}

208fn build_context_routing(
209    request: &ContextRequest,
210    scope: &DiffScopeReceipt,
211    candidate_paths: usize,
212    selected_paths: &[String],
213) -> Option<ContextRoutingReceipt> {
214    if scope.changed_paths.len() < OVERSIZED_CHANGE_PATHS {
215        return None;
216    }
217    let mut grouped_paths = BTreeMap::<String, Vec<String>>::new();
218    for path in &scope.changed_paths {
219        grouped_paths
220            .entry(context_path_group(path))
221            .or_default()
222            .push(path.clone());
223    }
224    if grouped_paths.len() < MIN_OVERSIZED_PATH_GROUPS {
225        return None;
226    }
227    let path_groups_total = grouped_paths.len();
228    let mut groups = grouped_paths.into_iter().collect::<Vec<_>>();
229    groups.sort_by(|left, right| {
230        right
231            .1
232            .len()
233            .cmp(&left.1.len())
234            .then_with(|| left.0.cmp(&right.0))
235    });
236    let selected_groups = selected_paths.iter().fold(
237        BTreeMap::<String, BTreeSet<&str>>::new(),
238        |mut paths, path| {
239            paths
240                .entry(context_path_group(path))
241                .or_default()
242                .insert(path);
243            paths
244        },
245    );
246    let selected_path_count = selected_paths
247        .iter()
248        .map(String::as_str)
249        .collect::<BTreeSet<_>>()
250        .len();
251    let strongest_selected_group = selected_groups
252        .values()
253        .map(BTreeSet::len)
254        .max()
255        .unwrap_or(0);
256    let weakly_concentrated = selected_path_count > 1
257        && strongest_selected_group.saturating_mul(2) <= selected_path_count;
258
259    let suggestions = groups
260        .iter()
261        .take(MAX_ROUTING_SUGGESTIONS)
262        .map(|(prefix, _)| ContextRoutingSuggestion {
263            include_paths: vec![if prefix == "<root>" {
264                "*".into()
265            } else {
266                format!("{prefix}/**")
267            }],
268        })
269        .collect();
270    let path_groups = groups
271        .into_iter()
272        .take(MAX_ROUTING_GROUPS)
273        .map(|(prefix, paths)| ContextPathGroup {
274            prefix,
275            changed_paths: paths.len(),
276        })
277        .collect();
278    Some(ContextRoutingReceipt {
279        candidate_paths,
280        changed_paths: scope.changed_paths.len(),
281        selected_paths: selected_path_count,
282        weakly_concentrated,
283        consistency: IndexConsistency::IndexedGeneration,
284        base_revision: request.base_revision.clone(),
285        known_hashes: request.known_hashes.clone(),
286        path_groups_total,
287        path_groups,
288        suggestions,
289    })
290}

292fn resolve_context_workflow(requested: ContextWorkflow, task: &str) -> ContextWorkflow {
293    if requested != ContextWorkflow::Auto {
294        return requested;
295    }
296    let task = task.to_ascii_lowercase();
297    if ["pull request", "contribution", "contributing"]
298        .iter()
299        .any(|signal| task.contains(signal))
300    {
301        ContextWorkflow::Contribution
302    } else if ["code review", "review this", "review the changes"]
303        .iter()
304        .any(|signal| task.contains(signal))
305    {
306        ContextWorkflow::Review
307    } else if ["investigate", "root cause", "diagnose"]
308        .iter()
309        .any(|signal| task.contains(signal))
310    {
311        ContextWorkflow::Investigation
312    } else {
313        ContextWorkflow::Implementation
314    }
315}

317fn workflow_path_role(path: &str, request: &ContextRequest) -> Option<(f64, &'static str)> {
318    let normalized = path.replace('\\', "/");
319    let lower = normalized.to_ascii_lowercase();
320    let mut best = WORKFLOW_PATHS
321        .matches(&normalized)
322        .into_iter()
323        .map(|index| WORKFLOW_PATH_ROLES[index])
324        .max_by(|left, right| left.0.total_cmp(&right.0));
325    if likely_owner_test(&lower, request) && best.is_none_or(|(score, _)| score < OWNER_TEST_SCORE)
326    {
327        best = Some((OWNER_TEST_SCORE, "owner_test"));
328    }
329    best
330}

350const WORKFLOW_PATH_ROLES: [(f64, &str); WORKFLOW_PATH_RULES.len()] = workflow_path_roles();

352const fn workflow_path_roles() -> [(f64, &'static str); WORKFLOW_PATH_RULES.len()] {
353    let mut roles = [(0.0, ""); WORKFLOW_PATH_RULES.len()];
354    let mut index = 0;
355    while index < WORKFLOW_PATH_RULES.len() {
356        roles[index] = (WORKFLOW_PATH_RULES[index].1, WORKFLOW_PATH_RULES[index].2);
357        index += 1;
358    }
359    roles
360}

362static WORKFLOW_PATHS: LazyLock<GlobSet> = LazyLock::new(|| {
363    let mut builder = GlobSetBuilder::new();
364    for (pattern, _, _) in WORKFLOW_PATH_RULES {
365        builder.add(

368                .literal_separator(true)
369                .build()
370                .expect("static workflow glob"),
371        );
372    }
373    builder.build().expect("static workflow glob set")

380fn owner_test_changed_path(path: &str, request: &ContextRequest) -> Option<String> {
381    let path = path.replace('\\', "/").to_ascii_lowercase();
382    let is_test = path.contains("/test")
383        || path.starts_with("test")
384        || path.contains("/spec")
385        || path.starts_with("spec");
386    if !is_test {
387        return None;
388    }
389    let test_name = path.rsplit('/').next().unwrap_or(&path);
390    let test_stem = test_name.split('.').next().unwrap_or(test_name);
391    request
392        .changed_paths
393        .iter()
394        .chain(&request.focus_paths)
395        .find_map(|changed| {
396            let normalized = changed.replace('\\', "/");
397            let stem = normalized
398                .rsplit('/')
399                .next()
400                .and_then(|name| name.split('.').next())?
401                .to_ascii_lowercase();
402            (stem.len() >= 3 && contains_filename_token(test_stem, &stem)).then(|| changed.clone())
403        })
404}

406fn contains_filename_token(path: &str, token: &str) -> bool {
407    path.match_indices(token).any(|(start, matched)| {
408        let end = start + matched.len();
409        let boundary =
410            |character: Option<char>| character.is_none_or(|value| !value.is_alphanumeric());
411        boundary(path[..start].chars().next_back()) && boundary(path[end..].chars().next())
412    })
413}

481            ContextTimedPhase::StoredExcerpt => &mut self.timings.stored_excerpt_ms,

624fn analyze_lexical_match(
625    hit: &ChunkHit,
626    matcher: &regex::Regex,
627    context_lines: usize,
628) -> Option<LexicalMatchFacts> {
629    let mut matches = matcher.find_iter(&hit.content);
630    let first = matches.next()?;
631    let occurrences = 1usize.saturating_add(
632        matches
633            .take(LEXICAL_OCCURRENCE_SATURATION.saturating_sub(1))
634            .count(),
635    );
636    let starts = line_starts(&hit.content);
637    let search_hit = chunk_search_hit_for_range(
638        hit,
639        first.start(),
640        first.end(),
641        context_lines,
642        false,
643        false,
644        &starts,
645    );
646    let local_start = byte_to_line(&starts, hit.content.len(), first.start());

694fn qualified_symbol_match(
695    concept: &str,
696    name: &str,
697    parent: Option<&str>,
698    signature: Option<&str>,
699) -> f64 {
700    if !concept.contains(['.', ':']) {
701        return 0.0;
702    }
703    let parts = concept
704        .split(|character: char| !character.is_alphanumeric() && character != '_')
705        .flat_map(identifier_words)
706        .map(|part| part.to_ascii_lowercase())
707        .filter(|part| part.chars().count() >= 2)
708        .collect::<HashSet<_>>();
709    if parts.len() < 2 {
710        return 0.0;
711    }
712    let haystack = format!(
713        "{} {} {}",
714        name,
715        parent.unwrap_or_default(),
716        signature.unwrap_or_default()
717    )
718    .to_ascii_lowercase();
719    f64::from(parts.iter().all(|part| haystack.contains(part)))
720}

722fn record_query_hit(
723    fusion: &mut HashMap<String, HashMap<String, f64>>,
724    path: &str,
725    fusion_key: &str,
726    weight: f64,
727    rank: usize,
728) {
729    if weight < MIN_CORROBORATED_QUERY_WEIGHT {
730        return;
731    }
732    const RRF_K: f64 = 60.0;
733    #[allow(clippy::cast_precision_loss)]
734    let score = weight * RRF_K / (RRF_K + rank as f64 + 1.0);
735    fusion
736        .entry(path.to_owned())
737        .or_default()
738        .entry(fusion_key.to_owned())
739        .and_modify(|current| *current = current.max(score))
740        .or_insert(score);
741}

743fn apply_query_fusion(
744    candidates: &mut [Candidate],
745    fusion: &HashMap<String, HashMap<String, f64>>,
746) {
747    for candidate in candidates {
748        let Some(matches) = fusion.get(&candidate.path) else {
749            continue;
750        };
751        if matches.len() > 1 {
752            let total = matches.values().sum::<f64>();
753            let strongest = matches.values().copied().fold(0.0, f64::max);
754            candidate.path_score += (total - strongest).min(0.2);
755            if !candidate
756                .match_kinds
757                .iter()
758                .any(|kind| kind == "multi-query")
759            {
760                candidate.match_kinds.push("multi-query".into());
761            }
762        }
763    }
764}

788fn corroborated_import_symbol<'a>(
789    symbols: Vec<SymbolRecord>,
790    queries: &'a [ContextQuery],
791    seed_concepts: &BTreeSet<String>,
792) -> Option<(SymbolRecord, &'a ContextQuery, f64)> {
793    let mut best: Option<(usize, usize, usize, SymbolRecord, &ContextQuery, f64)> = None;
794    for (query_rank, query) in queries.iter().enumerate() {
795        if query.concept_weight < MIN_CORROBORATED_QUERY_WEIGHT
796            || !seed_concepts.contains(&query.fusion_key)
797            || !(query.has_facet(FacetKind::ExactAtom)
798                || query.has_facet(FacetKind::Symbol)
799                || query.has_facet(FacetKind::Configuration))
800        {
801            continue;
802        }
803        for symbol in &symbols {
804            let exact = symbol.name.eq_ignore_ascii_case(&query.value);
805            let qualified = qualified_symbol_match(
806                &query.fusion_key,
807                &symbol.name,
808                symbol.parent.as_deref(),
809                symbol.signature.as_deref(),
810            ) > 0.0;
811            if !exact && !qualified {
812                continue;
813            }
814            let class = usize::from(qualified) * 2 + usize::from(exact);
815            let evidence = f64::from(exact) + f64::from(qualified) * 1.5;
816            let candidate = (
817                class,
818                usize::MAX - query_rank,
819                usize::MAX - symbol.start_line,
820                symbol.clone(),
821                query,
822                evidence,
823            );
824            if best.as_ref().is_none_or(|current| {
825                (candidate.0, candidate.1, candidate.2) > (current.0, current.1, current.2)
826            }) {
827                best = Some(candidate);
828            }
829        }
830    }
831    best.map(|(_, _, _, symbol, query, evidence)| (symbol, query, evidence))
832}

834fn import_seed_paths(
835    candidates: &[Candidate],
836    queries: &[ContextQuery],
837    tokenizer: crate::tokens::Tokenizer,
838) -> Vec<(String, BTreeSet<String>)> {
839    if low_cardinality_exact_query(queries) {
840        return Vec::new();
841    }
842    let mut paths = BTreeMap::<String, (f64, BTreeSet<String>)>::new();
843    for candidate in candidates {
844        if candidate.concept_weight < MIN_CORROBORATED_QUERY_WEIGHT || candidate.concepts.is_empty()
845        {
846            continue;
847        }
848        let token_count = candidate.token_count_with(tokenizer).max(1);
849        let score = candidate.score(&ranking::Weights::default(), token_count);
850        let entry = paths
851            .entry(candidate.path.clone())
852            .or_insert_with(|| (score, BTreeSet::new()));
853        entry.0 = entry.0.max(score);
854        entry.1.extend(candidate.concepts.iter().cloned());
855    }
856    let mut paths = paths.into_iter().collect::<Vec<_>>();
857    paths.sort_by(|left, right| {
858        right
859            .1
860            .0
861            .total_cmp(&left.1.0)
862            .then_with(|| left.0.cmp(&right.0))
863    });
864    paths
865        .into_iter()
866        .map(|(path, (_, concepts))| (path, concepts))
867        .collect()
868}

1220            .zip(symbol_excerpts)

1226                    hit.path,
1227                    excerpt.start_line,
1228                    excerpt.end_line,

1230                )

1285
1286        let focus_coverage_is_required =
1287            request.strict_focus_paths || request.minimum_fragments_per_focus_path.is_some();
1288        if focus_coverage_is_required || request.strict_changed_paths {
1289            coverage.strict_scope_satisfied = Some(
1290                (!focus_coverage_is_required
1291                    || coverage
1292                        .focus_path_coverage
1293                        .iter()
1294                        .all(|focus| focus.satisfied))
1295                    && coverage
1296                        .changed_path_coverage
1297                        .as_ref()
1298                        .is_none_or(|changed| changed.satisfied),
1299            );
1300        }
1301        Ok(())
1302    }

1325    fn append_import_symbol_candidates(
1326        &self,
1327        expansion: ImportExpansion<'_>,
1328        candidates: &mut Vec<Candidate>,
1329    ) -> Result<()> {
1330        let seed_paths = import_seed_paths(candidates, expansion.queries, self.config.tokenizer);
1331        let requested_paths = seed_paths
1332            .iter()
1333            .take(24)
1334            .map(|(path, _)| path.clone())
1335            .collect::<Vec<_>>();
1336        let targets =
1337            expansion
1338                .session
1339                .import_symbol_targets(&requested_paths, 32, MAX_IMPORT_SYMBOLS)?;
1340        let path_filter = PathFilter::new(
1341            &expansion.request.include_paths,
1342            &expansion.request.exclude_paths,
1343        )?;
1344        let mut pending = Vec::new();
1345        for target in targets {
1346            check_cancelled(expansion.cancellation)?;
1347            let Some((_, seed_concepts)) = seed_paths.get(target.seed_index) else {
1348                continue;
1349            };
1350            let target_path = &target.target_file.path;
1351            if !path_filter.allows(target_path) {
1352                continue;
1353            }
1354            let Some((symbol, query, exact)) =
1355                corroborated_import_symbol(target.symbols, expansion.queries, seed_concepts)
1356            else {
1357                continue;
1358            };
1359            pending.push((target.target_file, symbol, query.clone(), exact));
1360        }
1361        let excerpt_requests = pending
1362            .iter()
1363            .map(|(target_file, symbol, _, _)| AdaptiveExcerptRequest {
1364                file_id: target_file.id,
1365                declaration_start: symbol.start_line,
1366                declaration_end: symbol.end_line,
1367                matched_line: symbol.start_line,
1368                token_budget: excerpt_budget(
1369                    expansion.request.token_budget,
1370                    ContextExcerptKind::ImportSymbol,
1371                ),
1372            })
1373            .collect::<Vec<_>>();
1374        let excerpts = self.adaptive_context_excerpts(expansion.session, &excerpt_requests)?;
1375        let mut neighbor_count = 0usize;
1376        let mut neighbor_ranges = BTreeSet::new();
1377        for ((target_file, symbol, query, exact), excerpt) in pending.into_iter().zip(excerpts) {
1378            check_cancelled(expansion.cancellation)?;
1379            let Some(excerpt) = excerpt else { continue };
1380            let target_path = target_file.path;
1381            if !neighbor_ranges.insert((target_path.clone(), excerpt.start_line, excerpt.end_line))
1382            {
1383                continue;
1384            }
1385            let change_boost = Self::file_change_boost(
1386                Some(target_file.generation),
1387                &target_path,
1388                expansion.changed_paths,
1389                expansion.request.prior_repository_generation,
1390            );
1391            let candidate = Candidate::new(
1392                &target_path,
1393                excerpt.start_line,
1394                excerpt.end_line,
1395                excerpt.content,
1396            )
1397            .match_kind("import")
1398            .match_kind("symbol")
1399            .concept(&query.fusion_key, query.concept_weight)
1400            .representation("import_symbol")
1401            .symbol_name(symbol.name)
1402            .exact(exact)
1403            .symbol(1.0)
1404            .path_score(context_path_score(
1405                &target_path,
1406                expansion.terms,
1407                &expansion.request.task,
1408            ))
1409            .import_boost(1.0)
1410            .change_boost(change_boost);
1411            candidates.push(annotate_candidate(
1412                candidate,
1413                &query,
1414                "import_symbol",
1415                neighbor_count,
1416            ));
1417            neighbor_count += 1;
1418            if neighbor_count >= 24 {
1419                break;
1420            }
1421        }
1422        Ok(())
1423    }

1455    /// Resolve a diff scope from the request into a receipt, if one is supplied.
1456    ///
1457    /// A single `base_revision` resolves committed and working-tree paths since
1458    /// that revision, including untracked files. `BASE..HEAD` instead resolves
1459    /// an immutable revision range. Explicit `changed_paths` are merged with
1460    /// either result. When neither input is supplied, strict changed-path
1461    /// requests use the current working tree; otherwise `None` preserves
1462    /// task-only behavior.
1463    fn resolve_diff_scope(
1464        &self,
1465        request: &ContextRequest,
1466    ) -> Result<(Option<DiffScopeReceipt>, HashSet<String>)> {
1467        let has_base = request
1468            .base_revision
1469            .as_deref()
1470            .is_some_and(|rev| !rev.trim().is_empty());
1471        let has_paths = !request.changed_paths.is_empty();
1472        let revision = request
1473            .base_revision
1474            .as_deref()
1475            .filter(|revision| !revision.trim().is_empty());
1476        let immutable_range = revision.map(parse_revision_range).transpose()?.flatten();
1477        let git_result = match (revision, immutable_range) {
1478            (Some(_), Some((base, head))) => Some(git_diff_paths_between(
1479                &self.config.root,
1480                base,
1481                head,
1482                MAX_DIFF_CHANGED_PATHS,
1483            )?),
1484            (Some(revision), None) => Some(git_diff_paths(
1485                &self.config.root,
1486                revision,
1487                MAX_DIFF_CHANGED_PATHS,
1488            )?),
1489            (None, None) => None,
1490            (None, Some(_)) => unreachable!("a range comes from a revision"),
1491        };
1492        let working_tree_paths = git_changed_paths(&self.config.root, GIT_CHANGED_PATHS_MAX)
1493            .unwrap_or_else(|error| {
1494                tracing::debug!(%error, "working-tree signal unavailable");
1495                HashSet::new()
1496            });
1497        if !has_base && !has_paths && !request.strict_changed_paths {
1498            return Ok((None, working_tree_paths));
1499        }
1500        if let Some(git_result) = git_result {
1501            let mut changed_paths = request.changed_paths.clone();
1502            let mut resolved_paths = git_result.changed_paths;
1503            if immutable_range.is_none() {
1504                resolved_paths.extend(working_tree_paths.iter().cloned());
1505            }
1506            resolved_paths.sort();
1507            resolved_paths.dedup();
1508            for path in resolved_paths {
1509                if changed_paths.len() == MAX_DIFF_CHANGED_PATHS {
1510                    break;
1511                }
1512                if !changed_paths.contains(&path) {
1513                    changed_paths.push(path);
1514                }
1515            }
1516            changed_paths.sort();
1517            changed_paths.dedup();
1518            return Ok((
1519                Some(DiffScopeReceipt {
1520                    base_revision: Some(git_result.base_revision),
1521                    head_revision: Some(git_result.head_revision),
1522                    changed_paths,
1523                    indexed_changed_paths: 0,
1524                    evidence: None,
1525                }),
1526                working_tree_paths,
1527            ));
1528        }
1529        let mut resolved_paths = if has_paths {
1530            request.changed_paths.clone()
1531        } else {
1532            working_tree_paths.iter().cloned().collect::<Vec<_>>()
1533        };
1534        resolved_paths.sort();
1535        resolved_paths.dedup();
1536        Ok((
1537            Some(DiffScopeReceipt {
1538                base_revision: None,
1539                head_revision: None,
1540                changed_paths: resolved_paths,
1541                indexed_changed_paths: 0,
1542                evidence: None,
1543            }),
1544            working_tree_paths,
1545        ))
1546    }

1675    fn append_workflow_candidates(
1676        &self,
1677        session: &ReadSession,
1678        request: &ContextRequest,
1679        workflow: ContextWorkflow,
1680        cancellation: &CancellationToken,
1681        candidates: &mut Vec<Candidate>,
1682    ) -> Result<(Option<WorkflowReceipt>, Vec<String>)> {
1683        if !matches!(
1684            workflow,
1685            ContextWorkflow::Contribution | ContextWorkflow::Review
1686        ) {
1687            return Ok((None, Vec::new()));
1688        }
1689
1690        let mut matches = Vec::new();
1691        let path_filter = PathFilter::new(&request.include_paths, &request.exclude_paths)?;
1692        let mut path_excluded = Vec::new();
1693        let mut cursor = None;
1694        let mut scanned_files = 0;
1695        let mut scan_truncated = false;
1696        loop {
1697            check_cancelled(cancellation)?;
1698            let page = session.list_files(512, cursor)?;
1699            let Some(last) = page.last() else {
1700                break;
1701            };
1702            cursor = Some(last.id);
1703            for file in page {
1704                if scanned_files == MAX_WORKFLOW_SCAN_FILES {
1705                    scan_truncated = true;
1706                    break;
1707                }
1708                scanned_files += 1;
1709                if let Some((score, family)) = workflow_path_role(&file.path, request) {
1710                    if path_filter.allows(&file.path) {
1711                        matches.push((file, score, family));
1712                    } else {
1713                        path_excluded.push(file.path);
1714                    }
1715                }
1716            }
1717            if scan_truncated {
1718                break;
1719            }
1720        }
1721        let count = |family| {
1722            matches
1723                .iter()
1724                .filter(|(_, _, candidate_family)| *candidate_family == family)
1725                .count()
1726        };
1727        let guidance_candidates = count("guidance");
1728        let template_candidates = count("template");
1729        let validation_candidates = count("validation");
1730        let owner_test_candidates = count("owner_test");

1732        for (family, candidates) in [
1733            ("guidance", guidance_candidates),
1734            ("template", template_candidates),
1735            ("validation", validation_candidates),
1736            ("owner_test", owner_test_candidates),
1737        ] {
1738            if candidates == 0 {
1739                missing_families.push(family.to_owned());
1740            }
1741        }
1742        if scan_truncated {
1743            missing_families.push("repository_scan_truncated".to_owned());
1744        }
1745        matches.sort_by(|left, right| {
1746            right
1747                .1
1748                .total_cmp(&left.1)
1749                .then_with(|| left.0.path.cmp(&right.0.path))
1750        });

1754            .iter()

1758                desired_end_line: 80,
1759                required_start_line: 1,
1760                required_end_line: 1,
1761                max_lines: 80,
1762            })
1763            .collect::<Vec<_>>();
1764        for ((file, score, family), excerpt) in matches
1765            .into_iter()
1766            .zip(self.stored_excerpts(session, &excerpt_requests)?)
1767        {
1768            let Some(excerpt) = excerpt else { continue };
1769            candidates.push(
1770                Candidate::new(
1771                    file.path,
1772                    excerpt.start_line,
1773                    excerpt.end_line,
1774                    excerpt.content,
1775                )
1776                .match_kind(format!("workflow_{family}"))
1777                .representation("workflow")
1778                .path_score(score)
1779                .focus_boost(score),
1780            );
1781        }
1782        Ok((
1783            Some(WorkflowReceipt {
1784                guidance_candidates,
1785                template_candidates,
1786                validation_candidates,
1787                owner_test_candidates,
1788                missing_families,
1789            }),
1790            path_excluded,
1791        ))

1794    fn build_diff_evidence(
1795        &self,
1796        session: &ReadSession,
1797        request: &ContextRequest,
1798        scope: &DiffScopeReceipt,
1799        cancellation: &CancellationToken,
1800    ) -> Result<DiffEvidenceReceipt> {
1801        let mut changed_symbols = Vec::new();
1802        let mut relationships = BTreeSet::new();
1803        let mut gaps = Vec::new();
1804        let changed_hunks = if let Some(base_revision) = &scope.base_revision {
1805            let immutable_range = request
1806                .base_revision
1807                .as_deref()
1808                .and_then(|revision| parse_revision_range(revision).ok().flatten())
1809                .is_some();
1810            let mut hunks = if immutable_range {
1811                let head_revision = scope.head_revision.as_deref().ok_or_else(|| {
1812                    Error::InternalFailure("immutable diff scope has no head revision".into())
1813                })?;
1814                git_diff_hunks_between(
1815                    &self.config.root,
1816                    base_revision,
1817                    head_revision,
1818                    MAX_DIFF_EVIDENCE_SYMBOLS + 1,
1819                )?
1820            } else {
1821                git_diff_hunks(
1822                    &self.config.root,
1823                    base_revision,
1824                    MAX_DIFF_EVIDENCE_SYMBOLS + 1,
1825                )?
1826            };
1827            if hunks.len() > MAX_DIFF_EVIDENCE_SYMBOLS {
1828                gaps.push("changed_hunk_evidence_truncated".into());
1829                hunks.truncate(MAX_DIFF_EVIDENCE_SYMBOLS);
1830            }
1831            hunks
1832                .into_iter()
1833                .map(|hunk| DiffHunkEvidence {
1834                    path: hunk.path,
1835                    start_line: hunk.start_line,
1836                    end_line: hunk.end_line,
1837                })
1838                .collect::<Vec<_>>()
1839        } else {
1840            gaps.push("hunk_ranges_unavailable_for_explicit_paths".into());
1841            Vec::new()
1842        };
1843        let scoped_paths = scope
1844            .changed_paths
1845            .iter()
1846            .take(MAX_DIFF_EVIDENCE_PATHS)
1847            .collect::<Vec<_>>();
1848        if scope.changed_paths.len() > scoped_paths.len() {
1849            gaps.push("changed_path_evidence_truncated".into());
1850        }
1851
1852        for path in &scoped_paths {
1853            check_cancelled(cancellation)?;
1854            let Some(file) = session.find_file(path)? else {
1855                gaps.push(format!("{path}:not_indexed_or_deleted"));
1856                continue;
1857            };
1858            if !file.structurally_complete {
1859                gaps.push(format!("{path}:structural_coverage_incomplete"));
1860            }
1861            let symbols = session.get_symbols_for_file(file.id, MAX_DIFF_EVIDENCE_SYMBOLS)?;
1862            let path_hunks = changed_hunks
1863                .iter()
1864                .filter(|hunk| hunk.path == **path)
1865                .collect::<Vec<_>>();
1866            let symbols = symbols
1867                .into_iter()
1868                .filter(|symbol| {
1869                    path_hunks.is_empty()
1870                        || path_hunks.iter().any(|hunk| {
1871                            hunk.start_line <= hunk.end_line
1872                                && symbol.start_line <= hunk.end_line
1873                                && symbol.end_line >= hunk.start_line
1874                        })
1875                })
1876                .collect::<Vec<_>>();
1877            if symbols.is_empty() && file.structurally_complete {
1878                gaps.push(format!("{path}:no_indexed_definitions"));
1879            }
1880            for symbol in symbols {
1881                if changed_symbols.len() == MAX_DIFF_EVIDENCE_SYMBOLS {
1882                    gaps.push("changed_symbol_evidence_truncated".into());
1883                    break;
1884                }
1885                let mut references = session.search_references(
1886                    &symbol.name,
1887                    true,
1888                    MAX_REFERENCES_PER_CHANGED_SYMBOL + 1,
1889                )?;
1890                if references.len() > MAX_REFERENCES_PER_CHANGED_SYMBOL {
1891                    gaps.push(format!(
1892                        "{path}:{}:reference_evidence_truncated",
1893                        symbol.name
1894                    ));
1895                    references.truncate(MAX_REFERENCES_PER_CHANGED_SYMBOL);
1896                }
1897                for reference in references {
1898                    if reference.path != **path {
1899                        relationships.insert((
1900                            (**path).clone(),
1901                            reference.path,
1902                            "reference".to_owned(),
1903                        ));
1904                    }
1905                }
1906                changed_symbols.push(DiffSymbolEvidence {
1907                    path: (**path).clone(),
1908                    name: symbol.name,
1909                    kind: symbol.kind,
1910                    start_line: symbol.start_line,
1911                    end_line: symbol.end_line,
1912                });
1913            }
1914            for importer in session.affected_importers(&[(**path).clone()])? {
1915                if importer != **path {
1916                    relationships.insert(((**path).clone(), importer, "importer".to_owned()));
1917                }
1918            }
1919        }
1920
1921        let mut cursor = None;
1922        let mut scanned_owner_test_files = 0;
1923        let mut owner_test_scan_truncated = false;
1924        loop {
1925            check_cancelled(cancellation)?;
1926            let page = session.list_files(512, cursor)?;
1927            let Some(last) = page.last() else {
1928                break;
1929            };
1930            cursor = Some(last.id);
1931            for file in page {
1932                if scanned_owner_test_files == MAX_OWNER_TEST_SCAN_FILES {
1933                    owner_test_scan_truncated = true;
1934                    break;
1935                }

2554                );
2555            }

2557            response.workflow_receipt = workflow_receipt;
2558            response.meta.freshness = self.freshness();
2559            response.meta.repository_id = self.repository_id();
2560            if let Some(mut scope) = diff_scope.clone() {
2561                let mut indexed = 0usize;
2562                for path in &scope.changed_paths {
2563                    if session.find_file(path)?.is_some() {
2564                        indexed += 1;
2565                    }
2566                }
2567                scope.indexed_changed_paths = indexed;
2568                scope.evidence = Some(self.build_diff_evidence(
2569                    session,
2570                    &scoped_request,
2571                    &scope,
2572                    cancellation,
2573                )?);
2574                response.routing = build_context_routing(
2575                    &request,
2576                    &scope,
2577                    candidate_path_count,
2578                    &selected_paths,
2579                );
2580                if let Some(routing) = &response.routing {
2581                    let concentration = if routing.weakly_concentrated {
2582                        "; selected evidence is weakly concentrated"
2583                    } else {
2584                        ""
2585                    };
2586                    response.warnings.push(format!(
2587                        "context spans {} changed paths across {} path groups{concentration}",
2588                        routing.changed_paths, routing.path_groups_total
2589                    ));
2590                }
2591                response.diff_scope = Some(scope);
2592            }
2593            if !request.plan_only {
2594                let receipt_candidates = response
2595                    .fragments
2596                    .iter()
2597                    .map(|fragment| {
2598                        ReceiptEvidence::new(
2599                            fragment.path.clone(),
2600                            fragment.start_line,
2601                            fragment.end_line,
2602                            fragment.content_hash.clone(),
2603                            Some(&fragment.content),
2604                        )
2605                    })
2606                    .collect::<Vec<_>>();

2608                    request.receipt_id.as_deref(),
2609                    generation,
2610                    &receipt_candidates,
2611                )?;
2612                response.fragments = response
2613                    .fragments
2614                    .into_iter()
2615                    .zip(&receipt.decisions)
2616                    .filter_map(|(fragment, decision)| {
2617                        matches!(
2618                            decision,
2619                            ReceiptDecision::Return | ReceiptDecision::ReturnNearDuplicate
2620                        )
2621                        .then_some(fragment)
2622                    })
2623                    .collect();

2625                    .fragments
2626                    .iter()
2627                    .map(|fragment| fragment.content_hash.clone())
2628                    .collect();

2630                    .fragments
2631                    .iter()
2632                    .map(|fragment| self.config.tokenizer.count(&fragment.content))
2633                    .sum();
2634                response.meta.emitted_tokens = response.meta.source_tokens;
2635                receipt.apply_meta(&mut response.meta);
2636                if response.meta.receipt_near_duplicates > 0 {
2637                    response.warnings.push(format!(
2638                        "{} returned fragments are semantic near-duplicates of prior receipt evidence",
2639                        response.meta.receipt_near_duplicates
2640                    ));
2641                }
2642                if response.fragments.is_empty() {
2643                    if response.meta.receipt_suppressed_exact
2644                        + response.meta.receipt_suppressed_overlap
2645                        > 0
2646                    {
2647                        response
2648                            .warnings
2649                            .push("all selected evidence was already covered by the receipt".into());
2650                    } else {
2651                        response
2652                            .warnings
2653                            .push("no relevant indexed evidence found".into());
2654                    }
2655                }
2656            }
2657            self.finalize_response(&mut response)?;
2658            let baseline_source_tokens = if request.plan_only {
2659                None
2660            } else {
2661                let paths = response
2662                    .fragments
2663                    .iter()
2664                    .map(|fragment| fragment.path.clone())
2665                    .collect::<BTreeSet<_>>()
2666                    .into_iter()
2667                    .collect::<Vec<_>>();
2668                session.whole_file_source_tokens(&paths, self.config.tokenizer.name())?
2669            };
2670            if let Some(started) = ranking_started {
2671                phases.timings.ranking_finalize_ms =
2672                    started.elapsed().as_secs_f64() * 1_000.0;
2673            }
2674            let (phases, timings, primitive_keys) =
2675                phases.finish(generated_candidates.len());
2676            Ok((
2677                ContextEvaluation {
2678                    response,
2679                    generated_candidate_paths,
2680                    generated_candidates,
2681                    phases,
2682                    timings,
2683                    primitive_keys,
2684                },
2685                baseline_source_tokens,
2686            ))
2687        })
2688    }
