129fn retain_ranked_focus_candidate(candidates: &mut Vec<FocusCandidate>, candidate: FocusCandidate) {
130    if let Some(existing) = candidates.iter_mut().find(|existing| {
131        existing.path == candidate.path
132            && existing.start_line == candidate.start_line
133            && existing.end_line == candidate.end_line
134    }) {
135        if candidate.relevance > existing.relevance {
136            *existing = candidate;
137        }
138    } else {
139        candidates.push(candidate);
140    }
141    candidates.sort_by(|left, right| {
142        right
143            .relevance
144            .total_cmp(&left.relevance)
145            .then_with(|| left.path.cmp(&right.path))
146            .then_with(|| left.start_line.cmp(&right.start_line))
147            .then_with(|| left.end_line.cmp(&right.end_line))
148    });
149    candidates.truncate(MAX_CONTEXT_FOCUS_CANDIDATES_PER_PATTERN);
150}

152fn parse_revision_range(revision: &str) -> Result<Option<(&str, &str)>> {
153    let Some((base, head)) = revision.split_once("..") else {
154        return Ok(None);
155    };
156    if base.trim().is_empty()
157        || head.trim().is_empty()
158        || base.ends_with('.')
159        || head.starts_with('.')
160        || head.contains("..")
161    {
162        return Err(Error::InvalidInput {
163            field: "base revision",
164            reason: "revision range must be BASE..HEAD",
165        });
166    }
167    Ok(Some((base, head)))
168}

203impl ContextPathScorer {
204    fn new(terms: &[String], task: &str) -> Self {
205        let code_token_parts = facets::legacy_code_tokens(task)
206            .into_iter()
207            .filter(|token| {
208                token.contains("::")
209                    || token
210                        .split('.')
211                        .any(|part| part.chars().next().is_some_and(char::is_uppercase))
212            })
213            .map(|code_token| {
214                expand_terms(&code_token)
215                    .into_iter()
216                    .map(|part| part.to_ascii_lowercase())
217                    .filter(|part| part.chars().count() >= 2)
218                    .collect::<HashSet<_>>()
219                    .into_iter()
220                    .collect()
221            })
222            .collect();
223        Self {
224            terms: terms.iter().map(|term| term.to_ascii_lowercase()).collect(),
225            code_token_parts,
226            languages: [
227                task_mentions_language(task, "javascript"),
228                task_mentions_language(task, "typescript"),
229                task_mentions_language(task, "python"),
230                task_mentions_language(task, "rust"),
231                task_mentions_language(task, "go"),
232            ],
233        }
234    }
235
236    fn score(&self, path: &str) -> f64 {
237        let path = path.to_lowercase();
238        let mut score = self
239            .terms
240            .iter()
241            .filter(|term| path.contains(term.as_str()))
242            .count() as f64;
243        for parts in &self.code_token_parts {
244            let matched_parts = parts.iter().filter(|part| path.contains(*part)).count();
245            if matched_parts >= 2 {
246                #[allow(clippy::cast_precision_loss)]
247                {
248                    score += (matched_parts * matched_parts) as f64;
249                }
250            }
251        }
252        for (mentioned, component, extensions) in [
253            (self.languages[0], "/js/", &["js", "jsx", "mjs", "cjs"][..]),
254            (self.languages[1], "/ts/", &["ts", "tsx"][..]),
255            (self.languages[2], "/python/", &["py", "pyw"][..]),
256            (self.languages[3], "/rust/", &["rs"][..]),
257            (self.languages[4], "/go/", &["go"][..]),
258        ] {
259            let extension_matches = path
260                .rsplit_once('.')
261                .is_some_and(|(_, extension)| extensions.contains(&extension));
262            let component = component.trim_matches('/');
263            let component_matches = path
264                .split('/')
265                .any(|path_component| path_component == component);
266            if mentioned && (extension_matches || component_matches) {
267                // An explicit language name in the task is strong repository-scope
268                // evidence. Keep this above an exact-name match in another
269                // language so common names such as `Point` do not dominate.
270                score += 12.0;
271            }
272        }
273        score
274    }
275}

277fn context_path_group(path: &str) -> String {
278    let mut components = path.split('/').filter(|component| !component.is_empty());
279    let first = components.next().unwrap_or("<root>");
280    let second = components.next();
281    if second.is_none() {
282        return "<root>".into();
283    }
284    if matches!(
285        first,
286        "src" | "lib" | "app" | "apps" | "crates" | "packages"
287    ) && let Some(second) = second
288    {
289        format!("{first}/{second}")
290    } else {
291        first.to_owned()
292    }
293}

295fn build_context_routing(
296    request: &ContextRequest,
297    scope: &DiffScopeReceipt,
298    candidate_paths: usize,
299    selected_paths: &[String],
300) -> Option<ContextRoutingReceipt> {
301    if scope.changed_paths.len() < OVERSIZED_CHANGE_PATHS {
302        return None;
303    }
304    let mut grouped_paths = BTreeMap::<String, Vec<String>>::new();
305    for path in &scope.changed_paths {
306        grouped_paths
307            .entry(context_path_group(path))
308            .or_default()
309            .push(path.clone());
310    }
311    if grouped_paths.len() < MIN_OVERSIZED_PATH_GROUPS {
312        return None;
313    }
314    let path_groups_total = grouped_paths.len();
315    let mut groups = grouped_paths.into_iter().collect::<Vec<_>>();
316    groups.sort_by(|left, right| {
317        right
318            .1
319            .len()
320            .cmp(&left.1.len())
321            .then_with(|| left.0.cmp(&right.0))
322    });
323    let selected_groups = selected_paths.iter().fold(
324        BTreeMap::<String, BTreeSet<&str>>::new(),
325        |mut paths, path| {
326            paths
327                .entry(context_path_group(path))
328                .or_default()
329                .insert(path);
330            paths
331        },
332    );
333    let selected_path_count = selected_paths
334        .iter()
335        .map(String::as_str)
336        .collect::<BTreeSet<_>>()
337        .len();
338    let strongest_selected_group = selected_groups
339        .values()
340        .map(BTreeSet::len)
341        .max()
342        .unwrap_or(0);
343    let weakly_concentrated = selected_path_count > 1
344        && strongest_selected_group.saturating_mul(2) <= selected_path_count;
345
346    let suggestions = groups
347        .iter()
348        .take(MAX_ROUTING_SUGGESTIONS)
349        .map(|(prefix, _)| ContextRoutingSuggestion {
350            include_paths: vec![if prefix == "<root>" {
351                "*".into()
352            } else {
353                format!("{prefix}/**")
354            }],
355        })
356        .collect();
357    let path_groups = groups
358        .into_iter()
359        .take(MAX_ROUTING_GROUPS)
360        .map(|(prefix, paths)| ContextPathGroup {
361            prefix,
362            changed_paths: paths.len(),
363        })
364        .collect();
365    Some(ContextRoutingReceipt {
366        candidate_paths,
367        changed_paths: scope.changed_paths.len(),
368        selected_paths: selected_path_count,
369        weakly_concentrated,
370        consistency: IndexConsistency::IndexedGeneration,
371        base_revision: request.base_revision.clone(),
372        known_hashes: request.known_hashes.clone(),
373        path_groups_total,
374        path_groups,
375        suggestions,
376    })
377}

379fn resolve_context_workflow(requested: ContextWorkflow, task: &str) -> ContextWorkflow {
380    if requested != ContextWorkflow::Auto {
381        return requested;
382    }
383    let task = task.to_ascii_lowercase();
384    if ["pull request", "contribution", "contributing"]
385        .iter()
386        .any(|signal| task.contains(signal))
387    {
388        ContextWorkflow::Contribution
389    } else if ["code review", "review this", "review the changes"]
390        .iter()
391        .any(|signal| task.contains(signal))
392    {
393        ContextWorkflow::Review
394    } else if ["investigate", "root cause", "diagnose"]
395        .iter()
396        .any(|signal| task.contains(signal))
397    {
398        ContextWorkflow::Investigation
399    } else {
400        ContextWorkflow::Implementation
401    }

404fn workflow_path_role(path: &str, request: &ContextRequest) -> Option<(f64, &'static str)> {
405    let normalized = path.replace('\\', "/");
406    let lower = normalized.to_ascii_lowercase();
407    let mut best = WORKFLOW_PATHS
408        .matches(&normalized)
409        .into_iter()
410        .map(|index| WORKFLOW_PATH_ROLES[index])
411        .max_by(|left, right| left.0.total_cmp(&right.0));
412    if likely_owner_test(&lower, request) && best.is_none_or(|(score, _)| score < OWNER_TEST_SCORE)
413    {
414        best = Some((OWNER_TEST_SCORE, "owner_test"));
415    }
416    best
417}

439const fn workflow_path_roles() -> [(f64, &'static str); WORKFLOW_PATH_RULES.len()] {
440    let mut roles = [(0.0, ""); WORKFLOW_PATH_RULES.len()];
441    let mut index = 0;
442    while index < WORKFLOW_PATH_RULES.len() {
443        roles[index] = (WORKFLOW_PATH_RULES[index].1, WORKFLOW_PATH_RULES[index].2);
444        index += 1;
445    }
446    roles
447}

467fn owner_test_changed_path(path: &str, request: &ContextRequest) -> Option<String> {
468    let path = path.replace('\\', "/").to_ascii_lowercase();
469    let is_test = path.contains("/test")
470        || path.starts_with("test")
471        || path.contains("/spec")
472        || path.starts_with("spec");
473    if !is_test {
474        return None;
475    }
476    let test_name = path.rsplit('/').next().unwrap_or(&path);
477    let test_stem = test_name.split('.').next().unwrap_or(test_name);
478    request
479        .changed_paths
480        .iter()
481        .chain(&request.focus_paths)
482        .find_map(|changed| {
483            let normalized = changed.replace('\\', "/");
484            let stem = normalized
485                .rsplit('/')
486                .next()
487                .and_then(|name| name.split('.').next())?
488                .to_ascii_lowercase();
489            (stem.len() >= 3 && contains_filename_token(test_stem, &stem)).then(|| changed.clone())
490        })
491}

493fn contains_filename_token(path: &str, token: &str) -> bool {
494    path.match_indices(token).any(|(start, matched)| {
495        let end = start + matched.len();
496        let boundary =
497            |character: Option<char>| character.is_none_or(|value| !value.is_alphanumeric());
498        boundary(path[..start].chars().next_back()) && boundary(path[end..].chars().next())
499    })
500}

545        phase: ContextTimedPhase,
546        operation: impl FnOnce() -> Result<T>,
547    ) -> Result<T> {
548        if !self.enabled {
549            return operation();
550        }
551        let started = Instant::now();
552        let output = operation()?;
553        self.record_elapsed(phase, Some(started));
554        Ok(output)
555    }

557    fn record_elapsed(&mut self, phase: ContextTimedPhase, started: Option<Instant>) {
558        let Some(started) = started else { return };
559        let elapsed_ms = started.elapsed().as_secs_f64() * 1_000.0;
560        let target = match phase {
561            ContextTimedPhase::ExactSymbolLookup => &mut self.timings.exact_symbol_lookup_ms,
562            ContextTimedPhase::SymbolSearch => &mut self.timings.symbol_search_ms,
563            ContextTimedPhase::ReferenceSearch => &mut self.timings.reference_search_ms,
564            ContextTimedPhase::LexicalSearch => &mut self.timings.lexical_search_ms,
565            ContextTimedPhase::LexicalVerify => &mut self.timings.lexical_verify_ms,
566            ContextTimedPhase::EnclosingLookup => &mut self.timings.enclosing_lookup_ms,
567            ContextTimedPhase::AdaptiveExcerpt => &mut self.timings.adaptive_excerpt_ms,
568            ContextTimedPhase::StoredExcerpt => &mut self.timings.stored_excerpt_ms,
569        };
570        *target += elapsed_ms;
571    }

615            .adaptive_excerpt_requests

787fn qualified_symbol_match(
788    concept: &str,
789    name: &str,
790    parent: Option<&str>,
791    signature: Option<&str>,
792) -> f64 {
793    if !concept.contains(['.', ':']) {
794        return 0.0;
795    }
796    let parts = concept
797        .split(|character: char| !character.is_alphanumeric() && character != '_')
798        .flat_map(identifier_words)
799        .map(|part| part.to_ascii_lowercase())
800        .filter(|part| part.chars().count() >= 2)
801        .collect::<HashSet<_>>();
802    if parts.len() < 2 {
803        return 0.0;
804    }
805    let haystack = format!(
806        "{} {} {}",
807        name,
808        parent.unwrap_or_default(),
809        signature.unwrap_or_default()
810    )
811    .to_ascii_lowercase();
812    f64::from(parts.iter().all(|part| haystack.contains(part)))
813}

815fn record_query_hit(
816    fusion: &mut HashMap<String, HashMap<String, f64>>,
817    path: &str,
818    fusion_key: &str,
819    weight: f64,
820    rank: usize,
821) {
822    if weight < MIN_CORROBORATED_QUERY_WEIGHT {
823        return;
824    }
825    const RRF_K: f64 = 60.0;
826    #[allow(clippy::cast_precision_loss)]
827    let score = weight * RRF_K / (RRF_K + rank as f64 + 1.0);
828    fusion
829        .entry(path.to_owned())
830        .or_default()
831        .entry(fusion_key.to_owned())
832        .and_modify(|current| *current = current.max(score))
833        .or_insert(score);
834}

837    candidates: &mut [Candidate],
838    fusion: &HashMap<String, HashMap<String, f64>>,
839) {

846            let strongest = matches.values().copied().fold(0.0, f64::max);

848            if !candidate
849                .match_kinds
850                .iter()
851                .any(|kind| kind == "multi-query")
852            {
853                candidate.match_kinds.push("multi-query".into());
854            }
855        }
856    }

881fn corroborated_import_symbol<'a>(
882    symbols: Vec<SymbolRecord>,
883    queries: &'a [ContextQuery],
884    seed_concepts: &BTreeSet<String>,
885) -> Option<(SymbolRecord, &'a ContextQuery, f64)> {
886    let mut best: Option<(usize, usize, usize, SymbolRecord, &ContextQuery, f64)> = None;
887    for (query_rank, query) in queries.iter().enumerate() {
888        if query.concept_weight < MIN_CORROBORATED_QUERY_WEIGHT
889            || !seed_concepts.contains(&query.fusion_key)
890            || !(query.has_facet(FacetKind::ExactAtom)
891                || query.has_facet(FacetKind::Symbol)
892                || query.has_facet(FacetKind::Configuration))
893        {
894            continue;
895        }
896        for symbol in &symbols {
897            let exact = symbol.name.eq_ignore_ascii_case(&query.value);
898            let qualified = qualified_symbol_match(
899                &query.fusion_key,
900                &symbol.name,
901                symbol.parent.as_deref(),
902                symbol.signature.as_deref(),
903            ) > 0.0;
904            if !exact && !qualified {
905                continue;
906            }
907            let class = usize::from(qualified) * 2 + usize::from(exact);
908            let evidence = f64::from(exact) + f64::from(qualified) * 1.5;
909            let candidate = (
910                class,
911                usize::MAX - query_rank,
912                usize::MAX - symbol.start_line,
913                symbol.clone(),
914                query,
915                evidence,
916            );
917            if best.as_ref().is_none_or(|current| {
918                (candidate.0, candidate.1, candidate.2) > (current.0, current.1, current.2)
919            }) {
920                best = Some(candidate);
921            }
922        }
923    }
924    best.map(|(_, _, _, symbol, query, evidence)| (symbol, query, evidence))
925}

927fn import_seed_paths(
928    candidates: &[Candidate],
929    queries: &[ContextQuery],
930    tokenizer: crate::tokens::Tokenizer,
931) -> Vec<(String, BTreeSet<String>)> {
932    if low_cardinality_exact_query(queries) {
933        return Vec::new();
934    }
935    let mut paths = BTreeMap::<String, (f64, BTreeSet<String>)>::new();
936    for candidate in candidates {
937        if candidate.concept_weight < MIN_CORROBORATED_QUERY_WEIGHT || candidate.concepts.is_empty()
938        {
939            continue;
940        }
941        let token_count = candidate.token_count_with(tokenizer).max(1);
942        let score = candidate.score(&ranking::Weights::default(), token_count);
943        let entry = paths
944            .entry(candidate.path.clone())
945            .or_insert_with(|| (score, BTreeSet::new()));
946        entry.0 = entry.0.max(score);
947        entry.1.extend(candidate.concepts.iter().cloned());
948    }
949    let mut paths = paths.into_iter().collect::<Vec<_>>();
950    paths.sort_by(|left, right| {
951        right
952            .1
953            .0
954            .total_cmp(&left.1.0)
955            .then_with(|| left.0.cmp(&right.0))
956    });
957    paths
958        .into_iter()
959        .map(|(path, (_, concepts))| (path, concepts))
960        .collect()
961}

1283            .iter()
1284            .chain(&request.must_include_symbols)
1285        {
1286            if seen_exact_names.insert(name.clone()) {
1287                exact_names.push(name.clone());
1288            }
1289        }
1290        phases.counters.exact_symbol_names = exact_names.len();
1291        let required_names = request
1292            .must_include_symbols
1293            .iter()
1294            .map(String::as_str)
1295            .collect::<HashSet<_>>();
1296        let mut exact_presence = HashSet::new();
1297        let mut allowed_required_hits = HashMap::<String, SymbolHit>::new();
1298        for names in exact_names.chunks(MAX_EXACT_SYMBOL_BATCH_NAMES) {
1299            check_cancelled(cancellation)?;
1300            phases.counters.exact_symbol_batches =
1301                phases.counters.exact_symbol_batches.saturating_add(1);
1302            let results = phases.measure(ContextTimedPhase::ExactSymbolLookup, || {
1303                session.find_symbols_exact_batch(names, MAX_IMPORT_SYMBOLS)
1304            })?;
1305            for (name, hits) in names.iter().zip(results) {
1306                phases.record_primitive("exact_symbol", || {
1307                    format!("case_sensitive:true:limit:{MAX_IMPORT_SYMBOLS}:name:{name}")
1308                });
1309                phases.counters.exact_symbol_hits =
1310                    phases.counters.exact_symbol_hits.saturating_add(hits.len());
1311                if hits.is_empty() {
1312                    continue;
1313                }
1314                exact_presence.insert(name.clone());
1315                if required_names.contains(name.as_str())
1316                    && let Some(hit) = hits.into_iter().find(|hit| path_filter.allows(&hit.path))
1317                {
1318                    allowed_required_hits.insert(name.clone(), hit);
1319                }
1320            }
1321        }
1322        for symbol in &request.focus_symbols {
1323            check_cancelled(cancellation)?;
1324            if !exact_presence.contains(symbol) {
1325                coverage.unmatched_focus_symbols.push(symbol.clone());
1326            }
1327        }
1328        let mut required_symbol_hits = Vec::<(String, SymbolHit)>::new();
1329        for symbol in &request.must_include_symbols {
1330            check_cancelled(cancellation)?;
1331            if !exact_presence.contains(symbol) {
1332                coverage.unmatched_must_include_symbols.push(symbol.clone());
1333                continue;
1334            }
1335            if let Some(hit) = allowed_required_hits.get(symbol).cloned() {
1336                required_symbol_hits.push((symbol.clone(), hit));
1337            }
1338        }
1339        let required_symbol_budget = request
1340            .token_budget
1341            .saturating_div(required_symbol_hits.len().max(1))
1342            .max(1);
1343        let symbol_excerpt_requests = required_symbol_hits
1344            .iter()
1345            .map(|(_, hit)| AdaptiveExcerptRequest {
1346                file_id: hit.symbol.file_id,
1347                declaration_start: hit.symbol.start_line,
1348                declaration_end: hit.symbol.end_line,
1349                matched_line: hit.symbol.start_line,
1350                token_budget: required_symbol_budget,
1351            })
1352            .collect::<Vec<_>>();
1353        phases.record_adaptive_excerpts(&symbol_excerpt_requests);
1354        let symbol_excerpts = phases.measure(ContextTimedPhase::AdaptiveExcerpt, || {
1355            self.adaptive_context_excerpts(session, &symbol_excerpt_requests)
1356        })?;
1357        for (((symbol, hit), excerpt), rank) in required_symbol_hits
1358            .into_iter()
1359            .zip(symbol_excerpts)
1360            .zip(0usize..)
1361        {
1362            let Some(excerpt) = excerpt else { continue };
1363            candidates.push(
1364                Candidate::new(
1365                    hit.path,
1366                    excerpt.start_line,
1367                    excerpt.end_line,
1368                    excerpt.content,
1369                )
1370                .match_kind("must_symbol")
1371                .concept(format!("must:symbol:{symbol}"), 2.0)
1372                .representation("required_symbol")
1373                .symbol_name(hit.symbol.name)
1374                .target_range(hit.symbol.start_line, hit.symbol.end_line)
1375                .exact(2.0)
1376                .symbol(2.0)
1377                .focus_boost(2.0)
1378                .channel("must_symbol", rank),
1379            );
1380        }
1381
1382        Ok(ContextConstraintExpansion {
1383            coverage,
1384            focus_paths: focus_path_files
1385                .into_iter()
1386                .zip(focus_path_matches)
1387                .zip(focus_path_eligible)
1388                .map(
1389                    |((files, indexed_paths), eligible_paths)| FocusPathResolution {
1390                        files,
1391                        indexed_paths,
1392                        eligible_paths,
1393                    },
1394                )
1395                .collect(),
1396        })
1397    }

1399    fn append_focus_candidates(
1400        &self,
1401        expansion: FocusExpansion<'_>,
1402        candidates: &mut Vec<Candidate>,
1403        phases: &mut ContextPhaseTracker,
1404    ) -> Result<Vec<String>> {
1405        let FocusExpansion {
1406            session,
1407            request,
1408            queries,
1409            path_scorer,
1410            resolutions,
1411            cancellation,
1412        } = expansion;
1413        let mut warnings = Vec::new();
1414        if !request.strict_focus_paths && request.minimum_fragments_per_focus_path.is_none() {
1415            return Ok(warnings);
1416        }
1417        for ((pattern, resolution), pattern_index) in
1418            request.focus_paths.iter().zip(resolutions).zip(0usize..)
1419        {
1420            check_cancelled(cancellation)?;
1421            if resolution.eligible_paths > MAX_CONTEXT_FOCUS_FILES_PER_PATTERN {
1422                warnings.push(format!(
1423                    "focus pattern `{pattern}` matched {} eligible indexed paths; \
1424                     candidate generation inspected the first {} paths",
1425                    resolution.eligible_paths, MAX_CONTEXT_FOCUS_FILES_PER_PATTERN
1426                ));
1427            }
1428            if resolution.files.is_empty() {
1429                if resolution.indexed_paths > 0 {
1430                    warnings.push(format!(
1431                        "focus pattern `{pattern}` has no candidate-eligible indexed path after \
1432                         include, exclude, generated-artifact, and strict changed-scope policy"
1433                    ));
1434                }
1435                continue;
1436            }
1437
1438            let mut semantic = Vec::new();
1439            let mut fallback = Vec::new();
1440            for file in &resolution.files {
1441                check_cancelled(cancellation)?;
1442                phases.record_primitive("focus_file_chunks", || {
1443                    format!(
1444                        "file_id:{}:limit:{MAX_CONTEXT_FOCUS_CHUNKS_PER_FILE}",
1445                        file.id
1446                    )
1447                });
1448                let chunks =
1449                    session.get_chunks_for_file(file.id, MAX_CONTEXT_FOCUS_CHUNKS_PER_FILE)?;
1450                phases.record_primitive("focus_file_symbols", || {
1451                    format!(
1452                        "file_id:{}:limit:{MAX_CONTEXT_FOCUS_SYMBOLS_PER_FILE}",
1453                        file.id
1454                    )
1455                });
1456                let symbols =
1457                    session.get_symbols_for_file(file.id, MAX_CONTEXT_FOCUS_SYMBOLS_PER_FILE)?;
1458
1459                for symbol in symbols {
1460                    let Some(chunk) = chunks.iter().find(|chunk| {
1461                        chunk.start_line <= symbol.start_line && chunk.end_line >= symbol.start_line
1462                    }) else {
1463                        continue;
1464                    };
1465                    let symbol_start = symbol.start_line.saturating_sub(chunk.start_line);
1466                    let symbol_lines = symbol
1467                        .end_line
1468                        .min(chunk.end_line)
1469                        .saturating_sub(symbol.start_line)
1470                        .saturating_add(1);
1471                    let symbol_content = chunk
1472                        .content
1473                        .lines()
1474                        .skip(symbol_start)
1475                        .take(symbol_lines)
1476                        .collect::<Vec<_>>()
1477                        .join("\n");
1478                    let searchable = format!(
1479                        "{} {} {} {} {}",
1480                        symbol.name,
1481                        symbol.kind,
1482                        symbol.parent.as_deref().unwrap_or_default(),
1483                        symbol.signature.as_deref().unwrap_or_default(),
1484                        symbol_content
1485                    );
1486                    let relevance = focus_text_relevance(&searchable, queries);
1487                    if relevance == 0.0 {
1488                        continue;
1489                    }
1490                    let candidate = Candidate::new(
1491                        &file.path,
1492                        chunk.start_line,
1493                        chunk.end_line,
1494                        &chunk.content,
1495                    )
1496                    .match_kind("focus_symbol")
1497                    .concept(format!("focus:path:{pattern}"), 2.0)
1498                    .representation("focus_symbol")
1499                    .symbol_name(symbol.name)
1500                    .target_range(symbol.start_line, symbol.end_line)
1501                    .exact(relevance.min(4.0))
1502                    .symbol(1.0)
1503                    .path_score(path_scorer.score(&file.path))
1504                    .focus_boost(2.0);
1505                    retain_ranked_focus_candidate(
1506                        &mut semantic,
1507                        FocusCandidate {
1508                            relevance: relevance + 1.0,
1509                            path: file.path.clone(),
1510                            start_line: chunk.start_line,
1511                            end_line: chunk.end_line,
1512                            candidate,
1513                        },
1514                    );
1515                }
1516
1517                let mut file_fallback = None;
1518                for chunk in chunks {
1519                    let relevance = focus_text_relevance(&chunk.content, queries);
1520                    let candidate =
1521                        Candidate::new(&file.path, chunk.start_line, chunk.end_line, chunk.content)
1522                            .match_kind(if relevance > 0.0 {
1523                                "focus_text"
1524                            } else {
1525                                "focus_fallback"
1526                            })
1527                            .concept(format!("focus:path:{pattern}"), 2.0)
1528                            .representation(if relevance > 0.0 {
1529                                "focus_text"
1530                            } else {
1531                                "focus_fallback"
1532                            })
1533                            .exact(relevance.min(4.0))
1534                            .path_score(path_scorer.score(&file.path))
1535                            .focus_boost(2.0);
1536                    let candidate = FocusCandidate {
1537                        relevance,
1538                        path: file.path.clone(),
1539                        start_line: chunk.start_line,
1540                        end_line: chunk.end_line,
1541                        candidate,
1542                    };
1543                    if relevance > 0.0 {
1544                        retain_ranked_focus_candidate(&mut semantic, candidate);
1545                    } else if file_fallback
1546                        .as_ref()
1547                        .is_none_or(|best: &FocusCandidate| candidate.start_line < best.start_line)
1548                    {
1549                        file_fallback = Some(candidate);
1550                    }
1551                }
1552                fallback.extend(file_fallback);
1553            }
1554
1555            let selected = if semantic.is_empty() {
1556                &mut fallback
1557            } else {
1558                &mut semantic
1559            };
1560            selected.sort_by(|left, right| {
1561                right
1562                    .relevance
1563                    .total_cmp(&left.relevance)
1564                    .then_with(|| left.path.cmp(&right.path))
1565                    .then_with(|| left.start_line.cmp(&right.start_line))
1566                    .then_with(|| left.end_line.cmp(&right.end_line))
1567            });
1568            let mut retained_ranges = BTreeSet::new();
1569            let mut retained = 0usize;
1570            for mut candidate in selected.drain(..) {
1571                if !retained_ranges.insert((
1572                    candidate.path.clone(),
1573                    candidate.start_line,
1574                    candidate.end_line,
1575                )) {
1576                    continue;
1577                }
1578                candidate.candidate = candidate.candidate.channel(
1579                    "focus_local",
1580                    pattern_index
1581                        .saturating_mul(MAX_CONTEXT_FOCUS_CANDIDATES_PER_PATTERN)
1582                        .saturating_add(retained),
1583                );
1584                candidates.push(candidate.candidate);
1585                retained = retained.saturating_add(1);
1586                if retained == MAX_CONTEXT_FOCUS_CANDIDATES_PER_PATTERN {
1587                    break;
1588                }
1589            }
1590            if retained == 0 {
1591                warnings.push(format!(
1592                    "focus pattern `{pattern}` matched indexed files without bounded chunk evidence"
1593                ));
1594            }
1595        }
1596        Ok(warnings)
1597    }

1599    fn finalize_strict_scope_coverage(
1600        &self,
1601        session: &ReadSession,
1602        request: &ContextRequest,
1603        selected_paths: &[String],
1604        coverage: &mut ContextCoverageReceipt,
1605    ) -> Result<()> {
1606        for focus in &mut coverage.focus_path_coverage {
1607            let matcher = PathMatcher::new(std::slice::from_ref(&focus.pattern))?;
1608            focus.selected_fragments = selected_paths
1609                .iter()
1610                .filter(|path| matcher.is_match(path))
1611                .count();
1612            focus.satisfied =
1613                focus.indexed_paths > 0 && focus.selected_fragments >= focus.minimum_fragments;
1614        }
1615
1616        if request.strict_changed_paths {
1617            let changed_paths = request
1618                .changed_paths
1619                .iter()
1620                .map(String::as_str)
1621                .collect::<HashSet<_>>();
1622            let mut indexed_paths = 0usize;
1623            for path in &request.changed_paths {
1624                if session.find_file(path)?.is_some() {
1625                    indexed_paths = indexed_paths.saturating_add(1);
1626                }
1627            }
1628            let selected_fragments = selected_paths
1629                .iter()
1630                .filter(|path| changed_paths.contains(path.as_str()))
1631                .count();
1632            coverage.changed_path_coverage = Some(ContextChangedPathCoverage {
1633                resolved_paths: changed_paths.len(),
1634                indexed_paths,
1635                selected_fragments,
1636                satisfied: !changed_paths.is_empty() && indexed_paths > 0 && selected_fragments > 0,
1637            });
1638        }
1639
1640        let focus_coverage_is_required =
1641            request.strict_focus_paths || request.minimum_fragments_per_focus_path.is_some();
1642        if focus_coverage_is_required || request.strict_changed_paths {
1643            coverage.strict_scope_satisfied = Some(
1644                (!focus_coverage_is_required
1645                    || coverage
1646                        .focus_path_coverage
1647                        .iter()
1648                        .all(|focus| focus.satisfied))
1649                    && coverage
1650                        .changed_path_coverage
1651                        .as_ref()
1652                        .is_none_or(|changed| changed.satisfied),
1653            );
1654        }
1655        Ok(())
1656    }

1680        &self,
1681        expansion: ImportExpansion<'_>,
1682        candidates: &mut Vec<Candidate>,
1683    ) -> Result<()> {
1684        let seed_paths = import_seed_paths(candidates, expansion.queries, self.config.tokenizer);
1685        let requested_paths = seed_paths
1686            .iter()
1687            .take(24)
1688            .map(|(path, _)| path.clone())
1689            .collect::<Vec<_>>();
1690        let targets =
1691            expansion
1692                .session
1693                .import_symbol_targets(&requested_paths, 32, MAX_IMPORT_SYMBOLS)?;
1694        let path_filter = PathFilter::new(
1695            &expansion.request.include_paths,
1696            &expansion.request.exclude_paths,
1697        )?;
1698        let mut pending = Vec::new();
1699        for target in targets {
1700            check_cancelled(expansion.cancellation)?;
1701            let Some((_, seed_concepts)) = seed_paths.get(target.seed_index) else {
1702                continue;
1703            };
1704            let target_path = &target.target_file.path;
1705            if !path_filter.allows(target_path) {
1706                continue;
1707            }
1708            let Some((symbol, query, exact)) =
1709                corroborated_import_symbol(target.symbols, expansion.queries, seed_concepts)
1710            else {
1711                continue;
1712            };
1713            pending.push((target.target_file, symbol, query.clone(), exact));
1714        }
1715        let excerpt_requests = pending
1716            .iter()
1717            .map(|(target_file, symbol, _, _)| AdaptiveExcerptRequest {
1718                file_id: target_file.id,
1719                declaration_start: symbol.start_line,
1720                declaration_end: symbol.end_line,
1721                matched_line: symbol.start_line,
1722                token_budget: excerpt_budget(
1723                    expansion.request.token_budget,
1724                    ContextExcerptKind::ImportSymbol,
1725                ),
1726            })
1727            .collect::<Vec<_>>();
1728        let excerpts = self.adaptive_context_excerpts(expansion.session, &excerpt_requests)?;
1729        let mut neighbor_count = 0usize;
1730        let mut neighbor_ranges = BTreeSet::new();
1731        for ((target_file, symbol, query, exact), excerpt) in pending.into_iter().zip(excerpts) {
1732            check_cancelled(expansion.cancellation)?;
1733            let Some(excerpt) = excerpt else { continue };
1734            let target_path = target_file.path;
1735            if !neighbor_ranges.insert((target_path.clone(), excerpt.start_line, excerpt.end_line))
1736            {
1737                continue;
1738            }
1739            let change_boost = Self::file_change_boost(
1740                Some(target_file.generation),
1741                &target_path,
1742                expansion.changed_paths,
1743                expansion.request.prior_repository_generation,
1744            );
1745            let candidate = Candidate::new(
1746                &target_path,
1747                excerpt.start_line,
1748                excerpt.end_line,
1749                excerpt.content,
1750            )
1751            .match_kind("import")
1752            .match_kind("symbol")
1753            .concept(&query.fusion_key, query.concept_weight)
1754            .representation("import_symbol")
1755            .symbol_name(symbol.name)
1756            .exact(exact)
1757            .symbol(1.0)
1758            .path_score(context_path_score(
1759                &target_path,
1760                expansion.terms,
1761                &expansion.request.task,
1762            ))
1763            .import_boost(1.0)
1764            .change_boost(change_boost);
1765            candidates.push(annotate_candidate(
1766                candidate,
1767                &query,
1768                "import_symbol",
1769                neighbor_count,
1770            ));
1771            neighbor_count += 1;
1772            if neighbor_count >= 24 {
1773                break;
1774            }
1775        }
1776        Ok(())
1777    }

1809    /// Resolve a diff scope from the request into a receipt, if one is supplied.
1810    ///
1811    /// A single `base_revision` resolves committed and working-tree paths since
1812    /// that revision, including untracked files. `BASE..HEAD` instead resolves
1813    /// an immutable revision range. Explicit `changed_paths` are merged with
1814    /// either result. When neither input is supplied, strict changed-path
1815    /// either result unless strict changed-path scope makes the explicit paths
1816    /// authoritative. When neither input is supplied, strict changed-path
1817    /// requests use the current working tree; otherwise `None` preserves
1818    /// task-only behavior.
1819    fn resolve_diff_scope(
1820        &self,
1821        request: &ContextRequest,
1822    ) -> Result<(Option<DiffScopeReceipt>, HashSet<String>, bool)> {
1823        let has_base = request
1824            .base_revision
1825            .as_deref()
1826            .is_some_and(|rev| !rev.trim().is_empty());
1827        let has_paths = !request.changed_paths.is_empty();
1828        let revision = request
1829            .base_revision
1830            .as_deref()
1831            .filter(|revision| !revision.trim().is_empty());
1832        let immutable_range = revision.map(parse_revision_range).transpose()?.flatten();
1833        let explicit_hard_scope = has_paths && request.strict_changed_paths;
1834        let git_result = match (revision, immutable_range) {
1835            (Some(_), Some((base, head))) if explicit_hard_scope => {
1836                Some(git_diff_identity(&self.config.root, base, Some(head))?)
1837            }
1838            (Some(revision), None) if explicit_hard_scope => {
1839                Some(git_diff_identity(&self.config.root, revision, None)?)
1840            }
1841            (Some(_), Some((base, head))) => Some(git_diff_paths_between(
1842                &self.config.root,
1843                base,
1844                head,
1845                MAX_DIFF_CHANGED_PATHS,
1846            )?),
1847            (Some(revision), None) => Some(git_diff_paths(
1848                &self.config.root,
1849                revision,
1850                MAX_DIFF_CHANGED_PATHS,
1851            )?),
1852            (None, None) => None,
1853            (None, Some(_)) => unreachable!("a range comes from a revision"),
1854        };
1855        let working_tree_status = git_working_tree_status(&self.config.root, GIT_CHANGED_PATHS_MAX);
1856        if !working_tree_status.available {
1857            tracing::debug!("working-tree signal unavailable");
1858        }
1859        let working_tree_paths = working_tree_status.changed_paths;
1860        let working_tree_state_available = working_tree_status.available;
1861        if !has_base && !has_paths && !request.strict_changed_paths {
1862            return Ok((None, working_tree_paths, working_tree_state_available));
1863        }
1864        if let Some(git_result) = git_result {
1865            let mut changed_paths = request.changed_paths.clone();
1866            if !explicit_hard_scope {
1867                let mut resolved_paths = git_result.changed_paths;
1868                if immutable_range.is_none() {
1869                    resolved_paths.extend(working_tree_paths.iter().cloned());
1870                }
1871                resolved_paths.sort();
1872                resolved_paths.dedup();
1873                for path in resolved_paths {
1874                    if changed_paths.len() == MAX_DIFF_CHANGED_PATHS {
1875                        break;
1876                    }
1877                    if !changed_paths.contains(&path) {
1878                        changed_paths.push(path);
1879                    }
1880                }
1881            }
1882            changed_paths.sort();
1883            changed_paths.dedup();
1884            return Ok((
1885                Some(DiffScopeReceipt {
1886                    base_revision: Some(git_result.base_revision),
1887                    head_revision: Some(git_result.head_revision),
1888                    changed_paths,
1889                    indexed_changed_paths: 0,
1890                    evidence: None,
1891                }),
1892                working_tree_paths,
1893                working_tree_state_available,
1894            ));
1895        }
1896        let mut resolved_paths = if has_paths {
1897            request.changed_paths.clone()
1898        } else {
1899            working_tree_paths.iter().cloned().collect::<Vec<_>>()
1900        };
1901        resolved_paths.sort();
1902        resolved_paths.dedup();

2376        let mut gaps = Vec::new();

2381            .is_some();

2383            let head_revision = if immutable_range {
2384                Some(scope.head_revision.as_deref().ok_or_else(|| {
2385                    Error::InternalFailure("immutable diff scope has no head revision".into())

2391                &self.config.root,
2392                base_revision,

2394                &scope.changed_paths,
2395                MAX_DIFF_EVIDENCE_SYMBOLS + 1,

2398                gaps.push("changed_hunk_evidence_truncated".into());

2400            }
2401            hunks
2402                .into_iter()

2404                    path: hunk.path,
2405                    start_line: hunk.start_line,
2406                    end_line: hunk.end_line,

2412        };
2413        let scoped_paths = scope
2414            .changed_paths
2415            .iter()
2416            .take(MAX_DIFF_EVIDENCE_PATHS)

2418        if scope.changed_paths.len() > scoped_paths.len() {
2419            gaps.push("changed_path_evidence_truncated".into());

2423            check_cancelled(cancellation)?;
2424            let Some(file) = session.find_file(path)? else {

2441                            hunk.start_line <= hunk.end_line
2442                                && symbol.start_line <= hunk.end_line
2443                                && symbol.end_line >= hunk.start_line
2444                        })
2445                })
2446                .collect::<Vec<_>>();

2465                    references.truncate(MAX_REFERENCES_PER_CHANGED_SYMBOL);
2466                }
2467                for reference in references {
2468                    if reference.path != **path {
2469                        relationships.insert((
2470                            (**path).clone(),
2471                            reference.path,
2472                            "reference".to_owned(),

2474                    }
2475                }
2476                changed_symbols.push(DiffSymbolEvidence {
2477                    path: (**path).clone(),
2478                    name: symbol.name,
2479                    kind: symbol.kind,
2480                    start_line: symbol.start_line,
2481                    end_line: symbol.end_line,
2482                });
2483            }
2484            for importer in session.affected_importers(&[(**path).clone()])? {
2485                if importer != **path {
2486                    relationships.insert(((**path).clone(), importer, "importer".to_owned()));
2487                }
2488            }
2489        }
2490
2491        let mut cursor = None;
2492        let mut scanned_owner_test_files = 0;
2493        let mut owner_test_scan_truncated = false;
2494        loop {
2495            check_cancelled(cancellation)?;
2496            let page = session.list_files(512, cursor)?;
2497            let Some(last) = page.last() else {
2498                break;
2499            };
2500            cursor = Some(last.id);
2501            for file in page {
2502                if scanned_owner_test_files == MAX_OWNER_TEST_SCAN_FILES {
2503                    owner_test_scan_truncated = true;
2504                    break;
2505                }
2506                scanned_owner_test_files += 1;
2507                if let Some(changed_path) = owner_test_changed_path(&file.path, request) {
2508                    relationships.insert((changed_path, file.path, "test_name_match".to_owned()));
2509                }
2510            }
2511            if owner_test_scan_truncated {
2512                break;
2513            }
2514        }
2515        if owner_test_scan_truncated {
2516            gaps.push("owner_test_scan_truncated".into());
2517        }
2518
2519        let semantic_change = if workflow == ContextWorkflow::Review && !request.plan_only {
2520            let semantic_paths = scoped_paths
2521                .iter()
2522                .map(|path| (*path).clone())
2523                .collect::<Vec<_>>();

2528                let head_revision = scope.head_revision.as_deref().ok_or_else(|| {

2538                )

2699        &self,
2700        response: &mut ContextResponse,
2701        request: &ContextRequest,
2702        max_response_tokens: usize,
2703    ) -> Result<()> {
2704        self.finalize_response(response)?;
2705        if self.context_response_fits(response, request, max_response_tokens)? {
2706            return Ok(());
2707        }
2708
2709        response.omitted.clear();
2710        response.omission_summary.by_path.clear();
2711        response.omission_summary.by_language_or_file_type.clear();
2712        response.omission_summary.by_reason.clear();

2714        if self.context_response_fits(response, request, max_response_tokens)? {
2715            self.finalize_response(response)?;
2716            return Ok(());
2717        }
2718
2719        if let Some(scope) = &mut response.diff_scope {
2720            scope.evidence = None;
2721        }
2722        if self.context_response_fits(response, request, max_response_tokens)? {
2723            self.finalize_response(response)?;
2724            return Ok(());
2725        }
2726
2727        response.routing = None;
2728        if self.context_response_fits(response, request, max_response_tokens)? {
2729            self.finalize_response(response)?;
2730            return Ok(());
2731        }
2732
2733        if let Some(plan) = &mut response.plan {
2734            for candidate in &mut plan.candidates {
2735                candidate.reasons.clear();
2736            }
2737        }
2738        for fragment in &mut response.fragments {
2739            fragment.reason.clear();
2740        }
2741        if self.context_response_fits(response, request, max_response_tokens)? {
2742            self.finalize_response(response)?;
2743            return Ok(());
2744        }
2745
2746        let can_reduce_selected = request.include_paths.is_empty()
2747            && request.must_include_paths.is_empty()
2748            && request.must_include_symbols.is_empty()
2749            && request.focus_paths.is_empty()
2750            && request.focus_symbols.is_empty()
2751            && !request.strict_focus_paths
2752            && request.minimum_fragments_per_focus_path.is_none()
2753            && request.base_revision.is_none()
2754            && request.changed_paths.is_empty()
2755            && !request.strict_changed_paths
2756            && response.handoff_manifest.is_none();
2757        if can_reduce_selected {
2758            let selected = response
2759                .plan
2760                .as_ref()
2761                .map_or(response.fragments.len(), |plan| plan.candidates.len());
2762            let omission_reserve = response
2763                .omission_summary
2764                .budget_or_result_limit
2765                .saturating_add(selected);
2766            let budget = ResponseBudget::new(&self.config.tokenizer, max_response_tokens);
2767            let keep = budget.largest_fitting_prefix(selected, |keep| {
2768                let mut candidate = response.clone();
2769                Self::trim_context_selection(&mut candidate, keep);
2770                candidate.omission_summary.budget_or_result_limit = omission_reserve;
2771                Self::refresh_context_omission_warning(&mut candidate);
2772                self.context_response_tokens_with_receipt_reserve(&candidate, request)
2773            })?;
2774            if let Some(keep) = keep {
2775                Self::trim_context_selection(response, keep);
2776                self.finalize_response(response)?;
2777                if self.context_response_fits(response, request, max_response_tokens)? {
2778                    return Ok(());
2779                }
2780                return Err(Error::InternalFailure(
2781                    "context prefix fitting violated its monotonic sizing reserve".into(),
2782                ));
2783            }
2784            Self::trim_context_selection(response, 0);
2785        }
2786
2787        let minimum = self.context_response_tokens_with_receipt_reserve(response, request)?;
2788        Err(Error::RequestLimitExceeded {
2789            field: "max_response_tokens",
2790            requested: minimum,
2791            limit: max_response_tokens,
2792        })
2793    }
