67    async fn rebase_receipt_execute(
68        &self,
69        request: ReceiptRebaseRequest,
70        execution: RetrievalExecution,
71    ) -> Result<ReceiptRebaseResponse> {
72        let operation = TokenAccountingOperation::ReceiptRebase;
73        let RetrievalExecution {
74            consistency,
75            options,
76            cancellation,
77        } = execution;
78        self.observe_service_result(operation, self.validate_call_options(options))?;
79        self.observe_service_result(operation, validate_rebase_request(&request))?;
80        if let Some(consistency) = consistency {
81            let consistency_result = self
82                .apply_consistency_with_initial_deadline(
83                    consistency,
84                    cancellation.clone(),
85                    options.initial_reconciliation_deadline(),
86                )
87                .await;
88            self.observe_service_result(operation, consistency_result)?;
89        }
90        let this = self.clone();
91        let result = self
92            .blocking_executor
93            .run(cancellation, move |cancellation| {
94                this.rebase_receipt_sync(request, options, cancellation)
95            })
96            .await;
97        self.observe_service_result(operation, result)
98    }
99
100    fn rebase_receipt_sync(
101        &self,
102        request: ReceiptRebaseRequest,
103        options: ServiceCallOptions,
104        cancellation: &CancellationToken,
105    ) -> Result<ReceiptRebaseResponse> {
106        check_cancelled(cancellation)?;
107        validate_rebase_request(&request)?;
108        let source = self
109            .storage
110            .load_receipt_rebase_source(&request.receipt_id)?;
111        let classification = self.consistent(|session| {
112            let generation = session.generation();
113            if source.repository_generation >= generation {
114                return Err(Error::InvalidInput {
115                    field: "receipt_id",
116                    reason: "must belong to an earlier repository generation",
117                });
118            }
119            classify_receipt(self, session, generation, &source, cancellation)
120        })?;
121        let requested_samples = request
122            .max_samples_per_outcome
123            .unwrap_or(DEFAULT_RECEIPT_REBASE_SAMPLES_PER_OUTCOME);
124        let mut selected = None;
125        for sample_limit in (0..=requested_samples).rev() {
126            let response = build_response(self, &source, &classification, sample_limit);
127            if self.response_fits(&response, options)? {
128                selected = Some(response);
129                break;
130            }
131        }
132        let mut response = if let Some(response) = selected {
133            response
134        } else {
135            let response = build_response(self, &source, &classification, 0);
136            let limit = options
137                .max_response_tokens()
138                .expect("response fitting fails only with a configured limit");
139            return Err(self.response_budget_error(&response, limit)?);
140        };
141        check_cancelled(cancellation)?;
142        let receipt_id = self.storage.persist_rebased_receipt(
143            &source,
144            classification.generation,
145            &classification.carried,
146        )?;
147        response.meta.receipt_id = Some(receipt_id);
148        self.finalize_response(&mut response)?;
149        self.record_token_savings_classified(
150            TokenAccountingOperation::ReceiptRebase,
151            None,
152            &response.meta,
153            TokenSavingsRequestClass::Useful,
154        );
155        Ok(response)
156    }
157}

159fn validate_rebase_request(request: &ReceiptRebaseRequest) -> Result<()> {
160    super::validation::validate_input(&request.receipt_id, "receipt_id", 128)?;
161    if request
162        .max_samples_per_outcome
163        .is_some_and(|value| value > MAX_RECEIPT_REBASE_SAMPLES_PER_OUTCOME)
164    {
165        return Err(Error::RequestLimitExceeded {
166            field: "max_samples_per_outcome",
167            requested: request.max_samples_per_outcome.unwrap_or_default(),
168            limit: MAX_RECEIPT_REBASE_SAMPLES_PER_OUTCOME,
169        });
170    }
171    Ok(())
172}

174fn classify_receipt(
175    services: &Services,
176    session: &IndexReadSnapshot,
177    generation: u64,
178    source: &ReceiptRebaseSource,
179    cancellation: &CancellationToken,
180) -> Result<ReceiptRebaseClassification> {
181    let mut order = (0..source.evidence.len()).collect::<Vec<_>>();
182    order.sort_by(|left, right| {
183        source.evidence[*left]
184            .path
185            .cmp(&source.evidence[*right].path)
186            .then_with(|| left.cmp(right))
187    });
188    let mut outcomes = vec![ReceiptRebaseOutcomeKind::Unmapped; source.evidence.len()];
189    let mut cursor = 0usize;
190    let mut live_bytes = 0u64;
191    while cursor < order.len() {
192        check_cancelled(cancellation)?;
193        let path = &source.evidence[order[cursor]].path;
194        let end = order[cursor..]
195            .iter()
196            .position(|index| source.evidence[*index].path != *path)
197            .map_or(order.len(), |offset| cursor.saturating_add(offset));
198        let indices = &order[cursor..end];
199        let Some(indexed) = session.find_file(path)? else {
200            for index in indices {
201                outcomes[*index] = ReceiptRebaseOutcomeKind::Missing;
202            }
203            cursor = end;
204            continue;
205        };
206        let (text, observed_bytes) = load_exact_current_text(
207            services,
208            path,
209            &indexed.content_hash,
210            MAX_REBASE_LIVE_BYTES.saturating_sub(live_bytes),
211        );
212        live_bytes = live_bytes.saturating_add(observed_bytes);
213        let Some(text) = text else {
214            cursor = end;
215            continue;
216        };
217        let line_starts = crate::text::line_starts(&text);
218        for index in indices {
219            check_cancelled(cancellation)?;
220            let evidence = &source.evidence[*index];
221            if evidence.start_line == 0
222                || evidence.end_line < evidence.start_line
223                || evidence.end_line > line_starts.len()
224            {
225                continue;
226            }
227            let (start, end) = crate::text::line_range_to_byte_range(
228                &line_starts,
229                text.len(),
230                evidence.start_line,
231                evidence.end_line,
232            );
233            let source_matches = crate::text::hash(&text[start..end]) == evidence.content_hash;
234            let structural_matches = if source_matches {
235                Some(true)
236            } else {
237                session.receipt_structural_hash_matches(
238                    indexed.id,
239                    evidence.start_line,
240                    evidence.end_line,
241                    &evidence.content_hash,
242                )?
243            };
244            outcomes[*index] = match structural_matches {
245                Some(true) => ReceiptRebaseOutcomeKind::Carried,
246                Some(false) => ReceiptRebaseOutcomeKind::Changed,
247                None => ReceiptRebaseOutcomeKind::Unmapped,
248            };
249        }
250        cursor = end;
251    }
252    let mut counts = ReceiptRebaseCounts::default();
253    let mut carried = Vec::new();
254    for (evidence, outcome) in source.evidence.iter().zip(&outcomes) {
255        match outcome {
256            ReceiptRebaseOutcomeKind::Carried => {
257                counts.carried = counts.carried.saturating_add(1);
258                carried.push(evidence.clone());
259            }
260            ReceiptRebaseOutcomeKind::Changed => {
261                counts.changed = counts.changed.saturating_add(1);
262            }
263            ReceiptRebaseOutcomeKind::Missing => {
264                counts.missing = counts.missing.saturating_add(1);
265            }
266            ReceiptRebaseOutcomeKind::Unmapped => {
267                counts.unmapped = counts.unmapped.saturating_add(1);
268            }
269        }
270    }
271    Ok(ReceiptRebaseClassification {
272        generation,
273        outcomes_blake3: classification_digest(source, generation, &outcomes),
274        outcomes,
275        carried,
276        counts,
277    })
278}

280fn load_exact_current_text(
281    services: &Services,
282    path: &str,
283    indexed_content_hash: &str,
284    remaining_bytes: u64,
285) -> (Option<String>, u64) {
286    let mut file = match open_live_file(services, path) {
287        Ok(file) => file,
288        Err(_) => return (None, 0),
289    };
290    let metadata_bytes = match file.metadata() {
291        Ok(metadata) => metadata.len(),
292        Err(_) => return (None, 0),
293    };
294    if metadata_bytes > services.config.max_file_bytes || metadata_bytes > remaining_bytes {
295        return (None, 0);
296    }
297    let mut bytes = Vec::new();
298    if (&mut file)
299        .take(metadata_bytes)
300        .read_to_end(&mut bytes)
301        .is_err()
302    {
303        return (None, 0);
304    }
305    let payload_bytes = u64::try_from(bytes.len()).unwrap_or(u64::MAX);
306    if payload_bytes != metadata_bytes {
307        return (None, payload_bytes);
308    }
309    let final_metadata_bytes = match file.metadata() {
310        Ok(metadata) => metadata.len(),
311        Err(_) => return (None, payload_bytes),
312    };
313    if final_metadata_bytes != metadata_bytes
314        || crate::text::hash_bytes(&bytes) != indexed_content_hash
315    {
316        return (None, payload_bytes);
317    }
318    (String::from_utf8(bytes).ok(), payload_bytes)
319}

321fn classification_digest(
322    source: &ReceiptRebaseSource,
323    generation: u64,
324    outcomes: &[ReceiptRebaseOutcomeKind],
325) -> String {
326    fn update(hasher: &mut blake3::Hasher, bytes: &[u8]) {
327        hasher.update(&(bytes.len() as u64).to_le_bytes());
328        hasher.update(bytes);
329    }
330    let mut hasher = blake3::Hasher::new();
331    hasher.update(b"leantoken-receipt-rebase-v1\0");
332    update(&mut hasher, source.receipt_id.as_bytes());
333    hasher.update(&source.repository_generation.to_le_bytes());
334    hasher.update(&generation.to_le_bytes());
335    for (ordinal, (evidence, outcome)) in source.evidence.iter().zip(outcomes).enumerate() {
336        hasher.update(&(ordinal as u64).to_le_bytes());
337        update(&mut hasher, evidence.path.as_bytes());
338        hasher.update(&(evidence.start_line as u64).to_le_bytes());
339        hasher.update(&(evidence.end_line as u64).to_le_bytes());
340        update(&mut hasher, evidence.content_hash.as_bytes());
341        hasher.update(&[match outcome {
342            ReceiptRebaseOutcomeKind::Carried => 0,
343            ReceiptRebaseOutcomeKind::Changed => 1,
344            ReceiptRebaseOutcomeKind::Missing => 2,
345            ReceiptRebaseOutcomeKind::Unmapped => 3,
346        }]);
347    }
348    hasher.finalize().to_hex().to_string()
349}

351fn build_response(
352    services: &Services,
353    source: &ReceiptRebaseSource,
354    classification: &ReceiptRebaseClassification,
355    sample_limit: usize,
356) -> ReceiptRebaseResponse {
357    let mut samples = ReceiptRebaseSamples::default();
358    for (ordinal, (evidence, outcome)) in source
359        .evidence
360        .iter()
361        .zip(&classification.outcomes)
362        .enumerate()
363    {
364        let bucket = match outcome {
365            ReceiptRebaseOutcomeKind::Carried => &mut samples.carried,
366            ReceiptRebaseOutcomeKind::Changed => &mut samples.changed,
367            ReceiptRebaseOutcomeKind::Missing => &mut samples.missing,
368            ReceiptRebaseOutcomeKind::Unmapped => &mut samples.unmapped,
369        };
370        if bucket.len() < sample_limit {
371            bucket.push(ReceiptRebaseSample {
372                ordinal,
373                path: evidence.path.clone(),
374                start_line: evidence.start_line,
375                end_line: evidence.end_line,
376            });
377        }
378    }
379    let sampled = samples
380        .carried
381        .len()
382        .saturating_add(samples.changed.len())
383        .saturating_add(samples.missing.len())
384        .saturating_add(samples.unmapped.len());
385    let mut meta = services.meta(classification.generation, 0, None);
386    meta.receipt_id = Some(RECEIPT_ID_RESPONSE_RESERVE.into());
387    ReceiptRebaseResponse {
388        source_receipt_id: source.receipt_id.clone(),
389        source_repository_generation: source.repository_generation,
390        counts: classification.counts.clone(),
391        samples,
392        samples_complete: sampled == classification.counts.total(),
393        outcomes_blake3: classification.outcomes_blake3.clone(),
394        meta,
395    }
396}

403    fn live_validation_refuses_a_file_before_crossing_the_cumulative_bound() {

405        let content = "0123456789";

407        let config = crate::Config::discover(root.path(), Some(root.path().join("index.sqlite")))
408            .expect("config");
409        let services = Services::open(config).expect("services");
410        let digest = crate::text::hash(content);
411
412        assert_eq!(
413            load_exact_current_text(&services, "bounded.rs", &digest, 9),
414            (None, 0)
415        );
416        assert_eq!(
417            load_exact_current_text(&services, "bounded.rs", &digest, 10),
418            (Some(content.into()), 10)
419        );
