67    async fn rebase_receipt_execute(
68        &self,
69        request: ReceiptRebaseRequest,
70        execution: RetrievalExecution,
71    ) -> Result<ReceiptRebaseResponse> {
72        let operation = TokenAccountingOperation::ReceiptRebase;
73        let RetrievalExecution {
74            consistency,
75            options,
76            cancellation,
77        } = execution;
78        self.observe_service_result(operation, self.validate_call_options(options))?;
79        let request = self.observe_service_result(operation, parse_rebase_request(request))?;
80        if let Some(consistency) = consistency {
81            let consistency_result = self
82                .apply_consistency_with_initial_deadline(
83                    consistency,
84                    cancellation.clone(),
85                    options.initial_reconciliation_deadline(),
86                )
87                .await;
88            self.observe_service_result(operation, consistency_result)?;
89        }
90        let this = self.clone();
91        let result = self
92            .blocking_executor
93            .run(cancellation, move |cancellation| {
94                this.rebase_receipt_sync(request, options, cancellation)
95            })
96            .await;
97        self.observe_service_result(operation, result)
98    }
99
100    fn rebase_receipt_sync(
101        &self,
102        request: ParsedReceiptRebaseRequest,
103        options: ServiceCallOptions,
104        cancellation: &CancellationToken,
105    ) -> Result<ReceiptRebaseResponse> {
106        check_cancelled(cancellation)?;
107        let source = self
108            .storage
109            .load_receipt_rebase_source(&request.receipt_id)?;
110        let classification = self.consistent(|session| {
111            let generation = session.generation();
112            if source.repository_generation >= generation {
113                return Err(Error::InvalidInput {
114                    field: "receipt_id",
115                    reason: "must belong to an earlier repository generation",
116                });
117            }
118            classify_receipt(self, session, generation, &source, cancellation)
119        })?;
120        let requested_samples = request.samples_per_outcome;
121        let mut selected = None;
122        for sample_limit in (0..=requested_samples).rev() {
123            let response = build_response(self, &source, &classification, sample_limit);
124            if self.response_fits_with_receipt_reserve(&response, 0, options)? {
125                selected = Some(response);
126                break;
127            }
128        }
129        let mut response = if let Some(response) = selected {
130            response
131        } else {
132            let response = build_response(self, &source, &classification, 0);
133            let limit = options
134                .max_response_tokens()
135                .expect("response fitting fails only with a configured limit");
136            return Err(
137                self.response_budget_error_with_receipt_reserve(&response, 0, limit, options)?
138            );
139        };
140        check_cancelled(cancellation)?;
141        let receipt_id = self.storage.persist_rebased_receipt(
142            &source,
143            classification.generation,
144            &classification.carried,
145        )?;
146        response.meta.receipt_id = Some(receipt_id);
147        self.finalize_bounded_response(&mut response, options)?;
148        self.record_token_savings_classified(
149            TokenAccountingOperation::ReceiptRebase,
150            None,
151            &response.meta,
152            TokenSavingsRequestClass::Useful,
153        );
154        Ok(response)
155    }

163fn parse_rebase_request(request: ReceiptRebaseRequest) -> Result<ParsedReceiptRebaseRequest> {
164    super::validation::validate_input(&request.receipt_id, "receipt_id", 128)?;
165    if request
166        .max_samples_per_outcome
167        .is_some_and(|value| value > MAX_RECEIPT_REBASE_SAMPLES_PER_OUTCOME)
168    {
169        return Err(Error::RequestLimitExceeded {
170            field: "max_samples_per_outcome",
171            requested: request.max_samples_per_outcome.unwrap_or_default(),
172            limit: MAX_RECEIPT_REBASE_SAMPLES_PER_OUTCOME,
173        });
174    }
175    Ok(ParsedReceiptRebaseRequest {
176        receipt_id: request.receipt_id,
177        samples_per_outcome: request
178            .max_samples_per_outcome
179            .unwrap_or(DEFAULT_RECEIPT_REBASE_SAMPLES_PER_OUTCOME),
180    })
181}

183fn classify_receipt(
184    services: &Services,
185    session: &IndexReadSnapshot,
186    generation: u64,
187    source: &ReceiptRebaseSource,
188    cancellation: &CancellationToken,
189) -> Result<ReceiptRebaseClassification> {
190    let mut order = (0..source.evidence.len()).collect::<Vec<_>>();
191    order.sort_by(|left, right| {
192        source.evidence[*left]
193            .path
194            .cmp(&source.evidence[*right].path)
195            .then_with(|| left.cmp(right))
196    });
197    let mut outcomes = vec![ReceiptRebaseOutcomeKind::Unmapped; source.evidence.len()];
198    let mut cursor = 0usize;
199    let mut live_bytes = 0u64;
200    while cursor < order.len() {
201        check_cancelled(cancellation)?;
202        let path = &source.evidence[order[cursor]].path;
203        let end = order[cursor..]
204            .iter()
205            .position(|index| source.evidence[*index].path != *path)
206            .map_or(order.len(), |offset| cursor.saturating_add(offset));
207        let indices = &order[cursor..end];
208        let Some(indexed) = session.find_file(path)? else {
209            for index in indices {
210                outcomes[*index] = ReceiptRebaseOutcomeKind::Missing;
211            }
212            cursor = end;
213            continue;
214        };
215        let (text, observed_bytes) = load_exact_current_text(
216            services,
217            path,
218            &indexed.content_hash,
219            MAX_REBASE_LIVE_BYTES.saturating_sub(live_bytes),
220        );
221        live_bytes = live_bytes.saturating_add(observed_bytes);
222        let Some(text) = text else {
223            cursor = end;
224            continue;
225        };
226        let line_starts = crate::text::line_starts(&text);
227        for index in indices {
228            check_cancelled(cancellation)?;
229            let evidence = &source.evidence[*index];
230            if evidence.start_line == 0
231                || evidence.end_line < evidence.start_line
232                || evidence.end_line > line_starts.len()
233            {
234                continue;
235            }
236            let (start, end) = crate::text::line_range_to_byte_range(
237                &line_starts,
238                text.len(),
239                evidence.start_line,
240                evidence.end_line,
241            );
242            let source_matches = crate::text::hash(&text[start..end]) == evidence.content_hash;
243            let structural_matches = if source_matches {
244                Some(true)
245            } else {
246                session.receipt_structural_hash_matches(
247                    indexed.id,
248                    evidence.start_line,
249                    evidence.end_line,
250                    &evidence.content_hash,
251                )?
252            };
253            outcomes[*index] = match structural_matches {
254                Some(true) => ReceiptRebaseOutcomeKind::Carried,
255                Some(false) => ReceiptRebaseOutcomeKind::Changed,
256                None => ReceiptRebaseOutcomeKind::Unmapped,
257            };
258        }
259        cursor = end;
260    }
261    let mut counts = ReceiptRebaseCounts::default();
262    let mut carried = Vec::new();
263    for (evidence, outcome) in source.evidence.iter().zip(&outcomes) {
264        match outcome {
265            ReceiptRebaseOutcomeKind::Carried => {
266                counts.carried = counts.carried.saturating_add(1);
267                carried.push(evidence.clone());
268            }
269            ReceiptRebaseOutcomeKind::Changed => {
270                counts.changed = counts.changed.saturating_add(1);
271            }
272            ReceiptRebaseOutcomeKind::Missing => {
273                counts.missing = counts.missing.saturating_add(1);
274            }
275            ReceiptRebaseOutcomeKind::Unmapped => {
276                counts.unmapped = counts.unmapped.saturating_add(1);
277            }
278        }
279    }
280    Ok(ReceiptRebaseClassification {
281        generation,
282        outcomes_blake3: classification_digest(source, generation, &outcomes),
283        outcomes,
284        carried,
285        counts,
286    })
287}

289fn load_exact_current_text(
290    services: &Services,
291    path: &str,
292    indexed_content_hash: &str,
293    remaining_bytes: u64,
294) -> (Option<String>, u64) {
295    let mut file = match open_live_file(services, path) {
296        Ok(file) => file,
297        Err(_) => return (None, 0),
298    };
299    let metadata_bytes = match file.metadata() {
300        Ok(metadata) => metadata.len(),
301        Err(_) => return (None, 0),
302    };
303    if metadata_bytes > services.config.max_file_bytes || metadata_bytes > remaining_bytes {
304        return (None, 0);
305    }
306    let mut bytes = Vec::new();
307    if (&mut file)
308        .take(metadata_bytes)
309        .read_to_end(&mut bytes)
310        .is_err()
311    {
312        return (None, 0);
313    }
314    let payload_bytes = u64::try_from(bytes.len()).unwrap_or(u64::MAX);
315    if payload_bytes != metadata_bytes {
316        return (None, payload_bytes);
317    }
318    let final_metadata_bytes = match file.metadata() {
319        Ok(metadata) => metadata.len(),
320        Err(_) => return (None, payload_bytes),
321    };
322    if final_metadata_bytes != metadata_bytes
323        || crate::text::hash_bytes(&bytes) != indexed_content_hash
324    {
325        return (None, payload_bytes);
326    }
327    (String::from_utf8(bytes).ok(), payload_bytes)
328}

330fn classification_digest(
331    source: &ReceiptRebaseSource,
332    generation: u64,
333    outcomes: &[ReceiptRebaseOutcomeKind],
334) -> String {
335    fn update(hasher: &mut blake3::Hasher, bytes: &[u8]) {
336        hasher.update(&(bytes.len() as u64).to_le_bytes());
337        hasher.update(bytes);
338    }
339    let mut hasher = blake3::Hasher::new();
340    hasher.update(b"leantoken-receipt-rebase-v1\0");
341    update(&mut hasher, source.receipt_id.as_bytes());
342    hasher.update(&source.repository_generation.to_le_bytes());
343    hasher.update(&generation.to_le_bytes());
344    for (ordinal, (evidence, outcome)) in source.evidence.iter().zip(outcomes).enumerate() {
345        hasher.update(&(ordinal as u64).to_le_bytes());
346        update(&mut hasher, evidence.path.as_bytes());
347        hasher.update(&(evidence.start_line as u64).to_le_bytes());
348        hasher.update(&(evidence.end_line as u64).to_le_bytes());
349        update(&mut hasher, evidence.content_hash.as_bytes());
350        hasher.update(&[match outcome {
351            ReceiptRebaseOutcomeKind::Carried => 0,
352            ReceiptRebaseOutcomeKind::Changed => 1,
353            ReceiptRebaseOutcomeKind::Missing => 2,
354            ReceiptRebaseOutcomeKind::Unmapped => 3,
355        }]);
356    }
357    hasher.finalize().to_hex().to_string()
358}

360fn build_response(
361    services: &Services,
362    source: &ReceiptRebaseSource,
363    classification: &ReceiptRebaseClassification,
364    sample_limit: usize,
365) -> ReceiptRebaseResponse {
366    let mut samples = ReceiptRebaseSamples::default();
367    for (ordinal, (evidence, outcome)) in source
368        .evidence
369        .iter()
370        .zip(&classification.outcomes)
371        .enumerate()
372    {
373        let bucket = match outcome {
374            ReceiptRebaseOutcomeKind::Carried => &mut samples.carried,
375            ReceiptRebaseOutcomeKind::Changed => &mut samples.changed,
376            ReceiptRebaseOutcomeKind::Missing => &mut samples.missing,
377            ReceiptRebaseOutcomeKind::Unmapped => &mut samples.unmapped,
378        };
379        if bucket.len() < sample_limit {
380            bucket.push(ReceiptRebaseSample {
381                ordinal,
382                path: evidence.path.clone(),
383                start_line: evidence.start_line,
384                end_line: evidence.end_line,
385            });
386        }
387    }
388    let sampled = samples
389        .carried
390        .len()
391        .saturating_add(samples.changed.len())
392        .saturating_add(samples.missing.len())
393        .saturating_add(samples.unmapped.len());
394    let mut meta = services.meta(classification.generation, 0, None);
395    meta.receipt_id = Some(RECEIPT_ID_RESPONSE_RESERVE.into());
396    ReceiptRebaseResponse {
397        source_receipt_id: source.receipt_id.clone(),
398        source_repository_generation: source.repository_generation,
399        counts: classification.counts.clone(),
400        samples,
401        samples_complete: sampled == classification.counts.total(),
402        outcomes_blake3: classification.outcomes_blake3.clone(),
403        meta,
404    }
405}

418        let services = Services::open(config).expect("services");

423            (None, 0)
424        );
425        assert_eq!(
426            load_exact_current_text(&services, "bounded.rs", &digest, 10),
427            (Some(content.into()), 10)
