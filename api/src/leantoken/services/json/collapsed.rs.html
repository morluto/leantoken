1//! Collapsed projection: bounded array sampling and object shrinking.
2
3use serde_json::{Map, Value, json};
4
5use super::projection::ProjectionState;
6
7pub(super) fn collapsed_item_count(value: &Value, array_sample_size: usize) -> usize {
8    match value {
9        Value::Array(values) => 1usize.saturating_add(
10            values
11                .iter()
12                .take(array_sample_size)
13                .map(|value| collapsed_item_count(value, array_sample_size))
14                .fold(0usize, usize::saturating_add),
15        ),
16        Value::Object(values) => 1usize.saturating_add(
17            values
18                .values()
19                .map(|value| collapsed_item_count(value, array_sample_size))
20                .fold(0usize, usize::saturating_add),
21        ),
22        _ => 1,
23    }
24}
25
26pub(super) fn collapse_json(value: &Value, state: &mut ProjectionState) -> Value {
27    if !super::projection::take_item(state) {
28        return Value::Null;
29    }
30    match value {
31        Value::Array(values) => {
32            let mut sample = Vec::new();
33            for value in values.iter().take(state.array_sample_size()) {
34                if state.remaining() == 0 {
35                    state.mark_incomplete();
36                    break;
37                }
38                sample.push(collapse_json(value, state));
39            }
40            let omitted = values.len().saturating_sub(sample.len());
41            json!({
42                "$array": {
43                    "count": values.len(),
44                    "sample": sample,
45                    "omitted": omitted,
46                }
47            })
48        }
49        Value::Object(values) => {
50            let mut projected = Map::new();
51            for (key, value) in values {
52                if state.remaining() == 0 {
53                    state.mark_incomplete();
54                    break;
55                }
56                projected.insert(key.clone(), collapse_json(value, state));
57            }
58            Value::Object(projected)
59        }
60        _ => value.clone(),
61    }
62}
