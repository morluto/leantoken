1//! Cursor state and normalized query identity for paginated JSON key projections.
2
3use serde_json::json;
4
5use super::execution::JsonExecutionOptions;
6use crate::model::JsonOperation;
7use crate::services::Services;
8use crate::services::cursor::{CursorEnvelope, CursorKind, StreamId, StreamIdentityBuilder};
9use crate::text::CONTENT_FINGERPRINT_HEX_LEN;
10use crate::{Error, Result};
11
12pub(super) const MAX_JSON_CURSOR_BYTES: usize = 256;
13const JSON_CURSOR_GENERATION: u64 = 0;
14
15pub(super) struct JsonCursor {
16    envelope: CursorEnvelope,
17}
18
19impl JsonCursor {
20    pub(super) fn offset_for(&self, source_hash: &str, stream_id: StreamId) -> Result<usize> {
21        let payload =
22            self.envelope
23                .payload_for(CursorKind::JsonKeys, JSON_CURSOR_GENERATION, stream_id)?;
24        if payload.len() != CONTENT_FINGERPRINT_HEX_LEN + size_of::<u64>() {
25            return Err(Error::StaleCursor);
26        }
27        let (cursor_source_hash, offset) = payload.split_at(CONTENT_FINGERPRINT_HEX_LEN);
28        if cursor_source_hash != source_hash.as_bytes() {
29            return Err(Error::StaleCursor);
30        }
31        let offset = u64::from_le_bytes(offset.try_into().map_err(|_| Error::StaleCursor)?);
32        let offset = usize::try_from(offset).map_err(|_| Error::StaleCursor)?;
33        if offset == 0 {
34            return Err(Error::StaleCursor);
35        }
36        Ok(offset)
37    }
38}
39
40fn is_fingerprint(value: &str) -> bool {
41    value.len() == CONTENT_FINGERPRINT_HEX_LEN && value.bytes().all(|byte| byte.is_ascii_hexdigit())
42}
43
44pub(super) fn decode_json_cursor(cursor: &str) -> Result<JsonCursor> {
45    CursorEnvelope::parse(cursor, MAX_JSON_CURSOR_BYTES).map(|envelope| JsonCursor { envelope })
46}
47
48pub(super) fn json_query_hash(
49    operation: &JsonOperation,
50    execution: JsonExecutionOptions,
51) -> Result<String> {
52    let serialized = serde_json::to_string(&json!({
53        "operation": operation,
54        "depth": execution.depth(),
55        "order": "depth_then_pointer",
56    }))
57    .map_err(|error| Error::SerializationFailure(error.to_string()))?;
58    Ok(crate::text::hash(&serialized))
59}
60
61pub(super) fn json_stream_id(services: &Services, query_hash: &str) -> StreamId {
62    let mut stream = StreamIdentityBuilder::for_service(services, CursorKind::JsonKeys);
63    stream.field_str("query_hash", query_hash);
64    stream.finish()
65}
66
67pub(super) fn make_json_cursor(
68    stream_id: StreamId,
69    source_hash: &str,
70    offset: usize,
71) -> Result<String> {
72    if !is_fingerprint(source_hash) || offset == 0 {
73        return Err(Error::OperationFailure(
74            "invalid JSON continuation state".into(),
75        ));
76    }
77    let mut payload = Vec::with_capacity(CONTENT_FINGERPRINT_HEX_LEN + size_of::<u64>());
78    payload.extend_from_slice(source_hash.as_bytes());
79    payload.extend_from_slice(
80        &u64::try_from(offset)
81            .map_err(|_| Error::OperationFailure("JSON cursor offset overflow".into()))?
82            .to_le_bytes(),
83    );
84    CursorEnvelope::new(
85        CursorKind::JsonKeys,
86        JSON_CURSOR_GENERATION,
87        stream_id,
88        payload,
89    )
90    .map(CursorEnvelope::encode)
91}
