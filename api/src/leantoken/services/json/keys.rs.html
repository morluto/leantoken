72fn collect_all_keys(
73    value: &Value,
74    pointer: &str,
75    depth: usize,
76    max_depth: Option<usize>,
77    keys: &mut BTreeMap<String, (usize, &'static str)>,
78) {
79    if !keys.contains_key(pointer) {
80        keys.insert(pointer.to_owned(), (depth, json_type(value)));
81    }
82    if max_depth.is_some_and(|maximum| depth >= maximum) {
83        return;
84    }
85    match value {
86        Value::Object(values) => {
87            for (key, value) in values {
88                let pointer = format!("{pointer}/{}", escape_pointer(key));
89                collect_all_keys(value, &pointer, depth.saturating_add(1), max_depth, keys);
90            }
91        }
92        Value::Array(values) => {
93            let pointer = format!("{pointer}/*");
94            for value in values {
95                collect_all_keys(value, &pointer, depth.saturating_add(1), max_depth, keys);
96            }
97        }
98        _ => {}
99    }
100}

102pub(super) fn key_entries(
103    value: &Value,
104    max_depth: Option<usize>,
105    order: JsonKeyOrder,
106) -> Vec<Value> {
107    let mut keys = BTreeMap::new();
108    collect_all_keys(value, "", 0, max_depth, &mut keys);
109    let mut entries = keys.into_iter().collect::<Vec<_>>();
110    if order == JsonKeyOrder::DepthThenPointer {
111        entries.sort_by(
112            |(left_pointer, (left_depth, _)), (right_pointer, (right_depth, _))| {
113                left_depth
114                    .cmp(right_depth)
115                    .then_with(|| left_pointer.cmp(right_pointer))
116            },
117        );
118    }
119    entries
120        .into_iter()
121        .map(|(pointer, (_, value_type))| json!({"pointer": pointer, "type": value_type}))
122        .collect()
123}

125fn key_prefix_tokens(services: &Services, entries: &[Value], length: usize) -> Result<usize> {
126    json_tokens(services, &Value::Array(entries[..length].to_vec()))
127}
128
129fn largest_key_prefix_within_tokens(
130    services: &Services,
131    entries: &[Value],
132    max_tokens: usize,
133) -> Result<(usize, usize)> {
134    if entries.is_empty() {
135        return Ok((0, json_tokens(services, &Value::Array(Vec::new()))?));
136    }
137    let full_tokens = key_prefix_tokens(services, entries, entries.len())?;
138    if full_tokens <= max_tokens {
139        return Ok((entries.len(), full_tokens));
140    }
141
142    let mut lower = 0usize;
143    let mut upper = entries.len();
144    while lower < upper {
145        let middle = lower.saturating_add(upper).saturating_add(1) / 2;
146        if key_prefix_tokens(services, entries, middle)? <= max_tokens {
147            lower = middle;
148        } else {
149            upper = middle.saturating_sub(1);
150        }
151    }
152    let tokens = key_prefix_tokens(services, entries, lower)?;
153    Ok((lower, tokens))
154}

156pub(super) fn project_key_page(
157    services: &Services,
158    value: &Value,
159    max_items: usize,
160    max_tokens: usize,
161    context: KeyProjectionContext<'_>,
162) -> Result<KeyProjectionPage> {
163    let entries = key_entries(
164        value,
165        context.execution.depth(),
166        context.execution.key_order(),
167    );
168    let total_items = entries.len();
169    let offset = match context.cursor {
170        Some(cursor) if cursor.matches(context.source_hash, context.query_hash) => cursor.offset(),
171        Some(_) => return Err(Error::StaleCursor),
172        None => 0,
173    };
174    if offset > total_items || (offset == total_items && offset != 0) {
175        return Err(Error::StaleCursor);
176    }
177
178    let page_end = offset.saturating_add(max_items).min(total_items);
179    let candidates = &entries[offset..page_end];
180    let (returned_items, projected_tokens) =
181        largest_key_prefix_within_tokens(services, candidates, max_tokens)?;
182    if returned_items == 0 && !candidates.is_empty() {
183        return Err(Error::RequestLimitExceeded {
184            field: "one projected JSON item tokens",
185            requested: key_prefix_tokens(services, candidates, 1)?,
186            limit: max_tokens,
187        });
188    }
189
190    let consumed = offset.saturating_add(returned_items);
191    let remaining_items = total_items.saturating_sub(consumed);
192    let incomplete_reason = (remaining_items > 0).then_some(if returned_items < candidates.len() {
193        crate::model::JsonIncompleteReason::MaxTokens
194    } else {
195        crate::model::JsonIncompleteReason::MaxItems
196    });
197    let next_cursor = (remaining_items > 0).then(|| {
198        make_json_cursor(
199            context.execution.cursor_version(),
200            context.source_hash,
201            context.query_hash,
202            consumed,
203        )
204    });
205    Ok(KeyProjectionPage {
206        value: Value::Array(candidates[..returned_items].to_vec()),
207        total_items,
208        returned_items,
209        remaining_items,
210        incomplete_reason,
211        next_cursor,
212        projected_tokens,
213    })
214}

216/// Streaming keys projection used by the standard `project_json` path. Honors
217/// the shared `ProjectionState` item budget and deduplicates pointers.
218pub(super) fn collect_keys(
219    value: &Value,
220    pointer: &str,
221    keys: &mut BTreeMap<String, &'static str>,
222    state: &mut ProjectionState,
223) {
224    if !keys.contains_key(pointer) {
225        if !take_item(state) {
226            return;
227        }
228        keys.insert(pointer.to_owned(), json_type(value));
229    }
230    match value {
231        Value::Object(values) => {
232            for (key, value) in values {
233                let pointer = format!("{pointer}/{}", escape_pointer(key));
234                collect_keys(value, &pointer, keys, state);
235            }
236        }
237        Value::Array(values) => {
238            let pointer = format!("{pointer}/*");
239            for value in values {
240                collect_keys(value, &pointer, keys, state);
241            }
242        }
243        _ => {}
244    }
