13/// Work counters for one schema projection call, used by diagnostics and the
14/// schema profile example to measure candidate reuse.
15#[derive(Debug, Clone, Default, PartialEq, Eq)]
16pub(super) struct SchemaProjectionCounters {
17    /// Number of `build_schema_breadth_first` invocations.
18    pub(super) schema_builds: usize,
19    /// Number of `serde_json::to_string` calls on schema candidates.
20    pub(super) schema_serializations: usize,
21    /// Number of tokenizer count calls on serialized schema candidates.
22    pub(super) schema_token_counts: usize,
23    /// Number of times a cached `MeasuredProjection` was reused.
24    pub(super) cache_hits: usize,
25}
26
27/// Cached projection of one schema candidate: the rendered JSON value, its
28/// serialized string, and the exact tokenizer count over that string. The
29/// serialized string is retained so the final selection does not need to be
30/// re-serialized for token accounting.
31pub(super) struct MeasuredProjection {
32    value: Value,
33    /// Cached serialized representation. Retained for the projection cache
34    /// contract so callers that need the serialized form (e.g. response
35    /// fitting diagnostics) can reuse it without re-serializing.
36    #[allow(dead_code)]
37    serialized: String,
38    tokens: usize,
39}
40
41impl MeasuredProjection {
42    fn measure(
43        services: &Services,
44        value: Value,
45        counters: &mut SchemaProjectionCounters,
46    ) -> Result<Self> {
47        let serialized = serde_json::to_string(&value)
48            .map_err(|error| crate::Error::SerializationFailure(error.to_string()))?;
49        counters.schema_serializations = counters.schema_serializations.saturating_add(1);
50        let tokens = services.config.tokenizer.count(&serialized);
51        counters.schema_token_counts = counters.schema_token_counts.saturating_add(1);
52        Ok(Self {
53            value,
54            serialized,
55            tokens,
56        })
57    }
58
59    pub(super) fn tokens(&self) -> usize {
60        self.tokens
61    }
62
63    pub(super) fn into_value(self) -> Value {
64        self.value
65    }
66}

68/// Request-local plan that memoizes `MeasuredProjection` results by `max_items`
69/// across the binary fit loop so already-computed candidates are reused.
70pub(super) struct SchemaProjectionPlan<'a> {
71    services: &'a Services,
72    cache: BTreeMap<usize, MeasuredProjection>,
73    counters: SchemaProjectionCounters,
74}
75
76impl<'a> SchemaProjectionPlan<'a> {
77    fn new(services: &'a Services) -> Self {
78        Self {
79            services,
80            cache: BTreeMap::new(),
81            counters: SchemaProjectionCounters::default(),
82        }
83    }
84
85    /// Build and measure a breadth-first schema at `max_items`, or return a
86    /// previously cached measurement. The schema value, serialized string, and
87    /// exact token count are all reused on cache hits.
88    fn candidate(&mut self, value: &Value, max_items: usize) -> Result<&MeasuredProjection> {
89        if !self.cache.contains_key(&max_items) {
90            let built = build_schema_breadth_first(value, max_items);
91            self.counters.schema_builds = self.counters.schema_builds.saturating_add(1);
92            let measured = MeasuredProjection::measure(self.services, built, &mut self.counters)?;
93            self.cache.insert(max_items, measured);
94        } else {
95            self.counters.cache_hits = self.counters.cache_hits.saturating_add(1);
96        }
97        Ok(self
98            .cache
99            .get(&max_items)
100            .expect("entry was just inserted or cached"))
101    }
102
103    fn counters(&self) -> &SchemaProjectionCounters {
104        &self.counters
105    }
106}

142enum SchemaNodeKind {
143    Scalar(&'static str),
144    Object(BTreeMap<String, usize>),
145    Array { count: usize, variants: Vec<usize> },
146}
147
148struct SchemaNode {
149    kind: SchemaNodeKind,
150    pointer: String,
151}
152
153fn schema_node(value: &Value, pointer: String) -> SchemaNode {
154    let kind = match value {
155        Value::Object(_) => SchemaNodeKind::Object(BTreeMap::new()),
156        Value::Array(values) => SchemaNodeKind::Array {
157            count: values.len(),
158            variants: Vec::new(),
159        },
160        _ => SchemaNodeKind::Scalar(json_type(value)),
161    };
162    SchemaNode { kind, pointer }
163}
164
165fn schema_node_has_children(value: &Value) -> bool {
166    match value {
167        Value::Object(values) => !values.is_empty(),
168        Value::Array(values) => !values.is_empty(),
169        _ => false,
170    }
171}
172
173fn schema_frontier(queued: &VecDeque<(&Value, usize, String)>, omitted: &mut BTreeSet<String>) {
174    for (value, _, omission_root) in queued {
175        if schema_node_has_children(value) {
176            omitted.insert(omission_root.clone());
177        }
178    }
179}
180
181fn render_schema_node(nodes: &[SchemaNode], node_id: usize) -> Value {
182    match &nodes[node_id].kind {
183        SchemaNodeKind::Scalar(value_type) => json!({"type": value_type}),
184        SchemaNodeKind::Object(children) => {
185            let properties = children
186                .iter()
187                .map(|(key, child)| (key.clone(), render_schema_node(nodes, *child)))
188                .collect::<Map<_, _>>();
189            json!({"type": "object", "properties": properties})
190        }
191        SchemaNodeKind::Array { count, variants } => {
192            let mut unique = BTreeMap::new();
193            for variant in variants {
194                let schema = render_schema_node(nodes, *variant);
195                let key = serde_json::to_string(&schema).unwrap_or_default();
196                unique.entry(key).or_insert(schema);
197            }
198            let items = match unique.len() {
199                0 => json!({}),
200                1 => unique.into_values().next().unwrap_or_else(|| json!({})),
201                _ => json!({"anyOf": unique.into_values().collect::<Vec<_>>()}),
202            };
203            json!({"type": "array", "count": count, "items": items})
204        }
205    }
206}

208pub(super) fn build_schema_breadth_first(value: &Value, max_items: usize) -> Value {
209    debug_assert!(max_items > 0);
210    let mut nodes = vec![schema_node(value, String::new())];
211    let mut queue = VecDeque::from([(value, 0usize, String::new())]);
212    let mut omitted = BTreeSet::new();
213
214    'build: while let Some((current, node_id, omission_root)) = queue.pop_front() {
215        match current {
216            Value::Object(values) => {
217                for (key, child) in values {
218                    let pointer = format!("{}/{}", nodes[node_id].pointer, escape_pointer(key));
219                    let child_omission_root = if node_id == 0 {
220                        pointer.clone()
221                    } else {
222                        omission_root.clone()
223                    };
224                    if nodes.len() >= max_items {
225                        omitted.insert(child_omission_root);
226                        for remaining_key in values.keys().skip_while(|value| *value != key).skip(1)
227                        {
228                            omitted.insert(if node_id == 0 {
229                                format!(
230                                    "{}/{}",
231                                    nodes[node_id].pointer,
232                                    escape_pointer(remaining_key)
233                                )
234                            } else {
235                                omission_root.clone()
236                            });
237                        }
238                        schema_frontier(&queue, &mut omitted);
239                        break 'build;
240                    }
241                    let child_id = nodes.len();
242                    nodes.push(schema_node(child, pointer));
243                    let SchemaNodeKind::Object(children) = &mut nodes[node_id].kind else {
244                        unreachable!("object source has object schema node");
245                    };
246                    children.insert(key.clone(), child_id);
247                    if schema_node_has_children(child) {
248                        queue.push_back((child, child_id, child_omission_root));
249                    }
250                }
251            }
252            Value::Array(values) => {
253                let pointer = format!("{}/*", nodes[node_id].pointer);
254                let child_omission_root = if node_id == 0 {
255                    pointer.clone()
256                } else {
257                    omission_root.clone()
258                };
259                for child in values {
260                    if nodes.len() >= max_items {
261                        omitted.insert(child_omission_root.clone());
262                        schema_frontier(&queue, &mut omitted);
263                        break 'build;
264                    }
265                    let child_id = nodes.len();
266                    nodes.push(schema_node(child, pointer.clone()));
267                    let SchemaNodeKind::Array { variants, .. } = &mut nodes[node_id].kind else {
268                        unreachable!("array source has array schema node");
269                    };
270                    variants.push(child_id);
271                    if schema_node_has_children(child) {
272                        queue.push_back((child, child_id, child_omission_root.clone()));
273                    }
274                }
275            }
276            _ => {}
277        }
278    }
279
280    let mut schema = render_schema_node(&nodes, 0);
281    if !omitted.is_empty()
282        && let Value::Object(root) = &mut schema
283    {
284        let omitted_subtree_count = omitted.len();
285        let omitted_subtree_pointers = omitted
286            .into_iter()
287            .take(MAX_SCHEMA_OMITTED_POINTERS)
288            .collect::<Vec<_>>();
289        root.insert(
290            "x-leantoken-incomplete".into(),
291            json!({
292                "omitted_subtree_count": omitted_subtree_count,
293                "omitted_subtree_pointers": omitted_subtree_pointers,
294            }),
295        );
296    }
297    schema
298}

300pub(super) fn project_schema_page(
301    services: &Services,
302    value: &Value,
303    max_items: usize,
304    max_tokens: usize,
305) -> Result<SchemaProjection> {
306    let total_items = count_nodes(value);
307    let item_limit = total_items.min(max_items).max(1);
308    let mut plan = SchemaProjectionPlan::new(services);
309    let item_limited = plan.candidate(value, item_limit)?;
310    let item_limited_tokens = item_limited.tokens();
311    let (returned_items, schema, projected_tokens) = if item_limited_tokens <= max_tokens {
312        (
313            item_limit,
314            plan.cache
315                .remove(&item_limit)
316                .expect("item_limit was just measured")
317                .into_value(),
318            item_limited_tokens,
319        )
320    } else {
321        let root = plan.candidate(value, 1)?;
322        let root_tokens = root.tokens();
323        if root_tokens > max_tokens {
324            return Err(crate::Error::RequestLimitExceeded {
325                field: "one projected JSON item tokens",
326                requested: root_tokens,
327                limit: max_tokens,
328            });
329        }
330        let mut lower = 1usize;
331        let mut upper = item_limit.saturating_sub(1);
332        let mut best_items = 1usize;
333        let mut best_key = 1usize;
334        let mut best_tokens = root_tokens;
335        while lower <= upper {
336            let middle = lower.saturating_add(upper.saturating_sub(lower) / 2);
337            let candidate = plan.candidate(value, middle)?;
338            let candidate_tokens = candidate.tokens();
339            if candidate_tokens <= max_tokens {
340                lower = middle.saturating_add(1);
341                best_items = middle;
342                best_key = middle;
343                best_tokens = candidate_tokens;
344            } else {
345                upper = middle.saturating_sub(1);
346            }
347        }
348        let schema = plan
349            .cache
350            .remove(&best_key)
351            .expect("best key was cached during binary search")
352            .into_value();
353        (best_items, schema, best_tokens)
354    };
355    let counters = plan.counters().clone();
356    let remaining_items = total_items.saturating_sub(returned_items);
357    let incomplete_reason = (remaining_items > 0).then_some(if returned_items < item_limit {
358        crate::model::JsonIncompleteReason::MaxTokens
359    } else {
360        crate::model::JsonIncompleteReason::MaxItems
361    });
362    Ok(SchemaProjection {
363        value: schema,
364        total_items,
365        returned_items,
366        remaining_items,
367        incomplete_reason,
368        projected_tokens,
369        counters,
370    })
