13pub(super) fn effective_context_response_profile(
14    request: &ContextRequest,
15    options: ServiceCallOptions,
16) -> Result<ContextResponseProfile> {
17    match (
18        options.context_response_profile(),
19        request.verbose_diagnostics,
20    ) {
21        (Some(ContextResponseProfile::Compact | ContextResponseProfile::Balanced), true) => {
22            Err(Error::InvalidInput {
23                field: "response_profile",
24                reason: "verbose_diagnostics=true requires response_profile=explain",
25            })
26        }
27        (Some(profile), _) => Ok(profile),
28        (None, verbose) => Ok(ContextResponseProfile::from_legacy_verbose(verbose)),
29    }
30}

39pub(super) fn merge_selected_coverage(
40    mut coverage: ContextCoverageReceipt,
41    response: &mut ContextResponse,
42) -> ContextCoverageReceipt {
43    coverage.covered_must_include_paths =
44        std::mem::take(&mut response.coverage.covered_must_include_paths);
45    coverage.covered_must_include_symbols =
46        std::mem::take(&mut response.coverage.covered_must_include_symbols);
47    coverage.partial_must_include_symbols =
48        std::mem::take(&mut response.coverage.partial_must_include_symbols);
49    coverage.uncovered_must_include_paths =
50        std::mem::take(&mut response.coverage.uncovered_must_include_paths);
51    coverage.uncovered_must_include_symbols =
52        std::mem::take(&mut response.coverage.uncovered_must_include_symbols);
53    for (target, selected) in coverage
54        .required_evidence
55        .iter_mut()
56        .zip(std::mem::take(&mut response.coverage.required_evidence))
57    {
58        target.matched_queries = selected.matched_queries;
59        target.unmatched_queries = selected.unmatched_queries;
60        target.selected_fragments = selected.selected_fragments;
61        target.satisfied = target.indexed_paths > 0 && selected.satisfied;
62    }
63    for (target, selected) in coverage
64        .focus_path_coverage
65        .iter_mut()
66        .zip(std::mem::take(&mut response.coverage.focus_path_coverage))
67    {
68        if let Some(mut selected) = selected.diagnostics {
69            selected.eligible_paths = target
70                .diagnostics
71                .as_ref()
72                .map_or(0, |diagnostics| diagnostics.eligible_paths);
73            target.diagnostics = Some(selected);
74        }
75    }
76    if !coverage.required_evidence.is_empty() {
77        coverage.evidence_scope_satisfied =
78            Some(coverage.required_evidence.iter().all(|item| item.satisfied));
79    }
80    coverage
81        .uncovered_must_include_paths
82        .retain(|pattern| !coverage.unmatched_must_include_paths.contains(pattern));
83    coverage
84        .uncovered_must_include_symbols
85        .retain(|symbol| !coverage.unmatched_must_include_symbols.contains(symbol));
86    coverage
87}

89pub(super) fn selected_paths(response: &ContextResponse) -> Vec<String> {
90    response.plan.as_ref().map_or_else(
91        || {
92            response
93                .fragments
94                .iter()
95                .map(|fragment| fragment.path.clone())
96                .collect()
97        },
98        |plan| {
99            plan.candidates
100                .iter()
101                .map(|candidate| candidate.path.clone())
102                .collect()
103        },
104    )
105}

107pub(super) fn append_coverage_warnings(response: &mut ContextResponse) {
108    let uncovered = response
109        .coverage
110        .uncovered_must_include_paths
111        .len()
112        .saturating_add(response.coverage.uncovered_must_include_symbols.len());
113    if uncovered > 0 {
114        response.warnings.push(format!(
115            "{uncovered} indexed must-cover requirements were not selected"
116        ));
117    }
118    let partial = response.coverage.partial_must_include_symbols.len();
119    if partial > 0 {
120        let subject = if partial == 1 {
121            "1 required symbol was".to_owned()
122        } else {
123            format!("{partial} required symbols were")
124        };
125        response.warnings.push(format!(
126            "{subject} returned only partially; inspect target ranges and truncated fragments"
127        ));
128    }
129    let unmatched = response
130        .coverage
131        .unmatched_must_include_paths
132        .len()
133        .saturating_add(response.coverage.unmatched_must_include_symbols.len());
134    if unmatched > 0 {
135        response.warnings.push(format!(
136            "{unmatched} must-cover requirements matched no indexed evidence"
137        ));
138    }
139    let unsatisfied_evidence = response
140        .coverage
141        .required_evidence
142        .iter()
143        .filter(|item| !item.satisfied)
144        .count();
145    if unsatisfied_evidence > 0 {
146        response.warnings.push(format!(
147            "{unsatisfied_evidence} required evidence contracts lacked matching selected evidence"
148        ));
149    }
150    let bounded_evidence_paths = response
151        .coverage
152        .required_evidence
153        .iter()
154        .filter(|item| item.indexed_paths > item.inspected_paths)
155        .count();
156    if bounded_evidence_paths > 0 {
157        response.warnings.push(format!(
158            "{bounded_evidence_paths} required evidence contracts matched more indexed paths than the bounded local inspection covered"
159        ));
160    }
161    let unmatched_hints = response
162        .coverage
163        .unmatched_focus_paths
164        .len()
165        .saturating_add(response.coverage.unmatched_focus_symbols.len())
166        .saturating_add(response.coverage.unmatched_include_paths.len());
167    if unmatched_hints > 0 {
168        response.warnings.push(format!(
169            "{unmatched_hints} focus or include constraints matched no indexed evidence"
170        ));
171    }
172    let underfilled_focus_paths = response
173        .coverage
174        .focus_path_coverage
175        .iter()
176        .filter(|focus| !focus.satisfied)
177        .count();
178    if underfilled_focus_paths > 0 {
179        response.warnings.push(format!(
180            "{underfilled_focus_paths} focus path constraints did not meet minimum fragment coverage"
181        ));
182    }
183    if response
184        .coverage
185        .changed_path_coverage
186        .as_ref()
187        .is_some_and(|changed| !changed.satisfied)
188    {
189        response
190            .warnings
191            .push("strict changed-path scope produced no indexed selected evidence".into());
192    }
193}

195impl Services {
196    pub(super) fn finalize_context_delivery(
197        &self,
198        response: &mut ContextResponse,
199        finalization: ContextResponseFinalization<'_>,
200    ) -> Result<Option<usize>> {
201        let ContextResponseFinalization {
202            session,
203            request,
204            options,
205            generation,
206        } = finalization;
207        if let Some(max_response_tokens) = options.max_response_tokens() {
208            self.fit_context_response(response, request, max_response_tokens)?;
209        }
210        if !request.plan_only {
211            let receipt_candidates = response
212                .fragments
213                .iter()
214                .map(|fragment| {
215                    ReceiptEvidence::new(
216                        fragment.path.clone(),
217                        fragment.start_line,
218                        fragment.end_line,
219                        fragment.content_hash.clone(),
220                        Some(&fragment.content),
221                    )
222                })
223                .collect::<Vec<_>>();
224            let receipt = self.evaluate_receipt(
225                request.receipt_id.as_deref(),
226                generation,
227                &receipt_candidates,
228            )?;
229            response.fragments = response
230                .fragments
231                .drain(..)
232                .zip(&receipt.decisions)
233                .filter_map(|(fragment, decision)| {
234                    matches!(
235                        decision,
236                        ReceiptDecision::Return | ReceiptDecision::ReturnNearDuplicate
237                    )
238                    .then_some(fragment)
239                })
240                .collect();
241            response.receipt.fragment_hashes = response
242                .fragments
243                .iter()
244                .map(|fragment| fragment.content_hash.clone())
245                .collect();
246            response.meta.source_tokens = response
247                .fragments
248                .iter()
249                .map(|fragment| self.config.tokenizer.count(&fragment.content))
250                .sum();
251            receipt.apply_meta(&mut response.meta);
252            if response.meta.receipt_near_duplicates > 0 {
253                response.warnings.push(format!(
254                    "{} returned fragments are semantic near-duplicates of prior receipt evidence",
255                    response.meta.receipt_near_duplicates
256                ));
257            }
258            if response.fragments.is_empty() {
259                if response.meta.receipt_suppressed_exact + response.meta.receipt_suppressed_overlap
260                    > 0
261                {
262                    response
263                        .warnings
264                        .push("all selected evidence was already covered by the receipt".into());
265                } else if response.omission_summary.budget_or_result_limit == 0 {
266                    response
267                        .warnings
268                        .push("no relevant indexed evidence found".into());
269                }
270            }
271        }
272        if let Some(manifest) = &mut response.handoff_manifest {
273            manifest.receipt_id.clone_from(&response.meta.receipt_id);
274        }
275        self.finalize_response(response)?;
276        if let Some(max_response_tokens) = options.max_response_tokens()
277            && response.meta.total_response_tokens > max_response_tokens
278        {
279            return Err(Error::ResponseAccountingInvariant(
280                "context response exceeded its fitted serialized-response budget".into(),
281            ));
282        }
283        if request.plan_only {
284            return Ok(None);
285        }
286        let paths = response
287            .fragments
288            .iter()
289            .map(|fragment| fragment.path.clone())
290            .collect::<BTreeSet<_>>()
291            .into_iter()
292            .collect::<Vec<_>>();
293        session.whole_file_source_tokens(&paths, self.config.tokenizer.name())
294    }
