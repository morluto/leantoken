37pub(super) fn retain_required_evidence_plan(
38    plans: &mut Vec<RequiredEvidenceExcerptPlan>,
39    plan: RequiredEvidenceExcerptPlan,
40) {
41    if plans
42        .iter()
43        .any(|existing| existing.path == plan.path && existing.matched_line == plan.matched_line)
44    {
45        return;
46    }
47    let insertion = plans
48        .binary_search_by(|existing| {
49            plan.relevance
50                .total_cmp(&existing.relevance)
51                .then_with(|| existing.path.cmp(&plan.path))
52                .then_with(|| existing.matched_line.cmp(&plan.matched_line))
53        })
54        .unwrap_or_else(|index| index);
55    if insertion < MAX_CONTEXT_FOCUS_CANDIDATES_PER_PATTERN {
56        plans.insert(insertion, plan);
57        plans.truncate(MAX_CONTEXT_FOCUS_CANDIDATES_PER_PATTERN);
58    }
59}

70pub(super) fn retain_focus_file(files: &mut Vec<FileRecord>, file: &FileRecord) {
71    let insertion = files

80pub(super) fn retain_required_evidence_files(
81    file: &FileRecord,
82    matchers: &[PathMatcher],
83    path_filter: &PathFilter,
84    strict_changed_paths: Option<&HashSet<&str>>,
85    path_matches: &mut [usize],
86    path_files: &mut [Vec<FileRecord>],
87) {
88    for ((matcher, matches), files) in matchers
89        .iter()
90        .zip(path_matches.iter_mut())
91        .zip(path_files.iter_mut())
92    {
93        if !matcher.is_match(&file.path) {
94            continue;
95        }
96        *matches = matches.saturating_add(1);
97        if path_filter.allows(&file.path)
98            && strict_changed_paths.is_none_or(|paths| paths.contains(file.path.as_str()))
99        {
100            retain_focus_file(files, file);
101        }
102    }
103}

105pub(super) fn focus_text_relevance(text: &str, queries: &[ContextQuery]) -> f64 {
106    let normalized = text.to_lowercase();
107    queries
108        .iter()
109        .filter(|query| !query.has_facet(FacetKind::TestIntent))
110        .filter_map(|query| {
111            normalized
112                .contains(&query.value.to_lowercase())
113                .then_some(query.weight)
114        })
115        .sum()
116}

118pub(super) fn required_evidence_query_matches(text: &str, queries: &[String]) -> Vec<usize> {
119    let normalized = text.to_lowercase();
120    queries
121        .iter()
122        .enumerate()
123        .filter_map(|(index, query)| normalized.contains(&query.to_lowercase()).then_some(index))
124        .collect()
125}

127pub(super) fn retain_ranked_focus_candidate(
128    candidates: &mut Vec<FocusCandidate>,
129    candidate: FocusCandidate,
130) {
131    if let Some(existing) = candidates.iter_mut().find(|existing| {
132        existing.path == candidate.path
133            && existing.start_line == candidate.start_line
134            && existing.end_line == candidate.end_line
135    }) {
136        if candidate.relevance > existing.relevance {
137            *existing = candidate;
138        }
139    } else {
140        candidates.push(candidate);
141    }
142    candidates.sort_by(|left, right| {
143        right
144            .relevance
145            .total_cmp(&left.relevance)
146            .then_with(|| left.path.cmp(&right.path))
147            .then_with(|| left.start_line.cmp(&right.start_line))
148            .then_with(|| left.end_line.cmp(&right.end_line))
149    });
150    candidates.truncate(MAX_CONTEXT_FOCUS_CANDIDATES_PER_PATTERN);
151}
