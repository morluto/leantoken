2    pub(super) fn context_response_with_receipt_reserve(
3        &self,
4        response: &ContextResponse,
5        request: &ContextRequest,
6    ) -> Result<ContextResponse> {
7        let mut sized = response.clone();
8        if !request.plan_only {
9            let receipt_id = request
10                .receipt_id
11                .clone()
12                .unwrap_or_else(|| crate::receipt::RECEIPT_ID_RESPONSE_RESERVE.into());
13            let selected = sized.fragments.len();
14            sized.meta.receipt_id = Some(receipt_id.clone());
15            sized.meta.receipt_suppressed_exact = selected;
16            sized.meta.receipt_suppressed_overlap = selected;
17            sized.meta.receipt_near_duplicates = selected;
18            sized.warnings.push(format!(
19                "{selected} returned fragments are semantic near-duplicates of prior receipt evidence"
20            ));
21            sized
22                .warnings
23                .push("all selected evidence was already covered by the receipt".into());
24            if let Some(manifest) = &mut sized.handoff_manifest {
25                manifest.receipt_id = Some(receipt_id);
26            }
27        }
28        set_routing_consistency(&mut sized, IndexConsistency::ReconcileWorkingTree);
29        self.finalize_response(&mut sized)?;
30        Ok(sized)
31    }

33    pub(super) fn context_response_tokens_with_receipt_reserve(
34        &self,
35        response: &ContextResponse,
36        request: &ContextRequest,
37    ) -> Result<usize> {
38        let sized = self.context_response_with_receipt_reserve(response, request)?;
39        let budget = ResponseBudget::new(&self.config.tokenizer, usize::MAX);
40        let serialized_tokens = budget.serialized_tokens(&sized)?;
41        debug_assert_eq!(serialized_tokens, sized.meta.total_response_tokens);
42        Ok(serialized_tokens)
43    }

45    pub(super) fn context_response_budget_error(
46        &self,
47        response: &ContextResponse,
48        request: &ContextRequest,
49        provided_max_response_tokens: usize,
50    ) -> Result<Error> {
51        let minimum_required_response_tokens =
52            self.context_response_tokens_with_receipt_reserve(response, request)?;
53        let mut mandatory = response.clone();
54        set_routing_consistency(&mut mandatory, IndexConsistency::ReconcileWorkingTree);
55        self.finalize_response(&mut mandatory)?;
56        Ok(Self::response_budget_exceeded(
57            &mandatory.meta,
58            provided_max_response_tokens,
59            minimum_required_response_tokens,
60        ))
61    }

75    pub(super) fn refresh_context_omission_warning(response: &mut ContextResponse) {
76        response.warnings.retain(|warning| {
77            warning
78                .strip_suffix(" omitted")
79                .is_none_or(|count| count.parse::<usize>().is_err())
80        });
81        let omitted = response
82            .omission_summary
83            .path_excluded
84            .saturating_add(response.omission_summary.known_hash)
85            .saturating_add(response.omission_summary.budget_or_result_limit);
86        if omitted > 0 {
87            response.warnings.insert(0, format!("{omitted} omitted"));
88        }
89    }

91    pub(super) fn trim_context_selection(response: &mut ContextResponse, keep: usize) {
92        let (removed, removed_tokens) = if let Some(plan) = &mut response.plan {
93            let removed = plan.candidates.len().saturating_sub(keep);
94            let removed_tokens = plan
95                .candidates
96                .iter()
97                .skip(keep)
98                .map(|candidate| candidate.estimated_tokens)
99                .sum::<usize>();
100            plan.candidates.truncate(keep);
101            plan.estimated_source_tokens =
102                plan.estimated_source_tokens.saturating_sub(removed_tokens);
103            plan.result_complete &= removed == 0;
104            (removed, removed_tokens)
105        } else {
106            let removed = response.fragments.len().saturating_sub(keep);
107            let removed_tokens = response
108                .fragments
109                .iter()
110                .skip(keep)
111                .map(|fragment| fragment.token_count)
112                .sum::<usize>();
113            response.fragments.truncate(keep);
114            response.receipt.fragment_hashes.truncate(keep);
115            (removed, removed_tokens)
116        };
117        response.meta.source_tokens = response.meta.source_tokens.saturating_sub(removed_tokens);
118        response.omission_summary.budget_or_result_limit = response
119            .omission_summary
120            .budget_or_result_limit
121            .saturating_add(removed);
122        Self::refresh_context_omission_warning(response);
123    }

125    pub(super) fn fit_context_response(
126        &self,
127        response: &mut ContextResponse,
128        request: &ContextRequest,
129        max_response_tokens: usize,
130    ) -> Result<()> {
131        self.finalize_response(response)?;
132        if self.context_response_fits(response, request, max_response_tokens)? {
133            return Ok(());
134        }
135
136        response.omitted.clear();
137        response.omission_summary.by_path.clear();
138        response.omission_summary.by_language_or_file_type.clear();
139        response.omission_summary.by_reason.clear();
140        response.omission_summary.by_score_band.clear();
141        if self.context_response_fits(response, request, max_response_tokens)? {
142            self.finalize_response(response)?;
143            return Ok(());
144        }
145
146        if let Some(scope) = &mut response.diff_scope {
147            scope.evidence = None;
148        }
149        if self.context_response_fits(response, request, max_response_tokens)? {
150            self.finalize_response(response)?;
151            return Ok(());
152        }
153
154        response.routing = None;
155        if self.context_response_fits(response, request, max_response_tokens)? {
156            self.finalize_response(response)?;
157            return Ok(());
158        }
159
160        if let Some(plan) = &mut response.plan {
161            for candidate in &mut plan.candidates {
162                candidate.reasons.clear();
163            }
164        }
165        for fragment in &mut response.fragments {
166            fragment.reason.clear();
167        }
168        if self.context_response_fits(response, request, max_response_tokens)? {
169            self.finalize_response(response)?;
170            return Ok(());
171        }
172
173        let can_reduce_selected = request.include_paths.is_empty()
174            && request.must_include_paths.is_empty()
175            && request.must_include_symbols.is_empty()
176            && request.required_evidence.is_empty()
177            && request.focus_paths.is_empty()
178            && request.focus_symbols.is_empty()
179            && !request.strict_focus_paths
180            && request.minimum_fragments_per_focus_path.is_none()
181            && request.base_revision.is_none()
182            && request.changed_paths.is_empty()
183            && !request.strict_changed_paths
184            && response.handoff_manifest.is_none();
185        if can_reduce_selected {
186            let selected = response
187                .plan
188                .as_ref()
189                .map_or(response.fragments.len(), |plan| plan.candidates.len());
190            let omission_reserve = response
191                .omission_summary
192                .budget_or_result_limit
193                .saturating_add(selected);
194            let budget = ResponseBudget::new(&self.config.tokenizer, max_response_tokens);
195            let keep = budget.largest_fitting_prefix(selected, |keep| {
196                let mut candidate = response.clone();
197                Self::trim_context_selection(&mut candidate, keep);
198                candidate.omission_summary.budget_or_result_limit = omission_reserve;
199                Self::refresh_context_omission_warning(&mut candidate);
200                self.context_response_tokens_with_receipt_reserve(&candidate, request)
201            })?;
202            if let Some(keep) = keep {
203                Self::trim_context_selection(response, keep);
204                self.finalize_response(response)?;
205                if self.context_response_fits(response, request, max_response_tokens)? {
206                    return Ok(());
207                }
208                return Err(Error::ResponseAccountingInvariant(
209                    "context prefix fitting violated its monotonic sizing reserve".into(),
210                ));
211            }
212            Self::trim_context_selection(response, 0);
213        }
214
215        Err(self.context_response_budget_error(response, request, max_response_tokens)?)
216    }

218    pub(super) fn finalize_context_pipeline(
219        &self,
220        finalization: ContextFinalization<'_>,
221        batch: CandidateBatch,
222        mut phases: ContextPhaseTracker,
223    ) -> Result<(ContextEvaluation, Option<usize>)> {
224        let ContextFinalization {
225            session,
226            request,
227            scoped_request,
228            handoff,
229            options,
230            response_profile,
231            cancellation,
232            diagnostics,
233            generation,
234            diff_scope,
235            working_tree_state,
236            working_tree_paths,
237            working_tree_modified,
238            working_tree_untracked,
239            commit_revision,
240            branch,
241            resolved_workflow,
242        } = finalization;
243        let CandidateBatch {
244            candidates,
245            path_excluded_candidates,
246            coverage,
247            warnings: focus_generation_warnings,
248            workflow_receipt,
249            ..
250        } = batch;
251        let ranking_started = phases.timer();
252        let candidate_path_count = candidates
253            .iter()
254            .map(|candidate| candidate.path.as_str())
255            .collect::<BTreeSet<_>>()
256            .len();
257        let generated_candidate_paths = if diagnostics == CandidateDiagnostics::Collect {
258            candidates
259                .iter()
260                .map(|candidate| candidate.path.clone())
261                .collect::<BTreeSet<_>>()
262                .into_iter()
263                .collect()
264        } else {
265            Vec::new()
266        };
267        let generated_candidates = if diagnostics == CandidateDiagnostics::Collect {
268            candidates
269                .iter()
270                .map(|candidate| {
271                    let token_count = candidate.token_count_with(self.config.tokenizer).max(1);
272                    ContextCandidateEvaluation {
273                        path: candidate.path.clone(),
274                        start_line: candidate.start_line,
275                        end_line: candidate.end_line,
276                        representation: candidate.representation.clone(),
277                        match_kinds: candidate.match_kinds.clone(),
278                        concepts: candidate.concepts.clone(),
279                        concept_weight: candidate.concept_weight,
280                        score: candidate.score(&ranking::Weights::default(), token_count),
281                        token_count,
282                    }
283                })
284                .collect()
285        } else {
286            Vec::new()
287        };
288        let mut response = ranking::select_with_tokenizer_and_context_exclusions(
289            candidates,
290            scoped_request,
291            generation,
292            self.config.tokenizer,
293            &self.config.context_exclude_paths,
294            &path_excluded_candidates,
295        );
296        response.effective_response_profile = response_profile;
297        let mut coverage = response::merge_selected_coverage(coverage, &mut response);
298        let selected_paths = response::selected_paths(&response);
299        self.finalize_strict_scope_coverage(
300            session,
301            scoped_request,
302            &selected_paths,
303            &mut coverage,
304        )?;
305        response.coverage = coverage;
306        response.warnings.extend(focus_generation_warnings);
307        response::append_coverage_warnings(&mut response);
308        response.workflow = resolved_workflow;
309        response.workflow_receipt = workflow_receipt;
310        response.meta.freshness = self.freshness();
311        response.meta.repository_id = self.repository_id();
312        let provenance = RepositoryProvenance {
313            commit_revision: commit_revision.map(str::to_owned),
314            branch: branch.map(str::to_owned),
315            working_tree_state: if working_tree_state == HandoffWorkingTreeState::Unknown {
316                RepositoryWorkingTreeState::Unknown
317            } else if working_tree_modified {
318                RepositoryWorkingTreeState::Modified
319            } else if working_tree_untracked {
320                RepositoryWorkingTreeState::Untracked
321            } else {
322                RepositoryWorkingTreeState::Clean
323            },
324            repository_generation: generation,
325            freshness: response.meta.freshness.clone(),
326            status: if commit_revision.is_some()
327                && branch.is_some()
328                && working_tree_state != HandoffWorkingTreeState::Unknown
329            {
330                RepositoryProvenanceStatus::Available
331            } else {
332                RepositoryProvenanceStatus::Unavailable
333            },
334        };
335        response.provenance = Some(provenance.clone());
336        if let Some(scope) = diff_scope {
337            let mut scope = scope.clone();
338            let mut indexed = 0usize;
339            for path in &scope.changed_paths {
340                if session.find_file(path)?.is_some() {
341                    indexed += 1;
342                }
343            }
344            scope.indexed_changed_paths = indexed;
345            scope.evidence = (response_profile != ContextResponseProfile::Compact
346                && (!request.plan_only || request.explain_diagnostics))
347                .then(|| {
348                    self.build_diff_evidence(
349                        session,
350                        scoped_request,
351                        &scope,
352                        resolved_workflow,
353                        cancellation,
354                    )
355                })
356                .transpose()?;
357            response.routing =
358                build_context_routing(request, &scope, candidate_path_count, &selected_paths);
359            if let Some(routing) = &response.routing {
360                let concentration = if routing.weakly_concentrated {
361                    "; selected evidence is weakly concentrated"
362                } else {
363                    ""
364                };
365                response.warnings.push(format!(
366                    "context spans {} changed paths across {} path groups{concentration}",
367                    routing.changed_paths, routing.path_groups_total
368                ));
369            }
370            response.diff_scope = Some(scope);
371        }
372        if let Some(handoff) = &handoff {
373            let evidence = response
374                .fragments
375                .iter()
376                .map(|fragment| HandoffEvidence {
377                    path: fragment.path.clone(),
378                    start_line: fragment.start_line,
379                    end_line: fragment.end_line,
380                    content_hash: fragment.content_hash.clone(),
381                })
382                .collect::<Vec<_>>();
383            // A diff-scoped handoff must identify the requested immutable head.
384            // Keep the ordinary response provenance as the current snapshot,
385            // but make the handoff copy internally consistent with that scope.
386            let handoff_commit_revision = response
387                .diff_scope
388                .as_ref()
389                .and_then(|scope| scope.head_revision.clone())
390                .or_else(|| provenance.commit_revision.clone());
391            let handoff_provenance = handoff_commit_revision.as_ref().map_or_else(
392                || provenance.clone(),
393                |commit_revision| {
394                    let mut scoped = provenance.clone();
395                    scoped.commit_revision = Some(commit_revision.clone());
396                    scoped.status = if scoped.branch.is_some()
397                        && scoped.working_tree_state != RepositoryWorkingTreeState::Unknown
398                    {
399                        RepositoryProvenanceStatus::Available
400                    } else {
401                        RepositoryProvenanceStatus::Unavailable
402                    };
403                    scoped
404                },
405            );
406            let commit_revision_available = handoff_commit_revision.is_some();
407            response.handoff_manifest = Some(handoff::build(
408                request,
409                handoff,
410                &response,
411                evidence,
412                HandoffProvenance {
413                    commit_revision: handoff_commit_revision,
414                    commit_revision_available,
415                    working_tree_state: if commit_revision_available {
416                        working_tree_state
417                    } else {
418                        HandoffWorkingTreeState::Unknown
419                    },
420                    working_tree_paths: working_tree_paths.to_vec(),
421                    provenance: Some(handoff_provenance),
422                },
423            ));
424        }
425        let baseline_source_tokens = self.finalize_context_delivery(
426            &mut response,
427            response::ContextResponseFinalization {
428                session,
429                request,
430                options,
431                generation,
432            },
433        )?;
434        if let Some(started) = ranking_started {
435            phases.timings.ranking_finalize_ms = started.elapsed().as_secs_f64() * 1_000.0;
436        }
437        let (phases, timings, primitive_keys) = phases.finish(generated_candidates.len());
438        Ok((
439            ContextEvaluation {
440                response,
441                generated_candidate_paths,
442                generated_candidates,
443                phases,
444                timings,
445                primitive_keys,
446            },
447            baseline_source_tokens,
448        ))
449    }
