1impl Services {
2    pub(in crate::services::context) fn append_workflow_candidates(
3        &self,
4        session: &ReadSession,
5        request: &ContextRequest,
6        workflow: ContextWorkflow,
7        cancellation: &CancellationToken,
8        candidates: &mut Vec<Candidate>,
9    ) -> Result<(Option<WorkflowReceipt>, Vec<String>)> {
10        if !matches!(
11            workflow,
12            ContextWorkflow::Contribution | ContextWorkflow::Review
13        ) {
14            return Ok((None, Vec::new()));
15        }
16
17        let mut matches = Vec::new();
18        let path_filter = PathFilter::new(&request.include_paths, &request.exclude_paths)?;
19        let mut path_excluded = Vec::new();
20        let mut cursor = None;
21        let mut scanned_files = 0;
22        let mut scan_truncated = false;
23        loop {
24            check_cancelled(cancellation)?;
25            let page = session.list_files(512, cursor)?;
26            let Some(last) = page.last() else {
27                break;
28            };
29            cursor = Some(last.id);
30            for file in page {
31                if scanned_files == MAX_WORKFLOW_SCAN_FILES {
32                    scan_truncated = true;
33                    break;
34                }
35                scanned_files += 1;
36                if let Some((score, family)) = workflow_path_role(&file.path, request) {
37                    if path_filter.allows(&file.path) {
38                        matches.push((file, score, family));
39                    } else {
40                        path_excluded.push(file.path);
41                    }
42                }
43            }
44            if scan_truncated {
45                break;
46            }
47        }
48        let count = |family| {
49            matches
50                .iter()
51                .filter(|(_, _, candidate_family)| *candidate_family == family)
52                .count()
53        };
54        let guidance_candidates = count("guidance");
55        let template_candidates = count("template");
56        let validation_candidates = count("validation");
57        let owner_test_candidates = count("owner_test");
58        let mut missing_families = Vec::new();
59        for (family, candidates) in [
60            ("guidance", guidance_candidates),
61            ("template", template_candidates),
62            ("validation", validation_candidates),
63            ("owner_test", owner_test_candidates),
64        ] {
65            if candidates == 0 {
66                missing_families.push(family.to_owned());
67            }
68        }
69        if scan_truncated {
70            missing_families.push("repository_scan_truncated".to_owned());
71        }
72        matches.sort_by(|left, right| {
73            right
74                .1
75                .total_cmp(&left.1)
76                .then_with(|| left.0.path.cmp(&right.0.path))
77        });
78        matches.truncate(24);
79
80        let excerpt_requests = matches
81            .iter()
82            .map(|(file, _, _)| StoredExcerptRequest {
83                file_id: file.id,
84                desired_start_line: 1,
85                desired_end_line: 80,
86                required_start_line: 1,
87                required_end_line: 1,
88                max_lines: 80,
89            })
90            .collect::<Vec<_>>();
91        for ((file, score, family), excerpt) in matches
92            .into_iter()
93            .zip(self.stored_excerpts(session, &excerpt_requests)?)
94        {
95            let Some(excerpt) = excerpt else { continue };
96            candidates.push(
97                Candidate::new(
98                    file.path,
99                    excerpt.start_line,
100                    excerpt.end_line,
101                    excerpt.content,
102                )
103                .match_kind(format!("workflow_{family}"))
104                .representation("workflow")
105                .path_score(score)
106                .focus_boost(score),
107            );
108        }
109        Ok((
110            Some(WorkflowReceipt {
111                guidance_candidates,
112                template_candidates,
113                validation_candidates,
114                owner_test_candidates,
115                missing_families,
116            }),
117            path_excluded,
118        ))
119    }
120}
