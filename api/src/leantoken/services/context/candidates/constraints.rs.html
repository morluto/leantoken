2    pub(in crate::services::context) fn append_constraint_candidates(
3        &self,
4        expansion: ConstraintCandidateExpansion<'_>,
5        candidates: &mut Vec<Candidate>,
6        phases: &mut ContextPhaseTracker,
7    ) -> Result<ContextConstraintExpansion> {
8        let ConstraintCandidateExpansion {
9            session,
10            request,
11            queries: _,
12            path_scorer: _,
13            cancellation,
14        } = expansion;
15        let mut coverage = ContextCoverageReceipt::default();
16        let mut focus_path_matches = vec![0usize; request.focus_paths.len()];
17        let mut focus_path_files = vec![Vec::new(); request.focus_paths.len()];
18        let mut focus_path_eligible = vec![0usize; request.focus_paths.len()];
19        let mut include_path_matches = vec![false; request.include_paths.len()];
20        let mut required_path_matches = vec![false; request.must_include_paths.len()];
21        let mut required_path_files = vec![None::<FileRecord>; request.must_include_paths.len()];
22        let mut evidence_path_matches = vec![0usize; request.required_evidence.len()];
23        let mut evidence_path_files = vec![Vec::new(); request.required_evidence.len()];
24        let focus_matchers = request
25            .focus_paths
26            .iter()
27            .map(|pattern| PathMatcher::new(std::slice::from_ref(pattern)))
28            .collect::<Result<Vec<_>>>()?;
29        let include_matchers = request
30            .include_paths
31            .iter()
32            .map(|pattern| PathMatcher::new(std::slice::from_ref(pattern)))
33            .collect::<Result<Vec<_>>>()?;
34        let required_matchers = request
35            .must_include_paths
36            .iter()
37            .map(|pattern| PathMatcher::new(std::slice::from_ref(pattern)))
38            .collect::<Result<Vec<_>>>()?;
39        let evidence_matchers = request
40            .required_evidence
41            .iter()
42            .map(|requirement| PathMatcher::new(std::slice::from_ref(&requirement.path)))
43            .collect::<Result<Vec<_>>>()?;
44        let path_filter = PathFilter::new(&request.include_paths, &request.exclude_paths)?;
45        let context_exclude_paths = PathMatcher::new_lossy(&self.config.context_exclude_paths);
46        let strict_changed_paths = request.strict_changed_paths.then(|| {
47            request
48                .changed_paths
49                .iter()
50                .map(String::as_str)
51                .collect::<HashSet<_>>()
52        });
53
54        if !request.focus_paths.is_empty()
55            || !request.include_paths.is_empty()
56            || !request.must_include_paths.is_empty()
57            || !request.required_evidence.is_empty()
58        {
59            let mut cursor = None;
60            loop {
61                check_cancelled(cancellation)?;
62                let page = session.list_files(512, cursor)?;
63                let Some(last) = page.last() else {
64                    break;
65                };
66                cursor = Some(last.id);
67                for file in page {
68                    let focus_eligible = if focus_matchers.is_empty() {
69                        false
70                    } else {
71                        let explicitly_included = !request.include_paths.is_empty()
72                            && include_matchers
73                                .iter()
74                                .any(|matcher| matcher.is_match(&file.path));
75                        path_filter.allows(&file.path)
76                            && strict_changed_paths
77                                .as_ref()
78                                .is_none_or(|paths| paths.contains(file.path.as_str()))
79                            && (!context_exclude_paths.is_match(&file.path) || explicitly_included)
80                    };
81                    for (index, matcher) in focus_matchers.iter().enumerate() {
82                        if matcher.is_match(&file.path) {
83                            focus_path_matches[index] = focus_path_matches[index].saturating_add(1);
84                            if focus_eligible {
85                                focus_path_eligible[index] =
86                                    focus_path_eligible[index].saturating_add(1);
87                                retain_focus_file(&mut focus_path_files[index], &file);
88                            }
89                        }
90                    }
91                    for (index, matcher) in include_matchers.iter().enumerate() {
92                        include_path_matches[index] |= matcher.is_match(&file.path);
93                    }
94                    for (index, matcher) in required_matchers.iter().enumerate() {
95                        if !matcher.is_match(&file.path) {
96                            continue;
97                        }
98                        required_path_matches[index] = true;
99                        if required_path_files[index].is_none() && path_filter.allows(&file.path) {
100                            required_path_files[index] = Some(file.clone());
101                        }
102                    }
103                    retain_required_evidence_files(
104                        &file,
105                        &evidence_matchers,
106                        &path_filter,
107                        strict_changed_paths.as_ref(),
108                        &mut evidence_path_matches,
109                        &mut evidence_path_files,
110                    );
111                }
112            }
113        }
114
115        coverage.unmatched_focus_paths = request
116            .focus_paths
117            .iter()
118            .zip(&focus_path_matches)
119            .filter(|(_, matched)| **matched == 0)
120            .map(|(pattern, _)| pattern.clone())
121            .collect();
122        let minimum_focus_fragments = request.minimum_fragments_per_focus_path.unwrap_or(1);
123        if !request.focus_paths.is_empty() {
124            coverage.focus_path_coverage = request
125                .focus_paths
126                .iter()
127                .zip(&focus_path_matches)
128                .zip(&focus_path_eligible)
129                .map(
130                    |((pattern, indexed_paths), eligible_paths)| ContextFocusPathCoverage {
131                        pattern: pattern.clone(),
132                        indexed_paths: *indexed_paths,
133                        minimum_fragments: minimum_focus_fragments,
134                        selected_fragments: 0,
135                        satisfied: false,
136                        diagnostics: request.verbose_diagnostics.then(|| {
137                            ContextFocusPathDiagnostics {
138                                eligible_paths: *eligible_paths,
139                                ..ContextFocusPathDiagnostics::default()
140                            }
141                        }),
142                    },
143                )
144                .collect();
145        }
146        coverage.unmatched_include_paths = request
147            .include_paths
148            .iter()
149            .zip(include_path_matches)
150            .filter(|(_, matched)| !matched)
151            .map(|(pattern, _)| pattern.clone())
152            .collect();
153        coverage.unmatched_must_include_paths = request
154            .must_include_paths
155            .iter()
156            .zip(&required_path_matches)
157            .filter(|(_, matched)| !**matched)
158            .map(|(pattern, _)| pattern.clone())
159            .collect();
160        coverage.required_evidence = request
161            .required_evidence
162            .iter()
163            .zip(&evidence_path_matches)
164            .zip(&evidence_path_files)
165            .map(|((requirement, indexed_paths), inspected_files)| {
166                ContextRequiredEvidenceCoverage {
167                    path: requirement.path.clone(),
168                    indexed_paths: *indexed_paths,
169                    inspected_paths: inspected_files.len(),
170                    minimum_query_matches: requirement.minimum_query_matches,
171                    matched_queries: Vec::new(),
172                    unmatched_queries: requirement.queries.clone(),
173                    selected_fragments: 0,
174                    satisfied: false,
175                }
176            })
177            .collect();
178
179        self.append_required_path_candidates(expansion, required_path_files, candidates, phases)?;
180
181        let mut exact_names = Vec::new();
182        let mut seen_exact_names = HashSet::new();
183        for name in request
184            .focus_symbols
185            .iter()
186            .chain(&request.must_include_symbols)
187        {
188            if seen_exact_names.insert(name.clone()) {
189                exact_names.push(name.clone());
190            }
191        }
192        phases.counters.exact_symbol_names = exact_names.len();
193        let required_names = request
194            .must_include_symbols
195            .iter()
196            .map(String::as_str)
197            .collect::<HashSet<_>>();
198        let mut exact_presence = HashSet::new();
199        let mut allowed_required_hits = HashMap::<String, SymbolHit>::new();
200        for names in exact_names.chunks(MAX_EXACT_SYMBOL_BATCH_NAMES) {
201            check_cancelled(cancellation)?;
202            phases.counters.exact_symbol_batches =
203                phases.counters.exact_symbol_batches.saturating_add(1);
204            let results = phases.measure(ContextTimedPhase::ExactSymbolLookup, || {
205                session.find_symbols_exact_batch(names, MAX_IMPORT_SYMBOLS)
206            })?;
207            for (name, hits) in names.iter().zip(results) {
208                phases.record_primitive("exact_symbol", || {
209                    format!("case_sensitive:true:limit:{MAX_IMPORT_SYMBOLS}:name:{name}")
210                });
211                phases.counters.exact_symbol_hits =
212                    phases.counters.exact_symbol_hits.saturating_add(hits.len());
213                if hits.is_empty() {
214                    continue;
215                }
216                exact_presence.insert(name.clone());
217                if required_names.contains(name.as_str())
218                    && let Some(hit) = hits.into_iter().find(|hit| path_filter.allows(&hit.path))
219                {
220                    allowed_required_hits.insert(name.clone(), hit);
221                }
222            }
223        }
224        for symbol in &request.focus_symbols {
225            check_cancelled(cancellation)?;
226            if !exact_presence.contains(symbol) {
227                coverage.unmatched_focus_symbols.push(symbol.clone());
228            }
229        }
230        let mut required_symbol_hits = Vec::<(String, SymbolHit)>::new();
231        for symbol in &request.must_include_symbols {
232            check_cancelled(cancellation)?;
233            if !exact_presence.contains(symbol) {
234                coverage.unmatched_must_include_symbols.push(symbol.clone());
235                continue;
236            }
237            if let Some(hit) = allowed_required_hits.get(symbol).cloned() {
238                required_symbol_hits.push((symbol.clone(), hit));
239            }
240        }
241        let required_symbol_budget = request
242            .token_budget
243            .saturating_div(required_symbol_hits.len().max(1))
244            .max(1);
245        let symbol_excerpt_requests = required_symbol_hits
246            .iter()
247            .map(|(_, hit)| AdaptiveExcerptRequest {
248                file_id: hit.symbol.file_id,
249                declaration_start: hit.symbol.start_line,
250                declaration_end: hit.symbol.end_line,
251                matched_line: hit.symbol.start_line,
252                token_budget: required_symbol_budget,
253            })
254            .collect::<Vec<_>>();
255        phases.record_adaptive_excerpts(&symbol_excerpt_requests);
256        let symbol_excerpts = phases.measure(ContextTimedPhase::AdaptiveExcerpt, || {
257            self.adaptive_context_excerpts(session, &symbol_excerpt_requests)
258        })?;
259        for (((symbol, hit), excerpt), rank) in required_symbol_hits
260            .into_iter()
261            .zip(symbol_excerpts)
262            .zip(0usize..)
263        {
264            let Some(excerpt) = excerpt else { continue };
265            candidates.push(
266                Candidate::new(
267                    hit.path,
268                    excerpt.start_line,
269                    excerpt.end_line,
270                    excerpt.content,
271                )
272                .match_kind("must_symbol")
273                .concept(format!("must:symbol:{symbol}"), 2.0)
274                .representation("required_symbol")
275                .symbol_name(hit.symbol.name)
276                .target_range(hit.symbol.start_line, hit.symbol.end_line)
277                .exact(2.0)
278                .symbol(2.0)
279                .focus_boost(2.0)
280                .channel("must_symbol", rank),
281            );
282        }
283
284        self.append_required_evidence_candidates(
285            expansion,
286            &evidence_path_files,
287            candidates,
288            phases,
289        )?;
290
291        Ok(ContextConstraintExpansion {
292            coverage,
293            focus_paths: focus_path_files
294                .into_iter()
295                .zip(focus_path_matches)
296                .zip(focus_path_eligible)
297                .map(
298                    |((files, indexed_paths), eligible_paths)| FocusPathResolution {
299                        files,
300                        indexed_paths,
301                        eligible_paths,
302                    },
303                )
304                .collect(),
305        })
306    }

308    pub(in crate::services::context) fn append_required_path_candidates(
309        &self,
310        expansion: ConstraintCandidateExpansion<'_>,
311        required_path_files: Vec<Option<FileRecord>>,
312        candidates: &mut Vec<Candidate>,
313        phases: &mut ContextPhaseTracker,
314    ) -> Result<()> {
315        let ConstraintCandidateExpansion {
316            session,
317            request,
318            queries,
319            path_scorer,
320            cancellation,
321        } = expansion;
322        let required_path_entries = request
323            .must_include_paths
324            .iter()
325            .zip(required_path_files)
326            .filter_map(|(pattern, file)| file.map(|file| (pattern, file)))
327            .collect::<Vec<_>>();
328        let mut fallback_path_entries = Vec::new();
329        for (pattern, file) in &required_path_entries {
330            check_cancelled(cancellation)?;
331            phases.record_primitive("required_path_chunks", || {
332                format!(
333                    "file_id:{}:limit:{MAX_CONTEXT_FOCUS_CHUNKS_PER_FILE}",
334                    file.id
335                )
336            });
337            let chunks = session.get_chunks_for_file(file.id, MAX_CONTEXT_FOCUS_CHUNKS_PER_FILE)?;
338            let best = chunks
339                .into_iter()
340                .filter_map(|chunk| {
341                    let relevance = focus_text_relevance(&chunk.content, queries);
342                    (relevance > 0.0).then_some((relevance, chunk))
343                })
344                .max_by(|(left_score, left), (right_score, right)| {
345                    left_score
346                        .total_cmp(right_score)
347                        .then_with(|| right.start_line.cmp(&left.start_line))
348                });
349            if let Some((relevance, chunk)) = best {
350                candidates.push(
351                    Candidate::new(&file.path, chunk.start_line, chunk.end_line, chunk.content)
352                        .match_kind("must_path")
353                        .concept(format!("must:path:{pattern}"), 2.0)
354                        .representation("required_path")
355                        .exact(relevance.min(4.0))
356                        .path_score(path_scorer.score(&file.path))
357                        .focus_boost(2.0),
358                );
359            } else {
360                fallback_path_entries.push((*pattern, file.clone()));
361            }
362        }
363        let path_excerpt_requests = fallback_path_entries
364            .iter()
365            .map(|(_, file)| StoredExcerptRequest {
366                file_id: file.id,
367                desired_start_line: 1,
368                desired_end_line: 40,
369                required_start_line: 1,
370                required_end_line: 1,
371                max_lines: 40,
372            })
373            .collect::<Vec<_>>();
374        phases.record_stored_excerpts(&path_excerpt_requests);
375        let path_excerpts = phases.measure(ContextTimedPhase::StoredExcerpt, || {
376            self.stored_excerpts(session, &path_excerpt_requests)
377        })?;
378        for ((pattern, file), excerpt) in fallback_path_entries.into_iter().zip(path_excerpts) {
379            let Some(excerpt) = excerpt else { continue };
380            candidates.push(
381                Candidate::new(
382                    file.path,
383                    excerpt.start_line,
384                    excerpt.end_line,
385                    excerpt.content,
386                )
387                .match_kind("must_path_fallback")
388                .concept(format!("must:path:{pattern}"), 2.0)
389                .representation("required_path_fallback")
390                .exact(2.0)
391                .focus_boost(2.0),
392            );
393        }
394        Ok(())
395    }

397    pub(in crate::services::context) fn append_required_evidence_candidates(
398        &self,
399        expansion: ConstraintCandidateExpansion<'_>,
400        evidence_path_files: &[Vec<FileRecord>],
401        candidates: &mut Vec<Candidate>,
402        phases: &mut ContextPhaseTracker,
403    ) -> Result<()> {
404        let ConstraintCandidateExpansion {
405            session,
406            request,
407            queries: _,
408            path_scorer,
409            cancellation,
410        } = expansion;
411        for (requirement_index, (requirement, files)) in request
412            .required_evidence
413            .iter()
414            .zip(evidence_path_files)
415            .enumerate()
416        {
417            let mut retained = Vec::<RequiredEvidenceExcerptPlan>::new();
418            let normalized_queries = requirement
419                .queries
420                .iter()
421                .map(|query| query.to_lowercase())
422                .collect::<Vec<_>>();
423            for file in files {
424                check_cancelled(cancellation)?;
425                phases.record_primitive("required_evidence_chunks", || {
426                    format!(
427                        "file_id:{}:limit:{MAX_CONTEXT_FOCUS_CHUNKS_PER_FILE}",
428                        file.id
429                    )
430                });
431                for chunk in
432                    session.get_chunks_for_file(file.id, MAX_CONTEXT_FOCUS_CHUNKS_PER_FILE)?
433                {
434                    let relevance =
435                        required_evidence_query_matches(&chunk.content, &requirement.queries).len()
436                            as f64;
437                    for (line_offset, line) in chunk.content.lines().enumerate() {
438                        let normalized_line = line.to_lowercase();
439                        if !normalized_queries
440                            .iter()
441                            .any(|query| normalized_line.contains(query))
442                        {
443                            continue;
444                        }
445                        let matched_line = chunk.start_line.saturating_add(line_offset);
446                        retain_required_evidence_plan(
447                            &mut retained,
448                            RequiredEvidenceExcerptPlan {
449                                relevance,
450                                path: file.path.clone(),
451                                file_id: file.id,
452                                matched_line,
453                                requirement_index,
454                            },
455                        );
456                    }
457                }
458            }
459            let excerpt_requests = retained
460                .iter()
461                .map(|plan| StoredExcerptRequest {
462                    file_id: plan.file_id,
463                    desired_start_line: plan.matched_line.saturating_sub(19).max(1),
464                    desired_end_line: plan.matched_line.saturating_add(20),
465                    required_start_line: plan.matched_line,
466                    required_end_line: plan.matched_line,
467                    max_lines: 40,
468                })
469                .collect::<Vec<_>>();
470            phases.record_stored_excerpts(&excerpt_requests);
471            let excerpts = phases.measure(ContextTimedPhase::StoredExcerpt, || {
472                self.stored_excerpts(session, &excerpt_requests)
473            })?;
474            for (rank, (plan, excerpt)) in retained.into_iter().zip(excerpts).enumerate() {
475                let Some(excerpt) = excerpt else { continue };
476                let matched_queries =
477                    required_evidence_query_matches(&excerpt.content, &requirement.queries);
478                if matched_queries.is_empty() {
479                    continue;
480                }
481                let path_score = path_scorer.score(&plan.path);
482                let mut candidate = Candidate::new(
483                    plan.path,
484                    excerpt.start_line,
485                    excerpt.end_line,
486                    excerpt.content,
487                )
488                .match_kind("required_evidence")
489                .concept(format!("required:evidence:{}", plan.requirement_index), 2.0)
490                .representation("required_evidence")
491                .exact((matched_queries.len() as f64).min(4.0))
492                .path_score(path_score)
493                .focus_boost(2.0);
494                for query_index in matched_queries {
495                    candidate = candidate.match_kind(ranking::required_evidence_marker(
496                        plan.requirement_index,
497                        query_index,
498                    ));
499                }
500                candidates.push(candidate.channel("required_evidence", rank));
501            }
502        }
503        Ok(())
504    }
