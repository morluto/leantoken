26        let concept = query.fusion_key.as_str();

30            session.search_symbols(term, false, MAX_CONTEXT_HITS_PER_SOURCE)

32        phases.record_primitive("symbol_search", || {
33            format!("case_sensitive:false:limit:{MAX_CONTEXT_HITS_PER_SOURCE}:query:{term}")

35        phases.counters.symbol_candidates = phases
36            .counters

58                matched_line: hit.symbol.start_line,
59                token_budget: excerpt_budget(request.token_budget, ContextExcerptKind::Symbol),
60                selection_budget: request.token_budget,
61            })
62            .collect::<Vec<_>>();
63        phases.record_adaptive_excerpts(&symbol_excerpt_requests);
64        let symbol_excerpts = phases.measure(ContextTimedPhase::AdaptiveExcerpt, || {
65            self.adaptive_context_excerpts(session, &symbol_excerpt_requests)

67        for ((rank, hit), excerpt) in symbol_hits.into_iter().zip(symbol_excerpts) {
68            check_cancelled(cancellation)?;

88                    query.weight,

94                &hit.path,

105            .concept(concept, query.concept_weight)
106            .representation("symbol")
107            .symbol_name(hit.symbol.name)
108            .exact(exact + qualified * 1.5)
109            .symbol(1.0)
110            .path_score(path_scorer.score(&hit.path))
111            .change_boost(change_boost);
112            candidates.push(annotate_candidate(candidate, query, "symbol", rank));
113        }
114        let reference_results = if signals.caller {
115            phases.measure(ContextTimedPhase::ReferenceSearch, || {
116                session.search_references(term, false, MAX_CONTEXT_HITS_PER_SOURCE)
117            })?
118        } else {
119            Vec::new()
120        };
121        if signals.caller {
122            phases.record_primitive("reference_search", || {
123                format!("case_sensitive:false:limit:{MAX_CONTEXT_HITS_PER_SOURCE}:query:{term}")
124            });
125        }
126        phases.counters.reference_candidates = phases
127            .counters
128            .reference_candidates
129            .saturating_add(reference_results.len());
130        let mut reference_hits = Vec::new();
131        for (rank, hit) in reference_results.into_iter().enumerate() {
132            check_cancelled(cancellation)?;
133            if path_filter.allows(&hit.path)
134                && strict_changed_paths
135                    .as_ref()
136                    .is_none_or(|paths| paths.contains(hit.path.as_str()))
137            {

140                path_excluded_candidates.push(hit.path);

145            .map(|(_, hit)| (hit.reference.file_id, hit.reference.start_line))
146            .collect::<Vec<_>>();
147        phases.record_enclosing_locations(&reference_locations);
148        let enclosing = phases.measure(ContextTimedPhase::EnclosingLookup, || {
149            session.find_enclosing_symbols_batch(&reference_locations)

151        let mut adaptive_indices = Vec::new();
152        let mut adaptive_requests = Vec::new();
153        for (index, ((_, hit), symbol)) in reference_hits.iter().zip(enclosing).enumerate() {
154            if let Some(symbol) = symbol {
155                adaptive_indices.push(index);
156                adaptive_requests.push(AdaptiveExcerptRequest {
157                    file_id: hit.reference.file_id,
158                    declaration_start: symbol.start_line,

160                    matched_line: hit.reference.start_line,
161                    token_budget: excerpt_budget(
162                        request.token_budget,
163                        ContextExcerptKind::Reference,

172            self.adaptive_context_excerpts(session, &adaptive_requests)
173        })?;
174        for (index, excerpt) in adaptive_indices.into_iter().zip(hydrated_adaptive) {
175            adaptive_excerpts[index] = excerpt;
176        }
177        let mut fallback_indices = Vec::new();
178        let mut fallback_requests = Vec::new();
179        for (index, ((_, hit), adaptive)) in
180            reference_hits.iter().zip(&adaptive_excerpts).enumerate()
181        {
182            if adaptive.is_some() {
183                continue;
184            }
185            fallback_indices.push(index);
186            fallback_requests.push(StoredExcerptRequest {
187                file_id: hit.reference.file_id,
188                desired_start_line: hit.reference.start_line.saturating_sub(2).max(1),
189                desired_end_line: hit.reference.end_line.saturating_add(2),

200        for (index, excerpt) in fallback_indices.into_iter().zip(hydrated_fallback) {

208            check_cancelled(cancellation)?;

213                        hit.reference.start_line,

331                        end_line: facts.search_hit.end_line,
332                    },
333                    facts.matched_line,
334                    excerpt_budget(request.token_budget, ContextExcerptKind::Text),
335                    request.token_budget,
336                )
337            });
338            let Some(excerpt) = excerpt else { continue };
339            if query.fuse {
340                record_query_hit(
341                    query_fusion,
342                    &facts.search_hit.path,
343                    &query.fusion_key,
344                    query.weight,
345                    rank,
346                );
347            }
348            let change_boost = Self::file_change_boost(
349                Some(hit.generation),
350                &facts.search_hit.path,
351                changed_paths,
352                request.prior_repository_generation,
353            );
354            let candidate = Candidate::new(
355                &facts.search_hit.path,
356                excerpt.start_line,
357                excerpt.end_line,
358                excerpt.content,
359            )
360            .match_kind("text")
361            .concept(concept, query.concept_weight)
362            .exact(query.weight)
363            .bm25((-hit.score).max(0.0) * 1_000_000.0)
364            .path_score(path_scorer.score(&facts.search_hit.path))
365            .lexical_frequency_penalty((facts.occurrences.saturating_sub(5) as f64 / 20.0).min(1.0))
366            .change_boost(change_boost);
367            candidates.push(annotate_candidate(candidate, query, "text", rank));
368        }
369        Ok(())

372    fn context_lexical_hits(
373        &self,
374        session: &RepositoryGeneration,
375        request: &ContextRequest,
376        term: &str,
377        term_regex: &regex::Regex,
378        cancellation: &CancellationToken,
379    ) -> Result<(Vec<ChunkHit>, &'static str)> {
380        let folded = crate::symbol_identity::case_fold_literal_variants(term);
381        let folded_query = folded
382            .as_ref()
383            .filter(|variants| variants.expanded)
384            .map(crate::symbol_identity::case_fold_fts_query);
385        if folded.is_none() {
386            return Ok((
387                self.full_scan_literal_hits(LiteralFullScan {
388                    session,
389                    query: term,
390                    matcher: term_regex,
391                    include_paths: &request.include_paths,
392                    exclude_paths: &request.exclude_paths,
393                    max_candidates: MAX_CONTEXT_LEXICAL_HITS,
394                    max_tokens: request.token_budget,
395                    cancellation,
396                })?,
397                "unicode_case_fold_full_scan",
398            ));
399        }
400        if term.chars().count() < 3 {
401            let query = folded_query.unwrap_or_else(|| fts_quote(term));
402            return Ok((
403                session.search_word(&query, MAX_CONTEXT_LEXICAL_HITS)?,
404                if folded.is_some_and(|variants| variants.expanded) {
405                    "unicode_case_fold_word"
406                } else {
407                    "word"
408                },
409            ));
410        }
411        if let Some(expression) = folded_query {
412            return Ok((
413                session.search_trigram_expression(&expression, MAX_CONTEXT_LEXICAL_HITS)?,
414                "unicode_case_fold_trigram",
415            ));
416        }
417        Ok((
418            session.search_trigram(term, MAX_CONTEXT_LEXICAL_HITS)?,
419            "trigram",
420        ))
421    }
