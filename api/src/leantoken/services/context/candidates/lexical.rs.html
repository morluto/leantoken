365    fn context_lexical_hits(
366        &self,
367        expansion: QueryCandidateExpansion<'_>,
368        term_regex: &regex::Regex,
369        warnings: &mut Vec<String>,
370        path_excluded_candidates: &mut Vec<String>,
371    ) -> Result<(Vec<ChunkHit>, &'static str)> {
372        let QueryCandidateExpansion {
373            session,
374            request,
375            query,
376            cancellation,
377            ..
378        } = expansion;
379        let term = &query.value;
380        let folded = crate::symbol_identity::case_fold_literal_variants(term);
381        let folded_query = folded
382            .as_ref()
383            .filter(|variants| variants.expanded)
384            .map(crate::symbol_identity::case_fold_fts_query);
385        if folded.is_none() {
386            let scan = self.full_scan_literal_hits(LiteralFullScan {
387                session,
388                query: term,
389                matcher: term_regex,
390                include_paths: &request.include_paths,
391                exclude_paths: &request.exclude_paths,
392                max_candidates: MAX_CONTEXT_LEXICAL_HITS,
393                allows_path: &|path| {
394                    expansion
395                        .strict_changed_paths
396                        .is_none_or(|paths| paths.contains(path))
397                },
398                cancellation,
399            })?;
400            if let Some(error) = scan.limitation {
401                record_candidate_scan_limit(warnings, error)?;
402            }
403            return Ok((scan.hits, "unicode_case_fold_full_scan"));
404        }
405        if term.chars().count() < 3 {
406            let query = folded_query.unwrap_or_else(|| fts_quote(term));
407            return Ok((
408                context_ranked_hits(
409                    expansion,
410                    MAX_CONTEXT_LEXICAL_HITS,
411                    |offset, limit| session.search_word_page(&query, limit, offset),
412                    |hit: &ChunkHit| &hit.path,
413                    warnings,
414                    path_excluded_candidates,
415                )?,
416                if folded.is_some_and(|variants| variants.expanded) {
417                    "unicode_case_fold_word"
418                } else {
419                    "word"
420                },
421            ));
422        }
423        if let Some(expression) = folded_query {
424            return Ok((
425                context_ranked_hits(
426                    expansion,
427                    MAX_CONTEXT_LEXICAL_HITS,
428                    |offset, limit| {
429                        session.search_trigram_expression_page(&expression, limit, offset)
430                    },
431                    |hit: &ChunkHit| &hit.path,
432                    warnings,
433                    path_excluded_candidates,
434                )?,
435                "unicode_case_fold_trigram",
436            ));
437        }
438        Ok((
439            context_ranked_hits(
440                expansion,
441                MAX_CONTEXT_LEXICAL_HITS,
442                |offset, limit| session.search_trigram_page(term, limit, offset),
443                |hit: &ChunkHit| &hit.path,
444                warnings,
445                path_excluded_candidates,
446            )?,
447            "trigram",
448        ))
449    }

453fn context_ranked_hits<T>(
454    expansion: QueryCandidateExpansion<'_>,
455    limit: usize,
456    mut fetch: impl FnMut(usize, usize) -> Result<Vec<T>>,
457    path: impl Fn(&T) -> &str,
458    warnings: &mut Vec<String>,
459    path_excluded_candidates: &mut Vec<String>,
460) -> Result<Vec<T>> {
461    // Structural Unicode fallbacks verify a bounded materialized set. Do not
462    // repeat that whole scan for each page discarded by the scope filter.
463    let mut materialized =
464        if crate::symbol_identity::case_fold_literal_variants(&expansion.query.value).is_none() {
465            match fetch(0, crate::services::candidate_scan::MAX_FILTER_SCAN_ROWS) {
466                Ok(hits) => Some(hits.into_iter()),
467                Err(error) => {
468                    record_candidate_scan_limit(warnings, error)?;
469                    return Ok(Vec::new());
470                }
471            }
472        } else {
473            None
474        };
475    let mut recorded_exclusions = 0usize;
476    let result = crate::services::candidate_scan::collect_ranked_candidates(
477        limit,
478        expansion.cancellation,
479        |offset, page_limit| match &mut materialized {
480            Some(hits) => Ok(hits.by_ref().take(page_limit).collect()),
481            None => fetch(offset, page_limit),
482        },
483        |hit| {
484            let path = path(hit);
485            let allowed = expansion.path_filter.allows(path)
486                && expansion
487                    .strict_changed_paths
488                    .is_none_or(|paths| paths.contains(path));
489            // Omission diagnostics retain the old per-source bound even when
490            // scope filtering scans further to fill the candidate slots.
491            if !allowed && recorded_exclusions < limit {
492                path_excluded_candidates.push(path.to_owned());
493                recorded_exclusions += 1;
494            }
495            allowed
496        },
497    )?;
498    if result.scan_limited {
499        let warning = format!(
500            "candidate generation incomplete: ranked scan reached {} rows; narrow include_paths",
501            crate::services::candidate_scan::MAX_FILTER_SCAN_ROWS
502        );
503        if !warnings.contains(&warning) {
504            warnings.push(warning);
505        }
506    }
507    Ok(result.hits)
