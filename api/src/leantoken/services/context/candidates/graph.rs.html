2    pub(in crate::services::context) fn file_change_boost(
3        file_generation: Option<u64>,
4        path: &str,
5        changed_paths: &HashSet<String>,
6        prior_generation: Option<u64>,
7    ) -> f64 {
8        let mut boost = 0.0;
9
10        if let Some(prior) = prior_generation
11            && file_generation.is_some_and(|generation| generation > prior)
12        {
13            boost += 1.0;
14        }
15
16        if changed_paths.contains(path) {
17            boost += 1.0;
18        }
19
20        boost
21    }

23    pub(in crate::services::context) fn append_import_symbol_candidates(
24        &self,
25        expansion: ImportExpansion<'_>,
26        candidates: &mut Vec<Candidate>,
27    ) -> Result<()> {
28        let seed_paths = import_seed_paths(candidates, expansion.queries, self.config.tokenizer);
29        let requested_paths = seed_paths
30            .iter()
31            .take(24)
32            .map(|(path, _)| path.clone())
33            .collect::<Vec<_>>();
34        let targets =
35            expansion
36                .session
37                .import_symbol_targets(&requested_paths, 32, MAX_IMPORT_SYMBOLS)?;
38        let path_filter = PathFilter::new(
39            &expansion.request.include_paths,
40            &expansion.request.exclude_paths,
41        )?;
42        let mut pending = Vec::new();
43        for target in targets {
44            check_cancelled(expansion.cancellation)?;
45            let Some((_, seed_concepts)) = seed_paths.get(target.seed_index) else {
46                continue;
47            };
48            let target_path = &target.target_file.path;
49            if !path_filter.allows(target_path) {
50                continue;
51            }
52            let Some((symbol, query, exact)) =
53                corroborated_import_symbol(target.symbols, expansion.queries, seed_concepts)
54            else {
55                continue;
56            };
57            pending.push((target.target_file, symbol, query.clone(), exact));
58        }
59        let excerpt_requests = pending
60            .iter()
61            .map(|(target_file, symbol, _, _)| AdaptiveExcerptRequest {
62                file_id: target_file.id,
63                declaration_start: symbol.start_line,
64                declaration_end: symbol.end_line,
65                matched_line: symbol.start_line,
66                token_budget: excerpt_budget(
67                    expansion.request.token_budget,
68                    ContextExcerptKind::ImportSymbol,
69                ),
70            })
71            .collect::<Vec<_>>();
72        let excerpts = self.adaptive_context_excerpts(expansion.session, &excerpt_requests)?;
73        let mut neighbor_count = 0usize;
74        let mut neighbor_ranges = BTreeSet::new();
75        for ((target_file, symbol, query, exact), excerpt) in pending.into_iter().zip(excerpts) {
76            check_cancelled(expansion.cancellation)?;
77            let Some(excerpt) = excerpt else { continue };
78            let target_path = target_file.path;
79            if !neighbor_ranges.insert((target_path.clone(), excerpt.start_line, excerpt.end_line))
80            {
81                continue;
82            }
83            let change_boost = Self::file_change_boost(
84                Some(target_file.generation),
85                &target_path,
86                expansion.changed_paths,
87                expansion.request.prior_repository_generation,
88            );
89            let candidate = Candidate::new(
90                &target_path,
91                excerpt.start_line,
92                excerpt.end_line,
93                excerpt.content,
94            )
95            .match_kind("import")
96            .match_kind("symbol")
97            .concept(&query.fusion_key, query.concept_weight)
98            .representation("import_symbol")
99            .symbol_name(symbol.name)
100            .exact(exact)
101            .symbol(1.0)
102            .path_score(context_path_score(
103                &target_path,
104                expansion.terms,
105                &expansion.request.task,
106            ))
107            .import_boost(1.0)
108            .change_boost(change_boost);
109            candidates.push(annotate_candidate(
110                candidate,
111                &query,
112                "import_symbol",
113                neighbor_count,
114            ));
115            neighbor_count += 1;
116            if neighbor_count >= 24 {
117                break;
118            }
119        }
120        Ok(())
121    }

123    pub(in crate::services::context) fn apply_reverse_dependency_boost(
124        &self,
125        session: &IndexReadSnapshot,
126        queries: &[ContextQuery],
127        candidates: &mut [Candidate],
128    ) -> Result<()> {
129        let seed_paths = import_seed_paths(candidates, queries, self.config.tokenizer)
130            .into_iter()
131            .take(24)
132            .map(|(path, _)| path)
133            .collect::<Vec<_>>();
134        let importers = session
135            .affected_importers(&seed_paths)?
136            .into_iter()
137            .collect::<HashSet<_>>();
138        for candidate in candidates {
139            if importers.contains(&candidate.path) {
140                if !candidate
141                    .match_kinds
142                    .iter()
143                    .any(|kind| kind == "reverse-import")
144                {
145                    candidate.match_kinds.push("reverse-import".into());
146                }
147                candidate.import_boost = candidate.import_boost.max(1.0);
148            }
149        }
150        Ok(())
151    }
