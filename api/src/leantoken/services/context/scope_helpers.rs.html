1pub(super) fn parse_revision_range(revision: &str) -> Result<Option<(&str, &str)>> {
2    let Some((base, head)) = revision.split_once("..") else {
3        return Ok(None);
4    };
5    let base = base.trim();
6    let head = head.trim();
7    if base.trim().is_empty()
8        || head.trim().is_empty()
9        || base.ends_with('.')
10        || head.starts_with('.')
11        || head.contains("..")
12    {
13        return Err(Error::InvalidInput {
14            field: "base revision",
15            reason: "revision range must be BASE..HEAD",
16        });
17    }
18    Ok(Some((base, head)))
19}

55impl ContextPathScorer {
56    pub(super) fn new(terms: &[String], task: &str) -> Self {
57        let code_token_parts = facets::code_tokens(task)
58            .into_iter()
59            .filter(|token| {
60                token.contains("::")
61                    || token
62                        .split('.')
63                        .any(|part| part.chars().next().is_some_and(char::is_uppercase))
64            })
65            .map(|code_token| {
66                expand_terms(&code_token)
67                    .into_iter()
68                    .map(|part| part.to_ascii_lowercase())
69                    .filter(|part| part.chars().count() >= 2)
70                    .collect::<HashSet<_>>()
71                    .into_iter()
72                    .collect()
73            })
74            .collect();
75        Self {
76            terms: terms.iter().map(|term| term.to_ascii_lowercase()).collect(),
77            code_token_parts,
78            languages: [
79                task_mentions_language(task, "javascript"),
80                task_mentions_language(task, "typescript"),
81                task_mentions_language(task, "python"),
82                task_mentions_language(task, "rust"),
83                task_mentions_language(task, "go"),
84            ],
85            mcp_repository_intent: mcp_repository_intent(terms, task),
86        }
87    }

89    pub(super) fn score(&self, path: &str) -> f64 {
90        let path = path.to_lowercase();
91        let mut score = f64::from(
92            u32::try_from(
93                self.terms
94                    .iter()
95                    .filter(|term| path.contains(term.as_str()))
96                    .count(),
97            )
98            .unwrap_or(u32::MAX),
99        );
100        for parts in &self.code_token_parts {
101            let matched_parts = parts.iter().filter(|part| path.contains(*part)).count();
102            if matched_parts >= 2 {
103                let matched_parts = u32::try_from(matched_parts).unwrap_or(u32::MAX);
104                score += f64::from(matched_parts.saturating_mul(matched_parts));
105            }
106        }
107        for (mentioned, component, extensions) in [
108            (self.languages[0], "/js/", &["js", "jsx", "mjs", "cjs"][..]),
109            (self.languages[1], "/ts/", &["ts", "tsx"][..]),
110            (self.languages[2], "/python/", &["py", "pyw"][..]),
111            (self.languages[3], "/rust/", &["rs"][..]),
112            (self.languages[4], "/go/", &["go"][..]),
113        ] {
114            let extension_matches = path
115                .rsplit_once('.')
116                .is_some_and(|(_, extension)| extensions.contains(&extension));
117            let component = component.trim_matches('/');
118            let component_matches = path
119                .split('/')
120                .any(|path_component| path_component == component);
121            if mentioned && (extension_matches || component_matches) {
122                // An explicit language name in the task is strong repository-scope
123                // evidence. Keep this above an exact-name match in another
124                // language so common names such as `Point` do not dominate.
125                score += 12.0;
126            }
127        }
128        if self.mcp_repository_intent {
129            if path == "src/mcp.rs"
130                || path.starts_with("src/mcp/")
131                || path == "src/main/mcp_runtime.rs"
132            {
133                score += 14.0;
134            } else if path == "crates/test-suite/src/domains/protocol.rs"
135                || path == "crates/test-suite/src/domains/contracts.rs"
136                || path == "tests/mcp.rs"
137                || path.starts_with("tests/mcp/")
138            {
139                score += 8.0;
140            }
141        }
142        score
143    }
144}

146pub(super) fn mcp_repository_intent(terms: &[String], task: &str) -> bool {
147    let terms = terms
148        .iter()
149        .map(|term| term.to_ascii_lowercase())
150        .chain(
151            task.split(|character: char| !character.is_alphanumeric() && character != '_')
152                .filter(|term| !term.is_empty())
153                .map(str::to_ascii_lowercase),
154        )
155        .collect::<HashSet<_>>();
156    if terms.contains("mcp") {
157        return true;
158    }
159    let surface_terms = ["tool", "tools", "catalog", "schema", "schemas"];
160    let registration_terms = ["registration", "registered", "register", "server"];
161    surface_terms.iter().any(|term| terms.contains(*term))
162        && registration_terms.iter().any(|term| terms.contains(*term))
163}

165pub(super) fn context_path_group(path: &str) -> String {
166    let mut components = path.split('/').filter(|component| !component.is_empty());
167    let first = components.next().unwrap_or("<root>");
168    let second = components.next();
169    if second.is_none() {
170        return "<root>".into();
171    }
172    if matches!(
173        first,
174        "src" | "lib" | "app" | "apps" | "crates" | "packages"
175    ) && let Some(second) = second
176    {
177        format!("{first}/{second}")
178    } else {
179        first.to_owned()
180    }
181}

183pub(super) fn build_context_routing(
184    request: &ContextRequest,
185    scope: &DiffScopeReceipt,
186    candidate_paths: usize,
187    selected_paths: &[String],
188) -> Option<ContextRoutingReceipt> {
189    if scope.changed_paths.len() < OVERSIZED_CHANGE_PATHS {
190        return None;
191    }
192    let mut grouped_paths = BTreeMap::<String, Vec<String>>::new();
193    for path in &scope.changed_paths {
194        grouped_paths
195            .entry(context_path_group(path))
196            .or_default()
197            .push(path.clone());
198    }
199    if grouped_paths.len() < MIN_OVERSIZED_PATH_GROUPS {
200        return None;
201    }
202    let path_groups_total = grouped_paths.len();
203    let mut groups = grouped_paths.into_iter().collect::<Vec<_>>();
204    groups.sort_by(|left, right| {
205        right
206            .1
207            .len()
208            .cmp(&left.1.len())
209            .then_with(|| left.0.cmp(&right.0))
210    });
211    let selected_groups = selected_paths.iter().fold(
212        BTreeMap::<String, BTreeSet<&str>>::new(),
213        |mut paths, path| {
214            paths
215                .entry(context_path_group(path))
216                .or_default()
217                .insert(path);
218            paths
219        },
220    );
221    let selected_path_count = selected_paths
222        .iter()
223        .map(String::as_str)
224        .collect::<BTreeSet<_>>()
225        .len();
226    let strongest_selected_group = selected_groups
227        .values()
228        .map(BTreeSet::len)
229        .max()
230        .unwrap_or(0);
231    let weakly_concentrated = selected_path_count > 1
232        && strongest_selected_group.saturating_mul(2) <= selected_path_count;
233
234    let suggestions = groups
235        .iter()
236        .take(MAX_ROUTING_SUGGESTIONS)
237        .map(|(prefix, _)| ContextRoutingSuggestion {
238            include_paths: vec![if prefix == "<root>" {
239                "*".into()
240            } else {
241                format!("{prefix}/**")
242            }],
243        })
244        .collect();
245    let path_groups = groups
246        .into_iter()
247        .take(MAX_ROUTING_GROUPS)
248        .map(|(prefix, paths)| ContextPathGroup {
249            prefix,
250            changed_paths: paths.len(),
251        })
252        .collect();
253    Some(ContextRoutingReceipt {
254        candidate_paths,
255        changed_paths: scope.changed_paths.len(),
256        selected_paths: selected_path_count,
257        weakly_concentrated,
258        consistency: IndexConsistency::IndexedGeneration,
259        base_revision: request.base_revision.clone(),
260        known_hashes: request.known_hashes.clone(),
261        path_groups_total,
262        path_groups,
263        suggestions,
264    })
265}

267pub(super) fn resolve_context_workflow(requested: ContextWorkflow, task: &str) -> ContextWorkflow {
268    if requested != ContextWorkflow::Auto {
269        return requested;
270    }
271    let task = task.to_ascii_lowercase();
272    if ["pull request", "contribution", "contributing"]
273        .iter()
274        .any(|signal| task.contains(signal))
275    {
276        ContextWorkflow::Contribution
277    } else if ["code review", "review this", "review the changes"]
278        .iter()
279        .any(|signal| task.contains(signal))
280    {
281        ContextWorkflow::Review
282    } else if ["investigate", "root cause", "diagnose"]
283        .iter()
284        .any(|signal| task.contains(signal))
285    {
286        ContextWorkflow::Investigation
287    } else {
288        ContextWorkflow::Implementation
289    }
290}

292pub(super) fn workflow_path_role(
293    path: &str,
294    request: &ContextRequest,
295) -> Option<(f64, &'static str)> {
296    let normalized = path.replace('\\', "/");
297    let lower = normalized.to_ascii_lowercase();
298    let mut best = WORKFLOW_PATHS
299        .matches(&normalized)
300        .into_iter()
301        .map(|index| WORKFLOW_PATH_ROLES[index])
302        .max_by(|left, right| left.0.total_cmp(&right.0));
303    if likely_owner_test(&lower, request) && best.is_none_or(|(score, _)| score < OWNER_TEST_SCORE)
304    {
305        best = Some((OWNER_TEST_SCORE, "owner_test"));
306    }
307    best
308}

331pub(super) const fn workflow_path_roles() -> [(f64, &'static str); WORKFLOW_PATH_RULES.len()] {
332    let mut roles = [(0.0, ""); WORKFLOW_PATH_RULES.len()];
333    let mut index = 0;
334    while index < WORKFLOW_PATH_RULES.len() {
335        roles[index] = (WORKFLOW_PATH_RULES[index].1, WORKFLOW_PATH_RULES[index].2);
336        index += 1;
337    }
338    roles
339}

341static WORKFLOW_PATHS: LazyLock<GlobSet> = LazyLock::new(|| {
342    let mut builder = GlobSetBuilder::new();
343    for (pattern, _, _) in WORKFLOW_PATH_RULES {
344        builder.add(
345            GlobBuilder::new(pattern)
346                .case_insensitive(true)
347                .literal_separator(true)
348                .build()
349                .expect("static workflow glob"),
350        );
351    }
352    builder.build().expect("static workflow glob set")
353});

359pub(super) fn owner_test_changed_path(path: &str, request: &ContextRequest) -> Option<String> {
360    let path = path.replace('\\', "/").to_ascii_lowercase();
361    let is_test = path.contains("/test")
362        || path.starts_with("test")
363        || path.contains("/spec")
364        || path.starts_with("spec");
365    if !is_test {
366        return None;
367    }
368    let test_name = path.rsplit('/').next().unwrap_or(&path);
369    let test_stem = test_name.split('.').next().unwrap_or(test_name);
370    request
371        .changed_paths
372        .iter()
373        .chain(&request.focus_paths)
374        .find_map(|changed| {
375            let normalized = changed.replace('\\', "/");
376            let stem = normalized
377                .rsplit('/')
378                .next()
379                .and_then(|name| name.split('.').next())?
380                .to_ascii_lowercase();
381            (stem.len() >= 3 && contains_filename_token(test_stem, &stem)).then(|| changed.clone())
382        })
383}

385pub(super) fn contains_filename_token(path: &str, token: &str) -> bool {
386    path.match_indices(token).any(|(start, matched)| {
387        let end = start + matched.len();
388        let boundary =
389            |character: Option<char>| character.is_none_or(|value| !value.is_alphanumeric());
390        boundary(path[..start].chars().next_back()) && boundary(path[end..].chars().next())
391    })
392}
