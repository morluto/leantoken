13pub(super) fn parse_context_revision(revision: Option<&str>) -> Result<Option<ContextRevision>> {
14    let Some(revision) = revision else {
15        return Ok(None);
16    };
17    if revision.trim().is_empty() {
18        return Err(Error::InvalidInput {
19            field: "base revision",
20            reason: "must not be empty",
21        });
22    }
23    if revision != revision.trim() {
24        return Err(Error::InvalidInput {
25            field: "base revision",
26            reason: "must not have leading or trailing whitespace",
27        });
28    }
29    validate_input(revision, "base revision", MAX_BASE_REVISION_BYTES)?;
30    Ok(Some(match parse_revision_range(revision)? {
31        Some((base, head)) => ContextRevision::Range {
32            base: base.to_owned(),
33            head: head.to_owned(),
34        },
35        None => ContextRevision::Single(revision.to_owned()),
36    }))
37}

39pub(super) fn parse_revision_range(revision: &str) -> Result<Option<(&str, &str)>> {
40    let Some((base, head)) = revision.split_once("..") else {
41        return Ok(None);
42    };
43    let base = base.trim();
44    let head = head.trim();
45    if base.trim().is_empty()
46        || head.trim().is_empty()
47        || base.ends_with('.')
48        || head.starts_with('.')
49        || head.contains("..")
50    {
51        return Err(Error::InvalidInput {
52            field: "base revision",
53            reason: "revision range must be BASE..HEAD",
54        });
55    }
56    Ok(Some((base.trim(), head.trim())))
57}

94impl ContextPathScorer {
95    pub(super) fn new(terms: &[String], task: &str) -> Self {
96        let primary_task = facets::primary_task_text(task);
97        let code_token_parts = facets::code_tokens(task)
98            .into_iter()
99            .filter(|token| {
100                token.contains("::")
101                    || token
102                        .split('.')
103                        .any(|part| part.chars().next().is_some_and(char::is_uppercase))
104            })
105            .map(|code_token| {
106                expand_terms(&code_token)
107                    .into_iter()
108                    .map(|part| part.to_ascii_lowercase())
109                    .filter(|part| part.chars().count() >= 2)
110                    .collect::<HashSet<_>>()
111                    .into_iter()
112                    .collect()
113            })
114            .collect();
115        Self {
116            terms: terms.iter().map(|term| term.to_ascii_lowercase()).collect(),
117            code_token_parts,
118            languages: [
119                task_mentions_language(task, "javascript"),
120                task_mentions_language(task, "typescript"),
121                task_mentions_language(task, "python"),
122                task_mentions_language(task, "rust"),
123                task_mentions_language(task, "go"),
124            ],
125            mcp_repository_intent: mcp_repository_intent(&[], &primary_task),
126            owner_intent: !primary_task.is_empty(),
127        }
128    }

130    pub(super) fn score(&self, path: &str) -> f64 {
131        let normalized_path = path.to_lowercase();
132        let mut score = f64::from(
133            u32::try_from(
134                self.terms
135                    .iter()
136                    .filter(|term| normalized_path.contains(term.as_str()))
137                    .count(),
138            )
139            .unwrap_or(u32::MAX),
140        );
141        for parts in &self.code_token_parts {
142            let matched_parts = parts
143                .iter()
144                .filter(|part| normalized_path.contains(*part))
145                .count();
146            if matched_parts >= 2 {
147                let matched_parts = u32::try_from(matched_parts).unwrap_or(u32::MAX);
148                score += f64::from(matched_parts.saturating_mul(matched_parts));
149            }
150        }
151        for (mentioned, component, extensions) in [
152            (self.languages[0], "/js/", &["js", "jsx", "mjs", "cjs"][..]),
153            (self.languages[1], "/ts/", &["ts", "tsx"][..]),
154            (self.languages[2], "/python/", &["py", "pyw"][..]),
155            (self.languages[3], "/rust/", &["rs"][..]),
156            (self.languages[4], "/go/", &["go"][..]),
157        ] {
158            let extension_matches = normalized_path
159                .rsplit_once('.')
160                .is_some_and(|(_, extension)| extensions.contains(&extension));
161            let component = component.trim_matches('/');
162            let component_matches = normalized_path
163                .split('/')
164                .any(|path_component| path_component == component);
165            if mentioned && (extension_matches || component_matches) {
166                // An explicit language name in the task is strong repository-scope
167                // evidence. Keep this above an exact-name match in another
168                // language so common names such as `Point` do not dominate.
169                score += 12.0;
170            }
171        }
172        if self.mcp_repository_intent {
173            if normalized_path == "src/mcp.rs"
174                || normalized_path.starts_with("src/mcp/")
175                || normalized_path == "src/main/mcp_runtime.rs"
176            {
177                score += 14.0;
178            } else if normalized_path == "crates/test-suite/src/domains/protocol.rs"
179                || normalized_path == "crates/test-suite/src/domains/contracts.rs"
180                || normalized_path == "tests/mcp.rs"
181                || normalized_path.starts_with("tests/mcp/")
182            {
183                score += 8.0;
184            }
185        }
186        if self.owner_intent {
187            score += ranking::owner_path_prior(path);
188        }
189        score
190    }
191}

193pub(super) fn mcp_repository_intent(terms: &[String], task: &str) -> bool {
194    let terms = terms
195        .iter()
196        .map(|term| term.to_ascii_lowercase())
197        .chain(
198            task.split(|character: char| !character.is_alphanumeric() && character != '_')
199                .filter(|term| !term.is_empty())
200                .map(str::to_ascii_lowercase),
201        )
202        .collect::<HashSet<_>>();
203    if terms.contains("mcp") {
204        return true;
205    }
206    let surface_terms = ["tool", "tools", "catalog", "schema", "schemas"];
207    let registration_terms = ["registration", "registered", "register", "server"];
208    surface_terms.iter().any(|term| terms.contains(*term))
209        && registration_terms.iter().any(|term| terms.contains(*term))
210}

212pub(super) fn context_path_group(path: &str) -> String {
213    let mut components = path.split('/').filter(|component| !component.is_empty());
214    let first = components.next().unwrap_or("<root>");
215    let second = components.next();
216    if second.is_none() {
217        return "<root>".into();
218    }
219    if matches!(
220        first,
221        "src" | "lib" | "app" | "apps" | "crates" | "packages"
222    ) && let Some(second) = second
223    {
224        format!("{first}/{second}")
225    } else {
226        first.to_owned()
227    }
228}

230pub(super) fn build_context_routing(
231    request: &ContextRequest,
232    scope: &DiffScopeReceipt,
233    candidate_paths: usize,
234    selected_paths: &[String],
235) -> Option<ContextRoutingReceipt> {
236    if scope.changed_paths.len() < OVERSIZED_CHANGE_PATHS {
237        return None;
238    }
239    let mut grouped_paths = BTreeMap::<String, Vec<String>>::new();
240    for path in &scope.changed_paths {
241        grouped_paths
242            .entry(context_path_group(path))
243            .or_default()
244            .push(path.clone());
245    }
246    if grouped_paths.len() < MIN_OVERSIZED_PATH_GROUPS {
247        return None;
248    }
249    let path_groups_total = grouped_paths.len();
250    let mut groups = grouped_paths.into_iter().collect::<Vec<_>>();
251    groups.sort_by(|left, right| {
252        right
253            .1
254            .len()
255            .cmp(&left.1.len())
256            .then_with(|| left.0.cmp(&right.0))
257    });
258    let selected_groups = selected_paths.iter().fold(
259        BTreeMap::<String, BTreeSet<&str>>::new(),
260        |mut paths, path| {
261            paths
262                .entry(context_path_group(path))
263                .or_default()
264                .insert(path);
265            paths
266        },
267    );
268    let selected_path_count = selected_paths
269        .iter()
270        .map(String::as_str)
271        .collect::<BTreeSet<_>>()
272        .len();
273    let strongest_selected_group = selected_groups
274        .values()
275        .map(BTreeSet::len)
276        .max()
277        .unwrap_or(0);
278    let weakly_concentrated = selected_path_count > 1
279        && strongest_selected_group.saturating_mul(2) <= selected_path_count;
280
281    let suggestions = groups
282        .iter()
283        .take(MAX_ROUTING_SUGGESTIONS)
284        .map(|(prefix, _)| ContextRoutingSuggestion {
285            include_paths: vec![if prefix == "<root>" {
286                "*".into()
287            } else {
288                format!("{prefix}/**")
289            }],
290        })
291        .collect();
292    let path_groups = groups
293        .into_iter()
294        .take(MAX_ROUTING_GROUPS)
295        .map(|(prefix, paths)| ContextPathGroup {
296            prefix,
297            changed_paths: paths.len(),
298        })
299        .collect();
300    Some(ContextRoutingReceipt {
301        candidate_paths,
302        changed_paths: scope.changed_paths.len(),
303        changed_paths_complete: scope.changed_paths_complete,
304        changed_paths_limit: scope.changed_paths_limit,
305        selected_paths: selected_path_count,
306        weakly_concentrated,
307        consistency: IndexConsistency::IndexedGeneration,
308        base_revision: request.base_revision.clone(),
309        known_hashes: request.known_hashes.clone(),
310        path_groups_total,
311        path_groups,
312        suggestions,
313    })
314}

316pub(super) fn resolve_context_workflow(requested: ContextWorkflow, task: &str) -> ContextWorkflow {
317    if requested != ContextWorkflow::Auto {
318        return requested;
319    }
320    let task = task.to_ascii_lowercase();
321    if ["pull request", "contribution", "contributing"]
322        .iter()
323        .any(|signal| task.contains(signal))
324    {
325        ContextWorkflow::Contribution
326    } else if ["code review", "review this", "review the changes"]
327        .iter()
328        .any(|signal| task.contains(signal))
329    {
330        ContextWorkflow::Review
331    } else if ["investigate", "root cause", "diagnose"]
332        .iter()
333        .any(|signal| task.contains(signal))
334    {
335        ContextWorkflow::Investigation
336    } else {
337        ContextWorkflow::Implementation
338    }
339}

341pub(super) fn workflow_path_role(
342    path: &str,
343    request: &ContextRequest,
344) -> Option<(f64, &'static str)> {
345    let normalized = path.replace('\\', "/");
346    let lower = normalized.to_ascii_lowercase();
347    let mut best = WORKFLOW_PATHS
348        .matches(&normalized)
349        .into_iter()
350        .map(|index| WORKFLOW_PATH_ROLES[index])
351        .max_by(|left, right| left.0.total_cmp(&right.0));
352    if likely_owner_test(&lower, request) && best.is_none_or(|(score, _)| score < OWNER_TEST_SCORE)
353    {
354        best = Some((OWNER_TEST_SCORE, "owner_test"));
355    }
356    best
357}

380pub(super) const fn workflow_path_roles() -> [(f64, &'static str); WORKFLOW_PATH_RULES.len()] {
381    let mut roles = [(0.0, ""); WORKFLOW_PATH_RULES.len()];
382    let mut index = 0;
383    while index < WORKFLOW_PATH_RULES.len() {
384        roles[index] = (WORKFLOW_PATH_RULES[index].1, WORKFLOW_PATH_RULES[index].2);
385        index += 1;
386    }
387    roles
388}

390static WORKFLOW_PATHS: LazyLock<GlobSet> = LazyLock::new(|| {
391    let mut builder = GlobSetBuilder::new();
392    for (pattern, _, _) in WORKFLOW_PATH_RULES {
393        builder.add(

395                .case_insensitive(true)
396                .literal_separator(true)
397                .build()
398                .expect("static workflow glob"),
399        );
400    }
401    builder.build().expect("static workflow glob set")
402});

408pub(super) fn owner_test_changed_path(path: &str, request: &ContextRequest) -> Option<String> {
409    let path = path.replace('\\', "/").to_ascii_lowercase();
410    let is_test = path.contains("/test")
411        || path.starts_with("test")
412        || path.contains("/spec")
413        || path.starts_with("spec");
414    if !is_test {
415        return None;
416    }
417    let test_name = path.rsplit('/').next().unwrap_or(&path);
418    let test_stem = test_name.split('.').next().unwrap_or(test_name);
419    request
420        .changed_paths
421        .iter()
422        .chain(&request.focus_paths)
423        .find_map(|changed| {
424            let normalized = changed.replace('\\', "/");
425            let stem = normalized
426                .rsplit('/')
427                .next()
428                .and_then(|name| name.split('.').next())?
429                .to_ascii_lowercase();
430            (stem.len() >= 3 && contains_filename_token(test_stem, &stem)).then(|| changed.clone())
431        })
432}

434pub(super) fn contains_filename_token(path: &str, token: &str) -> bool {
435    path.match_indices(token).any(|(start, matched)| {
436        let end = start + matched.len();
437        let boundary =
438            |character: Option<char>| character.is_none_or(|value| !value.is_alphanumeric());
439        boundary(path[..start].chars().next_back()) && boundary(path[end..].chars().next())
440    })
441}
