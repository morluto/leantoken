1pub(super) fn parse_revision_range(revision: &str) -> Result<Option<(&str, &str)>> {
2    let Some((base, head)) = revision.split_once("..") else {
3        return Ok(None);
4    };
5    if base.trim().is_empty()
6        || head.trim().is_empty()
7        || base.ends_with('.')
8        || head.starts_with('.')
9        || head.contains("..")
10    {
11        return Err(Error::InvalidInput {
12            field: "base revision",
13            reason: "revision range must be BASE..HEAD",
14        });
15    }
16    Ok(Some((base, head)))
17}

53impl ContextPathScorer {
54    pub(super) fn new(terms: &[String], task: &str) -> Self {
55        let code_token_parts = facets::legacy_code_tokens(task)
56            .into_iter()
57            .filter(|token| {
58                token.contains("::")
59                    || token
60                        .split('.')
61                        .any(|part| part.chars().next().is_some_and(char::is_uppercase))
62            })
63            .map(|code_token| {
64                expand_terms(&code_token)
65                    .into_iter()
66                    .map(|part| part.to_ascii_lowercase())
67                    .filter(|part| part.chars().count() >= 2)
68                    .collect::<HashSet<_>>()
69                    .into_iter()
70                    .collect()
71            })
72            .collect();
73        Self {
74            terms: terms.iter().map(|term| term.to_ascii_lowercase()).collect(),
75            code_token_parts,
76            languages: [
77                task_mentions_language(task, "javascript"),
78                task_mentions_language(task, "typescript"),
79                task_mentions_language(task, "python"),
80                task_mentions_language(task, "rust"),
81                task_mentions_language(task, "go"),
82            ],
83            mcp_repository_intent: mcp_repository_intent(terms, task),
84        }
85    }

87    pub(super) fn score(&self, path: &str) -> f64 {
88        let path = path.to_lowercase();
89        let mut score = self
90            .terms
91            .iter()
92            .filter(|term| path.contains(term.as_str()))
93            .count() as f64;
94        for parts in &self.code_token_parts {
95            let matched_parts = parts.iter().filter(|part| path.contains(*part)).count();
96            if matched_parts >= 2 {
97                #[allow(clippy::cast_precision_loss)]
98                {
99                    score += (matched_parts * matched_parts) as f64;
100                }
101            }
102        }
103        for (mentioned, component, extensions) in [
104            (self.languages[0], "/js/", &["js", "jsx", "mjs", "cjs"][..]),
105            (self.languages[1], "/ts/", &["ts", "tsx"][..]),
106            (self.languages[2], "/python/", &["py", "pyw"][..]),
107            (self.languages[3], "/rust/", &["rs"][..]),
108            (self.languages[4], "/go/", &["go"][..]),
109        ] {
110            let extension_matches = path
111                .rsplit_once('.')
112                .is_some_and(|(_, extension)| extensions.contains(&extension));
113            let component = component.trim_matches('/');
114            let component_matches = path
115                .split('/')
116                .any(|path_component| path_component == component);
117            if mentioned && (extension_matches || component_matches) {
118                // An explicit language name in the task is strong repository-scope
119                // evidence. Keep this above an exact-name match in another
120                // language so common names such as `Point` do not dominate.
121                score += 12.0;
122            }
123        }
124        if self.mcp_repository_intent {
125            if path == "src/mcp.rs"
126                || path.starts_with("src/mcp/")
127                || path == "src/main/mcp_runtime.rs"
128            {
129                score += 14.0;
130            } else if path == "crates/test-suite/src/domains/protocol.rs"
131                || path == "crates/test-suite/src/domains/contracts.rs"
132                || path == "tests/mcp.rs"
133                || path.starts_with("tests/mcp/")
134            {
135                score += 8.0;
136            }
137        }
138        score
139    }
140}

142pub(super) fn mcp_repository_intent(terms: &[String], task: &str) -> bool {
143    let terms = terms
144        .iter()
145        .map(|term| term.to_ascii_lowercase())
146        .chain(
147            task.split(|character: char| !character.is_alphanumeric() && character != '_')
148                .filter(|term| !term.is_empty())
149                .map(str::to_ascii_lowercase),
150        )
151        .collect::<HashSet<_>>();
152    if terms.contains("mcp") {
153        return true;
154    }
155    let surface_terms = ["tool", "tools", "catalog", "schema", "schemas"];
156    let registration_terms = ["registration", "registered", "register", "server"];
157    surface_terms.iter().any(|term| terms.contains(*term))
158        && registration_terms.iter().any(|term| terms.contains(*term))
159}

161pub(super) fn context_path_group(path: &str) -> String {
162    let mut components = path.split('/').filter(|component| !component.is_empty());
163    let first = components.next().unwrap_or("<root>");
164    let second = components.next();
165    if second.is_none() {
166        return "<root>".into();
167    }
168    if matches!(
169        first,
170        "src" | "lib" | "app" | "apps" | "crates" | "packages"
171    ) && let Some(second) = second
172    {
173        format!("{first}/{second}")
174    } else {
175        first.to_owned()
176    }
177}

179pub(super) fn build_context_routing(
180    request: &ContextRequest,
181    scope: &DiffScopeReceipt,
182    candidate_paths: usize,
183    selected_paths: &[String],
184) -> Option<ContextRoutingReceipt> {
185    if scope.changed_paths.len() < OVERSIZED_CHANGE_PATHS {
186        return None;
187    }
188    let mut grouped_paths = BTreeMap::<String, Vec<String>>::new();
189    for path in &scope.changed_paths {
190        grouped_paths
191            .entry(context_path_group(path))
192            .or_default()
193            .push(path.clone());
194    }
195    if grouped_paths.len() < MIN_OVERSIZED_PATH_GROUPS {
196        return None;
197    }
198    let path_groups_total = grouped_paths.len();
199    let mut groups = grouped_paths.into_iter().collect::<Vec<_>>();
200    groups.sort_by(|left, right| {
201        right
202            .1
203            .len()
204            .cmp(&left.1.len())
205            .then_with(|| left.0.cmp(&right.0))
206    });
207    let selected_groups = selected_paths.iter().fold(
208        BTreeMap::<String, BTreeSet<&str>>::new(),
209        |mut paths, path| {
210            paths
211                .entry(context_path_group(path))
212                .or_default()
213                .insert(path);
214            paths
215        },
216    );
217    let selected_path_count = selected_paths
218        .iter()
219        .map(String::as_str)
220        .collect::<BTreeSet<_>>()
221        .len();
222    let strongest_selected_group = selected_groups
223        .values()
224        .map(BTreeSet::len)
225        .max()
226        .unwrap_or(0);
227    let weakly_concentrated = selected_path_count > 1
228        && strongest_selected_group.saturating_mul(2) <= selected_path_count;
229
230    let suggestions = groups
231        .iter()
232        .take(MAX_ROUTING_SUGGESTIONS)
233        .map(|(prefix, _)| ContextRoutingSuggestion {
234            include_paths: vec![if prefix == "<root>" {
235                "*".into()
236            } else {
237                format!("{prefix}/**")
238            }],
239        })
240        .collect();
241    let path_groups = groups
242        .into_iter()
243        .take(MAX_ROUTING_GROUPS)
244        .map(|(prefix, paths)| ContextPathGroup {
245            prefix,
246            changed_paths: paths.len(),
247        })
248        .collect();
249    Some(ContextRoutingReceipt {
250        candidate_paths,
251        changed_paths: scope.changed_paths.len(),
252        selected_paths: selected_path_count,
253        weakly_concentrated,
254        consistency: IndexConsistency::IndexedGeneration,
255        base_revision: request.base_revision.clone(),
256        known_hashes: request.known_hashes.clone(),
257        path_groups_total,
258        path_groups,
259        suggestions,
260    })
261}

263pub(super) fn resolve_context_workflow(requested: ContextWorkflow, task: &str) -> ContextWorkflow {
264    if requested != ContextWorkflow::Auto {
265        return requested;
266    }
267    let task = task.to_ascii_lowercase();
268    if ["pull request", "contribution", "contributing"]
269        .iter()
270        .any(|signal| task.contains(signal))
271    {
272        ContextWorkflow::Contribution
273    } else if ["code review", "review this", "review the changes"]
274        .iter()
275        .any(|signal| task.contains(signal))
276    {
277        ContextWorkflow::Review
278    } else if ["investigate", "root cause", "diagnose"]
279        .iter()
280        .any(|signal| task.contains(signal))
281    {
282        ContextWorkflow::Investigation
283    } else {
284        ContextWorkflow::Implementation
285    }
286}

288pub(super) fn workflow_path_role(
289    path: &str,
290    request: &ContextRequest,
291) -> Option<(f64, &'static str)> {
292    let normalized = path.replace('\\', "/");
293    let lower = normalized.to_ascii_lowercase();
294    let mut best = WORKFLOW_PATHS
295        .matches(&normalized)
296        .into_iter()
297        .map(|index| WORKFLOW_PATH_ROLES[index])
298        .max_by(|left, right| left.0.total_cmp(&right.0));
299    if likely_owner_test(&lower, request) && best.is_none_or(|(score, _)| score < OWNER_TEST_SCORE)
300    {
301        best = Some((OWNER_TEST_SCORE, "owner_test"));
302    }
303    best
304}

324pub(super) const WORKFLOW_PATH_ROLES: [(f64, &str); WORKFLOW_PATH_RULES.len()] =

327pub(super) const fn workflow_path_roles() -> [(f64, &'static str); WORKFLOW_PATH_RULES.len()] {
328    let mut roles = [(0.0, ""); WORKFLOW_PATH_RULES.len()];
329    let mut index = 0;
330    while index < WORKFLOW_PATH_RULES.len() {
331        roles[index] = (WORKFLOW_PATH_RULES[index].1, WORKFLOW_PATH_RULES[index].2);
332        index += 1;
333    }
334    roles
335}

337static WORKFLOW_PATHS: LazyLock<GlobSet> = LazyLock::new(|| {
338    let mut builder = GlobSetBuilder::new();
339    for (pattern, _, _) in WORKFLOW_PATH_RULES {
340        builder.add(
341            GlobBuilder::new(pattern)
342                .case_insensitive(true)
343                .literal_separator(true)
344                .build()
345                .expect("static workflow glob"),
346        );
347    }
348    builder.build().expect("static workflow glob set")
349});

355pub(super) fn owner_test_changed_path(path: &str, request: &ContextRequest) -> Option<String> {
356    let path = path.replace('\\', "/").to_ascii_lowercase();
357    let is_test = path.contains("/test")
358        || path.starts_with("test")
359        || path.contains("/spec")
360        || path.starts_with("spec");
361    if !is_test {
362        return None;
363    }
364    let test_name = path.rsplit('/').next().unwrap_or(&path);
365    let test_stem = test_name.split('.').next().unwrap_or(test_name);
366    request
367        .changed_paths
368        .iter()
369        .chain(&request.focus_paths)
370        .find_map(|changed| {
371            let normalized = changed.replace('\\', "/");
372            let stem = normalized
373                .rsplit('/')
374                .next()
375                .and_then(|name| name.split('.').next())?
376                .to_ascii_lowercase();
377            (stem.len() >= 3 && contains_filename_token(test_stem, &stem)).then(|| changed.clone())
378        })
379}

381pub(super) fn contains_filename_token(path: &str, token: &str) -> bool {
382    path.match_indices(token).any(|(start, matched)| {
383        let end = start + matched.len();
384        let boundary =
385            |character: Option<char>| character.is_none_or(|value| !value.is_alphanumeric());
386        boundary(path[..start].chars().next_back()) && boundary(path[end..].chars().next())
387    })
388}
