112pub(super) fn qualified_symbol_match(
113    concept: &str,
114    name: &str,
115    parent: Option<&str>,
116    signature: Option<&str>,
117) -> f64 {
118    if !concept.contains(['.', ':']) {
119        return 0.0;
120    }
121    let parts = concept
122        .split(|character: char| !character.is_alphanumeric() && character != '_')
123        .flat_map(identifier_words)
124        .map(|part| part.to_ascii_lowercase())
125        .filter(|part| part.chars().count() >= 2)
126        .collect::<HashSet<_>>();
127    if parts.len() < 2 {
128        return 0.0;
129    }
130    let haystack = format!(
131        "{} {} {}",
132        name,
133        parent.unwrap_or_default(),
134        signature.unwrap_or_default()
135    )
136    .to_ascii_lowercase();
137    f64::from(parts.iter().all(|part| haystack.contains(part)))
138}

140pub(super) fn record_query_hit(
141    fusion: &mut HashMap<String, HashMap<String, f64>>,
142    path: &str,
143    fusion_key: &str,
144    weight: f64,
145    rank: usize,
146) {
147    if weight < MIN_CORROBORATED_QUERY_WEIGHT {
148        return;
149    }
150    pub(super) const RRF_K: f64 = 60.0;
151    #[allow(clippy::cast_precision_loss)]
152    let score = weight * RRF_K / (RRF_K + rank as f64 + 1.0);
153    fusion
154        .entry(path.to_owned())
155        .or_default()
156        .entry(fusion_key.to_owned())
157        .and_modify(|current| *current = current.max(score))
158        .or_insert(score);
159}

161pub(super) fn apply_query_fusion(
162    candidates: &mut [Candidate],
163    fusion: &HashMap<String, HashMap<String, f64>>,
164) {
165    for candidate in candidates {
166        let Some(matches) = fusion.get(&candidate.path) else {
167            continue;
168        };
169        if matches.len() > 1 {
170            let total = matches.values().sum::<f64>();
171            let strongest = matches.values().copied().fold(0.0, f64::max);
172            candidate.path_score += (total - strongest).min(0.2);
173            if !candidate
174                .match_kinds
175                .iter()
176                .any(|kind| kind == "multi-query")
177            {
178                candidate.match_kinds.push("multi-query".into());
179            }
180        }
181    }
182}

206pub(super) fn corroborated_import_symbol<'a>(
207    symbols: Vec<SymbolRecord>,
208    queries: &'a [ContextQuery],
209    seed_concepts: &BTreeSet<String>,
210) -> Option<(SymbolRecord, &'a ContextQuery, f64)> {
211    let mut best: Option<(usize, usize, usize, SymbolRecord, &ContextQuery, f64)> = None;
212    for (query_rank, query) in queries.iter().enumerate() {
213        if query.concept_weight < MIN_CORROBORATED_QUERY_WEIGHT
214            || !seed_concepts.contains(&query.fusion_key)
215            || !(query.has_facet(FacetKind::ExactAtom)
216                || query.has_facet(FacetKind::Symbol)
217                || query.has_facet(FacetKind::Configuration))
218        {
219            continue;
220        }
221        for symbol in &symbols {
222            let exact = symbol.name.eq_ignore_ascii_case(&query.value);
223            let qualified = qualified_symbol_match(
224                &query.fusion_key,
225                &symbol.name,
226                symbol.parent.as_deref(),
227                symbol.signature.as_deref(),
228            ) > 0.0;
229            if !exact && !qualified {
230                continue;
231            }
232            let class = usize::from(qualified) * 2 + usize::from(exact);
233            let evidence = f64::from(exact) + f64::from(qualified) * 1.5;
234            let candidate = (
235                class,
236                usize::MAX - query_rank,
237                usize::MAX - symbol.start_line,
238                symbol.clone(),
239                query,
240                evidence,
241            );
242            if best.as_ref().is_none_or(|current| {
243                (candidate.0, candidate.1, candidate.2) > (current.0, current.1, current.2)
244            }) {
245                best = Some(candidate);
246            }
247        }
248    }
249    best.map(|(_, _, _, symbol, query, evidence)| (symbol, query, evidence))
250}

252pub(super) fn import_seed_paths(
253    candidates: &[Candidate],
254    queries: &[ContextQuery],
255    tokenizer: crate::tokens::Tokenizer,
256) -> Vec<(String, BTreeSet<String>)> {
257    if low_cardinality_exact_query(queries) {
258        return Vec::new();
259    }
260    let mut paths = BTreeMap::<String, (f64, BTreeSet<String>)>::new();
261    for candidate in candidates {
262        if candidate.concept_weight < MIN_CORROBORATED_QUERY_WEIGHT || candidate.concepts.is_empty()
263        {
264            continue;
265        }
266        let token_count = candidate.token_count_with(tokenizer).max(1);
267        let score = candidate.score(&ranking::Weights::default(), token_count);
268        let entry = paths
269            .entry(candidate.path.clone())
270            .or_insert_with(|| (score, BTreeSet::new()));
271        entry.0 = entry.0.max(score);
272        entry.1.extend(candidate.concepts.iter().cloned());
273    }
274    let mut paths = paths.into_iter().collect::<Vec<_>>();
275    paths.sort_by(|left, right| {
276        right
277            .1
278            .0
279            .total_cmp(&left.1.0)
280            .then_with(|| left.0.cmp(&right.0))
281    });
282    paths
283        .into_iter()
284        .map(|(path, (_, concepts))| (path, concepts))
285        .collect()
286}

297pub(super) fn task_mentions_language(task: &str, language: &str) -> bool {
298    task.split(|character: char| !character.is_alphanumeric())
299        .filter(|word| !word.is_empty())
300        .any(|word| {
301            if language == "go" {
302                word == "Go" || word.eq_ignore_ascii_case("golang")
303            } else {
304                word.eq_ignore_ascii_case(language)
305            }
306        })
307}
