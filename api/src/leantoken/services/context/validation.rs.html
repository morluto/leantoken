2    pub(super) fn validate_workflow_evidence(&self, evidence: &WorkflowEvidence) -> Result<()> {
3        let classes = [
4            ("failure_traces", &evidence.failure_traces),
5            ("symbols", &evidence.symbols),
6            ("paths", &evidence.paths),
7            ("test_intents", &evidence.test_intents),
8        ];
9        let mut total_bytes = 0usize;
10        for (field, values) in classes {
11            if values.len() > MAX_WORKFLOW_EVIDENCE_ITEMS_PER_CLASS {
12                return Err(Error::RequestLimitExceeded {
13                    field: "workflow_evidence items per class",
14                    requested: values.len(),
15                    limit: MAX_WORKFLOW_EVIDENCE_ITEMS_PER_CLASS,
16                });
17            }
18            for value in values {
19                validate_input(value, field, MAX_WORKFLOW_EVIDENCE_ITEM_BYTES)?;
20                if value.trim().is_empty() {
21                    return Err(Error::InvalidInput {
22                        field: "workflow_evidence",
23                        reason: "must not contain empty values",
24                    });
25                }
26                total_bytes = total_bytes.saturating_add(value.len());
27            }
28        }
29        if total_bytes > MAX_WORKFLOW_EVIDENCE_TOTAL_BYTES {
30            return Err(Error::RequestLimitExceeded {
31                field: "workflow_evidence total bytes",
32                requested: total_bytes,
33                limit: MAX_WORKFLOW_EVIDENCE_TOTAL_BYTES,
34            });
35        }
36        for path in &evidence.paths {
37            validate_relative(path)?;
38        }
39        Ok(())
40    }

42    pub(super) fn validate_context_request(
43        &self,
44        request: &ContextRequest,
45        handoff: Option<&HandoffManifestRequest>,
46    ) -> Result<()> {
47        validate_context_option_constraints(request, handoff)?;
48        if request.task.trim().is_empty() {
49            return Err(Error::InvalidInput {
50                field: "task",
51                reason: "must not be empty",
52            });
53        }
54        self.token_budget_limit(request.token_budget)?;
55        if let Some(max_fragments) = request.max_fragments {
56            self.result_limit(Some(max_fragments))?;
57        }
58        if let Some(minimum) = request.minimum_fragments_per_focus_path {
59            self.result_limit(Some(minimum))?;
60            if minimum > MAX_CONTEXT_FOCUS_CANDIDATES_PER_PATTERN {
61                return Err(Error::RequestLimitExceeded {
62                    field: "minimum_fragments_per_focus_path",
63                    requested: minimum,
64                    limit: MAX_CONTEXT_FOCUS_CANDIDATES_PER_PATTERN,
65                });
66            }
67        }
68        if request.focus_paths.len() > MAX_CONTEXT_FOCUS_PATTERNS {
69            return Err(Error::RequestLimitExceeded {
70                field: "focus_paths",
71                requested: request.focus_paths.len(),
72                limit: MAX_CONTEXT_FOCUS_PATTERNS,
73            });
74        }
75        validate_input(&request.task, "task", MAX_QUERY_BYTES)?;
76        if request
77            .include_paths
78            .iter()
79            .any(|pattern| pattern.trim().is_empty())
80        {
81            return Err(Error::InvalidInput {
82                field: "include paths",
83                reason: "must not contain empty patterns",
84            });
85        }
86        validate_glob_patterns(&request.include_paths)?;
87        if request
88            .must_include_paths
89            .iter()
90            .any(|pattern| pattern.trim().is_empty())
91        {
92            return Err(Error::InvalidInput {
93                field: "must include paths",
94                reason: "must not contain empty patterns",
95            });
96        }
97        validate_glob_patterns(&request.must_include_paths)?;
98        if request
99            .focus_paths
100            .iter()
101            .any(|pattern| pattern.trim().is_empty())
102        {
103            return Err(Error::InvalidInput {
104                field: "focus paths",
105                reason: "must not contain empty patterns",
106            });
107        }
108        validate_glob_patterns(&request.focus_paths)?;
109        if request
110            .exclude_paths
111            .iter()
112            .any(|pattern| pattern.trim().is_empty())
113        {
114            return Err(Error::InvalidInput {
115                field: "exclude paths",
116                reason: "must not contain empty patterns",
117            });
118        }
119        validate_glob_patterns(&request.exclude_paths)?;
120        if request.focus_symbols.len() > MAX_INPUT_ITEMS {
121            return Err(Error::LimitExceeded);
122        }
123        for symbol in &request.focus_symbols {
124            validate_input(symbol, "focus symbol", MAX_PATTERN_BYTES)?;
125            if symbol.trim().is_empty() {
126                return Err(Error::InvalidInput {
127                    field: "focus symbols",
128                    reason: "must not contain empty symbols",
129                });
130            }
131        }
132        if request.must_include_symbols.len() > MAX_INPUT_ITEMS {
133            return Err(Error::LimitExceeded);
134        }
135        for symbol in &request.must_include_symbols {
136            validate_input(symbol, "must include symbol", MAX_PATTERN_BYTES)?;
137            if symbol.trim().is_empty() {
138                return Err(Error::InvalidInput {
139                    field: "must include symbols",
140                    reason: "must not contain empty symbols",
141                });
142            }
143        }
144        if request.required_evidence.len() > MAX_CONTEXT_REQUIRED_EVIDENCE {
145            return Err(Error::RequestLimitExceeded {
146                field: "required_evidence",
147                requested: request.required_evidence.len(),
148                limit: MAX_CONTEXT_REQUIRED_EVIDENCE,
149            });
150        }
151        let mut evidence_query_bytes = 0usize;
152        for requirement in &request.required_evidence {
153            validate_glob_patterns(std::slice::from_ref(&requirement.path))?;
154            if requirement.path.trim().is_empty() {
155                return Err(Error::InvalidInput {
156                    field: "required_evidence path",
157                    reason: "must not be empty",
158                });
159            }
160            if requirement.queries.is_empty() {
161                return Err(Error::InvalidInput {
162                    field: "required_evidence queries",
163                    reason: "must not be empty",
164                });
165            }
166            if requirement.queries.len() > MAX_CONTEXT_EVIDENCE_QUERIES {
167                return Err(Error::RequestLimitExceeded {
168                    field: "required_evidence queries",
169                    requested: requirement.queries.len(),
170                    limit: MAX_CONTEXT_EVIDENCE_QUERIES,
171                });
172            }
173            if requirement.minimum_query_matches == 0
174                || requirement.minimum_query_matches > requirement.queries.len()
175            {
176                return Err(Error::InvalidInput {
177                    field: "required_evidence minimum_query_matches",
178                    reason: "must be between one and the number of queries",
179                });
180            }
181            for query in &requirement.queries {
182                validate_input(query, "required_evidence query", MAX_PATTERN_BYTES)?;
183                if query.trim().is_empty() {
184                    return Err(Error::InvalidInput {
185                        field: "required_evidence queries",
186                        reason: "must not contain empty queries",
187                    });
188                }
189                evidence_query_bytes = evidence_query_bytes.saturating_add(query.len());
190            }
191        }
192        if evidence_query_bytes > MAX_CONTEXT_EVIDENCE_QUERY_BYTES {
193            return Err(Error::RequestLimitExceeded {
194                field: "required_evidence query bytes",
195                requested: evidence_query_bytes,
196                limit: MAX_CONTEXT_EVIDENCE_QUERY_BYTES,
197            });
198        }
199        if request.known_hashes.len() > MAX_INPUT_ITEMS {
200            return Err(Error::LimitExceeded);
201        }
202        for hash in &request.known_hashes {
203            validate_input(hash, "known hash", 128)?;
204        }
205        if request.changed_paths.len() > MAX_DIFF_CHANGED_PATHS {
206            return Err(Error::LimitExceeded);
207        }
208        for path in &request.changed_paths {
209            validate_input(path, "changed path", MAX_PATH_BYTES)?;
210            validate_relative(path)?;
211        }
212        if let Some(revision) = request.base_revision.as_deref() {
213            validate_revision_field(revision)?;
214        }
215        for query in facets::plan(&request.task, MAX_CONTEXT_QUERIES)
216            .queries
217            .iter()
218            .filter(|query| !query.has_facet(FacetKind::TestIntent))
219        {
220            compile_literal_regex(&query.value, false)?;
221        }
222        if let Some(handoff) = handoff {
223            handoff::validate_request(handoff)?;
224        }
225        Ok(())
226    }
227}

229pub(super) fn validate_revision_field(revision: &str) -> Result<()> {
230    if revision.trim().is_empty() {
231        return Err(Error::InvalidInput {
232            field: "base revision",
233            reason: "must not be empty",
234        });
235    }
236    if revision != revision.trim() {
237        return Err(Error::InvalidInput {
238            field: "base revision",
239            reason: "must not have leading or trailing whitespace",
240        });
241    }
242    validate_input(revision, "base revision", MAX_BASE_REVISION_BYTES)?;
243    parse_revision_range(revision)?;
244    Ok(())
245}

247pub(super) fn validate_context_option_constraints(
248    request: &ContextRequest,
249    handoff: Option<&HandoffManifestRequest>,
250) -> Result<()> {
251    let mut violations = Vec::with_capacity(3);
252    if (request.strict_focus_paths || request.minimum_fragments_per_focus_path.is_some())
253        && request.focus_paths.is_empty()
254    {
255        violations.push(crate::InputViolation::new(
256            "focus paths",
257            "must not be empty when focus path constraints are enabled",
258        ));
259    }
260    if request.plan_only && request.receipt_id.is_some() {
261        violations.push(crate::InputViolation::new(
262            "receipt_id",
263            "must be omitted when plan_only is true",
264        ));
265    }
266    if request.plan_only && handoff.is_some() {
267        violations.push(crate::InputViolation::new(
268            "plan_only",
269            "cannot be combined with a handoff manifest",
270        ));
271    }
272    match violations.len() {
273        0 => Ok(()),
274        1 => {
275            let violation = violations[0];
276            Err(Error::InvalidInput {
277                field: violation.field,
278                reason: violation.reason,
279            })
280        }
281        _ => Err(Error::InvalidInputConstraints(crate::InputViolations::new(
282            violations,
283        ))),
284    }
285}
