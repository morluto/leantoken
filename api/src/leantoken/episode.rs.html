32/// Versioned input adapter for an existing redacted analyzer report.
33#[derive(Debug, Clone, Copy, PartialEq, Eq)]
34pub enum EpisodeAdapter {
35    /// Aggregate produced by `codex_multi_agent_suite`.
36    MultiAgentSuiteV1,
37    /// Classification produced by `model_ab_trajectory`.
38    ModelAbTrajectoryV1,
39    /// Version-two report produced by `mcp_wire_analyze`.
40    McpWireReportV2,
41    /// Publishable receipt produced by `codex_host_receipt`.
42    CodexHostReceiptV1,
43    /// Classification produced by `context_utilization`.
44    ContextUtilizationV1,
45}
46
47impl EpisodeAdapter {
48    fn name(self) -> &'static str {
49        match self {
50            Self::MultiAgentSuiteV1 => "multi_agent_suite",
51            Self::ModelAbTrajectoryV1 => "model_ab_trajectory",
52            Self::McpWireReportV2 => "mcp_wire_report",
53            Self::CodexHostReceiptV1 => "codex_host_receipt",
54            Self::ContextUtilizationV1 => "context_utilization",
55        }
56    }
57
58    const fn version(self) -> u32 {
59        match self {
60            Self::McpWireReportV2 => 2,
61            Self::MultiAgentSuiteV1
62            | Self::ModelAbTrajectoryV1
63            | Self::CodexHostReceiptV1
64            | Self::ContextUtilizationV1 => 1,
65        }
66    }
67}

403/// Normalize one existing, redacted analyzer report.
404///
405/// # Errors
406///
407/// Returns an error if the file exceeds the byte bound, has the wrong adapter
408/// schema, omits a required binding, violates a count bound, or is internally
409/// inconsistent.
410pub fn audit_episode(request: &EpisodeAuditRequest) -> Result<EpisodeAuditReport> {
411    let bytes = read_bounded(&request.input)?;
412    audit_episode_bytes(request.adapter, &bytes)
413}
414
415fn read_bounded(path: &Path) -> Result<Vec<u8>> {
416    read_bounded_with_limit(path, MAX_EPISODE_INPUT_BYTES)
417}
418
419fn read_bounded_with_limit(path: &Path, limit: u64) -> Result<Vec<u8>> {
420    let mut bytes = Vec::new();
421    File::open(path)?
422        .take(limit.saturating_add(1))
423        .read_to_end(&mut bytes)?;
424    if u64::try_from(bytes.len()).unwrap_or(u64::MAX) > limit {
425        return invalid(format!("episode input exceeds byte bound of {limit}"));
426    }
427    Ok(bytes)
428}

430fn audit_episode_bytes(adapter: EpisodeAdapter, bytes: &[u8]) -> Result<EpisodeAuditReport> {
431    if u64::try_from(bytes.len()).unwrap_or(u64::MAX) > MAX_EPISODE_INPUT_BYTES {
432        return invalid(format!(
433            "episode input exceeds byte bound of {MAX_EPISODE_INPUT_BYTES}"
434        ));
435    }
436    let value: Value = serde_json::from_slice(bytes).map_err(|error| {
437        Error::InvalidRequest(format!(
438            "invalid episode JSON at line {}, column {}",
439            error.line(),
440            error.column()
441        ))
442    })?;
443    let schema_version = required_u64(&value, "/schema_version")?;
444    if schema_version != u64::from(adapter.version()) {
445        return invalid(format!(
446            "adapter {} v{} requires input schema {}, found {schema_version}",
447            adapter.name(),
448            adapter.version(),
449            adapter.version()
450        ));
451    }
452    let mut report = match adapter {
453        EpisodeAdapter::MultiAgentSuiteV1 => normalize_suite(&value)?,
454        EpisodeAdapter::ModelAbTrajectoryV1 => normalize_trajectory(&value)?,
455        EpisodeAdapter::McpWireReportV2 => normalize_wire(&value)?,
456        EpisodeAdapter::CodexHostReceiptV1 => normalize_host_receipt(&value)?,
457        EpisodeAdapter::ContextUtilizationV1 => normalize_context_utilization(&value)?,
458    };
459    report.adapter = AdapterReceipt {
460        name: adapter.name(),
461        version: adapter.version(),
462        input_schema_version: u32::try_from(schema_version)
463            .map_err(|_| Error::InvalidRequest("input schema version does not fit u32".into()))?,
464    };
465    report.source.input_blake3 = blake3::hash(bytes).to_hex().to_string();
466    report.source.artifact_bindings.sort();
467    report.source.artifact_bindings.dedup();
468    if report.source.artifact_bindings.len() > MAX_ARTIFACT_BINDINGS {
469        return invalid(format!(
470            "artifact bindings exceed bound: {} > {MAX_ARTIFACT_BINDINGS}",
471            report.source.artifact_bindings.len()
472        ));
473    }
474    report.findings.sort_by_key(|finding| finding.code);
475    Ok(report)
476}

555fn normalize_suite(value: &Value) -> Result<EpisodeAuditReport> {
556    if !required_bool(value, "/consistency/complete_schedule")? {
557        return invalid("suite schedule is incomplete");
558    }
559    let run_count = bounded_count(required_u64(value, "/run_count")?, MAX_EPISODES, "episodes")?;
560    let samples = required_array(value, "/run_samples")?;
561    if samples.len() != run_count {
562        return invalid("run_count does not match run_samples length");
563    }
564    let arms = required_array(value, "/arms")?
565        .iter()
566        .enumerate()
567        .map(|(index, arm)| {
568            arm.as_str()
569                .filter(|arm| !arm.trim().is_empty())
570                .map(str::to_owned)
571                .ok_or_else(|| Error::InvalidRequest(format!("invalid arm at /arms/{index}")))
572        })
573        .collect::<Result<BTreeSet<_>>>()?;
574    if arms.len() > 64 {
575        return invalid("suite arm count exceeds bound of 64");
576    }
577    let candidate_arms = arms
578        .iter()
579        .filter(|arm| arm.contains("leantoken"))
580        .cloned()
581        .collect::<Vec<_>>();
582    if candidate_arms.len() != 1 {
583        return invalid("suite adapter requires exactly one LeanToken candidate arm");
584    }
585    let candidate_arm = &candidate_arms[0];
586    let baseline_arms = arms
587        .iter()
588        .filter(|arm| arm.as_str() == "thin-native")
589        .cloned()
590        .collect::<Vec<_>>();
591    if baseline_arms.len() != 1 {
592        return invalid("suite adapter requires the thin-native retrieval baseline");
593    }
594    let baseline_arm = &baseline_arms[0];
595
596    let mut schedule_indexes = BTreeSet::new();
597    let mut run_keys = BTreeSet::new();
598    let mut runs = Vec::with_capacity(run_count);
599    let mut bindings = standard_bindings(
600        value,
601        &[
602            ("suite_manifest", "/suite_manifest_blake3"),
603            ("aggregate_binary", "/aggregate_binary_blake3"),
604            ("receipt_set", "/receipt_set_blake3"),
605        ],
606    )?;
607    let receipt_binary_bindings = required_array(value, "/consistency/receipt_binary_blake3")?;
608    if receipt_binary_bindings.len() > 64 {
609        return invalid("suite receipt-binary bindings exceed bound of 64");
610    }
611    if run_count
612        .saturating_add(bindings.len())
613        .saturating_add(receipt_binary_bindings.len())
614        > MAX_ARTIFACT_BINDINGS
615    {
616        return invalid("suite artifact bindings exceed the normalized report bound");
617    }
618    for (index, digest) in receipt_binary_bindings.iter().enumerate() {
619        push_blake3_binding(
620            &mut bindings,
621            format!("receipt_binary_{index}"),
622            digest.as_str().ok_or_else(|| {
623                Error::InvalidRequest(format!(
624                    "invalid digest at /consistency/receipt_binary_blake3/{index}"
625                ))
626            })?,
627        )?;
628    }
629    for (index, sample) in samples.iter().enumerate() {
630        let prefix = format!("/run_samples/{index}");
631        let schedule_index = required_u64(sample, "/schedule_index")?;
632        let repetition = required_u64(sample, "/repetition")?;
633        let task_id = required_str(sample, "/task_id")?.to_owned();
634        let arm = required_str(sample, "/arm")?.to_owned();
635        if !arms.contains(&arm) {
636            return invalid(format!("{prefix}/arm is not declared by the suite"));
637        }
638        if !schedule_indexes.insert(schedule_index) {
639            return invalid("suite contains duplicate schedule_index values");
640        }
641        if !run_keys.insert((task_id.clone(), repetition, arm.clone())) {
642            return invalid("suite contains duplicate task/repetition/arm records");
643        }
644        push_blake3_binding(
645            &mut bindings,
646            format!("source_family_{schedule_index}"),
647            required_str(sample, "/source_family_blake3")?,
648        )?;
649        runs.push(SuiteRun {
650            task_id,
651            repetition,
652            arm,
653            success: required_bool(sample, "/success")?,
654            family_input: required_u64(sample, "/family_usage/total_input_tokens")?,
655            child_requests: required_u64(sample, "/child_provider_requests")?,
656            child_mcp_calls: required_u64(sample, "/child_mcp_calls")?,
657            child_failed_mcp_calls: required_u64(sample, "/child_failed_mcp_calls")?,
658            child_mcp_result_bytes: required_u64(sample, "/child_mcp_result_bytes")?,
659            child_mcp_source_tokens: required_u64(sample, "/child_mcp_source_tokens")?,
660            child_shell_calls: required_u64(sample, "/child_shell_calls")?,
661        });
662    }
663    let task_count = bounded_count(required_u64(value, "/task_count")?, MAX_EPISODES, "tasks")?;
664    let repetitions = bounded_count(
665        required_u64(value, "/repetitions")?,
666        MAX_EPISODES,
667        "repetitions",
668    )?;
669    let expected_runs = task_count
670        .checked_mul(repetitions)
671        .and_then(|count| count.checked_mul(arms.len()))
672        .ok_or_else(|| Error::InvalidRequest("suite schedule cardinality overflow".into()))?;
673    if expected_runs != run_count {
674        return invalid("suite task/arm/repetition cardinality does not match run_count");
675    }
676    let expected_schedule_indexes = (1..=u64::try_from(run_count)
677        .map_err(|_| Error::InvalidRequest("schedule count overflow".into()))?)
678        .collect::<BTreeSet<_>>();
679    if schedule_indexes != expected_schedule_indexes {
680        return invalid("suite schedule indexes must be contiguous and start at one");
681    }
682
683    let total_model_calls = checked_sum(runs.iter().map(|run| run.child_requests), "model calls")?;
684    bounded_count(total_model_calls, MAX_EPISODE_TOOL_CALLS, "model calls")?;
685    let total_tool_calls = checked_sum(
686        runs.iter()
687            .map(|run| {
688                run.child_mcp_calls
689                    .checked_add(run.child_shell_calls)
690                    .ok_or_else(|| Error::InvalidRequest("tool calls overflow".into()))
691            })
692            .collect::<Result<Vec<_>>>()?,
693        "tool calls",
694    )?;
695    bounded_count(total_tool_calls, MAX_EPISODE_TOOL_CALLS, "tool calls")?;
696    let total_provider_input =
697        checked_sum(runs.iter().map(|run| run.family_input), "provider input")?;
698    let total_result_bytes = checked_sum(
699        runs.iter().map(|run| run.child_mcp_result_bytes),
700        "MCP result bytes",
701    )?;
702    let total_source_tokens = checked_sum(
703        runs.iter().map(|run| run.child_mcp_source_tokens),
704        "MCP source tokens",
705    )?;
706    let total_failures = checked_sum(
707        runs.iter().map(|run| run.child_failed_mcp_calls),
708        "failed MCP calls",
709    )?;
710    let successes = u64::try_from(runs.iter().filter(|run| run.success).count())
711        .map_err(|_| Error::InvalidRequest("success count overflow".into()))?;
712
713    let candidate_runs = runs
714        .iter()
715        .filter(|run| &run.arm == candidate_arm)
716        .collect::<Vec<_>>();
717    let baseline_runs = runs
718        .iter()
719        .filter(|run| &run.arm == baseline_arm)
720        .collect::<Vec<_>>();
721    if candidate_runs.is_empty() || candidate_runs.len() != baseline_runs.len() {
722        return invalid("suite candidate and retrieval baseline are not equally populated");
723    }
724    let candidate_keys = candidate_runs
725        .iter()
726        .map(|run| (&run.task_id, run.repetition))
727        .collect::<BTreeSet<_>>();
728    let baseline_keys = baseline_runs
729        .iter()
730        .map(|run| (&run.task_id, run.repetition))
731        .collect::<BTreeSet<_>>();
732    if candidate_keys != baseline_keys {
733        return invalid("suite candidate and retrieval baseline samples are not paired");
734    }
735    let candidate_requests = checked_sum(
736        candidate_runs.iter().map(|run| run.child_requests),
737        "candidate model calls",
738    )?;
739    let candidate_input = checked_sum(
740        candidate_runs.iter().map(|run| run.family_input),
741        "candidate input",
742    )?;
743    let baseline_input = checked_sum(
744        baseline_runs.iter().map(|run| run.family_input),
745        "baseline input",
746    )?;
747    if baseline_input == 0 {
748        return invalid("suite retrieval baseline input must be positive");
749    }
750    let request_mean = exact_f64(candidate_requests, "candidate model calls")?
751        / exact_f64(
752            u64::try_from(candidate_runs.len())
753                .map_err(|_| Error::InvalidRequest("candidate count overflow".into()))?,
754            "candidate episode count",
755        )?;
756    let baseline_input_f64 = exact_f64(baseline_input, "baseline provider input")?;
757    let candidate_input_f64 = exact_f64(candidate_input, "candidate provider input")?;
758    let savings = (baseline_input_f64 - candidate_input_f64) / baseline_input_f64;
759
760    let arm_summaries = required_array(value, "/arm_summaries")?;
761    let declared_candidate = arm_summaries
762        .iter()
763        .find(|summary| optional_str(summary, "/arm") == Some(candidate_arm.as_str()))
764        .ok_or_else(|| Error::InvalidRequest("candidate arm summary is missing".into()))?;
765    verify_u64(
766        declared_candidate,
767        "/runs",
768        u64::try_from(candidate_runs.len())
769            .map_err(|_| Error::InvalidRequest("candidate count overflow".into()))?,
770    )?;
771    verify_u64(
772        declared_candidate,
773        "/successes",
774        u64::try_from(candidate_runs.iter().filter(|run| run.success).count())
775            .map_err(|_| Error::InvalidRequest("candidate success count overflow".into()))?,
776    )?;
777    verify_u64(
778        declared_candidate,
779        "/child_provider_requests/sum",
780        candidate_requests,
781    )?;
782    verify_f64(
783        declared_candidate,
784        "/child_provider_requests/mean",
785        request_mean,
786    )?;
787    verify_u64(
788        declared_candidate,
789        "/family_total_input/sum",
790        candidate_input,
791    )?;
792    let candidate_contract_violations = required_u64(declared_candidate, "/contract_violations")?;
793    let declared_all_violations = required_u64(value, "/consistency/contract_violation_count")?;
794    let summed_violations = checked_sum(
795        arm_summaries
796            .iter()
797            .map(|summary| required_u64(summary, "/contract_violations"))
798            .collect::<Result<Vec<_>>>()?,
799        "contract violations",
800    )?;
801    if summed_violations != declared_all_violations {
802        return invalid("suite contract violation summaries are inconsistent");
803    }
804
805    let comparisons = required_array(value, "/comparisons")?;
806    let declared_comparison = comparisons
807        .iter()
808        .find(|comparison| {
809            optional_str(comparison, "/baseline") == Some(baseline_arm.as_str())
810                && optional_str(comparison, "/candidate") == Some(candidate_arm.as_str())
811        })
812        .ok_or_else(|| {
813            Error::InvalidRequest(
814                "candidate versus retrieval-baseline comparison is missing".into(),
815            )
816        })?;
817    verify_f64(
818        declared_comparison,
819        "/weighted_total_input_savings_fraction",
820        savings,
821    )?;
822    verify_u64(
823        declared_comparison,
824        "/paired_runs",
825        u64::try_from(candidate_runs.len())
826            .map_err(|_| Error::InvalidRequest("candidate count overflow".into()))?,
827    )?;
828
829    let mut findings = vec![EpisodeFinding {
830        code: if candidate_arm.contains("context-bundle") {
831            "one_context_optional_search_contract"
832        } else {
833            "iterative_retrieval_contract"
834        },
835        occurrences: u64::try_from(candidate_runs.len())
836            .map_err(|_| Error::InvalidRequest("candidate count overflow".into()))?,
837        value: Some(request_mean),
838        unit: Some("mean_model_calls"),
839        evidence: FindingEvidence::AggregateExact,
840        detail: if candidate_arm.contains("context-bundle") {
841            "The frozen aggregate identifies the one-context plus optional-search candidate; contract violations are reported separately."
842        } else {
843            "The frozen aggregate identifies an iterative LeanToken retrieval candidate; the value is its mean child provider-request count."
844        },
845    }];
846    if savings < 0.0 {
847        findings.push(EpisodeFinding {
848            code: "provider_input_regression",
849            occurrences: u64::try_from(candidate_runs.len())
850                .map_err(|_| Error::InvalidRequest("candidate count overflow".into()))?,
851            value: Some(-savings),
852            unit: Some("fraction"),
853            evidence: FindingEvidence::AggregateExact,
854            detail: "Candidate family input exceeded its paired retrieval baseline in the complete redacted aggregate.",
855        });
856    }
857    if candidate_contract_violations > 0 {
858        findings.push(EpisodeFinding {
859            code: "retrieval_contract_violation",
860            occurrences: candidate_contract_violations,
861            value: None,
862            unit: None,
863            evidence: FindingEvidence::AggregateExact,
864            detail: "The frozen suite classifier recorded calls outside the candidate retrieval contract.",
865        });
866    }
867
868    let mut summary = empty_summary();
869    summary.successful_episodes = AuditCount::complete(successes);
870    summary.model_calls = AuditCount::subset(total_model_calls);
871    summary.tool_calls = AuditCount::subset(total_tool_calls);
872    summary.provider_input_tokens = AuditCount::complete(total_provider_input);
873    summary.mcp_result_bytes = AuditCount::subset(total_result_bytes);
874    summary.mcp_source_tokens = AuditCount::subset(total_source_tokens);
875    summary.failure_events = AuditCount::subset(total_failures);
876    Ok(base_report(
877        u64::try_from(run_count)
878            .map_err(|_| Error::InvalidRequest("episode count overflow".into()))?,
879        summary,
880        bindings,
881        classifier_coverage(&[], &[]),
882        findings,
883        vec![
884            "Model and tool counts cover child threads exposed by the aggregate, not every root-thread action.",
885            "Provider input is aggregate context volume; provider cache state is not controlled.",
886            "Task success uses the source suite's redacted path-set validator and does not establish patch correctness.",
887            "Contract labels come from the frozen suite arm and its aggregate violation accounting; this adapter does not reconstruct private tool arguments.",
888        ],
889    ))
890}

1249    summary.failure_events = AuditCount::subset(failures);
