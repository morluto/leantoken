3#[derive(Debug, Clone, Copy, Default, Deserialize, Serialize, JsonSchema, PartialEq, Eq)]
4#[serde(rename_all = "snake_case")]
5/// Response projection for structural outlines.
6pub(in crate::mcp) enum OutlineMcpProjection {
7    /// Preserve symbols, imports, and byte offsets.
8    #[default]
9    Full,
10    /// Return symbol signatures and line ranges without imports or byte offsets.
11    Signatures,
12}
13
14#[derive(Debug, Deserialize, JsonSchema)]
15#[serde(deny_unknown_fields)]
16pub(in crate::mcp) struct OutlineMcpRequest {
17    /// Expected opaque repository identity from an earlier response.
18    #[serde(default)]
19    #[schemars(schema_with = "expected_repository_id_schema")]
20    pub(in crate::mcp) expected_repository_id: Option<String>,
21    /// One to 256 repository-relative source files to outline.
22    #[schemars(length(min = 1, max = 256), inner(length(max = 4096)))]
23    pub(in crate::mcp) paths: Vec<String>,
24    /// Keep definitions whose names contain this value.
25    #[serde(default)]
26    #[schemars(length(max = 4096))]
27    pub(in crate::mcp) symbol_name: Option<String>,
28    /// Keep definitions of this exact syntax kind.
29    #[serde(default)]
30    #[schemars(length(max = 4096))]
31    pub(in crate::mcp) symbol_kind: Option<String>,
32    /// Maximum definitions and imports to return (default 20, maximum 100).
33    #[serde(default, deserialize_with = "deserialize_optional_limit")]
34    #[schemars(schema_with = "result_limit_schema", default = "default_result_option")]
35    pub(in crate::mcp) max_results: Option<usize>,
36    /// Maximum signature and import tokens (default 8000, maximum 32000).
37    #[serde(default, deserialize_with = "deserialize_optional_limit")]
38    #[schemars(schema_with = "token_limit_schema", default = "default_token_option")]
39    pub(in crate::mcp) max_tokens: Option<usize>,
40    /// Maximum tokens in the final serialized service response.
41    #[serde(default)]
42    #[schemars(schema_with = "response_token_limit_schema")]
43    pub(in crate::mcp) max_response_tokens: Option<usize>,
44    /// Suppress evidence already returned under this server-managed receipt.
45    #[serde(default)]
46    #[schemars(length(max = 128))]
47    pub(in crate::mcp) receipt_id: Option<String>,
48    /// Opaque cursor from a result-limited outline response.
49    #[serde(default)]
50    #[schemars(length(max = 256))]
51    pub(in crate::mcp) cursor: Option<String>,
52    /// Use `reconcile_working_tree` after edits; otherwise `indexed_generation`.
53    #[serde(default)]
54    #[schemars(schema_with = "index_consistency_schema")]
55    pub(in crate::mcp) consistency: IndexConsistency,
56    /// Response shape: `full` definitions (default) or compact `signatures`.
57    #[serde(default)]
58    pub(in crate::mcp) projection: OutlineMcpProjection,
59}
60
61impl OutlineMcpRequest {
62    pub(in crate::mcp) fn validate_limits(&self, limits: McpLimitPolicy) -> crate::Result<()> {
63        validate_optional_positive_limit("max_results", self.max_results, limits.max_results)?;
64        validate_optional_positive_limit("max_tokens", self.max_tokens, limits.max_output_tokens)?;
65        validate_optional_positive_limit(
66            "max_response_tokens",
67            self.max_response_tokens,
68            limits.max_response_tokens,
69        )
70    }
71
72    pub(in crate::mcp) fn into_parts(
73        self,
74    ) -> (
75        OutlineRequest,
76        OutlineMcpProjection,
77        IndexConsistency,
78        ServiceCallOptions,
79        Option<String>,
80    ) {
81        (
82            OutlineRequest {
83                paths: self.paths,
84                symbol_name: self.symbol_name,
85                symbol_kind: self.symbol_kind,
86                max_results: self.max_results,
87                max_tokens: self.max_tokens,
88                receipt_id: self.receipt_id,
89                cursor: self.cursor,
90            },
91            self.projection,
92            self.consistency,
93            service_call_options(self.max_response_tokens),
94            self.expected_repository_id,
95        )
96    }
