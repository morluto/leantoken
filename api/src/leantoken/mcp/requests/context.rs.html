118impl ContextMcpRequest {
119    pub(in crate::mcp) fn validate_limits(&self, limits: McpLimitPolicy) -> crate::Result<()> {
120        validate_optional_positive_limit(
121            "token_budget",
122            self.token_budget,
123            limits.max_output_tokens,
124        )?;
125        validate_optional_positive_limit(
126            "max_response_tokens",
127            self.max_response_tokens,
128            limits.max_response_tokens,
129        )?;
130        validate_optional_positive_limit("max_fragments", self.max_fragments, limits.max_results)?;
131        validate_optional_positive_limit(
132            "minimum_fragments_per_focus_path",
133            self.minimum_fragments_per_focus_path,
134            limits.max_results,
135        )
136    }
137
138    pub(in crate::mcp) fn into_parts(
139        self,
140        default_token_budget: usize,
141    ) -> (
142        ContextRequest,
143        ContextWorkflow,
144        WorkflowEvidence,
145        IndexConsistency,
146        ServiceCallOptions,
147        Option<String>,
148        Option<HandoffManifestRequest>,
149    ) {
150        let options = service_call_options_with_receipt(self.max_response_tokens, !self.plan_only);
151        let options = self.response_profile.map_or(options, |profile| {
152            options.with_context_response_profile(profile)
153        });
154        (
155            ContextRequest {
156                task: self.task.into_string(),
157                token_budget: self.token_budget.unwrap_or(default_token_budget),
158                include_paths: self.include_paths,
159                must_include_paths: self.must_include_paths,
160                must_include_symbols: self.must_include_symbols,
161                required_evidence: self.required_evidence,
162                max_fragments: self.max_fragments,
163                plan_only: self.plan_only,
164                focus_paths: self.focus_paths,
165                strict_focus_paths: self.strict_focus_paths,
166                minimum_fragments_per_focus_path: self.minimum_fragments_per_focus_path,
167                focus_symbols: self.focus_symbols,
168                exclude_paths: self.exclude_paths,
169                known_hashes: self.known_hashes,
170                receipt_id: self.receipt_id,
171                prior_repository_generation: self.prior_repository_generation,
172                base_revision: self.base_revision.map(NonEmptyText::into_string),
173                changed_paths: self.changed_paths,
174                strict_changed_paths: self.strict_changed_paths,
175                explain_diagnostics: false,
176            },
177            self.workflow,
178            self.workflow_evidence,
179            self.consistency,
180            options,
181            self.expected_repository_id,
182            self.handoff,
183        )
184    }
185}
