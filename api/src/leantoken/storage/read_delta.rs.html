58    pub(crate) fn read_delta_base_row_at(
59        &self,
60        target_key: &str,
61        content_hash: Option<&str>,
62        now_unix_millis: i64,
63    ) -> Result<Option<(String, ReadDeltaBase)>> {
64        // Fast path: use a reader-pool connection without holding the writer
65        // lock. Only fall back to the writer lock when a mutation is needed.
66        let mut read_conn = self.readers.get()?;
67        let read_tx = read_conn.transaction()?;
68        let row = load_read_delta_base(&read_tx, target_key, content_hash)?;
69        let Some((hash, base, created, last_access)) = row else {
70            return Ok(None);
71        };
72        let needs_stale_delete = created > now_unix_millis || last_access > now_unix_millis;
73        let needs_touch =
74            now_unix_millis.saturating_sub(last_access) < READ_DELTA_TOUCH_INTERVAL_MILLIS;
75        if needs_touch && !needs_stale_delete {
76            if crate::text::hash(&base.content) != hash {
77                return Err(Error::OperationFailure(
78                    "persistent read delta base hash mismatch".into(),
79                ));
80            }
81            return Ok(Some((hash, base)));
82        }
83        drop(read_tx);
84        drop(read_conn);
85
86        // Slow path: acquire the writer lock for mutations.
87        let mut connection = self
88            .writer
89            .lock()
90            .unwrap_or_else(|poisoned| poisoned.into_inner());
91        let transaction = connection.transaction_with_behavior(TransactionBehavior::Immediate)?;
92        prune_read_delta_bases(&transaction, now_unix_millis)?;
93        let row = load_read_delta_base(&transaction, target_key, content_hash)?;
94        let Some((hash, base, created, last_access)) = row else {
95            transaction.commit()?;
96            return Ok(None);
97        };
98        if created > now_unix_millis || last_access > now_unix_millis {
99            transaction.execute(
100                "DELETE FROM read_delta_bases
101                 WHERE target_key = ?1 AND content_hash = ?2",
102                params![target_key, hash],
103            )?;
104            transaction.commit()?;
105            return Ok(None);
106        }
107        if crate::text::hash(&base.content) != hash {
108            return Err(Error::OperationFailure(
109                "persistent read delta base hash mismatch".into(),
110            ));
111        }
112        if now_unix_millis.saturating_sub(last_access) >= READ_DELTA_TOUCH_INTERVAL_MILLIS {
113            let sequence = next_read_delta_access_sequence(&transaction)?;
114            transaction.execute(
115                "UPDATE read_delta_bases
116                 SET last_access_unix_millis = ?1,
117                     expires_unix_millis = ?2,
118                     access_sequence = ?3
119                 WHERE target_key = ?4 AND content_hash = ?5",
120                params![
121                    now_unix_millis,
122                    now_unix_millis.saturating_add(READ_DELTA_BASE_TTL_MILLIS),
123                    sequence,
124                    target_key,
125                    hash
126                ],
127            )?;
128        }
129        transaction.commit()?;
130        Ok(Some((hash, base)))
131    }

133    pub(crate) fn persist_read_delta_base_at(
134        &self,
135        target_key: &str,
136        content_hash: &str,
137        base: &ReadDeltaBase,
138        now_unix_millis: i64,
139    ) -> Result<bool> {
140        if now_unix_millis < 0 {
141            return Err(Error::OperationFailure(
142                "read delta base timestamp must be non-negative".into(),
143            ));
144        }
145        if base.content.len() > MAX_READ_DELTA_BASE_BYTES {
146            return Ok(false);
147        }
148        if crate::text::hash(&base.content) != content_hash {
149            return Err(Error::OperationFailure(
150                "read delta base content hash mismatch".into(),
151            ));
152        }
153        let logical_bytes = base.logical_bytes(target_key, content_hash);
154        if logical_bytes > MAX_TOTAL_READ_DELTA_BASE_BYTES {
155            return Ok(false);
156        }
157        let expires = now_unix_millis
158            .checked_add(READ_DELTA_BASE_TTL_MILLIS)
159            .ok_or_else(|| Error::OperationFailure("read delta base expiry overflow".into()))?;
160        let mut connection = self
161            .writer
162            .lock()
163            .unwrap_or_else(|poisoned| poisoned.into_inner());
164        let transaction = connection.transaction_with_behavior(TransactionBehavior::Immediate)?;
165        prune_read_delta_bases(&transaction, now_unix_millis)?;
166        let existing = transaction
167            .query_row(
168                "SELECT repository_generation,
169                        target_start_line, target_end_line,
170                        returned_start_line, returned_end_line,
171                        content, created_unix_millis, last_access_unix_millis
172                 FROM read_delta_bases
173                 WHERE target_key = ?1 AND content_hash = ?2",
174                params![target_key, content_hash],
175                |row| {
176                    Ok((
177                        i64_to_u64(row.get(0)?)?,
178                        i64_to_usize(row.get(1)?)?,
179                        i64_to_usize(row.get(2)?)?,
180                        i64_to_usize(row.get(3)?)?,
181                        i64_to_usize(row.get(4)?)?,
182                        row.get::<_, String>(5)?,
183                        row.get::<_, i64>(6)?,
184                        row.get::<_, i64>(7)?,
185                    ))
186                },
187            )
188            .optional()?;
189        if let Some((
190            generation,
191            target_start_line,
192            target_end_line,
193            returned_start_line,
194            returned_end_line,
195            stored_content,
196            created,
197            last_access,
198        )) = existing
199        {
200            if crate::text::hash(&stored_content) != content_hash {
201                return Err(Error::OperationFailure(
202                    "persistent read delta base hash mismatch".into(),
203                ));
204            }
205            if created > now_unix_millis || last_access > now_unix_millis {
206                transaction.execute(
207                    "DELETE FROM read_delta_bases
208                     WHERE target_key = ?1 AND content_hash = ?2",
209                    params![target_key, content_hash],
210                )?;
211            } else if generation == base.generation
212                && target_start_line == base.target_start_line
213                && target_end_line == base.target_end_line
214                && returned_start_line == base.returned_start_line
215                && returned_end_line == base.returned_end_line
216                && now_unix_millis.saturating_sub(last_access) < READ_DELTA_TOUCH_INTERVAL_MILLIS
217            {
218                transaction.commit()?;
219                return Ok(true);
220            } else {
221                let sequence = next_read_delta_access_sequence(&transaction)?;
222                let updated = transaction.execute(
223                    "UPDATE read_delta_bases
224                     SET repository_generation = ?1,
225                         target_start_line = ?2,
226                         target_end_line = ?3,
227                         returned_start_line = ?4,
228                         returned_end_line = ?5,
229                         last_access_unix_millis = ?6,
230                         expires_unix_millis = ?7,
231                         access_sequence = ?8
232                     WHERE target_key = ?9 AND content_hash = ?10",
233                    params![
234                        u64_to_i64(base.generation)?,
235                        usize_to_i64(base.target_start_line)?,
236                        usize_to_i64(base.target_end_line)?,
237                        usize_to_i64(base.returned_start_line)?,
238                        usize_to_i64(base.returned_end_line)?,
239                        now_unix_millis,
240                        expires,
241                        sequence,
242                        target_key,
243                        content_hash
244                    ],
245                )?;
246                if updated != 1 {
247                    return Err(Error::OperationFailure(
248                        "persistent read delta base update lost its row".into(),
249                    ));
250                }
251                transaction.commit()?;
252                return Ok(true);
253            }
254        }
255        loop {
256            let (count, bytes) = read_delta_usage(&transaction)?;
257            if count < MAX_READ_DELTA_BASES
258                && bytes.saturating_add(logical_bytes) <= MAX_TOTAL_READ_DELTA_BASE_BYTES
259            {
260                break;
261            }
262            if !evict_oldest_read_delta_base(&transaction)? {
263                transaction.commit()?;
264                return Ok(false);
265            }
266        }
267        let sequence = next_read_delta_access_sequence(&transaction)?;
268        transaction.execute(
269            "INSERT INTO read_delta_bases(
270                target_key, content_hash, repository_generation,
271                target_start_line, target_end_line,
272                returned_start_line, returned_end_line, content,
273                created_unix_millis, last_access_unix_millis,
274                expires_unix_millis, access_sequence, logical_bytes
275             ) VALUES (
276                ?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8, ?9, ?9, ?10, ?11, ?12
277             )",
278            params![
279                target_key,
280                content_hash,
281                u64_to_i64(base.generation)?,
282                usize_to_i64(base.target_start_line)?,
283                usize_to_i64(base.target_end_line)?,
284                usize_to_i64(base.returned_start_line)?,
285                usize_to_i64(base.returned_end_line)?,
286                base.content,
287                now_unix_millis,
288                expires,
289                sequence,
290                usize_to_i64(logical_bytes)?
291            ],
292        )?;
293        transaction.commit()?;
294        Ok(true)
295    }

300pub(crate) fn load_read_delta_base(
301    transaction: &Transaction<'_>,
302    target_key: &str,
303    content_hash: Option<&str>,
304) -> Result<Option<StoredReadDeltaBase>> {
305    let map = |row: &Row<'_>| {
306        Ok((
307            row.get(0)?,
308            ReadDeltaBase {
309                generation: i64_to_u64(row.get(1)?)?,
310                target_start_line: i64_to_usize(row.get(2)?)?,
311                target_end_line: i64_to_usize(row.get(3)?)?,
312                returned_start_line: i64_to_usize(row.get(4)?)?,
313                returned_end_line: i64_to_usize(row.get(5)?)?,
314                content: row.get(6)?,
315            },
316            row.get(7)?,
317            row.get(8)?,
318        ))
319    };
320    match content_hash {
321        Some(content_hash) => transaction
322            .query_row(
323                "SELECT content_hash, repository_generation,
324                        target_start_line, target_end_line,
325                        returned_start_line, returned_end_line, content,
326                        created_unix_millis, last_access_unix_millis
327                 FROM read_delta_bases
328                 WHERE target_key = ?1 AND content_hash = ?2",
329                params![target_key, content_hash],
330                map,
331            )
332            .optional()
333            .map_err(Into::into),
334        None => transaction
335            .query_row(
336                "SELECT content_hash, repository_generation,
337                        target_start_line, target_end_line,
338                        returned_start_line, returned_end_line, content,
339                        created_unix_millis, last_access_unix_millis
340                 FROM read_delta_bases
341                 WHERE target_key = ?1
342                 ORDER BY repository_generation DESC, access_sequence DESC, content_hash
343                 LIMIT 1",
344                [target_key],
345                map,
346            )
347            .optional()
348            .map_err(Into::into),
349    }
350}

352pub(crate) fn read_delta_usage(transaction: &Transaction<'_>) -> Result<(usize, usize)> {
353    transaction
354        .query_row(
355            "SELECT base_count, base_bytes FROM read_delta_base_usage WHERE id = 1",
356            [],
357            |row| Ok((i64_to_usize(row.get(0)?)?, i64_to_usize(row.get(1)?)?)),
358        )
359        .map_err(Into::into)
360}

362pub(crate) fn next_read_delta_access_sequence(transaction: &Transaction<'_>) -> Result<i64> {
363    let current: i64 = transaction.query_row(
364        "SELECT next_access_sequence FROM read_delta_base_usage WHERE id = 1",
365        [],
366        |row| row.get(0),
367    )?;
368    let next = current
369        .checked_add(1)
370        .ok_or_else(|| Error::OperationFailure("read delta access sequence overflow".into()))?;
371    let updated = transaction.execute(
372        "UPDATE read_delta_base_usage SET next_access_sequence = ?1 WHERE id = 1",
373        [next],
374    )?;
375    if updated != 1 {
376        return Err(Error::OperationFailure(
377            "persistent read delta usage row is missing".into(),
378        ));
379    }
380    Ok(next)
381}

383pub(crate) fn prune_read_delta_bases(
384    transaction: &Transaction<'_>,
385    now_unix_millis: i64,
386) -> Result<()> {
387    transaction.execute(
388        "DELETE FROM read_delta_bases WHERE expires_unix_millis <= ?1",
389        [now_unix_millis],
390    )?;
391    Ok(())
392}

394pub(crate) fn evict_oldest_read_delta_base(transaction: &Transaction<'_>) -> Result<bool> {
395    let candidate = transaction
396        .query_row(
397            "SELECT target_key, content_hash
398             FROM read_delta_bases
399             ORDER BY access_sequence, target_key, content_hash
400             LIMIT 1",
401            [],
402            |row| Ok((row.get::<_, String>(0)?, row.get::<_, String>(1)?)),
403        )
404        .optional()?;
405    let Some((target_key, content_hash)) = candidate else {
406        return Ok(false);
407    };
408    Ok(transaction.execute(
409        "DELETE FROM read_delta_bases
410         WHERE target_key = ?1 AND content_hash = ?2",
411        params![target_key, content_hash],
412    )? == 1)
413}
