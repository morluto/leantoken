41    pub(crate) fn whole_file_source_tokens(
42        &self,
43        paths: &[String],
44        tokenizer: &str,
45    ) -> Result<Option<usize>> {
46        if paths.is_empty() {
47            return Ok(Some(0));
48        }
49        let input = serde_json::to_string(paths)?;
50        let tokens: Option<i64> = self.conn.query_row(
51            "WITH requested(path) AS (
52                 SELECT DISTINCT CAST(value AS TEXT) FROM json_each(?1)
53             )
54             SELECT CASE
55                 WHEN COUNT(*) = SUM(files.source_tokenizer = ?2)
56                     THEN COALESCE(SUM(files.source_token_count), 0)
57                 ELSE NULL
58             END
59             FROM requested
60             JOIN files ON files.path = requested.path",
61            params![input, tokenizer],
62            |row| row.get(0),
63        )?;
64        Ok(tokens.map(i64_to_usize).transpose()?)
65    }
