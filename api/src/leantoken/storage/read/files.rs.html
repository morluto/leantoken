2    /// Return a row-id keyset page from this session's pinned snapshot.
3    ///
4    /// Use the final record's `id` as the next cursor. Cursors are storage-layer
5    /// values and should not be exposed as service cursors without binding them
6    /// to the repository generation and operation parameters.
7    pub fn list_files(&self, max_results: usize, cursor: Option<i64>) -> Result<Vec<FileRecord>> {
8        let limit = bounded_limit(max_results);
9        let mut stmt = self.conn.prepare_cached(
10            "SELECT id, path, language, size_bytes, modified_ns, content_hash, generation, structurally_complete FROM files WHERE (?1 IS NULL OR id > ?1) ORDER BY id LIMIT ?2",
11        )?;
12        let rows = stmt.query_map(params![cursor, limit], Storage::map_file)?;
13        let mut files = Vec::new();
14        for row in rows {
15            files.push(row?);
16        }
17        Ok(files)
18    }

20    /// Return a lean path projection page for fuzzy find scans.
21    ///
22    /// Use the final record's `id` as the next cursor. Omits hash, generation,
23    /// and modified metadata that find does not need.
24    pub(crate) fn list_file_paths(
25        &self,
26        max_results: usize,
27        cursor: Option<i64>,
28    ) -> Result<Vec<FilePathRecord>> {
29        let limit = bounded_limit(max_results);
30        let mut stmt = self.conn.prepare_cached(
31            "SELECT id, path, language, size_bytes FROM files
32             WHERE (?1 IS NULL OR id > ?1)
33             ORDER BY id
34             LIMIT ?2",
35        )?;
36        let rows = stmt.query_map(params![cursor, limit], |row| {
37            Ok(FilePathRecord {
38                id: row.get(0)?,
39                path: row.get(1)?,
40                language: row.get(2)?,
41                size_bytes: i64_to_u64(row.get(3)?)?,
42            })
43        })?;
44        Ok(rows.collect::<std::result::Result<Vec<_>, _>>()?)
45    }

47    /// SQL-paged file paths matching one or two SQLite `GLOB` patterns.
48    ///
49    /// Uses the `path_entries` file projection with a path keyset cursor. Callers
50    /// map globset patterns to SQL GLOB forms; brace expansion and other
51    /// unexpressible patterns keep the in-process globset fallback.
52    pub(crate) fn list_glob_paths(
53        &self,
54        primary: &str,
55        alternate: Option<&str>,
56        after: Option<&str>,
57        max_results: usize,
58    ) -> Result<Vec<PathRecord>> {
59        let mut stmt = self.conn.prepare_cached(
60            "SELECT path_entries.path, files.language, files.size_bytes
61             FROM path_entries
62             JOIN files ON files.id = path_entries.file_id
63             WHERE path_entries.kind = 1
64               AND (path_entries.path GLOB ?1
65                    OR (?2 IS NOT NULL AND path_entries.path GLOB ?2))
66               AND (?3 IS NULL OR path_entries.path > ?3)
67             ORDER BY path_entries.path
68             LIMIT ?4",
69        )?;
70        let rows = stmt.query_map(
71            params![primary, alternate, after, bounded_limit(max_results)],
72            |row| {
73                Ok(PathRecord {
74                    path: row.get(0)?,
75                    is_directory: false,
76                    language: row.get(1)?,
77                    size_bytes: row.get::<_, Option<i64>>(2)?.map(i64_to_u64).transpose()?,
78                })
79            },
80        )?;
81        Ok(rows.collect::<std::result::Result<Vec<_>, _>>()?)
82    }

84    pub(crate) fn regex_scan_files(
85        &self,
86        max_results: usize,
87        max_rows_scanned: usize,
88        include_paths: &[String],
89        exclude_paths: &[String],
90        mut allows_path: impl FnMut(&str) -> Result<bool>,
91    ) -> Result<Vec<(FileRecord, usize)>> {
92        let path_sql = scoped_regex_path_sql(include_paths, exclude_paths, 2);
93        let sql = format!(
94            "SELECT f.id, f.path, f.language, f.size_bytes, f.modified_ns,
95                    f.content_hash, f.generation, f.structurally_complete,
96                    COUNT(c.id)
97             FROM files f
98             LEFT JOIN chunks c ON c.file_id = f.id
99             WHERE 1 = 1 {path_clause}
100             GROUP BY f.id
101             ORDER BY f.id
102             LIMIT ?1",
103            path_clause = path_sql.clause,
104        );
105        let mut stmt = self.conn.prepare(&sql)?;
106        let mut bind = vec![rusqlite::types::Value::from(
107            i64::try_from(max_rows_scanned.saturating_add(1)).unwrap_or(i64::MAX),
108        )];
109        bind.extend(path_sql.params.into_iter().map(Into::into));
110        let mut rows = stmt.query(rusqlite::params_from_iter(bind))?;
111        let mut files = Vec::new();
112        let mut scanned = 0usize;
113        while let Some(row) = rows.next()? {
114            scanned += 1;
115            if scanned > max_rows_scanned {
116                return Err(Error::RetrievalLimitExceeded {
117                    kind: RetrievalLimitKind::RegexScopedRows,
118                    observed: scanned,
119                    limit: max_rows_scanned,
120                });
121            }
122            let path: String = row.get(1)?;
123            if !allows_path(&path)? {
124                continue;
125            }
126            if files.len() == max_results {
127                return Err(Error::RetrievalLimitExceeded {
128                    kind: RetrievalLimitKind::RegexFullScanFiles,
129                    observed: files.len().saturating_add(1),
130                    limit: max_results,
131                });
132            }
133            files.push((Storage::map_file(row)?, i64_to_usize(row.get(8)?)?));
134        }
135        Ok(files)
136    }

138    /// Read a lexicographically ordered keyset page from the relational path projection.
139    pub(crate) fn list_tree_paths(
140        &self,
141        root: &str,
142        max_depth: usize,
143        after: Option<&str>,
144        max_results: usize,
145    ) -> Result<Vec<PathRecord>> {
146        let root_depth = root.split('/').filter(|part| !part.is_empty()).count();
147        let depth_limit = i64::try_from(root_depth.saturating_add(max_depth)).unwrap_or(i64::MAX);
148        let mut stmt = self.conn.prepare_cached(
149            "SELECT path_entries.path, path_entries.kind, files.language, files.size_bytes
150             FROM path_entries
151             LEFT JOIN files ON files.id = path_entries.file_id
152             WHERE (?1 = '' OR path_entries.path = ?1
153                    OR substr(path_entries.path, 1, length(?1) + 1) = ?1 || '/')
154               AND path_entries.depth <= ?2
155               AND (?3 IS NULL OR path_entries.path > ?3)
156             ORDER BY path_entries.path
157             LIMIT ?4",
158        )?;
159        let rows = stmt.query_map(
160            params![root, depth_limit, after, bounded_limit(max_results)],
161            |row| {
162                let kind: i64 = row.get(1)?;
163                Ok(PathRecord {
164                    path: row.get(0)?,
165                    is_directory: kind == 0,
166                    language: row.get(2)?,
167                    size_bytes: row.get::<_, Option<i64>>(3)?.map(i64_to_u64).transpose()?,
168                })
169            },
170        )?;
171        Ok(rows.collect::<std::result::Result<Vec<_>, _>>()?)
172    }
