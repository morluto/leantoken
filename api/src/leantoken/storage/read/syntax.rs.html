2    pub(crate) fn receipt_structural_hash_matches(
3        &self,
4        file_id: i64,
5        start_line: usize,
6        end_line: usize,
7        content_hash: &str,
8    ) -> Result<Option<bool>> {
9        let limit = crate::receipt::MAX_REBASE_STRUCTURAL_CANDIDATES_PER_EVIDENCE;
10        let mut statement = self.conn.prepare_cached(
11            "SELECT content
12             FROM (
13                 SELECT COALESCE(signature, name) AS content, rowid AS stable_id, 0 AS kind
14                 FROM symbols
15                 WHERE file_id = ?1 AND start_line = ?2 AND end_line = ?3
16                 UNION ALL
17                 SELECT raw_target AS content, id AS stable_id, 1 AS kind
18                 FROM imports
19                 WHERE file_id = ?1 AND line = ?2 AND line = ?3
20             )
21             ORDER BY kind, stable_id
22             LIMIT ?4",
23        )?;
24        let mut rows = statement.query(params![
25            file_id,
26            usize_to_i64(start_line)?,
27            usize_to_i64(end_line)?,
28            usize_to_i64(limit.saturating_add(1))?,
29        ])?;
30        let mut count = 0usize;
31        while let Some(row) = rows.next()? {
32            count = count.saturating_add(1);
33            let content: String = row.get(0)?;
34            if crate::text::hash(&content) == content_hash {
35                return Ok(Some(true));
36            }
37        }
38        if count > limit {
39            return Ok(None);
40        }
41        Ok(Some(false))
42    }

57    /// Return the final indexed line for each file id, preserving request order.
58    pub(crate) fn file_end_lines_batch(&self, file_ids: &[i64]) -> Result<Vec<Option<usize>>> {
59        if file_ids.is_empty() {
60            return Ok(Vec::new());
61        }
62        let mut seen = HashSet::new();
63        let unique_file_ids = file_ids
64            .iter()
65            .copied()
66            .filter(|file_id| seen.insert(*file_id))
67            .collect::<Vec<_>>();
68        let input = serde_json::to_string(&unique_file_ids)?;
69        let mut stmt = self.conn.prepare_cached(
70            "WITH requested AS (
71                 SELECT CAST(value AS INTEGER) AS file_id
72                 FROM json_each(?1)
73             )
74             SELECT requested.file_id, MAX(chunks.end_line)
75             FROM requested
76             LEFT JOIN chunks ON chunks.file_id = requested.file_id
77             GROUP BY requested.file_id",
78        )?;
79        let rows = stmt.query_map(params![input], |row| {
80            Ok((
81                row.get::<_, i64>(0)?,
82                row.get::<_, Option<i64>>(1)?
83                    .map(i64_to_usize)
84                    .transpose()?,
85            ))
86        })?;
87        let end_lines = rows.collect::<std::result::Result<HashMap<_, _>, _>>()?;
88        Ok(file_ids
89            .iter()
90            .map(|file_id| end_lines.get(file_id).copied().flatten())
91            .collect())
92    }

94    /// Hydrate overlapping chunks for every requested range in one SQL query.
95    ///
96    /// The outer vector is aligned one-for-one with `ranges`, including duplicate
97    /// requests and ranges with no matches. Each inner vector is ordered by line.
98    pub(crate) fn get_chunks_overlapping_batch(
99        &self,
100        ranges: &[(i64, usize, usize)],
101    ) -> Result<Vec<Vec<ChunkRecord>>> {
102        if ranges.is_empty() {
103            return Ok(Vec::new());
104        }
105        let input = ranges
106            .iter()
107            .map(|(file_id, start_line, end_line)| {
108                serde_json::json!({
109                    "file_id": file_id,
110                    "start_line": start_line,
111                    "end_line": end_line,
112                })
113            })
114            .collect::<Vec<_>>();
115        let input = serde_json::to_string(&input)?;
116        let mut stmt = self.conn.prepare_cached(
117            "WITH requested AS (
118                 SELECT CAST(key AS INTEGER) AS request_index,
119                        CAST(value ->> 'file_id' AS INTEGER) AS file_id,
120                        CAST(value ->> 'start_line' AS INTEGER) AS start_line,
121                        CAST(value ->> 'end_line' AS INTEGER) AS end_line
122                 FROM json_each(?1)
123             )
124             SELECT requested.request_index,
125                    chunks.id, chunks.file_id, chunks.content,
126                    chunks.start_line, chunks.end_line,
127                    chunks.start_byte, chunks.end_byte, chunks.token_count
128             FROM requested
129             JOIN chunks
130               ON chunks.file_id = requested.file_id
131              AND chunks.end_line >= requested.start_line
132              AND chunks.start_line <= requested.end_line
133             ORDER BY requested.request_index, chunks.start_line",
134        )?;
135        let rows = stmt.query_map(params![input], |row| {
136            let request_index = i64_to_usize(row.get(0)?)?;
137            let chunk = ChunkRecord {
138                id: row.get(1)?,
139                file_id: row.get(2)?,
140                content: row.get(3)?,
141                start_line: i64_to_usize(row.get(4)?)?,
142                end_line: i64_to_usize(row.get(5)?)?,
143                start_byte: i64_to_usize(row.get(6)?)?,
144                end_byte: i64_to_usize(row.get(7)?)?,
145                token_count: i64_to_usize(row.get(8)?)?,
146            };
147            Ok((request_index, chunk))
148        })?;
149        let mut grouped = vec![Vec::new(); ranges.len()];
150        for row in rows {
151            let (request_index, chunk) = row?;
152            if let Some(chunks) = grouped.get_mut(request_index) {
153                chunks.push(chunk);
154            }
155        }
156        Ok(grouped)
157    }

172    pub(crate) fn get_symbols_for_file_filtered_page(
173        &self,
174        file_id: i64,
175        name: Option<&str>,
176        kind: Option<&str>,
177        max_results: usize,
178        offset: usize,
179    ) -> Result<Vec<SymbolRecord>> {
180        let limit = bounded_limit(max_results);
181        let offset = i64::try_from(offset).unwrap_or(i64::MAX);
182        let qualified =
183            name.is_some_and(|name| crate::symbol_identity::split_qualified_symbol(name).is_some());
184        let mut stmt = self.conn.prepare_cached(
185            "SELECT id, file_id, name, kind, parent, signature, start_line, end_line, start_byte, end_byte
186                 FROM symbols
187                 WHERE file_id = ?1
188                   AND (
189                       ?2 IS NULL
190                       OR instr(name, ?2) > 0
191                       OR (
192                           ?4
193                           AND parent IS NOT NULL
194                           AND instr(parent || '.' || name, ?2) > 0
195                       )
196                   )
197                   AND (?3 IS NULL OR kind = ?3)
198                 ORDER BY start_byte, id
199                 LIMIT ?5 OFFSET ?6",
200        )?;
201        let rows = stmt.query_map(
202            params![file_id, name, kind, qualified, limit, offset],
203            Storage::map_symbol,
204        )?;
205        Ok(rows.collect::<std::result::Result<Vec<_>, _>>()?)
206    }

239    pub(crate) fn find_symbol(
240        &self,
241        file_id: i64,
242        name: &str,
243    ) -> Result<crate::symbol_identity::SymbolResolution<SymbolRecord>> {
244        let qualified = crate::symbol_identity::split_qualified_symbol(name);
245        let sql = if qualified.is_some() {
246            "WITH matches AS (
247                 SELECT id, file_id, name, kind, parent, signature,
248                        start_line, end_line, start_byte, end_byte
249                 FROM symbols INDEXED BY symbols_name_idx
250                 WHERE file_id = ?1
251                   AND name = ?2 COLLATE NOCASE
252                   AND name = ?2 COLLATE BINARY
253                 UNION ALL
254                 SELECT id, file_id, name, kind, parent, signature,
255                        start_line, end_line, start_byte, end_byte
256                 FROM symbols INDEXED BY symbols_name_idx
257                 WHERE file_id = ?1
258                   AND name = ?4 COLLATE NOCASE
259                   AND name = ?4 COLLATE BINARY
260                   AND parent = ?3
261             )
262             SELECT id, file_id, name, kind, parent, signature,
263                    start_line, end_line, start_byte, end_byte
264             FROM matches
265             ORDER BY start_byte, id
266             LIMIT 2"
267        } else {
268            "SELECT id, file_id, name, kind, parent, signature,
269                    start_line, end_line, start_byte, end_byte
270             FROM symbols INDEXED BY symbols_name_idx
271             WHERE file_id = ?1
272               AND name = ?2 COLLATE NOCASE
273               AND name = ?2 COLLATE BINARY
274               AND ?3 IS NULL
275               AND ?4 IS NULL
276             ORDER BY start_byte, id
277             LIMIT 2"
278        };
279        let (parent, leaf_name) = qualified.unzip();
280        let mut stmt = self.conn.prepare_cached(sql)?;
281        let rows = stmt.query_map(
282            params![file_id, name, parent, leaf_name],
283            Storage::map_symbol,
284        )?;
285        let matches = rows.collect::<std::result::Result<Vec<_>, _>>()?;
286        Ok(crate::symbol_identity::resolve_symbol_matches(
287            matches.into_iter(),
288        ))
289    }

320    /// Find the narrowest enclosing symbol for every requested file/line pair.
321    ///
322    /// Results preserve input order and cardinality; `None` marks a location with
323    /// no enclosing declaration. Duplicate locations share one storage lookup.
324    pub(crate) fn find_enclosing_symbols_batch(
325        &self,
326        locations: &[(i64, usize)],
327    ) -> Result<Vec<Option<SymbolRecord>>> {
328        if locations.is_empty() {
329            return Ok(Vec::new());
330        }
331        let mut unique_indices = HashMap::new();
332        let mut unique_locations = Vec::new();
333        let location_mapping = locations
334            .iter()
335            .map(|location| {
336                *unique_indices.entry(*location).or_insert_with(|| {
337                    let unique_index = unique_locations.len();
338                    unique_locations.push(*location);
339                    unique_index
340                })
341            })
342            .collect::<Vec<_>>();
343        let input = unique_locations
344            .iter()
345            .map(|(file_id, line)| serde_json::json!({ "file_id": file_id, "line": line }))
346            .collect::<Vec<_>>();
347        let input = serde_json::to_string(&input)?;
348        let mut stmt = self.conn.prepare_cached(
349            "WITH requested AS (
350                 SELECT CAST(key AS INTEGER) AS request_index,
351                        CAST(value ->> 'file_id' AS INTEGER) AS file_id,
352                        CAST(value ->> 'line' AS INTEGER) AS line
353                 FROM json_each(?1)
354             )
355             SELECT requested.request_index,
356                    symbols.id, symbols.file_id, symbols.name, symbols.kind,
357                    symbols.parent, symbols.signature,
358                    symbols.start_line, symbols.end_line,
359                    symbols.start_byte, symbols.end_byte
360             FROM requested
361             JOIN symbols ON symbols.id = (
362                 SELECT enclosing.id
363                 FROM symbols AS enclosing
364                 WHERE enclosing.file_id = requested.file_id
365                   AND enclosing.start_line <= requested.line
366                   AND enclosing.end_line >= requested.line
367                 ORDER BY (enclosing.end_line - enclosing.start_line), enclosing.start_byte
368                 LIMIT 1
369             )
370             ORDER BY requested.request_index",
371        )?;
372        let rows = stmt.query_map(params![input], |row| {
373            let request_index = i64_to_usize(row.get(0)?)?;
374            let symbol = SymbolRecord {
375                id: row.get(1)?,
376                file_id: row.get(2)?,
377                name: row.get(3)?,
378                kind: row.get(4)?,
379                parent: row.get(5)?,
380                signature: row.get(6)?,
381                start_line: i64_to_usize(row.get(7)?)?,
382                end_line: i64_to_usize(row.get(8)?)?,
383                start_byte: i64_to_usize(row.get(9)?)?,
384                end_byte: i64_to_usize(row.get(10)?)?,
385            };
386            Ok((request_index, symbol))
387        })?;
388        let mut symbols = vec![None; unique_locations.len()];
389        for row in rows {
390            let (request_index, symbol) = row?;
391            if let Some(slot) = symbols.get_mut(request_index) {
392                *slot = Some(symbol);
393            }
394        }
395        Ok(location_mapping
396            .into_iter()
397            .map(|unique_index| symbols[unique_index].clone())
398            .collect())
399    }
