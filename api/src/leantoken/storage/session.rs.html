1use std::time::{Duration, Instant};
2
3impl Storage {
4    /// Open a read-only connection and begin a DEFERRED transaction so callers
5    /// observe one WAL snapshot until the session is dropped.
6    ///
7    /// Keep one session for every multi-query response. Dropping it rolls back
8    /// the read transaction and returns the connection to the bounded pool.
9    pub(super) fn begin_read(&self) -> Result<ReadSession> {
10        let checkout_started = Instant::now();
11        let conn = self.readers.get()?;
12        let checkout_wait = checkout_started.elapsed();
13        if checkout_wait >= Duration::from_millis(10) {
14            tracing::debug!(
15                wait_ms = checkout_wait.as_secs_f64() * 1_000.0,
16                "storage reader pool checkout waited"
17            );
18        }
19        // Under WAL, the first read in a DEFERRED transaction pins the snapshot
20        // for the rest of the connection's transaction lifetime.
21        conn.execute_batch("BEGIN DEFERRED")?;
22        Ok(ReadSession { conn })
23    }
24}
