13    ///
14    /// Indexing code that performs filesystem work before publication should use
15    /// [`Self::full_reconcile_at`] with the baseline captured before that work.

17        let baseline = self.meta()?;
18        self.full_reconcile_at(&baseline, config_hash, files)

21    /// Replace the complete index only if the generation and configuration used
22    /// to build the reconciliation plan are still current.
23    ///
24    /// On success, all derived rows and the next generation become visible in one
25    /// commit. A stale baseline returns [`Error::StaleReconciliation`] before any
26    /// mutation is published.
27    pub fn full_reconcile_at(
28        &self,
29        baseline: &MetaRecord,
30        config_hash: &str,
31        files: Vec<IndexedFile>,
32    ) -> Result<u64> {
33        self.publish_reconciliation_at(
34            baseline,
35            config_hash,
36            IndexingMode::Rebuild,
37            move |writer| {
38                for file in files {
39                    writer.replace(file)?;
40                }
41                Ok(())
42            },
43        )
44        .map(|(generation, ())| generation)
45    }

47    /// Atomically apply one repository reconciliation using an internally captured baseline.
48    ///
49    /// Unmentioned files remain unchanged; replacements, deletions, derived path
50    /// rows, import edges, and generation advancement become visible together.
51    /// Indexing code should prefer [`Self::reconcile_files_at`] when planning and
52    /// publication are separated by filesystem or parsing work.
53    pub fn reconcile_files(
54        &self,
55        config_hash: &str,
56        replacements: Vec<IndexedFile>,
57        deletions: &[String],
58    ) -> Result<u64> {
59        let baseline = self.meta()?;
60        self.reconcile_files_at(&baseline, config_hash, replacements, deletions)
61    }

63    /// Publish an incremental plan only if its source generation and config
64    /// still match the committed cache state.
65    ///
66    /// Replacements are whole-file units and deletions are repository-relative
67    /// paths. A no-op preserves the current generation when the configuration is
68    /// unchanged. Stale plans fail before publishing any mutation.
69    pub fn reconcile_files_at(
70        &self,
71        baseline: &MetaRecord,
72        config_hash: &str,
73        replacements: Vec<IndexedFile>,
74        deletions: &[String],
75    ) -> Result<u64> {
76        self.publish_reconciliation_at(
77            baseline,
78            config_hash,
79            IndexingMode::Reconcile,
80            move |writer| {
81                for path in deletions {
82                    writer.delete(path)?;
83                }
84                for file in replacements {
85                    writer.replace(file)?;
86                }
87                Ok(())
88            },
89        )
90        .map(|(generation, ())| generation)
91    }

170    pub(crate) fn publish_reconciliation_inner<T>(
171        &self,
172        baseline: &MetaRecord,
173        config_hash: &str,
174        mode: IndexingMode,
175        profiling: StorageProfiling,
176        mut observe: impl FnMut(ReconciliationPublicationPhase) -> Result<()>,
177        write: impl FnOnce(&mut ReconciliationWriter<'_, '_>) -> Result<T>,
178    ) -> Result<(u64, T, PublicationDiagnostics)> {
179        let mut writer = self
180            .writer
181            .lock()
182            .unwrap_or_else(|poisoned| poisoned.into_inner());
183        // Profiling uses a disposable writer connection so its temporary
184        // auto-checkpoint policy can never leak into ordinary publications.
185        // Holding the normal writer lock still serializes in-process mutation.
186        let mut profiled_connection = if profiling.is_collecting() {
187            let mut connection = Connection::open(&self.path)?;
188            Self::configure(&mut connection, DEFAULT_BUSY_TIMEOUT)?;
189            connection.busy_timeout(DEFAULT_BUSY_TIMEOUT)?;
190            connection.pragma_update(None, "wal_autocheckpoint", 0)?;
191            Some(connection)
192        } else {
193            None
194        };
195        let conn = match profiled_connection.as_mut() {
196            Some(connection) => connection,
197            None => &mut *writer,
198        };
199        (|| {
200            let mut diagnostics = PublicationDiagnostics::default();
201            let tx = conn.transaction_with_behavior(TransactionBehavior::Immediate)?;
202            let (current_generation, current_config, current_derivation): (i64, String, String) =
203                tx.query_row(
204                    "SELECT repository_generation, config_hash, derivation_fingerprint
205                 FROM meta WHERE id = 1",
206                    [],
207                    |row| Ok((row.get(0)?, row.get(1)?, row.get(2)?)),
208                )?;
209            verify_baseline(
210                baseline,
211                current_generation,
212                &current_config,
213                &current_derivation,
214            )?;
215            let runtime_derivation = crate::index_derivation::index_derivation_fingerprint();
216
217            // Initial and replacement publications can build the external-content
218            // FTS indexes once instead of maintaining them for every chunk mutation.
219            let bulk_fts = mode.is_rebuild() || current_generation == 0;
220            let mut trigger_guard = bulk_fts
221                .then(|| DatabaseTriggerGuard::disable(&tx))
222                .transpose()?;
223
224            if mode.is_rebuild() {
225                tx.execute("DELETE FROM files", [])?;
226                tx.execute("DELETE FROM path_entries", [])?;
227            }
228
229            let next_generation = current_generation
230                .checked_add(1)
231                .ok_or_else(|| Error::OperationFailure("repository generation exhausted".into()))?;
232            let mut writer = ReconciliationWriter {
233                transaction: &tx,
234                generation: next_generation,
235                mode,
236                replacements: 0,
237                deletions: HashSet::new(),
238                projection_refreshes: 0,
239            };
240            let (output, relational_write_ms, relational_write_bytes) =
241                measured_storage_phase(profiling, || write(&mut writer))?;
242            diagnostics.relational_write_ms = relational_write_ms;
243            diagnostics.relational_write_bytes = relational_write_bytes;
244            let changed = mode.is_rebuild()
245                || writer.replacements > 0
246                || !writer.deletions.is_empty()
247                || writer.projection_refreshes > 0
248                || current_config != config_hash
249                || current_derivation != runtime_derivation;
250            if !mode.is_rebuild() && !writer.deletions.is_empty() {
251                Self::remove_orphan_path_entries(&tx)?;
252            }
253            drop(writer);
254
255            if changed && bulk_fts {
256                observe(ReconciliationPublicationPhase::ChunkWordFts)?;
257                let (_, elapsed_ms, write_bytes) = measured_storage_phase(profiling, || {
258                    tx.execute(
259                        "INSERT INTO chunks_fts_word(chunks_fts_word) VALUES('rebuild')",
260                        [],
261                    )?;
262                    Ok(())
263                })?;
264                diagnostics.chunk_word_fts_rebuild_ms = elapsed_ms;
265                diagnostics.chunk_word_fts_rebuild_write_bytes = write_bytes;
266
267                observe(ReconciliationPublicationPhase::ChunkTrigramFts)?;
268                let (_, elapsed_ms, write_bytes) = measured_storage_phase(profiling, || {
269                    tx.execute(
270                        "INSERT INTO chunks_fts_trigram(chunks_fts_trigram) VALUES('rebuild')",
271                        [],
272                    )?;
273                    Ok(())
274                })?;
275                diagnostics.chunk_trigram_fts_rebuild_ms = elapsed_ms;
276                diagnostics.chunk_trigram_fts_rebuild_write_bytes = write_bytes;
277
278                observe(ReconciliationPublicationPhase::SymbolFts)?;
279                let (_, elapsed_ms, write_bytes) = measured_storage_phase(profiling, || {
280                    tx.execute(
281                        "INSERT INTO symbols_fts_trigram(symbols_fts_trigram) VALUES('rebuild')",
282                        [],
283                    )?;
284                    Ok(())
285                })?;
286                diagnostics.symbol_fts_rebuild_ms = elapsed_ms;
287                diagnostics.symbol_fts_rebuild_write_bytes = write_bytes;
288
289                observe(ReconciliationPublicationPhase::ReferenceFts)?;
290                let (_, elapsed_ms, write_bytes) = measured_storage_phase(profiling, || {
291                    tx.execute(
292                        "INSERT INTO symbol_refs_fts_trigram(symbol_refs_fts_trigram) VALUES('rebuild')",
293                        [],
294                    )?;
295                    Ok(())
296                })?;
297                diagnostics.reference_fts_rebuild_ms = elapsed_ms;
298                diagnostics.reference_fts_rebuild_write_bytes = write_bytes;
299            }
300            let published_generation = if changed {
301                tx.execute(
302                    "UPDATE meta
303                     SET config_hash = ?1,
304                         derivation_fingerprint = ?2,
305                         repository_generation = ?3,
306                         index_version = index_version + 1
307                     WHERE id = 1",
308                    params![config_hash, runtime_derivation, next_generation],
309                )?;
310                next_generation
311            } else {
312                current_generation
313            };
314            if let Some(guard) = trigger_guard.take() {
315                guard.restore()?;
316            }
317            drop(trigger_guard);
318            observe(ReconciliationPublicationPhase::CommitAndCheckpoint)?;
319            // A read-only COMMIT can invoke SQLite auto-checkpointing on a
320            // pre-existing WAL backlog. Rollback still releases BEGIN
321            // IMMEDIATE after the baseline check, without rewriting pages for
322            // a generation that did not change.
323            let (_, elapsed_ms, write_bytes) = measured_storage_phase(profiling, || {
324                if changed {
325                    Ok(tx.commit()?)
326                } else {
327                    Ok(tx.rollback()?)
328                }
329            })?;
330            diagnostics.commit_ms = elapsed_ms;
331            diagnostics.commit_write_bytes = write_bytes;
332
333            if profiling.is_collecting() {
334                diagnostics.post_commit_diagnostics_complete =
335                    populate_post_commit_diagnostics(conn, &self.path, changed, &mut diagnostics)
336                        .is_ok();
337            }
338            Ok((i64_to_u64(published_generation)?, output, diagnostics))
339        })()
340    }

342    pub(crate) fn insert_file(
343        tx: &Transaction,
344        file: &IndexedFile,
345        generation: i64,
346        source_tokens: Option<(&str, usize)>,
347    ) -> Result<()> {
348        let (source_tokenizer, source_token_count) = source_tokens.unwrap_or(("", 0));
349        tx.prepare_cached(
350            "INSERT INTO files(path, language, structurally_complete, size_bytes, modified_ns, content_hash, generation, source_token_count, source_tokenizer) VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8, ?9)",
351        )?
352        .execute(params![
353                &file.path,
354                file.language.as_deref(),
355                file.structurally_complete,
356                u64_to_i64(file.size_bytes)?,
357                file.modified_ns.map(u128_to_i64).transpose()?,
358                &file.content_hash,
359                generation,
360                usize_to_i64(source_token_count)?,
361                source_tokenizer,
362            ])?;
363        let file_id = tx.last_insert_rowid();
364        Self::insert_path_projection(tx, &file.path, file_id)?;
365
366        let mut insert_chunk = tx.prepare_cached(
367            "INSERT INTO chunks(file_id, content, start_line, end_line, start_byte, end_byte, token_count) VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7)",
368        )?;
369        for chunk in &file.chunks {
370            insert_chunk.execute(params![
371                file_id,
372                &chunk.content,
373                usize_to_i64(chunk.start_line)?,
374                usize_to_i64(chunk.end_line)?,
375                usize_to_i64(chunk.start_byte)?,
376                usize_to_i64(chunk.end_byte)?,
377                usize_to_i64(chunk.token_count)?,
378            ])?;
379        }
380        drop(insert_chunk);
381
382        let mut insert_symbol = tx.prepare_cached(
383            "INSERT INTO symbols(file_id, name, kind, parent, signature, start_line, end_line, start_byte, end_byte) VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8, ?9)",
384        )?;
385        for symbol in &file.symbols {
386            insert_symbol.execute(params![
387                file_id,
388                &symbol.name,
389                &symbol.kind,
390                symbol.parent.as_deref(),
391                symbol.signature.as_deref(),
392                usize_to_i64(symbol.start_line)?,
393                usize_to_i64(symbol.end_line)?,
394                usize_to_i64(symbol.start_byte)?,
395                usize_to_i64(symbol.end_byte)?,
396            ])?;
397        }
398        drop(insert_symbol);
399
400        let mut insert_reference = tx.prepare_cached(
401            "INSERT INTO symbol_refs(file_id, name, kind, role, enclosing_symbol, start_line, end_line, start_byte, end_byte) VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8, ?9)",
402        )?;
403        for reference in &file.references {
404            insert_reference.execute(params![
405                file_id,
406                &reference.name,
407                &reference.kind,
408                role_to_str(reference.role),
409                reference.enclosing_symbol.as_deref(),
410                usize_to_i64(reference.start_line)?,
411                usize_to_i64(reference.end_line)?,
412                usize_to_i64(reference.start_byte)?,
413                usize_to_i64(reference.end_byte)?,
414            ])?;
415        }
416        drop(insert_reference);
417
418        let mut insert_import = tx.prepare_cached(
419            "INSERT INTO imports(file_id, raw_target, resolved_path, line) VALUES (?1, ?2, ?3, ?4)",
420        )?;
421        let mut insert_import_candidate = tx.prepare_cached(
422            "INSERT INTO import_candidates(import_id, candidate_path, priority) VALUES (?1, ?2, ?3)",
423        )?;
424        for import in &file.imports {
425            insert_import.execute(params![
426                file_id,
427                &import.raw_target,
428                import.resolved_path.as_deref(),
429                usize_to_i64(import.line)?,
430            ])?;
431            let import_id = tx.last_insert_rowid();
432            for (priority, candidate_path) in import.candidate_paths.iter().enumerate() {
433                insert_import_candidate.execute(params![
434                    import_id,
435                    candidate_path,
436                    usize_to_i64(priority)?
437                ])?;
438            }
439        }
440
441        Ok(())
442    }
