1pub(crate) fn verify_baseline(
2    baseline: &MetaRecord,
3    current_generation: i64,
4    current_config: &str,
5    current_derivation: &str,
6) -> Result<()> {
7    let actual = i64_to_u64(current_generation)?;
8    if actual != baseline.repository_generation
9        || current_config != baseline.config_hash
10        || current_derivation != baseline.derivation_fingerprint
11    {
12        return Err(Error::StaleReconciliation {
13            expected: baseline.repository_generation,
14            actual,
15        });
16    }
17    Ok(())
18}

53impl StorageColumn {

55        match self {
56            Self::MetaRepositoryRoot
57            | Self::MetaRepositoryIdentity
58            | Self::MetaRepositoryGeneration
59            | Self::MetaDerivationFingerprint
60            | Self::MetaIndexScopeIncludes
61            | Self::MetaIndexScopeExcludes => StorageTable::Meta,
62            Self::FilesSizeBytes
63            | Self::FilesLanguage
64            | Self::FilesStructurallyComplete
65            | Self::FilesPath => StorageTable::Files,
66        }
67    }

70        match self {
71            Self::MetaRepositoryRoot => "repository_root",
72            Self::MetaRepositoryIdentity => "repository_identity",

74            Self::MetaDerivationFingerprint => "derivation_fingerprint",
75            Self::MetaIndexScopeIncludes => "index_scope_includes",

77            Self::FilesSizeBytes => "size_bytes",
78            Self::FilesLanguage => "language",
79            Self::FilesStructurallyComplete => "structurally_complete",
80            Self::FilesPath => "path",
81        }
82    }

85pub(crate) fn table_exists(conn: &Connection, table: StorageTable) -> Result<bool> {
86    Ok(conn.query_row(
87        "SELECT EXISTS(SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = ?1)",
88        [table.as_str()],
89        |row| row.get(0),
90    )?)
91}

93pub(crate) fn column_exists(conn: &Connection, column: StorageColumn) -> Result<bool> {
94    let sql = format!(
95        "SELECT EXISTS(SELECT 1 FROM pragma_table_info('{}') WHERE name = ?1)",
96        column.table().as_str()
97    );
98    Ok(conn.query_row(&sql, [column.as_str()], |row| row.get(0))?)
99}

101pub(crate) fn count_table_rows(conn: &Connection, table: StorageTable) -> Result<usize> {
102    if !table_exists(conn, table)? {
103        return Ok(0);
104    }
105    let sql = format!("SELECT count(*) FROM {}", table.as_str());
106    Ok(i64_to_usize(conn.query_row(&sql, [], |row| row.get(0))?)?)
107}

109pub(crate) fn repository_identity(path: &Path, index_scope_digest: Option<&str>) -> String {
110    let mut hasher = blake3::Hasher::new();
111    #[cfg(unix)]
112    {
113        use std::os::unix::ffi::OsStrExt;
114        hasher.update(path.as_os_str().as_bytes());
115    }
116    #[cfg(windows)]
117    {
118        use std::os::windows::ffi::OsStrExt;
119        for unit in path.as_os_str().encode_wide() {
120            hasher.update(&unit.to_le_bytes());
121        }
122    }
123    #[cfg(not(any(unix, windows)))]
124    hasher.update(path.to_string_lossy().as_bytes());
125    if let Some(scope_digest) = index_scope_digest {
126        hasher.update(b"\0leantoken-index-scope-v1\0");
127        hasher.update(scope_digest.as_bytes());
128    }
129    hasher.finalize().to_hex().to_string()
130}
