35    pub(crate) fn read_receipt(
36        &self,
37        requested_id: &str,
38        now_unix_millis: i64,
39    ) -> Result<StoredReceipt> {
40        if requested_id.len() > MAX_RECEIPT_ID_BYTES {
41            return Err(Error::UnknownReceipt(requested_id.to_owned()));
42        }
43        let mut conn = self.readers.get()?;
44        let tx = conn.transaction_with_behavior(TransactionBehavior::Deferred)?;
45        let namespace: String = tx.query_row(
46            "SELECT namespace FROM retrieval_receipt_usage WHERE id = 1",
47            [],
48            |row| row.get(0),
49        )?;
50        let Some(row_id) = parse_receipt_id(requested_id, &namespace) else {
51            return Err(Error::UnknownReceipt(requested_id.to_owned()));
52        };
53        let Some(receipt) = load_receipt_connection(&tx, row_id)? else {
54            return Err(Error::UnknownReceipt(requested_id.to_owned()));
55        };
56        let repository_identity: String = tx.query_row(
57            "SELECT repository_identity FROM meta WHERE id = 1",
58            [],
59            |row| row.get(0),
60        )?;
61        if receipt.repository_identity != repository_identity
62            || receipt.created_unix_millis > now_unix_millis
63            || receipt.last_access_unix_millis > now_unix_millis
64            || receipt.expires_unix_millis <= now_unix_millis
65        {
66            return Err(Error::UnknownReceipt(requested_id.to_owned()));
67        }
68        let evidence = load_receipt_evidence_connection(&tx, row_id)?;
69        if evidence.len() != receipt.evidence_count
70            || evidence.len() > MAX_EVIDENCE_PER_RECEIPT
71            || receipt.evidence_bytes > MAX_EVIDENCE_BYTES_PER_RECEIPT
72        {
73            return Err(Error::OperationFailure(
74                "retrieval receipt storage bounds are inconsistent".into(),
75            ));
76        }
77        tx.commit()?;
78        Ok(StoredReceipt {
79            receipt_id: requested_id.to_owned(),
80            repository_identity,
81            repository_generation: receipt.repository_generation,
82            created_unix_millis: receipt.created_unix_millis,
83            expires_unix_millis: receipt.expires_unix_millis,
84            complete: receipt.evidence_count < MAX_EVIDENCE_PER_RECEIPT
85                && receipt
86                    .evidence_bytes
87                    .saturating_add(MAX_RECEIPT_EVIDENCE_LOGICAL_BYTES)
88                    <= MAX_EVIDENCE_BYTES_PER_RECEIPT,
89            evidence,
90        })
91    }

109    pub(crate) fn evaluate_receipt_at(
110        &self,
111        requested_id: Option<&str>,
112        generation: u64,
113        candidates: &[ReceiptEvidence],
114        suppress_overlap: bool,
115        now_unix_millis: i64,
116    ) -> Result<ReceiptEvaluation> {
117        if requested_id.is_some_and(|id| id.len() > MAX_RECEIPT_ID_BYTES) {
118            return Err(Error::InputTooLong {
119                field: "receipt_id",
120                max_bytes: MAX_RECEIPT_ID_BYTES,
121            });
122        }
123        if now_unix_millis < 0 {
124            return Err(Error::OperationFailure(
125                "system clock precedes the Unix epoch".into(),
126            ));
127        }
128        if candidates
129            .iter()
130            .any(|evidence| evidence.logical_bytes() > MAX_EVIDENCE_BYTES_PER_RECEIPT)
131        {
132            return Err(Error::InputTooLong {
133                field: "receipt_evidence",
134                max_bytes: MAX_EVIDENCE_BYTES_PER_RECEIPT,
135            });
136        }
137        let expires_unix_millis = now_unix_millis
138            .checked_add(RECEIPT_TTL_MILLIS)
139            .ok_or_else(|| Error::OperationFailure("retrieval receipt expiry overflow".into()))?;
140        let mut conn = self
141            .writer
142            .lock()
143            .unwrap_or_else(|poisoned| poisoned.into_inner());
144        let tx = conn.transaction_with_behavior(TransactionBehavior::Immediate)?;
145        prune_expired_receipts(&tx, now_unix_millis)?;
146        let mut usage = receipt_usage(&tx)?;
147        let requested_existing = requested_id.is_some();
148
149        let (receipt_id, receipt) = if let Some(requested_id) = requested_id {
150            let Some(row_id) = parse_receipt_id(requested_id, &usage.namespace) else {
151                tx.commit()?;
152                return Err(Error::UnknownReceipt(requested_id.to_owned()));
153            };
154            let Some(receipt) = load_receipt(&tx, row_id)? else {
155                tx.commit()?;
156                return Err(Error::UnknownReceipt(requested_id.to_owned()));
157            };
158            if receipt.created_unix_millis > now_unix_millis
159                || receipt.last_access_unix_millis > now_unix_millis
160            {
161                tx.execute("DELETE FROM retrieval_receipts WHERE id = ?1", [row_id])?;
162                tx.commit()?;
163                return Err(Error::UnknownReceipt(requested_id.to_owned()));
164            }
165            if receipt.repository_generation != generation {
166                tx.commit()?;
167                return Err(Error::StaleReceipt {
168                    receipt_generation: receipt.repository_generation,
169                    repository_generation: generation,
170                });
171            }
172            let repository_identity: String = tx.query_row(
173                "SELECT repository_identity FROM meta WHERE id = 1",
174                [],
175                |row| row.get(0),
176            )?;
177            if receipt.repository_identity != repository_identity {
178                tx.commit()?;
179                return Err(Error::UnknownReceipt(requested_id.to_owned()));
180            }
181            (requested_id.to_owned(), receipt)
182        } else {
183            let repository_identity: String = tx.query_row(
184                "SELECT repository_identity FROM meta WHERE id = 1",
185                [],
186                |row| row.get(0),
187            )?;
188            let logical_bytes = RECEIPT_HEADER_FIXED_LOGICAL_BYTES
189                .checked_add(repository_identity.len())
190                .ok_or_else(|| {
191                    Error::OperationFailure("retrieval receipt byte accounting overflow".into())
192                })?;
193            if logical_bytes > MAX_TOTAL_RECEIPT_BYTES {
194                return Err(Error::OperationFailure(
195                    "repository identity exceeds the retrieval receipt byte quota".into(),
196                ));
197            }
198            while usage.receipt_count >= MAX_RECEIPTS
199                || usage.receipt_bytes.saturating_add(logical_bytes) > MAX_TOTAL_RECEIPT_BYTES
200            {
201                if !evict_oldest_receipt_except(&tx, None)? {
202                    return Err(Error::OperationFailure(
203                        "retrieval receipt quota could not evict a receipt".into(),
204                    ));
205                }
206                usage = receipt_usage(&tx)?;
207            }
208            let access_sequence = next_receipt_access_sequence(&tx, usage.next_access_sequence)?;
209            usage.next_access_sequence = access_sequence;
210            tx.execute(
211                "INSERT INTO retrieval_receipts(
212                    repository_identity,
213                    repository_generation,
214                    created_unix_millis,
215                    last_access_unix_millis,
216                    expires_unix_millis,
217                    access_sequence,
218                    logical_bytes
219                 ) VALUES (?1, ?2, ?3, ?3, ?4, ?5, ?6)",
220                params![
221                    repository_identity,
222                    u64_to_i64(generation)?,
223                    now_unix_millis,
224                    expires_unix_millis,
225                    access_sequence,
226                    usize_to_i64(logical_bytes)?
227                ],
228            )?;
229            let row_id = tx.last_insert_rowid();
230            let receipt_id = format_receipt_id(&usage.namespace, row_id);
231            (
232                receipt_id,
233                PersistentReceiptRow {
234                    id: row_id,
235                    repository_identity,
236                    repository_generation: generation,
237                    created_unix_millis: now_unix_millis,
238                    last_access_unix_millis: now_unix_millis,
239                    expires_unix_millis,
240                    evidence_count: 0,
241                    evidence_bytes: 0,
242                },
243            )
244        };
245
246        let previous = load_receipt_evidence(&tx, receipt.id)?;
247        let decisions = candidates
248            .iter()
249            .map(|candidate| decide(&previous, candidate, suppress_overlap))
250            .collect::<Vec<_>>();
251        let returned = candidates
252            .iter()
253            .zip(&decisions)
254            .filter(|(_, decision)| {
255                matches!(
256                    decision,
257                    ReceiptDecision::Return | ReceiptDecision::ReturnNearDuplicate
258                )
259            })
260            .map(|(candidate, _)| candidate.clone())
261            .collect::<Vec<_>>();
262        let mut append = Vec::new();
263        let mut append_bytes = 0usize;
264        for evidence in returned {
265            let logical_bytes = evidence.logical_bytes();
266            if receipt.evidence_count.saturating_add(append.len()) >= MAX_EVIDENCE_PER_RECEIPT
267                || receipt
268                    .evidence_bytes
269                    .saturating_add(append_bytes)
270                    .saturating_add(logical_bytes)
271                    > MAX_EVIDENCE_BYTES_PER_RECEIPT
272            {
273                break;
274            }
275            append_bytes = append_bytes.saturating_add(logical_bytes);
276            append.push((evidence, logical_bytes));
277        }
278
279        usage = receipt_usage(&tx)?;
280        while usage.evidence_count.saturating_add(append.len()) > MAX_TOTAL_EVIDENCE
281            || usage.evidence_bytes.saturating_add(append_bytes) > MAX_TOTAL_EVIDENCE_BYTES
282        {
283            if !evict_oldest_receipt_except(&tx, Some(receipt.id))? {
284                break;
285            }
286            usage = receipt_usage(&tx)?;
287        }
288        while usage.evidence_count.saturating_add(append.len()) > MAX_TOTAL_EVIDENCE
289            || usage.evidence_bytes.saturating_add(append_bytes) > MAX_TOTAL_EVIDENCE_BYTES
290        {
291            let Some((_, removed_bytes)) = append.pop() else {
292                break;
293            };
294            append_bytes = append_bytes.saturating_sub(removed_bytes);
295        }
296
297        let append_count = append.len();
298        for (index, (evidence, logical_bytes)) in append.into_iter().enumerate() {
299            let ordinal = receipt.evidence_count.checked_add(index).ok_or_else(|| {
300                Error::OperationFailure("retrieval receipt ordinal overflow".into())
301            })?;
302            tx.execute(
303                "INSERT INTO retrieval_receipt_evidence(
304                    receipt_id,
305                    ordinal,
306                    path,
307                    start_line,
308                    end_line,
309                    content_hash,
310                    semantic_signature,
311                    exact_only,
312                    logical_bytes
313                 ) VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8, ?9)",
314                params![
315                    receipt.id,
316                    usize_to_i64(ordinal)?,
317                    evidence.path,
318                    usize_to_i64(evidence.start_line)?,
319                    usize_to_i64(evidence.end_line)?,
320                    evidence.content_hash,
321                    evidence
322                        .semantic_signature
323                        .map(|signature| signature as i64),
324                    i64::from(evidence.exact_only),
325                    usize_to_i64(logical_bytes)?
326                ],
327            )?;
328        }
329        if append_count > 0 {
330            let append_count = usize_to_i64(append_count)?;
331            let append_bytes = usize_to_i64(append_bytes)?;
332            let updated = tx.execute(
333                "UPDATE retrieval_receipts
334                 SET evidence_count = evidence_count + ?1,
335                     evidence_bytes = evidence_bytes + ?2
336                 WHERE id = ?3",
337                params![append_count, append_bytes, receipt.id],
338            )?;
339            if updated != 1 {
340                return Err(Error::OperationFailure(
341                    "retrieval receipt disappeared before counter update".into(),
342                ));
343            }
344            tx.execute(
345                "UPDATE retrieval_receipt_usage
346                 SET evidence_count = evidence_count + ?1,
347                     evidence_bytes = evidence_bytes + ?2
348                 WHERE id = 1",
349                params![append_count, append_bytes],
350            )?;
351        }
352        let touch_due = now_unix_millis.saturating_sub(receipt.last_access_unix_millis)
353            >= RECEIPT_TOUCH_INTERVAL_MILLIS;
354        if requested_existing && (append_count > 0 || touch_due) {
355            let access_sequence = next_receipt_access_sequence(&tx, usage.next_access_sequence)?;
356            tx.execute(
357                "UPDATE retrieval_receipts
358                 SET last_access_unix_millis = ?1,
359                     expires_unix_millis = ?2,
360                     access_sequence = ?3
361                 WHERE id = ?4",
362                params![
363                    now_unix_millis,
364                    expires_unix_millis,
365                    access_sequence,
366                    receipt.id
367                ],
368            )?;
369        }
370        tx.commit()?;
371        Ok(ReceiptEvaluation {
372            receipt_id,
373            decisions,
374        })
375    }

384    pub(crate) fn load_receipt_rebase_source_at(
385        &self,
386        requested_id: &str,
387        now_unix_millis: i64,
388    ) -> Result<ReceiptRebaseSource> {
389        if requested_id.len() > MAX_RECEIPT_ID_BYTES {
390            return Err(Error::InputTooLong {
391                field: "receipt_id",
392                max_bytes: MAX_RECEIPT_ID_BYTES,
393            });
394        }
395        if now_unix_millis < 0 {
396            return Err(Error::OperationFailure(
397                "system clock precedes the Unix epoch".into(),
398            ));
399        }
400        let mut conn = self
401            .writer
402            .lock()
403            .unwrap_or_else(|poisoned| poisoned.into_inner());
404        let tx = conn.transaction_with_behavior(TransactionBehavior::Deferred)?;
405        let usage = receipt_usage(&tx)?;
406        let Some(row_id) = parse_receipt_id(requested_id, &usage.namespace) else {
407            tx.commit()?;
408            return Err(Error::UnknownReceipt(requested_id.to_owned()));
409        };
410        let Some(receipt) = load_receipt(&tx, row_id)? else {
411            tx.commit()?;
412            return Err(Error::UnknownReceipt(requested_id.to_owned()));
413        };
414        if receipt.created_unix_millis > now_unix_millis
415            || receipt.last_access_unix_millis > now_unix_millis
416            || receipt.expires_unix_millis <= now_unix_millis
417        {
418            tx.commit()?;
419            return Err(Error::UnknownReceipt(requested_id.to_owned()));
420        }
421        let repository_identity: String = tx.query_row(
422            "SELECT repository_identity FROM meta WHERE id = 1",
423            [],
424            |row| row.get(0),
425        )?;
426        if receipt.repository_identity != repository_identity {
427            tx.commit()?;
428            return Err(Error::UnknownReceipt(requested_id.to_owned()));
429        }
430        let evidence = load_receipt_evidence(&tx, receipt.id)?;
431        if evidence.len() != receipt.evidence_count {
432            return Err(Error::OperationFailure(
433                "retrieval receipt evidence count is inconsistent".into(),
434            ));
435        }
436        tx.commit()?;
437        Ok(ReceiptRebaseSource {
438            receipt_id: requested_id.to_owned(),
439            repository_identity,
440            repository_generation: receipt.repository_generation,
441            evidence,
442        })
443    }

459    pub(crate) fn persist_rebased_receipt_at(
460        &self,
461        source: &ReceiptRebaseSource,
462        expected_repository_generation: u64,
463        carried: &[ReceiptEvidence],
464        now_unix_millis: i64,
465    ) -> Result<String> {
466        if now_unix_millis < 0 {
467            return Err(Error::OperationFailure(
468                "system clock precedes the Unix epoch".into(),
469            ));
470        }
471        let expires_unix_millis = now_unix_millis
472            .checked_add(RECEIPT_TTL_MILLIS)
473            .ok_or_else(|| Error::OperationFailure("retrieval receipt expiry overflow".into()))?;
474        let mut conn = self
475            .writer
476            .lock()
477            .unwrap_or_else(|poisoned| poisoned.into_inner());
478        let tx = conn.transaction_with_behavior(TransactionBehavior::Immediate)?;
479        let mut usage = receipt_usage(&tx)?;
480        let Some(source_row_id) = parse_receipt_id(&source.receipt_id, &usage.namespace) else {
481            tx.commit()?;
482            return Err(Error::UnknownReceipt(source.receipt_id.clone()));
483        };
484        let Some(current_source) = load_receipt(&tx, source_row_id)? else {
485            tx.commit()?;
486            return Err(Error::UnknownReceipt(source.receipt_id.clone()));
487        };
488        if current_source.created_unix_millis > now_unix_millis
489            || current_source.last_access_unix_millis > now_unix_millis
490            || current_source.expires_unix_millis <= now_unix_millis
491        {
492            tx.commit()?;
493            return Err(Error::UnknownReceipt(source.receipt_id.clone()));
494        }
495        let (repository_identity, repository_generation): (String, u64) = tx.query_row(
496            "SELECT repository_identity, repository_generation FROM meta WHERE id = 1",
497            [],
498            |row| Ok((row.get(0)?, i64_to_u64(row.get(1)?)?)),
499        )?;
500        if repository_identity != source.repository_identity
501            || current_source.repository_identity != source.repository_identity
502        {
503            tx.commit()?;
504            return Err(Error::UnknownReceipt(source.receipt_id.clone()));
505        }
506        if repository_generation != expected_repository_generation {
507            return Err(Error::RetryableConflict(
508                crate::error::RetryableOperation::Retrieval,
509            ));
510        }
511        let current_evidence = load_receipt_evidence(&tx, source_row_id)?;
512        if current_source.repository_generation != source.repository_generation
513            || current_evidence != source.evidence
514        {
515            return Err(Error::RetryableConflict(
516                crate::error::RetryableOperation::Retrieval,
517            ));
518        }
519        let mut next_source = 0usize;
520        for evidence in carried {
521            let Some(offset) = source.evidence[next_source..]
522                .iter()
523                .position(|candidate| candidate == evidence)
524            else {
525                return Err(Error::OperationFailure(
526                    "rebased receipt evidence is not an ordered source subset".into(),
527                ));
528            };
529            next_source = next_source.saturating_add(offset).saturating_add(1);
530        }
531
532        tx.execute(
533            "DELETE FROM retrieval_receipts
534             WHERE expires_unix_millis <= ?1 AND id != ?2",
535            params![now_unix_millis, source_row_id],
536        )?;
537        usage = receipt_usage(&tx)?;
538        let logical_bytes = RECEIPT_HEADER_FIXED_LOGICAL_BYTES
539            .checked_add(repository_identity.len())
540            .ok_or_else(|| {
541                Error::OperationFailure("retrieval receipt byte accounting overflow".into())
542            })?;
543        let evidence_bytes = carried.iter().try_fold(0usize, |total, evidence| {
544            total.checked_add(evidence.logical_bytes()).ok_or_else(|| {
545                Error::OperationFailure("retrieval receipt byte accounting overflow".into())
546            })
547        })?;
548        if carried.len() > MAX_EVIDENCE_PER_RECEIPT
549            || evidence_bytes > MAX_EVIDENCE_BYTES_PER_RECEIPT
550            || logical_bytes > MAX_TOTAL_RECEIPT_BYTES
551        {
552            return Err(Error::OperationFailure(
553                "rebased retrieval receipt exceeds its source-derived quota".into(),
554            ));
555        }
556        while usage.receipt_count >= MAX_RECEIPTS
557            || usage.receipt_bytes.saturating_add(logical_bytes) > MAX_TOTAL_RECEIPT_BYTES
558            || usage.evidence_count.saturating_add(carried.len()) > MAX_TOTAL_EVIDENCE
559            || usage.evidence_bytes.saturating_add(evidence_bytes) > MAX_TOTAL_EVIDENCE_BYTES
560        {
561            if !evict_oldest_receipt_except(&tx, Some(source_row_id))? {
562                return Err(Error::OperationFailure(
563                    "retrieval receipt quota cannot preserve the rebase source".into(),
564                ));
565            }
566            usage = receipt_usage(&tx)?;
567        }
568        let access_sequence = next_receipt_access_sequence(&tx, usage.next_access_sequence)?;
569        tx.execute(
570            "INSERT INTO retrieval_receipts(
571                repository_identity,
572                repository_generation,
573                created_unix_millis,
574                last_access_unix_millis,
575                expires_unix_millis,
576                access_sequence,
577                logical_bytes,
578                evidence_count,
579                evidence_bytes
580             ) VALUES (?1, ?2, ?3, ?3, ?4, ?5, ?6, ?7, ?8)",
581            params![
582                repository_identity,
583                u64_to_i64(repository_generation)?,
584                now_unix_millis,
585                expires_unix_millis,
586                access_sequence,
587                usize_to_i64(logical_bytes)?,
588                usize_to_i64(carried.len())?,
589                usize_to_i64(evidence_bytes)?,
590            ],
591        )?;
592        let row_id = tx.last_insert_rowid();
593        for (ordinal, evidence) in carried.iter().enumerate() {
594            tx.execute(
595                "INSERT INTO retrieval_receipt_evidence(
596                    receipt_id,
597                    ordinal,
598                    path,
599                    start_line,
600                    end_line,
601                    content_hash,
602                    semantic_signature,
603                    exact_only,
604                    logical_bytes
605                 ) VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8, ?9)",
606                params![
607                    row_id,
608                    usize_to_i64(ordinal)?,
609                    evidence.path,
610                    usize_to_i64(evidence.start_line)?,
611                    usize_to_i64(evidence.end_line)?,
612                    evidence.content_hash,
613                    evidence
614                        .semantic_signature
615                        .map(|signature| signature as i64),
616                    1_i64,
617                    usize_to_i64(evidence.logical_bytes())?,
618                ],
619            )?;
620        }
621        if !carried.is_empty() {
622            tx.execute(
623                "UPDATE retrieval_receipt_usage
624                 SET evidence_count = evidence_count + ?1,
625                     evidence_bytes = evidence_bytes + ?2
626                 WHERE id = 1",
627                params![usize_to_i64(carried.len())?, usize_to_i64(evidence_bytes)?],
628            )?;
629        }
630        let receipt_id = format_receipt_id(&usage.namespace, row_id);
631        tx.commit()?;
632        Ok(receipt_id)
633    }
634}

638        "SELECT namespace,
639                next_access_sequence,
640                receipt_count,
641                receipt_bytes,
642                evidence_count,
643                evidence_bytes
644         FROM retrieval_receipt_usage

648            Ok(PersistentReceiptUsage {
649                namespace: row.get(0)?,
650                next_access_sequence: row.get(1)?,
651                receipt_count: i64_to_usize(row.get(2)?)?,
652                receipt_bytes: i64_to_usize(row.get(3)?)?,
653                evidence_count: i64_to_usize(row.get(4)?)?,
654                evidence_bytes: i64_to_usize(row.get(5)?)?,
655            })
656        },
657    )
658    .map_err(Into::into)
659}

661pub(crate) fn next_receipt_access_sequence(tx: &Transaction<'_>, current: i64) -> Result<i64> {
662    let next = current.checked_add(1).ok_or_else(|| {
663        Error::OperationFailure("retrieval receipt access sequence overflow".into())
664    })?;
665    let updated = tx.execute(
666        "UPDATE retrieval_receipt_usage
667         SET next_access_sequence = ?1
668         WHERE id = 1 AND next_access_sequence = ?2",
669        params![next, current],
670    )?;
671    if updated != 1 {
672        return Err(Error::OperationFailure(
673            "retrieval receipt access sequence changed unexpectedly".into(),
674        ));
675    }
676    Ok(next)
677}

727pub(crate) fn load_receipt_evidence_connection(
728    connection: &Connection,
729    receipt_id: i64,
730) -> Result<Vec<ReceiptEvidence>> {
731    let mut statement = connection.prepare(
732        "SELECT path, start_line, end_line, content_hash, semantic_signature, exact_only
733         FROM retrieval_receipt_evidence
734         WHERE receipt_id = ?1
735         ORDER BY ordinal",
736    )?;
737    statement
738        .query_map([receipt_id], |row| {
739            Ok(ReceiptEvidence {
740                path: row.get(0)?,
741                start_line: i64_to_usize(row.get(1)?)?,
742                end_line: i64_to_usize(row.get(2)?)?,
743                content_hash: row.get(3)?,
744                semantic_signature: row.get::<_, Option<i64>>(4)?.map(|value| value as u64),
745                exact_only: row.get::<_, i64>(5)? != 0,
746            })
747        })?
748        .collect::<rusqlite::Result<Vec<_>>>()
749        .map_err(Into::into)
750}

761pub(crate) fn evict_oldest_receipt_except(
762    tx: &Transaction<'_>,
763    retained_id: Option<i64>,
764) -> Result<bool> {
765    let candidate = match retained_id {
766        Some(retained_id) => tx
767            .query_row(
768                "SELECT id
769                 FROM retrieval_receipts
770                 WHERE access_sequence > 0 AND id != ?1
771                 ORDER BY access_sequence, id
772                 LIMIT 1",
773                [retained_id],
774                |row| row.get::<_, i64>(0),
775            )
776            .optional()?,
777        None => tx
778            .query_row(
779                "SELECT id
780                 FROM retrieval_receipts
781                 WHERE access_sequence > 0
782                 ORDER BY access_sequence, id
783                 LIMIT 1",
784                [],
785                |row| row.get::<_, i64>(0),
786            )
787            .optional()?,
788    };
789    let Some(candidate) = candidate else {
790        return Ok(false);
791    };
792    Ok(tx.execute("DELETE FROM retrieval_receipts WHERE id = ?1", [candidate])? == 1)
793}

795pub(crate) fn unix_millis(time: SystemTime) -> i64 {
796    time.duration_since(UNIX_EPOCH)
797        .ok()
798        .and_then(|duration| i64::try_from(duration.as_millis()).ok())
799        .unwrap_or(0)
800}
