1/// Return paths reported by `git status` as working-tree changes.
2///
3/// The result is capped at `max` entries to keep the call bounded. If the
4/// root is not a Git repository, `git` is unavailable, or `git status` fails,
5/// an empty set is returned so callers can safely proceed without a diff
6/// signal.
7pub fn git_changed_paths(root: &Path, max: usize) -> Result<HashSet<String>> {
8    git_changed_paths_with(root, max, Path::new("git"), Duration::from_millis(500))
9}

11/// Bounded working-tree paths plus whether `git status` completed successfully.
12pub(crate) struct GitWorkingTreeStatus {
13    pub(crate) changed_paths: HashSet<String>,
14    state: GitWorkingTreeState,
15}
16
17#[derive(Clone, Copy)]
18enum GitWorkingTreeState {
19    Unavailable(GitWorkingTreeChanges),
20    Available(GitWorkingTreeChanges),
21}
22
23#[derive(Clone, Copy)]
24enum GitWorkingTreeChanges {
25    Clean,
26    Modified,
27    Untracked,
28    ModifiedAndUntracked,
29}
30
31impl GitWorkingTreeStatus {
32    fn unavailable(changed_paths: HashSet<String>, modified: bool, untracked: bool) -> Self {
33        Self {
34            changed_paths,
35            state: GitWorkingTreeState::Unavailable(GitWorkingTreeChanges::from_flags(
36                modified, untracked,
37            )),
38        }
39    }
40
41    fn available(changed_paths: HashSet<String>, modified: bool, untracked: bool) -> Self {
42        Self {
43            changed_paths,
44            state: GitWorkingTreeState::Available(GitWorkingTreeChanges::from_flags(
45                modified, untracked,
46            )),
47        }
48    }
49
50    pub(crate) const fn is_available(&self) -> bool {
51        matches!(self.state, GitWorkingTreeState::Available(_))
52    }
53
54    pub(crate) const fn has_modified(&self) -> bool {
55        matches!(
56            self.state.changes(),
57            GitWorkingTreeChanges::Modified | GitWorkingTreeChanges::ModifiedAndUntracked
58        )
59    }
60
61    pub(crate) const fn has_untracked(&self) -> bool {
62        matches!(
63            self.state.changes(),
64            GitWorkingTreeChanges::Untracked | GitWorkingTreeChanges::ModifiedAndUntracked
65        )
66    }
67}
68
69impl GitWorkingTreeState {
70    const fn changes(self) -> GitWorkingTreeChanges {
71        match self {
72            Self::Unavailable(changes) | Self::Available(changes) => changes,
73        }
74    }
75}
76
77impl GitWorkingTreeChanges {
78    const fn from_flags(modified: bool, untracked: bool) -> Self {
79        match (modified, untracked) {
80            (false, false) => Self::Clean,
81            (true, false) => Self::Modified,
82            (false, true) => Self::Untracked,
83            (true, true) => Self::ModifiedAndUntracked,
84        }
85    }
86}

93pub(crate) fn git_changed_paths_with(
94    root: &Path,
95    max: usize,
96    program: &Path,
97    timeout: Duration,
98) -> Result<HashSet<String>> {
99    Ok(git_working_tree_status_with(root, max, program, timeout).changed_paths)

102pub(crate) fn git_working_tree_status_with(
103    root: &Path,
104    max: usize,
105    program: &Path,
106    timeout: Duration,
107) -> GitWorkingTreeStatus {
108    if max == 0 {
109        return GitWorkingTreeStatus::available(HashSet::new(), false, false);
110    }
111    let prefix = git_worktree_prefix(root);
112    let args = [
113        "-c",
114        "core.fsmonitor=false",
115        "status",
116        "--porcelain=v1",
117        "-z",
118        "--untracked-files=all",
119        "--no-renames",
120        "--",
121        ".",
122    ]
123    .map(str::to_owned);
124    let Ok(output) = run_git_capture(
125        root,
126        program,
127        &args,
128        GitCaptureOptions {
129            timeout,
130            field: "git status",
131            timeout_reason: "git status timed out",
132            failure_reason: "git status failed",
133            max_output_bytes: bounded_git_output(max, GIT_PATH_OUTPUT_BYTES_PER_RESULT),
134        },
135    ) else {
136        return GitWorkingTreeStatus::unavailable(HashSet::new(), false, false);
137    };
138    parse_git_status_observation(output.as_slice(), max, &prefix)
139}
140
141pub(crate) fn bounded_git_output(max_results: usize, bytes_per_result: usize) -> usize {
142    max_results
143        .saturating_mul(bytes_per_result)
144        .max(bytes_per_result)
145        .min(MAX_GIT_DISCOVERY_OUTPUT_BYTES)
146}

148pub(crate) fn git_worktree_prefix(root: &Path) -> String {
149    root.ancestors()
150        .find(|ancestor| ancestor.join(".git").exists())
151        .and_then(|worktree| root.strip_prefix(worktree).ok())
152        .map(slash_path)
153        .filter(|prefix| !prefix.is_empty())
154        .map(|prefix| format!("{prefix}/"))
155        .unwrap_or_default()
156}

167pub(crate) fn parse_git_status_observation<R: BufRead>(
168    mut reader: R,
169    max: usize,
170    prefix: &str,
171) -> GitWorkingTreeStatus {
172    if max == 0 {
173        return GitWorkingTreeStatus::available(HashSet::new(), false, false);
174    }
175    let mut changed = HashSet::new();
176    let mut modified = false;
177    let mut untracked = false;
178    let mut record = Vec::new();
179
180    loop {
181        record.clear();
182        match reader.read_until(0, &mut record) {
183            Ok(0) => break,
184            Ok(_) => {}
185            Err(_) => return GitWorkingTreeStatus::unavailable(changed, modified, untracked),
186        }
187
188        if record.last() == Some(&0) {
189            record.pop();
190        }
191        if record.len() < 4 || record.get(2) != Some(&b' ') {
192            continue;
193        }
194
195        let status = &record[..2];
196
197        // Ignore ignored files; keep modified, added, deleted, and untracked.
198        if status == b"!!" {
199            continue;
200        }
201
202        if status == b"??" {
203            untracked = true;
204        } else {
205            modified = true;
206        }
207
208        let Ok(path) = std::str::from_utf8(&record[3..]) else {
209            return GitWorkingTreeStatus::unavailable(HashSet::new(), modified, untracked);
210        };
211
212        let Some(path) = path.strip_prefix(prefix) else {
213            continue;
214        };
215        changed.insert(slash_path(Path::new(path)));
216        if changed.len() == max {
217            break;
218        }
219    }
220    GitWorkingTreeStatus::available(changed, modified, untracked)
221}
