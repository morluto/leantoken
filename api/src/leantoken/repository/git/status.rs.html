1/// Return paths reported by `git status` as working-tree changes.
2///
3/// The result is capped at `max` entries to keep the call bounded. If the
4/// root is not a Git repository, `git` is unavailable, or `git status` fails,
5/// an empty set is returned so callers can safely proceed without a diff
6/// signal.
7pub fn git_changed_paths(root: &Path, max: usize) -> Result<HashSet<String>> {
8    git_changed_paths_with(root, max, Path::new("git"), Duration::from_millis(500))
9}

11/// Bounded working-tree paths plus whether `git status` completed successfully.
12pub(crate) struct GitWorkingTreeStatus {
13    pub(crate) changed_paths: HashSet<String>,
14    pub(crate) available: bool,
15}

31fn git_working_tree_status_with(
32    root: &Path,
33    max: usize,
34    program: &Path,
35    timeout: Duration,
36) -> GitWorkingTreeStatus {
37    if max == 0 {
38        return GitWorkingTreeStatus {
39            changed_paths: HashSet::new(),
40            available: true,
41        };
42    }
43    let prefix = git_worktree_prefix(root);
44    let args = [
45        "-c",
46        "core.fsmonitor=false",
47        "status",
48        "--porcelain=v1",
49        "-z",
50        "--untracked-files=all",
51        "--no-renames",
52        "--",
53        ".",
54    ]
55    .map(str::to_owned);
56    let Ok(output) = run_git_capture(
57        root,
58        program,
59        &args,
60        GitCaptureOptions {
61            timeout,
62            field: "git status",
63            timeout_reason: "git status timed out",
64            failure_reason: "git status failed",
65            max_output_bytes: bounded_git_output(max, GIT_PATH_OUTPUT_BYTES_PER_RESULT),
66        },
67    ) else {
68        return GitWorkingTreeStatus {
69            changed_paths: HashSet::new(),
70            available: false,
71        };
72    };
73    let (changed_paths, available) = parse_git_status_observation(output.as_slice(), max, &prefix);
74    GitWorkingTreeStatus {
75        changed_paths,
76        available,
77    }
78}

80fn bounded_git_output(max_results: usize, bytes_per_result: usize) -> usize {
81    max_results
82        .saturating_mul(bytes_per_result)
83        .max(bytes_per_result)
84        .min(MAX_GIT_DISCOVERY_OUTPUT_BYTES)
85}

87fn git_worktree_prefix(root: &Path) -> String {
88    root.ancestors()
89        .find(|ancestor| ancestor.join(".git").exists())
90        .and_then(|worktree| root.strip_prefix(worktree).ok())
91        .map(slash_path)
92        .filter(|prefix| !prefix.is_empty())
93        .map(|prefix| format!("{prefix}/"))
94        .unwrap_or_default()
95}

98fn parse_git_status<R: BufRead>(mut reader: R, max: usize, prefix: &str) -> HashSet<String> {
99    parse_git_status_observation(&mut reader, max, prefix).0
100}

102fn parse_git_status_observation<R: BufRead>(
103    mut reader: R,
104    max: usize,
105    prefix: &str,
106) -> (HashSet<String>, bool) {
107    if max == 0 {
108        return (HashSet::new(), true);
109    }
110    let mut changed = HashSet::new();
111    let mut record = Vec::new();
112
113    loop {
114        record.clear();
115        match reader.read_until(0, &mut record) {
116            Ok(0) => break,
117            Ok(_) => {}
118            Err(_) => return (changed, false),
119        }
120
121        if record.last() == Some(&0) {
122            record.pop();
123        }
124        if record.len() < 4 || record.get(2) != Some(&b' ') {
125            continue;
126        }
127
128        let status = &record[..2];
129        let path = String::from_utf8_lossy(&record[3..]).into_owned();
130
131        // Ignore ignored files; keep modified, added, deleted, and untracked.
132        if status == b"!!" {
133            continue;
134        }
135
136        let Some(path) = path.strip_prefix(prefix) else {
137            continue;
138        };
139        changed.insert(slash_path(Path::new(path)));
140        if changed.len() == max {
141            break;
142        }
143    }
144    (changed, true)
145}
