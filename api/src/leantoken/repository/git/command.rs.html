28pub(crate) fn run_git_capture_bounded(
29    root: &Path,
30    program: &Path,
31    args: &[String],
32    input: Option<&[u8]>,
33    options: GitCaptureOptions,
34) -> Result<Vec<u8>> {
35    use std::io::Read;
36    use std::sync::{
37        Arc,
38        atomic::{AtomicBool, Ordering},
39    };
40    use std::thread;
41    use std::time::Instant;
42
43    let mut command = Command::new(program);
44    command
45        .args(args)
46        .current_dir(root)
47        .stdin(if input.is_some() {
48            Stdio::piped()
49        } else {
50            Stdio::null()
51        })
52        .stdout(Stdio::piped())
53        .stderr(Stdio::null());
54    let mut child = command.group_spawn().map_err(|_| Error::InvalidInput {
55        field: options.field,
56        reason: "git is unavailable",
57    })?;
58
59    let output_limit_exceeded = Arc::new(AtomicBool::new(false));
60    let reader_exceeded = Arc::clone(&output_limit_exceeded);
61    let (release_reader, reader_release) = std::sync::mpsc::channel();
62    let max_output_bytes = options.max_output_bytes;
63    let mut stdout = child
64        .inner()
65        .stdout
66        .take()
67        .ok_or_else(|| Error::OperationFailure("git stdout unavailable".into()))?;
68    let reader = thread::spawn(move || -> std::io::Result<Vec<u8>> {
69        let mut output = Vec::with_capacity(max_output_bytes.min(64 * 1024));
70        let mut chunk = [0u8; 8 * 1024];
71        loop {
72            let read = stdout.read(&mut chunk)?;
73            if read == 0 {
74                return Ok(output);
75            }
76            if output.len().saturating_add(read) > max_output_bytes {
77                reader_exceeded.store(true, Ordering::Release);
78                // Keep the pipe open without draining it until the parent has
79                // killed the producer. Otherwise a fast producer can observe
80                // SIGPIPE, let a shell wrapper continue to its next command,
81                // and perform work after the limit was crossed.
82                let _ = reader_release.recv();
83                return Ok(output);
84            }
85            output.extend_from_slice(&chunk[..read]);
86        }
87    });
88    let writer = input.map(|input| {
89        let input = input.to_vec();
90        let mut stdin = child.inner().stdin.take();
91        thread::spawn(move || -> std::io::Result<()> {
92            stdin
93                .as_mut()
94                .ok_or_else(|| std::io::Error::other("git stdin unavailable"))?
95                .write_all(&input)
96        })
97    });
98
99    enum ChildOutcome {
100        Exited(std::process::ExitStatus),
101        OutputLimit,
102        Timeout,
103        WaitError(std::io::Error),
104    }
105
106    let deadline = Instant::now() + options.timeout;
107    let outcome = loop {
108        if output_limit_exceeded.load(Ordering::Acquire) {
109            break ChildOutcome::OutputLimit;
110        }
111        match child.try_wait() {
112            Ok(Some(status)) => break ChildOutcome::Exited(status),
113            Ok(None) => {}
114            Err(error) => break ChildOutcome::WaitError(error),
115        }
116        if Instant::now() >= deadline {
117            break ChildOutcome::Timeout;
118        }
119        thread::sleep(Duration::from_millis(5));
120    };
121
122    // Terminate the whole process group even after the direct child exits:
123    // external helpers can otherwise retain stdout and block the reader join.
124    let _ = child.kill();
125    let _ = child.wait();
126    // If the reader crossed the bound just as the child exited, it may be
127    // waiting for this signal even though the polling loop observed normal
128    // exit first.
129    let _ = release_reader.send(());
130
131    let status = match outcome {
132        ChildOutcome::Exited(status) => status,
133        ChildOutcome::OutputLimit => {
134            let _ = reader.join();
135            if let Some(writer) = writer {
136                let _ = writer.join();
137            }
138            return Err(Error::RequestLimitExceeded {
139                field: "git output bytes",
140                requested: options.max_output_bytes.saturating_add(1),
141                limit: options.max_output_bytes,
142            });
143        }
144        ChildOutcome::Timeout => {
145            let _ = reader.join();
146            if let Some(writer) = writer {
147                let _ = writer.join();
148            }
149            return Err(Error::InvalidInput {
150                field: options.field,
151                reason: options.timeout_reason,
152            });
153        }
154        ChildOutcome::WaitError(error) => {
155            let _ = reader.join();
156            if let Some(writer) = writer {
157                let _ = writer.join();
158            }
159            return Err(error.into());
160        }
161    };
162    if let Some(writer) = writer {
163        writer
164            .join()
165            .map_err(|_| Error::OperationFailure("git stdin task panicked".into()))??;
166    }
167    let output = reader
168        .join()
169        .map_err(|_| Error::OperationFailure("git stdout task panicked".into()))??;
170    if output_limit_exceeded.load(Ordering::Acquire) {
171        return Err(Error::RequestLimitExceeded {
172            field: "git output bytes",
173            requested: options.max_output_bytes.saturating_add(1),
174            limit: options.max_output_bytes,
175        });
176    }
177    if !status.success() {
178        return Err(Error::InvalidInput {
179            field: options.field,
180            reason: options.failure_reason,
181        });
182    }
183    Ok(output)
184}
