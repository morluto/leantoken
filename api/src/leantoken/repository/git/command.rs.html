30pub(crate) fn run_git_capture_bounded(
31    root: &Path,
32    program: &Path,
33    args: &[String],
34    input: Option<&[u8]>,
35    options: GitCaptureOptions,
36) -> Result<Vec<u8>> {
37    let mut command = Command::new(program);
38    // Force Git to treat all pathspec arguments as literal filesystem
39    // paths, preventing pathspec magic (e.g. `:(literal)`, `:(exclude)`)
40    // in repository filenames from being interpreted as Git selectors.
41    // See issue #546: repository filenames can legitimately begin with
42    // pathspec magic syntax, but Git should never treat them as patterns.
43    command
44        .env_remove("GIT_GLOB_PATHSPECS")
45        .env_remove("GIT_ICASE_PATHSPECS")
46        .env_remove("GIT_NOGLOB_PATHSPECS")
47        .env("GIT_LITERAL_PATHSPECS", "1")
48        .args(args)
49        .current_dir(root);
50    let output = capture_stdout_bounded(
51        &mut command,
52        input,
53        CaptureOptions {
54            timeout: options.timeout,
55            max_stdout_bytes: options.max_output_bytes,
56        },
57    )
58    .map_err(|error| match error {
59        CaptureError::Spawn => Error::InvalidInput {
60            field: options.field,
61            reason: "git is unavailable",
62        },
63        CaptureError::MissingPipe(pipe) => {
64            Error::OperationFailure(format!("git {pipe} unavailable"))
65        }
66        CaptureError::StdoutLimit { limit } => Error::RequestLimitExceeded {
67            field: "git output bytes",
68            requested: limit.saturating_add(1),
69            limit,
70        },
71        CaptureError::Timeout => Error::InvalidInput {
72            field: options.field,
73            reason: options.timeout_reason,
74        },
75        CaptureError::Io(error) => error.into(),
76        CaptureError::WorkerPanicked(worker) => {
77            Error::OperationFailure(format!("git {worker} task panicked"))
78        }
79    })?;
80    if !output.status.success() {
81        return Err(Error::InvalidInput {
82            field: options.field,
83            reason: options.failure_reason,
84        });
85    }
86    Ok(output.stdout)
87}
