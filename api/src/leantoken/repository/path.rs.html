1pub fn resolve_existing(root: &Path, requested: &str) -> Result<PathBuf> {
2    let relative = validate_relative(requested)?;
3    let canonical = root.join(relative).canonicalize()?;
4    if !canonical.starts_with(root) {
5        return Err(Error::PathOutsideRoot(canonical));

14/// Validate and normalize a repository-relative request path.
15///
16/// Repository keys always use forward slashes, independent of the host
17/// platform. This helper therefore recognizes both separator styles before
18/// applying the relative-path contract.
19pub fn normalize_relative(requested: &str) -> Result<String> {
20    if requested.is_empty() || requested.contains('\0') {
21        return Err(Error::InvalidInput {
22            field: "path",
23            reason: "must be a non-empty relative path",
24        });
25    }
26    // `Path` only recognizes prefixes for the host platform. Reject common
27    // Windows absolute forms explicitly so a request has the same contract on
28    // Linux, macOS, and Windows.
29    let bytes = requested.as_bytes();
30    let has_windows_drive = bytes.len() >= 2 && bytes[0].is_ascii_alphabetic() && bytes[1] == b':';
31    let has_windows_root = requested.starts_with('\\');
32    if has_windows_drive || has_windows_root {
33        return Err(Error::PathOutsideRoot(PathBuf::from(requested)));
34    }
35    let normalized = requested.replace('\\', "/");
36    if normalized.starts_with('/') {
37        return Err(Error::PathOutsideRoot(PathBuf::from(requested)));
38    }
39    let path = Path::new(&normalized);
40    if path.is_absolute() {
41        return Err(Error::PathOutsideRoot(path.to_path_buf()));
42    }
43    let mut components = Vec::new();
44    for component in normalized.split('/') {
45        match component {
46            "" | "." => {}
47            ".." => return Err(Error::PathOutsideRoot(path.to_path_buf())),
48            component => components.push(component),
49        }
50    }
51    if components.is_empty() {
52        return Ok(".".to_owned());
53    }
54    Ok(components.join("/"))
55}

57pub fn slash_path(path: &Path) -> String {
58    path.components()
59        .filter_map(|component| match component {
60            Component::Normal(value) => Some(value.to_string_lossy().into_owned()),
61            _ => None,
62        })
63        .collect::<Vec<_>>()
64        .join("/")
65}

67pub(crate) fn checked_slash_path(path: &Path) -> Result<String> {
68    path.components()
69        .filter_map(|component| match component {
70            Component::Normal(value) => Some(
71                value
72                    .to_str()
73                    .map(str::to_owned)
74                    .ok_or_else(|| Error::UnsupportedPathEncoding(path.to_path_buf())),
75            ),
76            _ => None,
77        })
78        .collect::<Result<Vec<_>>>()
79        .map(|components| components.join("/"))
80}
