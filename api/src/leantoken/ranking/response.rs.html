13pub(in crate::ranking) fn build_context_plan(
14    params: BuildContextPlanParams<'_>,
15) -> Option<ContextQueryPlan> {
16    let BuildContextPlanParams {
17        request,
18        focus_path_matchers,
19        selected,
20        candidate_paths_total,
21        estimated_source_tokens,
22        generated_artifact_warning,
23        result_complete,
24        policy,
25    } = params;
26    policy.is_plan().then(|| {
27        let minimum_fragments = policy.focus_minimum().unwrap_or(0);
28        let focus_coverage = request
29            .focus_paths
30            .iter()
31            .zip(focus_path_matchers)
32            .map(|(pattern, matcher)| {
33                let candidate_fragments = selected
34                    .iter()
35                    .filter(|candidate| matcher.is_match(&candidate.candidate.path))
36                    .count();
37                ContextPlanFocusCoverage {
38                    pattern: pattern.clone(),
39                    candidate_fragments,
40                    minimum_fragments,
41                    satisfied: candidate_fragments >= minimum_fragments,
42                }
43            })
44            .collect();
45        let candidates = selected
46            .iter()
47            .map(|scored| {
48                let (target_start_line, target_end_line) = scored.candidate.target_lines().unzip();
49                ContextPlanCandidate {
50                    path: scored.candidate.path.clone(),
51                    start_line: scored.candidate.start_line,
52                    end_line: scored.candidate.end_line,
53                    target_start_line,
54                    target_end_line,
55                    truncated: scored.candidate.target_truncated(),
56                    representation: scored.candidate.representation.clone(),
57                    score: (scored.score * 10_000.0).round() / 10_000.0,
58                    reasons: scored
59                        .candidate
60                        .reason()
61                        .split("; ")
62                        .map(str::to_owned)
63                        .collect(),
64                    estimated_tokens: scored.token_count,
65                }
66            })
67            .collect();
68        ContextQueryPlan {
69            candidates,
70            candidate_paths_total,
71            estimated_source_tokens,
72            focus_coverage,
73            generated_artifact_warning,
74            result_complete,
75        }
76    })
77}

79pub(in crate::ranking) fn materialize_context_fragments(
80    selected: &[ScoredCandidate],
81    estimated_source_tokens: usize,
82    policy: ContextSelectionPolicy,
83) -> (Vec<ContextFragment>, Vec<String>, usize) {
84    if policy.is_plan() {
85        return (Vec::new(), Vec::new(), 0);
86    }
87    let fragments = selected
88        .iter()
89        .map(|scored| {
90            let (target_start_line, target_end_line) = scored.candidate.target_lines().unzip();
91            ContextFragment {
92                path: scored.candidate.path.clone(),
93                start_line: scored.candidate.start_line,
94                end_line: scored.candidate.end_line,
95                target_start_line,
96                target_end_line,
97                truncated: scored.candidate.target_truncated(),
98                representation: scored.candidate.representation.clone(),
99                content: scored.candidate.content.clone(),
100                content_hash: scored.content_hash.clone(),
101                score: (scored.score * 10_000.0).round() / 10_000.0,
102                reason: scored.candidate.reason(),
103                token_count: scored.token_count,
104            }
105        })
106        .collect();
107    let fragment_hashes = selected
108        .iter()
109        .map(|scored| scored.content_hash.clone())
110        .collect();
111    (fragments, fragment_hashes, estimated_source_tokens)
112}

128pub(in crate::ranking) fn finalize_context_response(
129    params: FinalizeContextParams<'_>,
130) -> ContextResponse {
131    let FinalizeContextParams {
132        request,
133        repository_generation,
134        tokenizer,
135        plan,
136        fragments,
137        fragment_hashes,
138        emitted_tokens,
139        omitted,
140        omission_summary,
141        coverage,
142        warnings,
143    } = params;
144    let task_hash = blake3::hash(request.task.as_bytes()).to_hex().to_string();
145    let receipt = EvidenceReceipt {
146        task_fingerprint: task_hash[..32].to_string(),
147        fragment_hashes,
148    };
149    let meta = ResponseMeta {
150        repository_id: String::new(),
151        repository_generation,
152        freshness: Freshness::Current,
153        index_scope: crate::model::IndexScopeMode::Full,
154        index_scope_digest: None,
155        source_tokens: emitted_tokens,
156        protocol_tokens: 0,
157        path_and_metadata_tokens: 0,
158        total_response_tokens: 0,
159        tokenizer: tokenizer.name().into(),
160        token_count_exact: tokenizer.is_exact(),
161        receipt_id: None,
162        receipt_suppressed_exact: 0,
163        receipt_suppressed_overlap: 0,
164        receipt_near_duplicates: 0,
165        next_cursor: None,
166    };
167    let mut response = ContextResponse {
168        effective_response_profile: if request.explain_diagnostics {
169            ContextResponseProfile::Explain
170        } else {
171            ContextResponseProfile::Balanced
172        },
173        workflow: crate::model::ContextWorkflow::Implementation,
174        workflow_receipt: None,
175        plan,
176        fragments,
177        receipt,
178        diff_scope: None,
179        omitted,
180        omission_summary,
181        coverage,
182        routing: None,
183        handoff_manifest: None,
184        provenance: None,
185        warnings,
186        meta,
187    };
188    let accounting = tokens::response_token_accounting(&response, emitted_tokens, &tokenizer)
189        .expect("context response metadata is serializable");
190    response.meta.protocol_tokens = accounting.protocol_tokens;
191    response.meta.path_and_metadata_tokens = accounting.path_and_metadata_tokens;
192    response.meta.total_response_tokens = accounting.total_response_tokens;
193    response
194}
