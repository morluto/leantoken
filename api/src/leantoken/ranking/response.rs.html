2pub(in crate::ranking) fn build_context_plan(
3    request: &ContextRequest,
4    selected: &[ScoredCandidate],
5    candidate_paths_total: usize,
6    estimated_source_tokens: usize,
7    generated_artifact_warning: bool,
8    result_complete: bool,
9) -> Option<ContextQueryPlan> {
10    request.plan_only.then(|| {
11        let minimum_fragments = request.minimum_fragments_per_focus_path.unwrap_or(1);
12        let focus_coverage = request
13            .focus_paths
14            .iter()
15            .map(|pattern| {
16                let matcher = PathMatcher::new(std::slice::from_ref(pattern))
17                    .expect("focus paths are validated at request admission");
18                let candidate_fragments = selected
19                    .iter()
20                    .filter(|candidate| matcher.is_match(&candidate.candidate.path))
21                    .count();
22                ContextPlanFocusCoverage {
23                    pattern: pattern.clone(),
24                    candidate_fragments,
25                    minimum_fragments,
26                    satisfied: candidate_fragments >= minimum_fragments,
27                }
28            })
29            .collect();
30        let candidates = selected
31            .iter()
32            .map(|scored| ContextPlanCandidate {
33                path: scored.candidate.path.clone(),
34                start_line: scored.candidate.start_line,
35                end_line: scored.candidate.end_line,
36                target_start_line: scored.candidate.target_start_line,
37                target_end_line: scored.candidate.target_end_line,
38                truncated: scored.candidate.target_truncated(),
39                representation: scored.candidate.representation.clone(),
40                score: (scored.score * 10_000.0).round() / 10_000.0,
41                reasons: scored
42                    .candidate
43                    .reason()
44                    .split("; ")
45                    .map(str::to_owned)
46                    .collect(),
47                estimated_tokens: scored.token_count,
48            })
49            .collect();
50        ContextQueryPlan {
51            candidates,
52            candidate_paths_total,
53            estimated_source_tokens,
54            focus_coverage,
55            generated_artifact_warning,
56            result_complete,
57        }
58    })
59}

61pub(in crate::ranking) fn materialize_context_fragments(
62    request: &ContextRequest,
63    selected: &[ScoredCandidate],
64    estimated_source_tokens: usize,
65) -> (Vec<ContextFragment>, Vec<String>, usize) {
66    if request.plan_only {
67        return (Vec::new(), Vec::new(), 0);
68    }
69    let fragments = selected
70        .iter()
71        .map(|scored| ContextFragment {
72            path: scored.candidate.path.clone(),
73            start_line: scored.candidate.start_line,
74            end_line: scored.candidate.end_line,
75            target_start_line: scored.candidate.target_start_line,
76            target_end_line: scored.candidate.target_end_line,
77            truncated: scored.candidate.target_truncated(),
78            representation: scored.candidate.representation.clone(),
79            content: scored.candidate.content.clone(),
80            content_hash: scored.content_hash.clone(),
81            score: (scored.score * 10_000.0).round() / 10_000.0,
82            reason: scored.candidate.reason(),
83            token_count: scored.token_count,
84        })
85        .collect();
86    let fragment_hashes = selected
87        .iter()
88        .map(|scored| scored.content_hash.clone())
89        .collect();
90    (fragments, fragment_hashes, estimated_source_tokens)
91}

107pub(in crate::ranking) fn finalize_context_response(
108    params: FinalizeContextParams<'_>,
109) -> ContextResponse {
110    let FinalizeContextParams {
111        request,
112        repository_generation,
113        tokenizer,
114        plan,
115        fragments,
116        fragment_hashes,
117        emitted_tokens,
118        omitted,
119        omission_summary,
120        coverage,
121        warnings,
122    } = params;
123    let task_hash = blake3::hash(request.task.as_bytes()).to_hex().to_string();
124    let receipt = EvidenceReceipt {
125        task_fingerprint: task_hash[..32].to_string(),
126        fragment_hashes,
127    };
128    let meta = ResponseMeta {
129        repository_id: String::new(),
130        repository_generation,
131        freshness: Freshness::Current,
132        index_scope: crate::model::IndexScopeMode::Full,
133        index_scope_digest: None,
134        source_tokens: emitted_tokens,
135        protocol_tokens: 0,
136        path_and_metadata_tokens: 0,
137        total_response_tokens: 0,
138        tokenizer: tokenizer.name().into(),
139        token_count_exact: tokenizer.is_exact(),
140        receipt_id: None,
141        receipt_suppressed_exact: 0,
142        receipt_suppressed_overlap: 0,
143        receipt_near_duplicates: 0,
144        next_cursor: None,
145    };
146    let mut response = ContextResponse {
147        effective_response_profile: if request.explain_diagnostics {
148            ContextResponseProfile::Explain
149        } else {
150            ContextResponseProfile::Balanced
151        },
152        workflow: crate::model::ContextWorkflow::Implementation,
153        workflow_receipt: None,
154        plan,
155        fragments,
156        receipt,
157        diff_scope: None,
158        omitted,
159        omission_summary,
160        coverage,
161        routing: None,
162        handoff_manifest: None,
163        warnings,
164        meta,
165    };
166    let accounting = tokens::response_token_accounting(&response, emitted_tokens, &tokenizer)
167        .expect("context response metadata is serializable");
168    response.meta.protocol_tokens = accounting.protocol_tokens;
169    response.meta.path_and_metadata_tokens = accounting.path_and_metadata_tokens;
170    response.meta.total_response_tokens = accounting.total_response_tokens;
171    response
172}
