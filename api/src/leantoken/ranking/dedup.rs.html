2/// Remove content-identical candidates and candidates whose line ranges
3/// overlap the same file by at least the module's overlap threshold. The higher-scored
4/// copy is kept.
5#[must_use]
6pub fn deduplicate(candidates: Vec<ScoredCandidate>) -> Vec<ScoredCandidate> {
7    deduplicate_with_options(candidates, &Weights::default())
8}

10pub(crate) fn deduplicate_with_options(
11    candidates: Vec<ScoredCandidate>,
12    weights: &Weights,
13) -> Vec<ScoredCandidate> {
14    let mut sorted = candidates;
15    sorted.sort_by(|a, b| {
16        let ord = b.candidate.exact.total_cmp(&a.candidate.exact);
17        if ord != Ordering::Equal {
18            return ord;
19        }
20        let ord = b.score.total_cmp(&a.score);
21        if ord != Ordering::Equal {
22            return ord;
23        }
24        let ord = a.candidate.path.cmp(&b.candidate.path);
25        if ord != Ordering::Equal {
26            return ord;
27        }
28        a.candidate.start_line.cmp(&b.candidate.start_line)
29    });
30
31    let mut kept: Vec<ScoredCandidate> = Vec::with_capacity(sorted.len());
32    let mut seen_hashes: HashMap<(String, String), usize> = HashMap::new();
33    let mut kept_by_path: HashMap<String, Vec<usize>> = HashMap::new();
34
35    for candidate in sorted {
36        let hash_key = (
37            candidate.candidate.path.clone(),
38            candidate.content_hash.clone(),
39        );
40        if let Some(existing) = seen_hashes.get(&hash_key).copied() {
41            merge_scored_candidate(&mut kept[existing], &candidate, weights);
42            continue;
43        }
44
45        let candidate_lines = candidate.candidate.line_count();
46        let duplicate = kept_by_path
47            .get(&candidate.candidate.path)
48            .and_then(|indices| {
49                indices.iter().copied().find(|&index| {
50                    let existing = &kept[index];
51
52                    // Non-overlapping ranges cannot be duplicates.
53                    if candidate.candidate.end_line < existing.candidate.start_line
54                        || candidate.candidate.start_line > existing.candidate.end_line
55                    {
56                        return false;
57                    }
58
59                    let overlap_start = candidate
60                        .candidate
61                        .start_line
62                        .max(existing.candidate.start_line);
63                    let overlap_end = candidate
64                        .candidate
65                        .end_line
66                        .min(existing.candidate.end_line);
67                    let overlap_lines = overlap_end - overlap_start + 1;
68                    let min_lines = candidate_lines.min(existing.candidate.line_count());
69
70                    overlap_lines >= min_lines.div_ceil(2)
71                })
72            });
73        if let Some(existing) = duplicate {
74            merge_scored_candidate(&mut kept[existing], &candidate, weights);
75            continue;
76        }
77
78        let kept_index = kept.len();
79        seen_hashes.insert(hash_key, kept_index);
80        kept_by_path
81            .entry(candidate.candidate.path.clone())
82            .or_default()
83            .push(kept_index);
84        kept.push(candidate);
85    }
86
87    kept.sort_by(|a, b| {
88        b.score
89            .total_cmp(&a.score)
90            .then_with(|| a.candidate.path.cmp(&b.candidate.path))
91            .then_with(|| a.candidate.start_line.cmp(&b.candidate.start_line))
92    });
93    kept
94}

96pub(in crate::ranking) fn merge_scored_candidate(
97    existing: &mut ScoredCandidate,
98    duplicate: &ScoredCandidate,
99    weights: &Weights,
100) {
101    merge_candidate_signals(&mut existing.candidate, &duplicate.candidate);
102    existing.score = existing.candidate.score(weights, existing.token_count);
103    existing.marginal_score = existing.score / bounded_count_f64(existing.token_count);
104}

106pub(in crate::ranking) fn merge_candidate_signals(existing: &mut Candidate, duplicate: &Candidate) {
107    for kind in &duplicate.match_kinds {
108        if !existing.match_kinds.contains(kind) {
109            existing.match_kinds.push(kind.clone());
110        }
111    }
112    for concept in &duplicate.concepts {
113        if !existing.concepts.contains(concept) {
114            existing.concepts.push(concept.clone());
115        }
116    }
117    existing.concept_weight = existing.concept_weight.max(duplicate.concept_weight);
118    if existing.target_start_line.is_none() && duplicate.target_start_line.is_some() {
119        existing.symbol_name.clone_from(&duplicate.symbol_name);
120        existing.target_start_line = duplicate.target_start_line;
121        existing.target_end_line = duplicate.target_end_line;
122        existing
123            .representation
124            .clone_from(&duplicate.representation);
125    } else if existing.symbol_name.is_none() {
126        existing.symbol_name.clone_from(&duplicate.symbol_name);
127    }
128    existing.exact = existing.exact.max(duplicate.exact);
129    existing.symbol = existing.symbol.max(duplicate.symbol);
130    existing.reference = existing.reference.max(duplicate.reference);
131    existing.bm25 = existing.bm25.max(duplicate.bm25);
132    existing.path_score = existing.path_score.max(duplicate.path_score);
133    existing.lexical_frequency_penalty = existing
134        .lexical_frequency_penalty
135        .min(duplicate.lexical_frequency_penalty);
136    existing.size_score = existing.size_score.max(duplicate.size_score);
137    existing.focus_boost = existing.focus_boost.max(duplicate.focus_boost);
138    existing.import_boost = existing.import_boost.max(duplicate.import_boost);
139    existing.change_boost = existing.change_boost.max(duplicate.change_boost);
140}
