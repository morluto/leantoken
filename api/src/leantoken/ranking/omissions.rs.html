2/// Divisor for the per-file diversity cap. A 1,200-token context may include
3/// two non-overlapping regions from one file, while tiny budgets still prefer
4/// breadth.
5pub(in crate::ranking) const DIVERSITY_DIVISOR: usize = 600;
6pub(in crate::ranking) const MAX_OMITTED_DETAILS: usize = 1;
7pub(in crate::ranking) const MAX_OMISSION_FACETS: usize = 12;
8pub(in crate::ranking) const MIN_RELATIVE_CONTEXT_SCORE: f64 = 0.25;

10pub(in crate::ranking) fn increment_facet(
11    counts: &mut HashMap<String, usize>,
12    value: impl Into<String>,
13) {
14    let count = counts.entry(value.into()).or_default();
15    *count = count.saturating_add(1);
16}

18pub(in crate::ranking) fn bounded_facets(
19    counts: HashMap<String, usize>,
20) -> Vec<ContextOmissionFacet> {
21    let mut facets = counts
22        .into_iter()
23        .map(|(value, count)| ContextOmissionFacet { value, count })
24        .collect::<Vec<_>>();
25    facets.sort_by(|left, right| {
26        right
27            .count
28            .cmp(&left.count)
29            .then_with(|| left.value.cmp(&right.value))
30    });
31    if facets.len() > MAX_OMISSION_FACETS {
32        let other = facets
33            .drain(MAX_OMISSION_FACETS - 1..)
34            .map(|facet| facet.count)
35            .sum();
36        facets.push(ContextOmissionFacet {
37            value: "[other]".into(),
38            count: other,
39        });
40    }
41    facets
42}

44pub(in crate::ranking) fn candidate_file_type(path: &str) -> String {
45    Path::new(path)
46        .extension()
47        .and_then(|extension| extension.to_str())
48        .filter(|extension| !extension.is_empty())
49        .map_or_else(
50            || "[no extension]".into(),
51            |extension| format!(".{}", extension.to_ascii_lowercase()),
52        )
53}

55pub(in crate::ranking) fn score_band(score: f64) -> &'static str {
56    if score >= 1.0 {
57        "score >= 1.0"
58    } else if score >= 0.5 {
59        "0.5 <= score < 1.0"
60    } else if score > 0.0 {
61        "0 < score < 0.5"
62    } else {
63        "score = 0"
64    }
65}

67pub(in crate::ranking) fn summarize_omissions(
68    path_omitted: &[ScoredCandidate],
69    known_omitted: &[ScoredCandidate],
70    limit_omitted: &[ScoredCandidate],
71    prefiltered_path_omissions: &[String],
72    focus_paths: &PathMatcher,
73    changed_paths: &HashSet<&str>,
74    explain_diagnostics: bool,
75) -> ContextOmissionSummary {
76    let path_excluded = path_omitted
77        .len()
78        .saturating_add(prefiltered_path_omissions.len());
79    let known_hash = known_omitted.len();
80    let budget_or_result_limit = limit_omitted.len();
81    if !explain_diagnostics {
82        return ContextOmissionSummary {
83            path_excluded,
84            known_hash,
85            budget_or_result_limit,
86            ..ContextOmissionSummary::default()
87        };
88    }
89
90    let mut paths = HashMap::new();
91    let mut file_types = HashMap::new();
92    let mut score_bands = HashMap::new();
93    let mut focused = 0usize;
94    let mut changed = 0usize;
95
96    let mut record = |path: &str, score: Option<f64>| {
97        increment_facet(&mut paths, path);
98        increment_facet(&mut file_types, candidate_file_type(path));
99        increment_facet(&mut score_bands, score.map_or("not scored", score_band));
100        focused = focused.saturating_add(usize::from(focus_paths.is_match(path)));
101        changed = changed.saturating_add(usize::from(changed_paths.contains(path)));
102    };
103    for candidate in path_omitted
104        .iter()
105        .chain(known_omitted)
106        .chain(limit_omitted)
107    {
108        record(&candidate.candidate.path, Some(candidate.score));
109    }
110    for path in prefiltered_path_omissions {
111        record(path, None);
112    }
113
114    let total = path_excluded
115        .saturating_add(known_hash)
116        .saturating_add(budget_or_result_limit);
117    let mut reasons = HashMap::new();
118    if path_excluded > 0 {
119        reasons.insert("path_excluded".into(), path_excluded);
120    }
121    if known_hash > 0 {
122        reasons.insert("known_hash".into(), known_hash);
123    }
124    if budget_or_result_limit > 0 {
125        reasons.insert("budget_or_result_limit".into(), budget_or_result_limit);
126    }
127
128    ContextOmissionSummary {
129        path_excluded,
130        known_hash,
131        budget_or_result_limit,
132        by_path: bounded_facets(paths),
133        by_language_or_file_type: bounded_facets(file_types),
134        by_reason: bounded_facets(reasons),
135        by_score_band: bounded_facets(score_bands),
136        focused,
137        not_focused: total.saturating_sub(focused),
138        changed,
139        not_changed: total.saturating_sub(changed),
140    }
141}

143pub(in crate::ranking) struct BuildOmissionsParams<'a> {
144    pub request: &'a ContextRequest,
145    pub path_omitted: Vec<ScoredCandidate>,
146    pub known_omitted: Vec<ScoredCandidate>,
147    pub limit_omitted: Vec<ScoredCandidate>,
148    pub prefiltered_path_omissions: &'a [String],
149    pub focus_paths: &'a PathMatcher,
150    pub changed_paths: &'a HashSet<&'a str>,
151    pub generated_artifact_warning: bool,
152}

154pub(in crate::ranking) fn build_context_omissions(
155    params: BuildOmissionsParams<'_>,
156) -> (ContextOmissionSummary, Vec<OmittedCandidate>, Vec<String>) {
157    let BuildOmissionsParams {
158        request,
159        path_omitted,
160        known_omitted,
161        mut limit_omitted,
162        prefiltered_path_omissions,
163        focus_paths,
164        changed_paths,
165        generated_artifact_warning,
166    } = params;
167    let omission_summary = summarize_omissions(
168        &path_omitted,
169        &known_omitted,
170        &limit_omitted,
171        prefiltered_path_omissions,
172        focus_paths,
173        changed_paths,
174        request.explain_diagnostics,
175    );
176    let mut omitted = path_omitted
177        .into_iter()
178        .map(|scored| OmittedCandidate {
179            path: scored.candidate.path,
180            start_line: scored.candidate.start_line,
181            end_line: scored.candidate.end_line,
182            reason: "path excluded".to_string(),
183        })
184        .chain(known_omitted.into_iter().map(|scored| OmittedCandidate {
185            path: scored.candidate.path,
186            start_line: scored.candidate.start_line,
187            end_line: scored.candidate.end_line,
188            reason: "known hash".to_string(),
189        }))
190        .collect::<Vec<_>>();
191    omitted.extend(limit_omitted.drain(..).map(|scored| OmittedCandidate {
192        path: scored.candidate.path,
193        start_line: scored.candidate.start_line,
194        end_line: scored.candidate.end_line,
195        reason: "budget or result limit".to_string(),
196    }));
197    if !request.explain_diagnostics {
198        omitted.clear();
199    }
200    omitted.truncate(MAX_OMITTED_DETAILS);
201
202    let omitted_count = omission_summary
203        .path_excluded
204        .saturating_add(omission_summary.known_hash)
205        .saturating_add(omission_summary.budget_or_result_limit);
206    let mut warnings = Vec::new();
207    if omitted_count > 0 {
208        warnings.push(format!("{omitted_count} omitted"));
209    }
210    if request.plan_only && generated_artifact_warning {
211        warnings.push(
212            "generated-artifact candidates matched context exclusion defaults; review their explicit inclusion before materializing source"
213                .into(),
214        );
215    }
216    (omission_summary, omitted, warnings)
217}
