2pub(in crate::ranking) fn greedy_select(
3    candidates: Vec<ScoredCandidate>,
4    budget: usize,
5    max_per_file: usize,
6    max_fragments: usize,
7    initial_file_counts: HashMap<String, usize>,
8) -> (Vec<ScoredCandidate>, Vec<ScoredCandidate>) {
9    let mut pool = candidates;
10    pool.sort_by(compare_utility);
11    let confidence_floor = pool.first().map_or(0.0, |candidate| {
12        candidate.score * MIN_RELATIVE_CONTEXT_SCORE
13    });
14
15    let mut selected = Vec::new();
16    let mut deferred = Vec::with_capacity(pool.len());
17    let mut omitted: Vec<ScoredCandidate> = Vec::with_capacity(pool.len());
18    let mut used_tokens = 0usize;
19    let mut file_counts: HashMap<String, usize> = initial_file_counts;
20    let mut covered_concepts = HashSet::new();
21    let mut concept_representations = HashSet::new();
22    let mut concept_paths = HashMap::new();
23
24    for candidate in pool {
25        let adds_concept = candidate
26            .candidate
27            .concepts
28            .iter()
29            .any(|concept| !covered_concepts.contains(concept));
30        if !adds_concept || candidate.candidate.concept_weight < 1.0 {
31            deferred.push(candidate);
32            continue;
33        }
34        let file_count = *file_counts.get(&candidate.candidate.path).unwrap_or(&0);
35        let remaining = budget.saturating_sub(used_tokens);
36
37        if candidate_fits(
38            &candidate,
39            remaining,
40            file_count,
41            max_per_file,
42            selected.len(),
43            max_fragments,
44        ) {
45            covered_concepts.extend(candidate.candidate.concepts.iter().cloned());
46            concept_representations.extend(
47                candidate
48                    .candidate
49                    .concepts
50                    .iter()
51                    .map(|concept| (concept.clone(), candidate.candidate.representation.clone())),
52            );
53            for concept in &candidate.candidate.concepts {
54                concept_paths
55                    .entry(concept.clone())
56                    .or_insert_with(|| candidate.candidate.path.clone());
57            }
58            push_selected(candidate, &mut selected, &mut used_tokens, &mut file_counts);
59        } else {
60            deferred.push(candidate);
61        }
62    }
63
64    deferred.sort_by(|left, right| {
65        let left_same_path = left.candidate.concepts.iter().any(|concept| {
66            concept_paths
67                .get(concept)
68                .is_some_and(|path| path == &left.candidate.path)
69        });
70        let right_same_path = right.candidate.concepts.iter().any(|concept| {
71            concept_paths
72                .get(concept)
73                .is_some_and(|path| path == &right.candidate.path)
74        });
75        right_same_path
76            .cmp(&left_same_path)
77            .then_with(|| compare_utility(left, right))
78    });
79    let mut remaining = Vec::with_capacity(deferred.len());
80    for candidate in deferred {
81        let adds_decisive_view = candidate.candidate.concept_weight >= 1.8
82            && candidate.candidate.concepts.iter().any(|concept| {
83                covered_concepts.contains(concept)
84                    && !concept_representations
85                        .contains(&(concept.clone(), candidate.candidate.representation.clone()))
86            });
87        let file_count = *file_counts.get(&candidate.candidate.path).unwrap_or(&0);
88        let remaining_tokens = budget.saturating_sub(used_tokens);
89        if adds_decisive_view
90            && candidate_fits(
91                &candidate,
92                remaining_tokens,
93                file_count,
94                max_per_file,
95                selected.len(),
96                max_fragments,
97            )
98        {
99            concept_representations.extend(
100                candidate
101                    .candidate
102                    .concepts
103                    .iter()
104                    .map(|concept| (concept.clone(), candidate.candidate.representation.clone())),
105            );
106            push_selected(candidate, &mut selected, &mut used_tokens, &mut file_counts);
107        } else {
108            remaining.push(candidate);
109        }
110    }
111
112    let mut fill = Vec::with_capacity(remaining.len());
113    for candidate in remaining {
114        let adds_concept = candidate
115            .candidate
116            .concepts
117            .iter()
118            .any(|concept| !covered_concepts.contains(concept));
119        let file_count = *file_counts.get(&candidate.candidate.path).unwrap_or(&0);
120        let remaining_tokens = budget.saturating_sub(used_tokens);
121        let confident =
122            candidate.candidate.concept_weight >= 1.0 || candidate.score >= confidence_floor;
123        if adds_concept
124            && confident
125            && candidate_fits(
126                &candidate,
127                remaining_tokens,
128                file_count,
129                max_per_file,
130                selected.len(),
131                max_fragments,
132            )
133        {
134            covered_concepts.extend(candidate.candidate.concepts.iter().cloned());
135            push_selected(candidate, &mut selected, &mut used_tokens, &mut file_counts);
136        } else {
137            fill.push(candidate);
138        }
139    }
140
141    for candidate in fill {
142        if candidate.candidate.concept_weight < 1.0 && candidate.score < confidence_floor {
143            omitted.push(candidate);
144            continue;
145        }
146        let file_count = *file_counts.get(&candidate.candidate.path).unwrap_or(&0);
147        let remaining = budget.saturating_sub(used_tokens);
148        if candidate_fits(
149            &candidate,
150            remaining,
151            file_count,
152            max_per_file,
153            selected.len(),
154            max_fragments,
155        ) {
156            push_selected(candidate, &mut selected, &mut used_tokens, &mut file_counts);
157        } else {
158            omitted.push(candidate);
159        }
160    }
161
162    (selected, omitted)
163}

165pub(in crate::ranking) fn candidate_fits(
166    candidate: &ScoredCandidate,
167    remaining_tokens: usize,
168    file_count: usize,
169    max_per_file: usize,
170    selected_count: usize,
171    max_fragments: usize,
172) -> bool {
173    candidate.token_count <= remaining_tokens
174        && file_count < max_per_file
175        && selected_count < max_fragments
176}

178pub(in crate::ranking) fn push_selected(
179    candidate: ScoredCandidate,
180    selected: &mut Vec<ScoredCandidate>,
181    used_tokens: &mut usize,
182    file_counts: &mut HashMap<String, usize>,
183) {
184    *used_tokens += candidate.token_count;
185    *file_counts
186        .entry(candidate.candidate.path.clone())
187        .or_insert(0) += 1;
188    selected.push(candidate);
189}

191pub(in crate::ranking) fn compare_utility(a: &ScoredCandidate, b: &ScoredCandidate) -> Ordering {
192    let ord = b.score.total_cmp(&a.score);
193    if ord != Ordering::Equal {
194        return ord;
195    }
196
197    let ord = b.marginal_score.total_cmp(&a.marginal_score);
198    if ord != Ordering::Equal {
199        return ord;
200    }
201
202    let ord = a.token_count.cmp(&b.token_count);
203    if ord != Ordering::Equal {
204        return ord;
205    }
206
207    let ord = a.candidate.path.cmp(&b.candidate.path);
208    if ord != Ordering::Equal {
209        return ord;
210    }
211
212    a.candidate.start_line.cmp(&b.candidate.start_line)
213}
