2pub(in crate::ranking) fn greedy_select(
3    candidates: Vec<ScoredCandidate>,
4    budget: usize,
5    max_per_file: usize,
6    max_fragments: usize,
7    initial_file_counts: HashMap<String, usize>,
8) -> (Vec<ScoredCandidate>, Vec<ScoredCandidate>) {
9    let mut pool = candidates;
10    pool.sort_by(compare_utility);
11    let confidence_floor = pool.first().map_or(0.0, |candidate| {
12        candidate.score * MIN_RELATIVE_CONTEXT_SCORE
13    });
14
15    let mut selected = Vec::new();
16    let mut deferred = Vec::with_capacity(pool.len());
17    let mut omitted: Vec<ScoredCandidate> = Vec::with_capacity(pool.len());
18    let mut used_tokens = 0usize;
19    let mut file_counts: HashMap<String, usize> = initial_file_counts;
20    let mut covered_concepts = HashSet::new();
21    let mut concept_representations = HashSet::new();
22    let mut concept_paths = HashMap::new();
23    let mut evidence_counts = EvidenceQuotaCounts::default();
24
25    let primary_candidate_fits = |candidate: &ScoredCandidate, path_class| {
26        context_path_class(&candidate.candidate.path) == path_class
27            && evidence_counts.allows(candidate, max_fragments)
28            && candidate_fits(
29                candidate,
30                budget,
31                *file_counts.get(&candidate.candidate.path).unwrap_or(&0),
32                max_per_file,
33                selected.len(),
34                max_fragments,
35            )
36    };
37    let primary_position_for = |path_class| {
38        let exact_position = pool.iter().position(|candidate| {
39            primary_candidate_fits(candidate, path_class)
40                && carries_specific_exact_primary_change(&candidate.candidate)
41                && carries_facet(&candidate.candidate, "primary_change")
42        });
43        let specific_baseline = pool.iter().find(|candidate| {
44            primary_candidate_fits(candidate, path_class)
45                && carries_specific_primary_change(&candidate.candidate)
46        });
47        exact_position.or_else(|| {
48            specific_baseline
49                .and_then(|baseline| {
50                    let owner_path = pool
51                        .iter()
52                        .enumerate()
53                        .filter(|(_, candidate)| {
54                            primary_candidate_fits(candidate, path_class)
55                                && carries_specific_primary_change(&candidate.candidate)
56                                && candidate.candidate.representation
57                                    == baseline.candidate.representation
58                                && carries_all_facet_values(
59                                    &candidate.candidate,
60                                    &baseline.candidate,
61                                    "primary_change",
62                                )
63                        })
64                        .max_by_key(|(position, candidate)| {
65                            (
66                                facet_value_count(&candidate.candidate, "primary_change"),
67                                std::cmp::Reverse(*position),
68                            )
69                        })
70                        .map(|(_, candidate)| candidate.candidate.path.as_str())?;
71                    pool.iter().position(|candidate| {
72                        candidate.candidate.path == owner_path
73                            && primary_candidate_fits(candidate, path_class)
74                            && carries_specific_primary_change(&candidate.candidate)
75                            && candidate.candidate.representation
76                                == baseline.candidate.representation
77                            && carries_all_facet_values(
78                                &candidate.candidate,
79                                &baseline.candidate,
80                                "primary_change",
81                            )
82                    })
83                })
84                .or_else(|| {
85                    let baseline = pool.iter().find(|candidate| {
86                        primary_candidate_fits(candidate, path_class)
87                            && carries_facet(&candidate.candidate, "primary_change")
88                    })?;
89                    let owner_path = pool
90                        .iter()
91                        .enumerate()
92                        .filter(|(_, candidate)| {
93                            primary_candidate_fits(candidate, path_class)
94                                && carries_facet(&candidate.candidate, "primary_change")
95                                && candidate.candidate.representation
96                                    == baseline.candidate.representation
97                                && carries_all_facet_values(
98                                    &candidate.candidate,
99                                    &baseline.candidate,
100                                    "primary_change",
101                                )
102                        })
103                        .max_by_key(|(position, candidate)| {
104                            (
105                                facet_value_count(&candidate.candidate, "primary_change"),
106                                std::cmp::Reverse(*position),
107                            )
108                        })
109                        .map(|(_, candidate)| candidate.candidate.path.as_str())?;
110                    pool.iter().position(|candidate| {
111                        candidate.candidate.path == owner_path
112                            && primary_candidate_fits(candidate, path_class)
113                            && carries_facet(&candidate.candidate, "primary_change")
114                            && candidate.candidate.representation
115                                == baseline.candidate.representation
116                            && carries_all_facet_values(
117                                &candidate.candidate,
118                                &baseline.candidate,
119                                "primary_change",
120                            )
121                    })
122                })
123        })
124    };
125    let primary_position = primary_position_for(ContextPathClass::Production)
126        .or_else(|| primary_position_for(ContextPathClass::Supporting));
127    let mut primary_owner = None;
128    if let Some(position) = primary_position {
129        let candidate = pool.remove(position);
130        primary_owner = Some((
131            candidate.candidate.path.clone(),
132            candidate.candidate.concepts.clone(),
133        ));
134        record_context_selection(
135            &candidate,
136            &mut covered_concepts,
137            &mut concept_representations,
138            &mut concept_paths,
139            &mut evidence_counts,
140        );
141        push_selected(candidate, &mut selected, &mut used_tokens, &mut file_counts);
142    }
143
144    let eligible_tests = pool
145        .iter()
146        .enumerate()
147        .filter(|(_, candidate)| {
148            context_path_class(&candidate.candidate.path) == ContextPathClass::Test
149                && (carries_specific_exact_atom(&candidate.candidate)
150                    || primary_owner.as_ref().is_some_and(|(path, _)| {
151                        owner_test_path_affinity(path, &candidate.candidate.path) > 0
152                    })
153                    || candidate
154                        .candidate
155                        .concepts
156                        .iter()
157                        .any(|concept| covered_concepts.contains(concept)))
158                && evidence_counts.allows(candidate, max_fragments)
159                && candidate_fits(
160                    candidate,
161                    budget.saturating_sub(used_tokens),
162                    *file_counts.get(&candidate.candidate.path).unwrap_or(&0),
163                    max_per_file,
164                    selected.len(),
165                    max_fragments,
166                )
167        })
168        .collect::<Vec<_>>();
169    let test_position = eligible_tests
170        .iter()
171        .map(|(position, candidate)| {
172            let owner_affinity = primary_owner.as_ref().map_or(0, |(path, _)| {
173                owner_test_path_affinity(path, &candidate.candidate.path)
174            });
175            (
176                *position,
177                owner_affinity,
178                usize::from(carries_specific_exact_atom(&candidate.candidate)),
179            )
180        })
181        .max_by_key(|(position, owner_affinity, exact)| {
182            (
183                usize::from(*owner_affinity >= 3),
184                *exact,
185                *owner_affinity,
186                std::cmp::Reverse(*position),
187            )
188        })
189        .map(|(position, _, _)| position);
190    if let Some(position) = test_position {
191        let candidate = pool.remove(position);
192        record_context_selection(
193            &candidate,
194            &mut covered_concepts,
195            &mut concept_representations,
196            &mut concept_paths,
197            &mut evidence_counts,
198        );
199        push_selected(candidate, &mut selected, &mut used_tokens, &mut file_counts);
200    }
201
202    if let Some(position) = pool.iter().position(|candidate| {
203        context_path_class(&candidate.candidate.path) != ContextPathClass::Auxiliary
204            && carries_facet(&candidate.candidate, "preserve_constraint")
205            && evidence_counts.allows(candidate, max_fragments)
206            && candidate_fits(
207                candidate,
208                budget.saturating_sub(used_tokens),
209                *file_counts.get(&candidate.candidate.path).unwrap_or(&0),
210                max_per_file,
211                selected.len(),
212                max_fragments,
213            )
214    }) {
215        let candidate = pool.remove(position);
216        record_context_selection(
217            &candidate,
218            &mut covered_concepts,
219            &mut concept_representations,
220            &mut concept_paths,
221            &mut evidence_counts,
222        );
223        push_selected(candidate, &mut selected, &mut used_tokens, &mut file_counts);
224    }
225
226    for candidate in pool {
227        let adds_concept = candidate
228            .candidate
229            .concepts
230            .iter()
231            .any(|concept| !covered_concepts.contains(concept));
232        if !adds_concept || candidate.candidate.concept_weight < 1.0 {
233            deferred.push(candidate);
234            continue;
235        }
236        let file_count = *file_counts.get(&candidate.candidate.path).unwrap_or(&0);
237        let remaining = budget.saturating_sub(used_tokens);
238
239        if evidence_counts.allows(&candidate, max_fragments)
240            && candidate_fits(
241                &candidate,
242                remaining,
243                file_count,
244                max_per_file,
245                selected.len(),
246                max_fragments,
247            )
248        {
249            covered_concepts.extend(candidate.candidate.concepts.iter().cloned());
250            concept_representations.extend(
251                candidate
252                    .candidate
253                    .concepts
254                    .iter()
255                    .map(|concept| (concept.clone(), candidate.candidate.representation.clone())),
256            );
257            for concept in &candidate.candidate.concepts {
258                concept_paths
259                    .entry(concept.clone())
260                    .or_insert_with(|| candidate.candidate.path.clone());
261            }
262            evidence_counts.record(&candidate);
263            push_selected(candidate, &mut selected, &mut used_tokens, &mut file_counts);
264        } else {
265            deferred.push(candidate);
266        }
267    }
268
269    deferred.sort_by(|left, right| {
270        let left_same_path = left.candidate.concepts.iter().any(|concept| {
271            concept_paths
272                .get(concept)
273                .is_some_and(|path| path == &left.candidate.path)
274        });
275        let right_same_path = right.candidate.concepts.iter().any(|concept| {
276            concept_paths
277                .get(concept)
278                .is_some_and(|path| path == &right.candidate.path)
279        });
280        right_same_path
281            .cmp(&left_same_path)
282            .then_with(|| compare_utility(left, right))
283    });
284    let mut remaining = Vec::with_capacity(deferred.len());
285    for candidate in deferred {
286        let adds_decisive_view = candidate.candidate.concept_weight >= 1.8
287            && candidate.candidate.concepts.iter().any(|concept| {
288                covered_concepts.contains(concept)
289                    && !concept_representations
290                        .contains(&(concept.clone(), candidate.candidate.representation.clone()))
291            });
292        let file_count = *file_counts.get(&candidate.candidate.path).unwrap_or(&0);
293        let remaining_tokens = budget.saturating_sub(used_tokens);
294        if adds_decisive_view
295            && evidence_counts.allows(&candidate, max_fragments)
296            && candidate_fits(
297                &candidate,
298                remaining_tokens,
299                file_count,
300                max_per_file,
301                selected.len(),
302                max_fragments,
303            )
304        {
305            concept_representations.extend(
306                candidate
307                    .candidate
308                    .concepts
309                    .iter()
310                    .map(|concept| (concept.clone(), candidate.candidate.representation.clone())),
311            );
312            evidence_counts.record(&candidate);
313            push_selected(candidate, &mut selected, &mut used_tokens, &mut file_counts);
314        } else {
315            remaining.push(candidate);
316        }
317    }
318
319    let mut fill = Vec::with_capacity(remaining.len());
320    for candidate in remaining {
321        let adds_concept = candidate
322            .candidate
323            .concepts
324            .iter()
325            .any(|concept| !covered_concepts.contains(concept));
326        let file_count = *file_counts.get(&candidate.candidate.path).unwrap_or(&0);
327        let remaining_tokens = budget.saturating_sub(used_tokens);
328        let confident =
329            candidate.candidate.concept_weight >= 1.0 || candidate.score >= confidence_floor;
330        if adds_concept
331            && confident
332            && evidence_counts.allows(&candidate, max_fragments)
333            && candidate_fits(
334                &candidate,
335                remaining_tokens,
336                file_count,
337                max_per_file,
338                selected.len(),
339                max_fragments,
340            )
341        {
342            covered_concepts.extend(candidate.candidate.concepts.iter().cloned());
343            evidence_counts.record(&candidate);
344            push_selected(candidate, &mut selected, &mut used_tokens, &mut file_counts);
345        } else {
346            fill.push(candidate);
347        }
348    }
349
350    for candidate in fill {
351        if candidate.candidate.concept_weight < 1.0 && candidate.score < confidence_floor {
352            omitted.push(candidate);
353            continue;
354        }
355        let file_count = *file_counts.get(&candidate.candidate.path).unwrap_or(&0);
356        let remaining = budget.saturating_sub(used_tokens);
357        if evidence_counts.allows(&candidate, max_fragments)
358            && candidate_fits(
359                &candidate,
360                remaining,
361                file_count,
362                max_per_file,
363                selected.len(),
364                max_fragments,
365            )
366        {
367            evidence_counts.record(&candidate);
368            push_selected(candidate, &mut selected, &mut used_tokens, &mut file_counts);
369        } else {
370            omitted.push(candidate);
371        }
372    }
373
374    (selected, omitted)
375}

377#[derive(Default)]
378struct EvidenceQuotaCounts {
379    auxiliary: usize,
380    failures: usize,
381    tests: usize,
382    preserve: usize,
383}
384
385fn record_context_selection(
386    candidate: &ScoredCandidate,
387    covered_concepts: &mut HashSet<String>,
388    concept_representations: &mut HashSet<(String, String)>,
389    concept_paths: &mut HashMap<String, String>,
390    evidence_counts: &mut EvidenceQuotaCounts,
391) {
392    covered_concepts.extend(candidate.candidate.concepts.iter().cloned());
393    concept_representations.extend(
394        candidate
395            .candidate
396            .concepts
397            .iter()
398            .map(|concept| (concept.clone(), candidate.candidate.representation.clone())),
399    );
400    for concept in &candidate.candidate.concepts {
401        concept_paths
402            .entry(concept.clone())
403            .or_insert_with(|| candidate.candidate.path.clone());
404    }
405    evidence_counts.record(candidate);
406}
407
408impl EvidenceQuotaCounts {
409    fn allows(&self, candidate: &ScoredCandidate, max_fragments: usize) -> bool {
410        let auxiliary_limit = usize::from(max_fragments > 0);
411        let failure_limit = max_fragments.min(2);
412        let test_limit = max_fragments.min(2);
413        let preserve_limit = max_fragments.min(2);
414        let path_class = context_path_class(&candidate.candidate.path);
415        (path_class != ContextPathClass::Auxiliary || self.auxiliary < auxiliary_limit)
416            && (!carries_facet(&candidate.candidate, "failure_trace")
417                || self.failures < failure_limit)
418            && (path_class != ContextPathClass::Test || self.tests < test_limit)
419            && (!carries_facet(&candidate.candidate, "preserve_constraint")
420                || self.preserve < preserve_limit)
421    }
422
423    fn record(&mut self, candidate: &ScoredCandidate) {
424        match context_path_class(&candidate.candidate.path) {
425            ContextPathClass::Auxiliary => self.auxiliary += 1,
426            ContextPathClass::Test => self.tests += 1,
427            ContextPathClass::Production | ContextPathClass::Supporting => {}
428        }
429        if carries_facet(&candidate.candidate, "preserve_constraint") {
430            self.preserve += 1;
431        }
432        if carries_facet(&candidate.candidate, "failure_trace") {
433            self.failures += 1;
434        }
435    }
436}

438pub(in crate::ranking) fn candidate_fits(
439    candidate: &ScoredCandidate,
440    remaining_tokens: usize,
441    file_count: usize,
442    max_per_file: usize,
443    selected_count: usize,
444    max_fragments: usize,
445) -> bool {
446    candidate.token_count <= remaining_tokens
447        && file_count < max_per_file
448        && selected_count < max_fragments
449}

451pub(in crate::ranking) fn push_selected(
452    candidate: ScoredCandidate,
453    selected: &mut Vec<ScoredCandidate>,
454    used_tokens: &mut usize,
455    file_counts: &mut HashMap<String, usize>,
456) {
457    *used_tokens += candidate.token_count;
458    *file_counts
459        .entry(candidate.candidate.path.clone())
460        .or_insert(0) += 1;
461    selected.push(candidate);
462}

464pub(in crate::ranking) fn compare_utility(a: &ScoredCandidate, b: &ScoredCandidate) -> Ordering {
465    let ord = b.score.total_cmp(&a.score);
466    if ord != Ordering::Equal {
467        return ord;
468    }
469
470    let ord = b.marginal_score.total_cmp(&a.marginal_score);
471    if ord != Ordering::Equal {
472        return ord;
473    }
474
475    let ord = a.token_count.cmp(&b.token_count);
476    if ord != Ordering::Equal {
477        return ord;
478    }
479
480    let ord = a.candidate.path.cmp(&b.candidate.path);
481    if ord != Ordering::Equal {
482        return ord;
483    }
484
485    a.candidate.start_line.cmp(&b.candidate.start_line)
