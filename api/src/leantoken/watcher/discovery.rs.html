3/// Count every directory that a recursive backend may register before
4/// enabling the watcher. Callback filtering does not reduce kernel watches.
5pub(super) fn inspect_watch_admission(
6    root: &Path,
7    directory_cap: usize,
8    entry_cap: usize,
9    cancellation: &CancellationToken,
10) -> WatchAdmission {
11    use ignore::WalkBuilder;
12    let mut builder = WalkBuilder::new(root);
13    builder.hidden(false);
14    builder.ignore(false);
15    builder.parents(false);
16    builder.git_ignore(false);
17    builder.git_exclude(false);
18    builder.git_global(false);
19    builder.follow_links(false);
20    let walker = builder.build();
21    let mut entries = 0usize;
22    let mut directories = 0usize;
23    for entry in walker {
24        if cancellation.is_cancelled() {
25            return WatchAdmission {
26                entries,
27                directories,
28                complete: false,
29                fallback_reason: Some(WatcherFallbackReason::AdmissionCancelled),
30            };
31        }
32        if entries >= entry_cap {
33            return WatchAdmission {
34                entries,
35                directories,
36                complete: false,
37                fallback_reason: Some(WatcherFallbackReason::AdmissionEntryLimit),
38            };
39        }
40        entries += 1;
41        match entry {
42            Ok(entry) => {
43                if !entry.file_type().map(|ft| ft.is_dir()).unwrap_or(false) {
44                    continue;
45                }
46                directories += 1;
47                if directories > directory_cap {
48                    return WatchAdmission {
49                        entries,
50                        directories,
51                        complete: false,
52                        fallback_reason: Some(WatcherFallbackReason::AdmissionDirectoryLimit),
53                    };
54                }
55            }
56            // Failure to inspect a subtree means the recursive backend's
57            // watch count cannot be proven bounded. Use polling instead.
58            Err(_) => {
59                return WatchAdmission {
60                    entries,
61                    directories,
62                    complete: false,
63                    fallback_reason: Some(WatcherFallbackReason::AdmissionError),
64                };
65            }
66        }
67    }
68    WatchAdmission {
69        entries,
70        directories,
71        complete: true,
72        fallback_reason: None,
73    }
74}

76pub(super) fn relative_path(
77    root: &Path,
78    path: &Path,
79    policy: &DiscoveryPolicy,
80    directory_hint: DirectoryHint,
81) -> std::result::Result<Option<String>, ()> {
82    if !path.starts_with(root) {
83        return Ok(None);
84    }
85    let rel = path.strip_prefix(root).map_err(|_| ())?;
86    let s = checked_slash_path(rel).map_err(|_| ())?;
87    if s.is_empty() {
88        Ok(None)
89    } else if directory_hint == DirectoryHint::Unknown {
90        // Rename notifications can refer to a source that has already
91        // disappeared. Admit the path before generated-tree filtering so the
92        // ambiguity reaches bounded reconciliation.
93        Ok(Some(s))
94    } else if !policy.includes_watch_path(&s, directory_hint == DirectoryHint::Yes || path.is_dir())
95    {
96        Ok(None)
97    } else {
98        Ok(Some(s))
99    }
100}

109pub(super) fn event_path_is_directory(event: &Event) -> DirectoryHint {
110    matches!(
111        event.kind,
112        EventKind::Create(CreateKind::Folder) | EventKind::Remove(RemoveKind::Folder)
113    )
114    .then_some(DirectoryHint::Yes)
115    .unwrap_or({
116        if matches!(event.kind, EventKind::Modify(ModifyKind::Name(_))) {
117            DirectoryHint::Unknown
118        } else {
119            DirectoryHint::No
120        }
121    })
