3impl Indexer {
4    pub(super) fn import_projections(
5        &self,
6        paths: &HashSet<String>,
7        source_path_overrides: &HashMap<String, String>,
8        repository_paths: &HashSet<String>,
9        cancellation: &CancellationToken,
10    ) -> Result<Vec<ImportProjection>> {
11        let mut paths = paths.iter().cloned().collect::<Vec<_>>();
12        paths.sort_unstable();
13        let seeds = self.storage.import_seeds_for_paths(&paths)?;
14        let mut projections = Vec::with_capacity(seeds.len());
15        for seed in seeds {
16            check_cancelled(cancellation)?;
17            let source_path = source_path_overrides
18                .get(&seed.source_path)
19                .map_or(seed.source_path.as_str(), String::as_str);
20            let candidate_paths = import_candidates(source_path, &seed.raw_target);
21            let resolved_path = resolve_import_candidates(&candidate_paths, repository_paths);
22            projections.push(ImportProjection {
23                id: seed.id,
24                file_id: seed.file_id,
25                resolved_path,
26                candidate_paths,
27            });
28        }
29        Ok(projections)
30    }
31
32    pub(super) fn affected_importers(
33        &self,
34        deletions: &HashSet<String>,
35        change_set: &ChangeSet,
36        cancellation: &CancellationToken,
37    ) -> Result<HashSet<String>> {
38        let membership_changes = change_set.membership_changes();
39        let mut affected = HashSet::new();
40        for importer_path in self.storage.affected_importers(&membership_changes)? {
41            check_cancelled(cancellation)?;
42            if deletions.contains(&importer_path) {
43                continue;
44            }
45            affected.insert(importer_path);
46        }
47        Ok(affected)
48    }
