3pub(super) const MAX_IMPORT_TARGET_BYTES: usize = 16 * 1024;
4// Every Rust module base expands to at most `path.rs` and `path/mod.rs`.
5pub(super) const MAX_RUST_MODULE_BASES: usize = MAX_IMPORT_CANDIDATES_PER_IMPORT / 2;
6// A go.mod must fit this bound or its module metadata is ignored.
7pub(super) const MAX_GO_MOD_BYTES: u64 = 64 * 1024;
8// The upward go.mod walk stops at this many ancestors regardless of path depth.
9pub(super) const MAX_GO_MOD_WALK_DEPTH: usize = 64;

11/// Module metadata parsed once per resolution run from indexed `go.mod`
12/// files, read through the repository root with a byte bound so projections
13/// depend only on the repository snapshot and never on the process working
14/// directory.
15pub(super) struct GoModuleIndex {
16    pub(super) modules: HashMap<String, String>,
17}
18
19impl GoModuleIndex {
20    pub(super) fn load(
21        repository_paths: &HashSet<String>,
22        repository_root: &Dir,
23        cancellation: &CancellationToken,
24    ) -> Result<Self> {
25        let mut modules = HashMap::new();
26        for path in repository_paths {
27            check_cancelled(cancellation)?;
28            if !is_go_mod_path(path) {
29                continue;
30            }
31            let bytes = match read_bounded(repository_root, path, MAX_GO_MOD_BYTES) {
32                Ok(Some(bytes)) => bytes,
33                // A go.mod deleted or renamed after the path snapshot was
34                // captured is stale metadata, not a reconciliation failure.
35                Ok(None) => continue,
36                Err(error) if error.kind() == std::io::ErrorKind::NotFound => continue,
37                Err(error) => return Err(error.into()),
38            };
39            let content = String::from_utf8_lossy(&bytes);
40            if let Some(module_path) = parse_module_directive(&content) {
41                modules.insert(path.clone(), module_path);
42            }
43        }
44        Ok(Self { modules })
45    }
46}

48pub(super) fn is_go_mod_path(path: &str) -> bool {
49    path == "go.mod" || path.ends_with("/go.mod")
50}

52fn parse_module_directive(content: &str) -> Option<String> {
53    for line in content.lines() {
54        let line = line.trim();
55        let Some(module_path) = line.strip_prefix("module ") else {
56            continue;
57        };
58        // A valid directive may carry a trailing `//` comment; go.mod module
59        // paths never contain `//`, so the first occurrence is the comment.
60        let module_path = module_path
61            .split("//")
62            .next()
63            .unwrap_or(module_path)
64            .trim()
65            .trim_matches('"')
66            .trim_matches('\'');
67        if module_path.is_empty() || module_path.len() > MAX_IMPORT_TARGET_BYTES {
68            continue;
69        }
70        return Some(module_path.to_owned());
71    }
72    None
73}

83pub(super) fn resolve_imports(
84    files: &mut [IndexedFile],
85    go_modules: &GoModuleIndex,
86    sorted_paths: &[String],
87    cancellation: &CancellationToken,
88) -> Result<()> {
89    for file in files {
90        check_cancelled(cancellation)?;
91        for import in &mut file.imports {
92            check_cancelled(cancellation)?;
93            let projection =
94                derive_import_projection(&file.path, &import.raw_target, sorted_paths, go_modules);
95            import.resolved_path = projection.resolved_path;
96            import.candidate_paths = projection.candidate_paths;
97        }
98    }
99    Ok(())
100}

116pub(super) fn import_candidates(
117    source_path: &str,
118    raw_target: &str,
119    sorted_paths: &[String],
120    go_modules: &GoModuleIndex,
121) -> Vec<String> {
122    if raw_target.len() > MAX_IMPORT_TARGET_BYTES {
123        return Vec::new();
124    }
125    let source = std::path::Path::new(source_path);
126    match ImportResolutionPolicy::for_source(source) {
127        ImportResolutionPolicy::Python => python_import_candidates(source, raw_target),
128        ImportResolutionPolicy::JavaScript => javascript_import_candidates(source_path, raw_target),
129        ImportResolutionPolicy::TypeScript => typescript_import_candidates(source_path, raw_target),
130        ImportResolutionPolicy::Rust => rust_import_candidates(source, raw_target),
131        ImportResolutionPolicy::Latex => latex_import_candidates(source_path, raw_target),
132        ImportResolutionPolicy::WebResource => {
133            web_resource_import_candidates(source_path, raw_target)
134        }
135        ImportResolutionPolicy::Go => {
136            go_import_candidates(source, raw_target, go_modules, sorted_paths)
137        }
138        ImportResolutionPolicy::Unsupported => Vec::new(),
139    }
140}

142#[derive(Debug, Clone, Copy, PartialEq, Eq)]
143enum ImportResolutionPolicy {
144    Python,
145    JavaScript,
146    TypeScript,
147    Rust,
148    Latex,
149    WebResource,
150    Go,
151    Unsupported,
152}

154impl ImportResolutionPolicy {
155    fn for_source(source: &std::path::Path) -> Self {
156        match source.extension().and_then(|extension| extension.to_str()) {
157            Some("py" | "pyi") => Self::Python,
158            Some("js" | "mjs" | "cjs" | "jsx") => Self::JavaScript,
159            Some("ts" | "mts" | "cts" | "tsx") => Self::TypeScript,
160            Some("rs") => Self::Rust,
161            Some("tex" | "ltx") => Self::Latex,
162            Some("html" | "htm") => Self::WebResource,
163            Some("go") => Self::Go,
164            _ => Self::Unsupported,
165        }
166    }
167}

169fn python_import_candidates(source: &std::path::Path, raw_target: &str) -> Vec<String> {
170    let mut candidates = Vec::new();
171    for base in python_module_paths(source, raw_target) {
172        let Some(base) = normalize_relative(&base) else {
173            continue;
174        };
175        // CPython checks a package directory before a same-named module file.
176        for extension in ["py", "pyi"] {
177            push_candidate(
178                &mut candidates,
179                base.join("__init__").with_extension(extension),
180            );
181        }
182        for extension in ["py", "pyi"] {
183            push_candidate(&mut candidates, base.with_extension(extension));
184        }
185    }
186    candidates
187}

189fn javascript_import_candidates(source_path: &str, raw_target: &str) -> Vec<String> {
190    let Some(base) = relative_import_base(source_path, raw_target) else {
191        return Vec::new();
192    };
193    let mut candidates = Vec::new();
194    if base.extension().is_some() {
195        push_candidate(&mut candidates, base);
196        return candidates;
197    }
198    for extension in ["js", "mjs", "cjs", "jsx"] {
199        push_candidate(&mut candidates, base.with_extension(extension));
200    }
201    for extension in ["js", "mjs", "cjs", "jsx"] {
202        push_candidate(
203            &mut candidates,
204            base.join("index").with_extension(extension),
205        );
206    }
207    candidates
208}

210fn typescript_import_candidates(source_path: &str, raw_target: &str) -> Vec<String> {
211    let Some(base) = relative_import_base(source_path, raw_target) else {
212        return Vec::new();
213    };
214    let mut candidates = Vec::new();
215    let extension = base.extension().and_then(|extension| extension.to_str());
216    let source_extension = std::path::Path::new(source_path)
217        .extension()
218        .and_then(|extension| extension.to_str());
219    // `.mts` establishes Node ESM semantics without consulting project
220    // configuration. Extensionless relative imports are invalid there, so do
221    // not apply the classic/bundler heuristic used for context-dependent `.ts`
222    // and `.tsx` sources.
223    if extension.is_none() && source_extension == Some("mts") {
224        return candidates;
225    }
226    let substitutions: &[&str] = match extension {
227        Some("js") => &["ts", "tsx", "d.ts", "js", "jsx"],
228        Some("jsx") => &["tsx", "d.ts", "jsx"],
229        Some("mjs") => &["mts", "d.mts", "mjs"],
230        Some("cjs") => &["cts", "d.cts", "cjs"],
231        Some(_) => &[],
232        // TypeScript may add the runtime `.js` family under classic,
233        // CommonJS, or bundler resolution, then substitute its source/type
234        // counterparts. It never implicitly adds `.mjs`/`.mts` or
235        // `.cjs`/`.cts`; those runtime extensions must be explicit.
236        None => &["ts", "tsx", "d.ts", "js", "jsx"],
237    };
238    if substitutions.is_empty() {
239        push_candidate(&mut candidates, base);
240        return candidates;
241    }
242    for substitution in substitutions {
243        push_candidate(&mut candidates, base.with_extension(substitution));
244    }
245    if extension.is_none() {
246        for substitution in substitutions {
247            push_candidate(
248                &mut candidates,
249                base.join("index").with_extension(substitution),
250            );
251        }
252    }
253    candidates
254}

287fn web_resource_import_candidates(source_path: &str, raw_target: &str) -> Vec<String> {
288    // HTML resource URLs have much broader semantics than repository paths.
289    // Preserve the existing safe subset: an explicit dot-relative URL with no
290    // query or fragment, resolved to exactly one repository-owned file.
291    if raw_target.contains(['?', '#']) {
292        return Vec::new();
293    }
294    let Some(base) = relative_import_base(source_path, raw_target) else {
295        return Vec::new();
296    };
297    let mut candidates = Vec::new();
298    push_candidate(&mut candidates, base);
299    candidates
300}

317fn push_candidate(candidates: &mut Vec<String>, candidate: std::path::PathBuf) {
318    if candidates.len() >= MAX_IMPORT_CANDIDATES_PER_IMPORT {
319        return;
320    }
321    let Some(candidate) = normalize_relative(&candidate) else {
322        return;
323    };
324    let candidate = candidate.to_string_lossy().replace('\\', "/");
325    if candidate.len() > MAX_IMPORT_CANDIDATE_PATH_BYTES {
326        return;
327    }
328    if !candidates.contains(&candidate) {
329        candidates.push(candidate);
330    }
331}

333pub(super) fn python_module_paths(
334    source: &std::path::Path,
335    raw_target: &str,
336) -> Vec<std::path::PathBuf> {
337    let level = raw_target.bytes().take_while(|byte| *byte == b'.').count();
338    let module = raw_target[level..].replace('.', "/");
339    if level == 0 {
340        let module = std::path::PathBuf::from(module);
341        let mut paths = Vec::with_capacity(2);
342        if let Some(root) = conventional_python_source_root(source) {
343            paths.push(root.join(&module));
344        }
345        paths.push(module);
346        paths.dedup();
347        return paths;
348    }
349
350    let mut base = source
351        .parent()
352        .unwrap_or_else(|| std::path::Path::new(""))
353        .to_path_buf();
354    for _ in 1..level {
355        if !base.pop() {
356            return Vec::new();
357        }
358    }
359    if !module.is_empty() {
360        base.push(module);
361    }
362    vec![base]
363}

365fn conventional_python_source_root(source: &std::path::Path) -> Option<std::path::PathBuf> {
366    let mut prefix = std::path::PathBuf::new();
367    let mut source_root = None;
368    for component in source.parent()?.components() {
369        let std::path::Component::Normal(component) = component else {
370            return None;
371        };
372        prefix.push(component);
373        if component == "src" {
374            source_root = Some(prefix.clone());
375        }
376    }
377    source_root
378}

380/// Decompose a raw Rust `use` target into candidate module paths.
381///
382/// Handles grouped imports (`a::{b, c}`), aliases (`a::b as c`), and
383/// leading path qualifiers (`crate`, `self`, `super`). For each concrete
384/// target, all module prefixes are tried from longest to shortest so that
385/// `module::symbol` resolves to `module` when the full `module/symbol`
386/// path does not exist.
387pub(super) fn rust_module_paths(
388    source: &std::path::Path,
389    raw_target: &str,
390) -> Vec<std::path::PathBuf> {
391    let trimmed = raw_target.trim();
392    let crate_root = rust_crate_roots(source)
393        .into_iter()
394        .next()
395        .unwrap_or_default();
396    let (stripped, roots) = if let Some(target) = trimmed.strip_prefix("crate::") {
397        (target, vec![crate_root.clone()])
398    } else if let Some(target) = trimmed.strip_prefix("self::") {
399        (target, vec![rust_source_module_dir(source, &crate_root)])
400    } else if trimmed.starts_with("super::") {
401        let mut target = trimmed;
402        let mut root = rust_source_module_dir(source, &crate_root);
403        while let Some(rest) = target.strip_prefix("super::") {
404            if root == crate_root || !root.pop() {
405                return Vec::new();
406            }
407            target = rest;
408        }
409        (target, vec![root])
410    } else {
411        (trimmed, vec![crate_root])
412    };
413
414    let mut targets = Vec::new();
415    let before_brace = stripped
416        .split('{')
417        .next()
418        .unwrap_or("")
419        .trim_end_matches(':');
420    let group_body = stripped.find('{').and_then(|_| {
421        stripped
422            .split('{')
423            .nth(1)
424            .and_then(|rest| rest.split('}').next())
425    });
426
427    if let Some(group) = group_body {
428        let prefix = before_brace.trim_end_matches(':');
429        for item in group.split(',') {
430            let item = item.trim();
431            let item = item.split(" as ").next().unwrap_or(item).trim();
432            if item.is_empty() {
433                continue;
434            }
435            let full = if prefix.is_empty() {
436                item.to_string()
437            } else {
438                format!("{prefix}::{item}")
439            };
440            targets.push(full);
441            if targets.len() > MAX_RUST_MODULE_BASES {
442                return Vec::new();
443            }
444        }
445    } else {
446        let single = stripped.split(" as ").next().unwrap_or(stripped).trim();
447        if !single.is_empty() {
448            targets.push(single.to_string());
449        }
450    }
451
452    let mut parsed_targets = Vec::with_capacity(targets.len());
453    let mut required_bases = 0usize;
454    for target in &targets {
455        let segments = target
456            .split("::")
457            .filter(|segment| !segment.is_empty())
458            .collect::<Vec<_>>();
459        if segments.is_empty() {
460            continue;
461        }
462        required_bases = match required_bases.checked_add(segments.len()) {
463            Some(total) if total <= MAX_RUST_MODULE_BASES => total,
464            _ => return Vec::new(),
465        };
466        parsed_targets.push(segments);
467    }
468
469    let mut bases = Vec::with_capacity(required_bases);
470    for segments in parsed_targets {
471        for prefix_len in (1..=segments.len()).rev() {
472            let path_str = segments[..prefix_len].join("/");
473            for root in &roots {
474                bases.push(root.join(&path_str));
475            }
476        }
477    }
478    bases
479}

481pub(super) fn rust_source_module_dir(
482    source: &std::path::Path,
483    crate_root: &std::path::Path,
484) -> std::path::PathBuf {
485    let parent = source.parent().unwrap_or_else(|| std::path::Path::new(""));
486    match source.file_stem().and_then(|stem| stem.to_str()) {
487        Some("lib" | "main" | "mod") | None => parent.to_path_buf(),
488        Some(_)
489            if parent.as_os_str().is_empty()
490                || (parent == crate_root
491                    && parent.file_name().is_some_and(|name| {
492                        matches!(
493                            name.to_str(),
494                            Some("bin" | "tests" | "examples" | "benches")
495                        )
496                    })) =>
497        {
498            parent.to_path_buf()
499        }
500        Some(stem) => parent.join(stem),
501    }
502}

504pub(super) fn rust_crate_roots(source: &std::path::Path) -> Vec<std::path::PathBuf> {
505    let components = source
506        .components()
507        .filter_map(|component| match component {
508            std::path::Component::Normal(component) => Some(component.to_os_string()),
509            _ => None,
510        })
511        .collect::<Vec<_>>();
512    let file_index = components.len().saturating_sub(1);
513    let mut root = std::path::PathBuf::new();
514
515    if let Some(index) = components[..file_index]
516        .iter()
517        .rposition(|component| component == "src")
518    {
519        for component in &components[..=index] {
520            root.push(component);
521        }
522        if components
523            .get(index + 1)
524            .is_some_and(|value| value == "bin")
525        {
526            root.push("bin");
527            if file_index > index + 2 {
528                root.push(&components[index + 2]);
529            }
530        }
531        return vec![root];
532    }
533
534    if let Some(index) = components[..file_index]
535        .iter()
536        .rposition(|component| matches!(component.to_str(), Some("tests" | "examples" | "benches")))
537    {
538        for component in &components[..=index] {
539            root.push(component);
540        }
541        if file_index > index + 1 {
542            root.push(&components[index + 1]);
543        }
544        return vec![root];
545    }
546
547    vec![
548        source
549            .parent()
550            .unwrap_or_else(|| std::path::Path::new(""))
551            .to_path_buf(),
552    ]
553}

555pub(super) fn resolve_import_candidates(
556    candidates: &[String],
557    sorted_paths: &[String],
558) -> Option<String> {
559    // Candidates are ordered from most-specific to least-specific. Return the
560    // first existing candidate; a more specific match always wins over a
561    // shorter prefix fallback. This preserves the conservative contract for
562    // same-priority candidates (e.g. exact file vs directory init) while
563    // allowing module prefix fallback for Rust imports.
564    for candidate in candidates {
565        if sorted_paths.binary_search(candidate).is_ok() {
566            return Some(candidate.clone());
567        }
568    }
569    None
570}

572pub(super) fn normalize_relative(path: &std::path::Path) -> Option<std::path::PathBuf> {
573    let mut normalized = std::path::PathBuf::new();
574    for component in path.components() {
575        match component {
576            std::path::Component::CurDir => {}
577            std::path::Component::Normal(part) => normalized.push(part),
578            std::path::Component::ParentDir => {
579                if !normalized.pop() {
580                    return None;
581                }
582            }
583            std::path::Component::RootDir | std::path::Component::Prefix(_) => return None,
584        }
585    }
586    Some(normalized)
587}

589/// Resolve Go imports using module-path to directory mapping.
590///
591/// Go imports are module paths like "example.com/acme/pkg/internal/service".
592/// We resolve them against the indexed snapshot: the nearest indexed `go.mod`
593/// ancestor (read once, bounded, through the repository root) supplies the
594/// `module` directive, and the module-relative remainder maps to a package
595/// directory. The candidate is the lexicographically smallest indexed `.go`
596/// file directly inside that package directory, so the projection always
597/// names an indexed file rather than a directory.
598fn go_import_candidates(
599    source: &std::path::Path,
600    raw_target: &str,
601    go_modules: &GoModuleIndex,
602    sorted_paths: &[String],
603) -> Vec<String> {
604    let target = raw_target.trim_matches(|c| c == '"' || c == '\'');
605    if target.is_empty() {
606        return Vec::new();
607    }
608    // Walk up from the source file to find the nearest indexed go.mod.
609    let mut dir = source.parent();
610    let mut depth = 0usize;
611    while let Some(parent) = dir {
612        if depth >= MAX_GO_MOD_WALK_DEPTH {
613            break;
614        }
615        depth += 1;
616        let go_mod_path = parent.join("go.mod").to_string_lossy().replace('\\', "/");
617        let Some(module_path) = go_modules.modules.get(&go_mod_path) else {
618            dir = parent.parent();
619            continue;
620        };
621        // The module prefix must end at a path segment boundary: an import of
622        // "example.com/acme2" must not match module "example.com/acme".
623        if target != module_path
624            && !(target.starts_with(module_path.as_str())
625                && target.as_bytes().get(module_path.len()) == Some(&b'/'))
626        {
627            break; // Nearest go.mod did not declare this module — stop walking up
628        }
629        let remainder = &target[module_path.len()..];
630        let remainder = remainder.strip_prefix('/').unwrap_or(remainder);
631        let go_mod_dir = std::path::Path::new(&go_mod_path)
632            .parent()
633            .unwrap_or_else(|| std::path::Path::new(""))
634            .to_string_lossy()
635            .into_owned();
636        // Build the package directory as a forward-slash string: indexed paths
637        // are slash-separated on every platform, so joining native `Path`
638        // buffers would leak backslashes on Windows.
639        let package_dir = if remainder.is_empty() {
640            go_mod_dir
641        } else {
642            match normalize_relative(std::path::Path::new(remainder)) {
643                Some(package_dir) => {
644                    let package_dir = package_dir.to_string_lossy().replace('\\', "/");
645                    if package_dir.is_empty() || go_mod_dir.is_empty() {
646                        package_dir
647                    } else {
648                        format!("{go_mod_dir}/{package_dir}")
649                    }
650                }
651                None => break,
652            }
653        };
654        return minimum_indexed_go_file(&package_dir, sorted_paths)
655            .into_iter()
656            .collect();
657    }
658    Vec::new()
659}

661/// The lexicographically smallest indexed `.go` file directly inside the
662/// package directory, or `None` when the package has no indexed Go files.
663fn minimum_indexed_go_file(package_dir: &str, sorted_paths: &[String]) -> Option<String> {
664    let prefix = if package_dir.is_empty() {
665        String::new()
666    } else {
667        format!("{package_dir}/")
668    };
669    let start = if prefix.is_empty() {
670        0
671    } else {
672        sorted_paths.partition_point(|path| path.as_str() < prefix.as_str())
673    };
674    for path in &sorted_paths[start..] {
675        if !path.starts_with(&prefix) {
676            break;
677        }
678        let direct = &path[prefix.len()..];
679        // Dependency imports never compile a package's test files, so a
680        // `_test.go` member cannot represent the package to readers.
681        if !direct.contains('/') && direct.ends_with(".go") && !direct.ends_with("_test.go") {
682            return Some(path.clone());
683        }
684    }
685    None
686}
