3pub(super) fn resolve_imports(
4    files: &mut [IndexedFile],
5    repository_paths: &HashSet<String>,
6    cancellation: &CancellationToken,
7) -> Result<()> {
8    for file in files {
9        check_cancelled(cancellation)?;
10        for import in &mut file.imports {
11            check_cancelled(cancellation)?;
12            import.candidate_paths = import_candidates(&file.path, &import.raw_target);
13            import.resolved_path =
14                resolve_import_candidates(&import.candidate_paths, repository_paths);
15        }
16    }
17    Ok(())
18}

32pub(super) fn import_candidates(source_path: &str, raw_target: &str) -> Vec<String> {
33    let source = std::path::Path::new(source_path);
34    let parent = source.parent().unwrap_or_else(|| std::path::Path::new(""));
35    let mut bases = Vec::new();
36
37    if matches!(
38        source.extension().and_then(|ext| ext.to_str()),
39        Some("py" | "pyi")
40    ) {
41        bases.extend(python_module_paths(source, raw_target));
42    } else if raw_target.starts_with('.') {
43        bases.push(parent.join(raw_target));
44    } else if source.extension().and_then(|ext| ext.to_str()) == Some("rs") {
45        bases.extend(rust_module_paths(source, raw_target));
46    } else if matches!(
47        source.extension().and_then(|ext| ext.to_str()),
48        Some("tex" | "ltx")
49    ) {
50        bases.push(parent.join(raw_target));
51    } else {
52        return Vec::new();
53    }
54
55    let init_file = match source.extension().and_then(|ext| ext.to_str()) {
56        Some("py" | "pyi") => Some("__init__"),
57        Some("rs") => Some("mod"),
58        _ => None,
59    };
60
61    let extensions: &[&str] = match source.extension().and_then(|ext| ext.to_str()) {
62        Some("js" | "mjs" | "cjs") => &["", "js", "mjs", "cjs"],
63        Some("ts" | "mts" | "cts" | "tsx") => &["", "ts", "tsx", "mts", "cts", "js"],
64        Some("py" | "pyi") => &["", "py", "pyi"],
65        Some("rs") => &["", "rs"],
66        Some("tex" | "ltx") => &["", "tex", "ltx"],
67        _ => &[""],
68    };
69    let mut matches = Vec::new();
70    for base in bases {
71        let Some(base) = normalize_relative(&base) else {
72            continue;
73        };
74        for extension in extensions {
75            let exact = if extension.is_empty() || base.extension().is_some() {
76                base.clone()
77            } else {
78                base.with_extension(extension)
79            };
80            let directory_init = match init_file {
81                Some(init) if extension.is_empty() => base.join(init),
82                Some(init) => base.join(init).with_extension(extension),
83                None if extension.is_empty() => base.join("index"),
84                None => base.join("index").with_extension(extension),
85            };
86            for candidate in [exact, directory_init] {
87                let candidate = if candidate.extension().is_some() || extension.is_empty() {
88                    candidate
89                } else {
90                    candidate.with_extension(extension)
91                };
92                let candidate = candidate.to_string_lossy().replace('\\', "/");
93                if !matches.contains(&candidate) {
94                    matches.push(candidate);
95                }
96            }
97        }
98    }
99    matches
100}

102pub(super) fn python_module_paths(
103    source: &std::path::Path,
104    raw_target: &str,
105) -> Vec<std::path::PathBuf> {
106    let level = raw_target.bytes().take_while(|byte| *byte == b'.').count();
107    let module = raw_target[level..].replace('.', "/");
108    if level == 0 {
109        return vec![std::path::PathBuf::from(module)];
110    }
111
112    let mut base = source
113        .parent()
114        .unwrap_or_else(|| std::path::Path::new(""))
115        .to_path_buf();
116    for _ in 1..level {
117        if !base.pop() {
118            return Vec::new();
119        }
120    }
121    if !module.is_empty() {
122        base.push(module);
123    }
124    vec![base]
125}

127/// Decompose a raw Rust `use` target into candidate module paths.
128///
129/// Handles grouped imports (`a::{b, c}`), aliases (`a::b as c`), and
130/// leading path qualifiers (`crate`, `self`, `super`). For each concrete
131/// target, all module prefixes are tried from longest to shortest so that
132/// `module::symbol` resolves to `module` when the full `module/symbol`
133/// path does not exist.
134pub(super) fn rust_module_paths(
135    source: &std::path::Path,
136    raw_target: &str,
137) -> Vec<std::path::PathBuf> {
138    let trimmed = raw_target.trim();
139    let (stripped, roots) = if let Some(target) = trimmed.strip_prefix("crate::") {
140        (target, rust_crate_roots(source))
141    } else if let Some(target) = trimmed.strip_prefix("self::") {
142        (target, vec![rust_source_module_dir(source)])
143    } else if trimmed.starts_with("super::") {
144        let mut target = trimmed;
145        let mut root = rust_source_module_dir(source);
146        while let Some(rest) = target.strip_prefix("super::") {
147            if !root.pop() {
148                return Vec::new();
149            }
150            target = rest;
151        }
152        (target, vec![root])
153    } else {
154        (trimmed, rust_crate_roots(source))
155    };
156
157    let mut targets = Vec::new();
158    let before_brace = stripped
159        .split('{')
160        .next()
161        .unwrap_or("")
162        .trim_end_matches(':');
163    let group_body = stripped.find('{').and_then(|_| {
164        stripped
165            .split('{')
166            .nth(1)
167            .and_then(|rest| rest.split('}').next())
168    });
169
170    if let Some(group) = group_body {
171        let prefix = before_brace.trim_end_matches(':');
172        for item in group.split(',') {
173            let item = item.trim();
174            let item = item.split(" as ").next().unwrap_or(item).trim();
175            if item.is_empty() {
176                continue;
177            }
178            let full = if prefix.is_empty() {
179                item.to_string()
180            } else {
181                format!("{prefix}::{item}")
182            };
183            targets.push(full);
184        }
185    } else {
186        let single = stripped.split(" as ").next().unwrap_or(stripped).trim();
187        if !single.is_empty() {
188            targets.push(single.to_string());
189        }
190    }
191
192    let mut bases = Vec::new();
193    for target in targets {
194        let segments: Vec<&str> = target.split("::").filter(|s| !s.is_empty()).collect();
195        if segments.is_empty() {
196            continue;
197        }
198        for prefix_len in (1..=segments.len()).rev() {
199            let path_str = segments[..prefix_len].join("/");
200            for root in &roots {
201                bases.push(root.join(&path_str));
202            }
203        }
204    }
205    bases
206}

208pub(super) fn rust_source_module_dir(source: &std::path::Path) -> std::path::PathBuf {
209    let parent = source.parent().unwrap_or_else(|| std::path::Path::new(""));
210    match source.file_stem().and_then(|stem| stem.to_str()) {
211        Some("lib" | "main" | "mod") | None => parent.to_path_buf(),
212        Some(stem) => parent.join(stem),
213    }
214}

216pub(super) fn rust_crate_roots(source: &std::path::Path) -> Vec<std::path::PathBuf> {
217    if source.starts_with("src") {
218        vec![std::path::PathBuf::from("src"), std::path::PathBuf::new()]
219    } else {
220        vec![std::path::PathBuf::new(), std::path::PathBuf::from("src")]
221    }
222}

224pub(super) fn resolve_import_candidates(
225    candidates: &[String],
226    repository_paths: &HashSet<String>,
227) -> Option<String> {
228    // Candidates are ordered from most-specific to least-specific. Return the
229    // first existing candidate; a more specific match always wins over a
230    // shorter prefix fallback. This preserves the conservative contract for
231    // same-priority candidates (e.g. exact file vs directory init) while
232    // allowing module prefix fallback for Rust imports.
233    for candidate in candidates {
234        if repository_paths.contains(candidate) {
235            return Some(candidate.clone());
236        }
237    }
238    None
239}

241pub(super) fn normalize_relative(path: &std::path::Path) -> Option<std::path::PathBuf> {
242    let mut normalized = std::path::PathBuf::new();
243    for component in path.components() {
244        match component {
245            std::path::Component::CurDir => {}
246            std::path::Component::Normal(part) => normalized.push(part),
247            std::path::Component::ParentDir => {
248                if !normalized.pop() {
249                    return None;
250                }
251            }
252            std::path::Component::RootDir | std::path::Component::Prefix(_) => return None,
253        }
254    }
255    Some(normalized)
256}
