3pub(super) const MAX_IMPORT_TARGET_BYTES: usize = 16 * 1024;
4// Every Rust module base expands to at most `path.rs` and `path/mod.rs`.
5pub(super) const MAX_RUST_MODULE_BASES: usize = MAX_IMPORT_CANDIDATES_PER_IMPORT / 2;

7pub(super) fn resolve_imports(
8    files: &mut [IndexedFile],
9    repository_paths: &HashSet<String>,
10    cancellation: &CancellationToken,
11) -> Result<()> {
12    for file in files {
13        check_cancelled(cancellation)?;
14        for import in &mut file.imports {
15            check_cancelled(cancellation)?;
16            let projection =
17                derive_import_projection(&file.path, &import.raw_target, repository_paths);
18            import.resolved_path = projection.resolved_path;
19            import.candidate_paths = projection.candidate_paths;
20        }
21    }
22    Ok(())
23}

25pub(super) fn derive_import_projection(
26    source_path: &str,
27    raw_target: &str,
28    repository_paths: &HashSet<String>,
29) -> ImportProjectionValue {
30    let candidate_paths = import_candidates(source_path, raw_target);

33        resolved_path,
34        candidate_paths,
35    }

38pub(super) fn import_candidates(source_path: &str, raw_target: &str) -> Vec<String> {
39    if raw_target.len() > MAX_IMPORT_TARGET_BYTES {
40        return Vec::new();
41    }
42    let source = std::path::Path::new(source_path);
43    match ImportResolutionPolicy::for_source(source) {
44        ImportResolutionPolicy::Python => python_import_candidates(source, raw_target),
45        ImportResolutionPolicy::JavaScript => javascript_import_candidates(source_path, raw_target),
46        ImportResolutionPolicy::TypeScript => typescript_import_candidates(source_path, raw_target),
47        ImportResolutionPolicy::Rust => rust_import_candidates(source, raw_target),
48        ImportResolutionPolicy::Latex => latex_import_candidates(source_path, raw_target),
49        ImportResolutionPolicy::WebResource => {
50            web_resource_import_candidates(source_path, raw_target)
51        }
52        ImportResolutionPolicy::Unsupported => Vec::new(),
53    }
54}

56#[derive(Debug, Clone, Copy, PartialEq, Eq)]
57enum ImportResolutionPolicy {
58    Python,
59    JavaScript,
60    TypeScript,
61    Rust,
62    Latex,
63    WebResource,
64    Unsupported,
65}
66
67impl ImportResolutionPolicy {
68    fn for_source(source: &std::path::Path) -> Self {
69        match source.extension().and_then(|extension| extension.to_str()) {
70            Some("py" | "pyi") => Self::Python,
71            Some("js" | "mjs" | "cjs" | "jsx") => Self::JavaScript,
72            Some("ts" | "mts" | "cts" | "tsx") => Self::TypeScript,
73            Some("rs") => Self::Rust,
74            Some("tex" | "ltx") => Self::Latex,
75            Some("html" | "htm") => Self::WebResource,
76            _ => Self::Unsupported,
77        }
78    }
79}

81fn python_import_candidates(source: &std::path::Path, raw_target: &str) -> Vec<String> {
82    let mut candidates = Vec::new();
83    for base in python_module_paths(source, raw_target) {
84        let Some(base) = normalize_relative(&base) else {
85            continue;
86        };
87        // CPython checks a package directory before a same-named module file.
88        for extension in ["py", "pyi"] {
89            push_candidate(
90                &mut candidates,
91                base.join("__init__").with_extension(extension),
92            );
93        }
94        for extension in ["py", "pyi"] {
95            push_candidate(&mut candidates, base.with_extension(extension));
96        }
97    }
98    candidates
99}

101fn javascript_import_candidates(source_path: &str, raw_target: &str) -> Vec<String> {
102    let Some(base) = relative_import_base(source_path, raw_target) else {
103        return Vec::new();
104    };
105    let mut candidates = Vec::new();
106    if base.extension().is_some() {
107        push_candidate(&mut candidates, base);
108        return candidates;
109    }
110    for extension in ["js", "mjs", "cjs", "jsx"] {
111        push_candidate(&mut candidates, base.with_extension(extension));
112    }
113    for extension in ["js", "mjs", "cjs", "jsx"] {
114        push_candidate(
115            &mut candidates,
116            base.join("index").with_extension(extension),
117        );
118    }
119    candidates
120}

122fn typescript_import_candidates(source_path: &str, raw_target: &str) -> Vec<String> {
123    let Some(base) = relative_import_base(source_path, raw_target) else {
124        return Vec::new();
125    };
126    let mut candidates = Vec::new();
127    let extension = base.extension().and_then(|extension| extension.to_str());
128    let source_extension = std::path::Path::new(source_path)
129        .extension()
130        .and_then(|extension| extension.to_str());
131    // `.mts` establishes Node ESM semantics without consulting project
132    // configuration. Extensionless relative imports are invalid there, so do
133    // not apply the classic/bundler heuristic used for context-dependent `.ts`
134    // and `.tsx` sources.
135    if extension.is_none() && source_extension == Some("mts") {
136        return candidates;
137    }
138    let substitutions: &[&str] = match extension {
139        Some("js") => &["ts", "tsx", "d.ts", "js", "jsx"],
140        Some("jsx") => &["tsx", "d.ts", "jsx"],
141        Some("mjs") => &["mts", "d.mts", "mjs"],
142        Some("cjs") => &["cts", "d.cts", "cjs"],
143        Some(_) => &[],
144        // TypeScript may add the runtime `.js` family under classic,
145        // CommonJS, or bundler resolution, then substitute its source/type
146        // counterparts. It never implicitly adds `.mjs`/`.mts` or
147        // `.cjs`/`.cts`; those runtime extensions must be explicit.
148        None => &["ts", "tsx", "d.ts", "js", "jsx"],
149    };
150    if substitutions.is_empty() {
151        push_candidate(&mut candidates, base);
152        return candidates;
153    }
154    for substitution in substitutions {
155        push_candidate(&mut candidates, base.with_extension(substitution));
156    }
157    if extension.is_none() {
158        for substitution in substitutions {
159            push_candidate(
160                &mut candidates,
161                base.join("index").with_extension(substitution),
162            );
163        }
164    }
165    candidates
166}

168fn rust_import_candidates(source: &std::path::Path, raw_target: &str) -> Vec<String> {
169    let mut candidates = Vec::new();
170    for base in rust_module_paths(source, raw_target) {
171        let Some(base) = normalize_relative(&base) else {
172            continue;
173        };
174        if base.extension().is_some() {
175            push_candidate(&mut candidates, base);
176        } else {
177            push_candidate(&mut candidates, base.with_extension("rs"));
178            push_candidate(&mut candidates, base.join("mod.rs"));
179        }
180    }
181    candidates
182}

184fn latex_import_candidates(source_path: &str, raw_target: &str) -> Vec<String> {
185    let Some(base) = source_relative_base(source_path, raw_target) else {
186        return Vec::new();
187    };
188    let mut candidates = Vec::new();
189    if base.extension().is_some() {
190        push_candidate(&mut candidates, base);
191    } else {
192        for extension in ["tex", "ltx"] {
193            push_candidate(&mut candidates, base.with_extension(extension));
194        }
195    }
196    candidates

199fn web_resource_import_candidates(source_path: &str, raw_target: &str) -> Vec<String> {
200    // HTML resource URLs have much broader semantics than repository paths.
201    // Preserve the existing safe subset: an explicit dot-relative URL with no
202    // query or fragment, resolved to exactly one repository-owned file.
203    if raw_target.contains(['?', '#']) {
204        return Vec::new();
205    }
206    let Some(base) = relative_import_base(source_path, raw_target) else {
207        return Vec::new();
208    };
209    let mut candidates = Vec::new();
210    push_candidate(&mut candidates, base);
211    candidates
212}

214fn relative_import_base(source_path: &str, raw_target: &str) -> Option<std::path::PathBuf> {
215    if !raw_target.starts_with('.') {
216        return None;
217    }
218    source_relative_base(source_path, raw_target)
219}

221fn source_relative_base(source_path: &str, raw_target: &str) -> Option<std::path::PathBuf> {
222    crate::repository::RepositoryPath::parse(source_path)
223        .ok()?
224        .join_relative(raw_target)
225        .ok()
226        .map(|path| std::path::PathBuf::from(path.as_str()))
227}

229fn push_candidate(candidates: &mut Vec<String>, candidate: std::path::PathBuf) {
230    if candidates.len() >= MAX_IMPORT_CANDIDATES_PER_IMPORT {
231        return;
232    }
233    let Some(candidate) = normalize_relative(&candidate) else {
234        return;
235    };
236    let candidate = candidate.to_string_lossy().replace('\\', "/");
237    if candidate.len() > MAX_IMPORT_CANDIDATE_PATH_BYTES {
238        return;
239    }
240    if !candidates.contains(&candidate) {
241        candidates.push(candidate);
242    }
243}

245pub(super) fn python_module_paths(
246    source: &std::path::Path,
247    raw_target: &str,
248) -> Vec<std::path::PathBuf> {
249    let level = raw_target.bytes().take_while(|byte| *byte == b'.').count();
250    let module = raw_target[level..].replace('.', "/");
251    if level == 0 {
252        let module = std::path::PathBuf::from(module);
253        let mut paths = Vec::with_capacity(2);
254        if let Some(root) = conventional_python_source_root(source) {
255            paths.push(root.join(&module));
256        }
257        paths.push(module);
258        paths.dedup();
259        return paths;
260    }
261
262    let mut base = source
263        .parent()
264        .unwrap_or_else(|| std::path::Path::new(""))
265        .to_path_buf();
266    for _ in 1..level {
267        if !base.pop() {
268            return Vec::new();
269        }
270    }
271    if !module.is_empty() {
272        base.push(module);
273    }
274    vec![base]
275}

277fn conventional_python_source_root(source: &std::path::Path) -> Option<std::path::PathBuf> {
278    let mut prefix = std::path::PathBuf::new();
279    let mut source_root = None;
280    for component in source.parent()?.components() {
281        let std::path::Component::Normal(component) = component else {
282            return None;
283        };
284        prefix.push(component);
285        if component == "src" {
286            source_root = Some(prefix.clone());
287        }
288    }
289    source_root
290}

292/// Decompose a raw Rust `use` target into candidate module paths.
293///
294/// Handles grouped imports (`a::{b, c}`), aliases (`a::b as c`), and
295/// leading path qualifiers (`crate`, `self`, `super`). For each concrete
296/// target, all module prefixes are tried from longest to shortest so that
297/// `module::symbol` resolves to `module` when the full `module/symbol`
298/// path does not exist.
299pub(super) fn rust_module_paths(
300    source: &std::path::Path,
301    raw_target: &str,
302) -> Vec<std::path::PathBuf> {
303    let trimmed = raw_target.trim();
304    let crate_root = rust_crate_roots(source)
305        .into_iter()
306        .next()
307        .unwrap_or_default();
308    let (stripped, roots) = if let Some(target) = trimmed.strip_prefix("crate::") {
309        (target, vec![crate_root.clone()])
310    } else if let Some(target) = trimmed.strip_prefix("self::") {
311        (target, vec![rust_source_module_dir(source, &crate_root)])
312    } else if trimmed.starts_with("super::") {
313        let mut target = trimmed;
314        let mut root = rust_source_module_dir(source, &crate_root);
315        while let Some(rest) = target.strip_prefix("super::") {
316            if root == crate_root || !root.pop() {
317                return Vec::new();
318            }
319            target = rest;
320        }
321        (target, vec![root])
322    } else {
323        (trimmed, vec![crate_root])
324    };
325
326    let mut targets = Vec::new();
327    let before_brace = stripped
328        .split('{')
329        .next()
330        .unwrap_or("")
331        .trim_end_matches(':');
332    let group_body = stripped.find('{').and_then(|_| {
333        stripped
334            .split('{')
335            .nth(1)
336            .and_then(|rest| rest.split('}').next())
337    });
338
339    if let Some(group) = group_body {
340        let prefix = before_brace.trim_end_matches(':');
341        for item in group.split(',') {
342            let item = item.trim();
343            let item = item.split(" as ").next().unwrap_or(item).trim();
344            if item.is_empty() {
345                continue;
346            }
347            let full = if prefix.is_empty() {
348                item.to_string()
349            } else {
350                format!("{prefix}::{item}")
351            };
352            targets.push(full);
353            if targets.len() > MAX_RUST_MODULE_BASES {
354                return Vec::new();
355            }
356        }
357    } else {
358        let single = stripped.split(" as ").next().unwrap_or(stripped).trim();
359        if !single.is_empty() {
360            targets.push(single.to_string());
361        }
362    }
363
364    let mut parsed_targets = Vec::with_capacity(targets.len());
365    let mut required_bases = 0usize;
366    for target in &targets {
367        let segments = target
368            .split("::")
369            .filter(|segment| !segment.is_empty())
370            .collect::<Vec<_>>();
371        if segments.is_empty() {
372            continue;
373        }
374        required_bases = match required_bases.checked_add(segments.len()) {
375            Some(total) if total <= MAX_RUST_MODULE_BASES => total,
376            _ => return Vec::new(),
377        };
378        parsed_targets.push(segments);
379    }
380
381    let mut bases = Vec::with_capacity(required_bases);
382    for segments in parsed_targets {
383        for prefix_len in (1..=segments.len()).rev() {
384            let path_str = segments[..prefix_len].join("/");
385            for root in &roots {
386                bases.push(root.join(&path_str));
387            }
388        }
389    }
390    bases
391}

393pub(super) fn rust_source_module_dir(
394    source: &std::path::Path,
395    crate_root: &std::path::Path,
396) -> std::path::PathBuf {
397    let parent = source.parent().unwrap_or_else(|| std::path::Path::new(""));
398    match source.file_stem().and_then(|stem| stem.to_str()) {
399        Some("lib" | "main" | "mod") | None => parent.to_path_buf(),
400        Some(_)
401            if parent.as_os_str().is_empty()
402                || (parent == crate_root
403                    && parent.file_name().is_some_and(|name| {
404                        matches!(
405                            name.to_str(),
406                            Some("bin" | "tests" | "examples" | "benches")
407                        )
408                    })) =>
409        {
410            parent.to_path_buf()
411        }
412        Some(stem) => parent.join(stem),
413    }
414}

416pub(super) fn rust_crate_roots(source: &std::path::Path) -> Vec<std::path::PathBuf> {
417    let components = source
418        .components()
419        .filter_map(|component| match component {
420            std::path::Component::Normal(component) => Some(component.to_os_string()),
421            _ => None,
422        })
423        .collect::<Vec<_>>();
424    let file_index = components.len().saturating_sub(1);
425    let mut root = std::path::PathBuf::new();
426
427    if let Some(index) = components[..file_index]
428        .iter()
429        .rposition(|component| component == "src")
430    {
431        for component in &components[..=index] {
432            root.push(component);
433        }
434        if components
435            .get(index + 1)
436            .is_some_and(|value| value == "bin")
437        {
438            root.push("bin");
439            if file_index > index + 2 {
440                root.push(&components[index + 2]);
441            }
442        }
443        return vec![root];
444    }
445
446    if let Some(index) = components[..file_index]
447        .iter()
448        .rposition(|component| matches!(component.to_str(), Some("tests" | "examples" | "benches")))
449    {
450        for component in &components[..=index] {
451            root.push(component);
452        }
453        if file_index > index + 1 {
454            root.push(&components[index + 1]);
455        }
456        return vec![root];
457    }
458
459    vec![
460        source
461            .parent()
462            .unwrap_or_else(|| std::path::Path::new(""))
463            .to_path_buf(),
464    ]
465}

467pub(super) fn resolve_import_candidates(
468    candidates: &[String],
469    repository_paths: &HashSet<String>,
470) -> Option<String> {
471    // Candidates are ordered from most-specific to least-specific. Return the
472    // first existing candidate; a more specific match always wins over a
473    // shorter prefix fallback. This preserves the conservative contract for
474    // same-priority candidates (e.g. exact file vs directory init) while
475    // allowing module prefix fallback for Rust imports.
476    for candidate in candidates {
477        if repository_paths.contains(candidate) {
478            return Some(candidate.clone());
479        }
480    }
481    None
482}

484pub(super) fn normalize_relative(path: &std::path::Path) -> Option<std::path::PathBuf> {
485    let mut normalized = std::path::PathBuf::new();
486    for component in path.components() {
487        match component {
488            std::path::Component::CurDir => {}
489            std::path::Component::Normal(part) => normalized.push(part),
490            std::path::Component::ParentDir => {
491                if !normalized.pop() {
492                    return None;
493                }
494            }
495            std::path::Component::RootDir | std::path::Component::Prefix(_) => return None,
496        }
497    }
498    Some(normalized)
