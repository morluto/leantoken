4    /// Construct an indexer whose dedicated worker pool is created on demand.
5    pub fn new(config: Arc<Config>, storage: Storage) -> Result<Self> {
6        Self::validate_config(&config)?;
7        let repository_root = Arc::new(Dir::open_ambient_dir(
8            &config.root,
9            cap_std::ambient_authority(),
10        )?);
11        let progress = IndexProgressRegistry::new(index_progress_cache_namespace(&config));
12        Ok(Self {
13            config,
14            storage,
15            pool: Arc::new(LazyWorkerPool::new()),
16            repository_root,
17            progress,
18        })
19    }

100    pub(super) fn reconcile_cancellable_report_inner(
101        &self,
102        rebuild: bool,
103        cancellation: &CancellationToken,
104        profiling: StorageProfiling,
105    ) -> Result<ProfiledIndexReport> {
106        for _ in 0..3 {
107            match self.reconcile_once(rebuild, cancellation, profiling) {
108                Err(Error::StaleReconciliation { .. }) => continue,
109                result => return result,
110            }
111        }
112        Err(Error::RetryableConflict(RetryableOperation::Reconciliation))
113    }

156    pub(super) fn reconcile_once_with_profiling_hooks(
157        &self,
158        rebuild: bool,
159        cancellation: &CancellationToken,
160        profiling: StorageProfiling,
161        before_preparation: impl FnOnce(),
162        after_publication: impl FnOnce(),
163    ) -> Result<ProfiledIndexReport> {
164        let total_started = Instant::now();
165        let process_write_before = (profiling == StorageProfiling::Collect)
166            .then(process_write_bytes)
167            .flatten();
168        check_cancelled(cancellation)?;
169        let baseline = self.storage.meta()?;
170        let mut progress =
171            (baseline.repository_generation == 0).then(|| self.progress.start(0, cancellation));
172
173        let discovery_started = Instant::now();
174        let discovery = discover_files_with_limits_policy_filter_and_progress(
175            &self.config.root,
176            self.config.discovery_limits(),
177            self.config.discovery_policy(),
178            cancellation,
179            |path| !self.config.is_database_artifact_path(path),
180            |stats| {
181                if let Some(progress) = &progress {
182                    progress.discovered(stats.walk_entries, stats.files, stats.total_source_bytes);
183                }
184            },
185        )?;
186        let discovery_elapsed = discovery_started.elapsed();
187        let discovery_stats = discovery.stats;
188        tracing::debug!(
189            walk_entries = discovery.stats.walk_entries,
190            files = discovery.stats.files,
191            total_source_bytes = discovery.stats.total_source_bytes,
192            max_depth = discovery.stats.max_depth,
193            "repository discovery completed"
194        );
195        let discovered = discovery.files;
196        if let Some(progress) = &progress {
197            progress.discovered(
198                discovery_stats.walk_entries,
199                discovery_stats.files,
200                discovery_stats.total_source_bytes,
201            );
202            progress.phase(IndexProgressPhase::HashAndPlan);
203        }
204        let planning_started = Instant::now();
205        check_cancelled(cancellation)?;
206        let existing = self.existing_files(cancellation)?;
207        let config_hash = self.config_hash();
208        let force = rebuild || baseline.config_hash != config_hash;
209
210        let mut repository_paths = HashSet::with_capacity(discovered.len());
211        for file in &discovered {
212            check_cancelled(cancellation)?;
213            repository_paths.insert(file.relative_path.clone());
214        }
215        let mut deletions = Vec::new();
216        for path in existing.keys() {
217            check_cancelled(cancellation)?;
218            if !repository_paths.contains(path) {
219                deletions.push(path.clone());
220            }
221        }
222
223        let mut unchanged = 0usize;
224        let mut candidates = Vec::new();
225        for file in discovered {
226            check_cancelled(cancellation)?;
227            // Size bounds the identity read; the indexed content hash is the
228            // authority. mtime can churn across checkouts, copy tools, and
229            // filesystems without changing the indexed repository view.
230            if !force
231                && let Some(record) = existing.get(&file.relative_path)
232                && record.size_bytes == file.size_bytes
233                && content_unchanged(
234                    &self.repository_root,
235                    &file.relative_path,
236                    &record.content_hash,
237                    self.config.max_file_bytes,
238                )
239            {
240                unchanged += 1;
241                continue;
242            }
243            candidates.push(file);
244        }
245        let planning_elapsed = planning_started.elapsed();
246
247        let mut removed_paths = deletions.into_iter().collect::<HashSet<_>>();
248        let mut source_bytes = PublishedSourceBytes::new(
249            &existing,
250            &removed_paths,
251            self.config.max_total_source_bytes,
252        );
253        let mut warnings = Vec::new();
254        let mut skip_reasons = IndexSkipReasonCounts::default();
255        let mut files_indexed = 0usize;
256        before_preparation();
257        if let Some(progress) = &progress {
258            progress.phase(IndexProgressPhase::Preparation);
259        }
260
261        // Phase 1: Preparation runs outside BEGIN IMMEDIATE so the SQLite
262        // writer lock is not held during filesystem reads, hashing, parsing,
263        // tokenization, or import resolution. Prepared records are flushed
264        // from each bounded batch into a storage-owned SQLite stage.
265        let mut staged = PreparedReconciliation::new(
266            &self.storage,
267            self.config.tokenizer.name(),
268            &baseline,
269            &config_hash,
270            rebuild,
271            profiling == StorageProfiling::Collect,
272        )?;
273        const MAX_INITIAL_REMOVALS_PER_STAGE: usize = 256;
274        for path in &removed_paths {
275            staged.stage_removal(path.clone());
276            if staged.pending_removals() >= MAX_INITIAL_REMOVALS_PER_STAGE {
277                staged.flush()?;
278            }
279        }
280        let preparation = self.prepare_candidate_batches_with_progress(
281            &candidates,
282            cancellation,
283            profiling,
284            || {
285                if let Some(progress) = &progress {
286                    progress.phase(IndexProgressPhase::Preparation);
287                }
288            },
289            |prepared| {
290                if let Some(progress) = &progress {
291                    progress.prepared_batch(prepared.len());
292                    progress.phase(IndexProgressPhase::RelationalWrite);
293                }
294                let mut indexed = Vec::with_capacity(prepared.len());
295                let mut source_token_counts = HashMap::with_capacity(prepared.len());
296                for result in prepared {
297                    check_cancelled(cancellation)?;
298                    match result {
299                        PreparedFile::Indexed(file, source_token_count, warning) => {
300                            source_bytes.replace(&file.path, file.size_bytes);
301                            source_token_counts.insert(file.path.clone(), source_token_count);
302                            indexed.push(*file);
303                            if let Some(warning) = warning {
304                                push_warning(&mut warnings, warning);
305                            }
306                        }
307                        PreparedFile::Binary(path) => {
308                            source_bytes.remove(&path);
309                            skip_reasons.binary = skip_reasons.binary.saturating_add(1);
310                            if existing.contains_key(&path) && removed_paths.insert(path.clone()) {
311                                staged.stage_removal(path);
312                            }
313                        }
314                        PreparedFile::Oversized(path) => {
315                            source_bytes.remove(&path);
316                            skip_reasons.oversized_during_read =
317                                skip_reasons.oversized_during_read.saturating_add(1);
318                            if existing.contains_key(&path) && removed_paths.insert(path.clone()) {
319                                staged.stage_removal(path);
320                            }
321                        }
322                        PreparedFile::Failed(path, error) => {
323                            skip_reasons.failed = skip_reasons.failed.saturating_add(1);
324                            push_warning(&mut warnings, format!("{path}: {error}"));
325                        }
326                    }
327                }
328                resolve_imports(&mut indexed, &repository_paths, cancellation)?;
329                let staged_files = indexed.len();
330                files_indexed = files_indexed.saturating_add(indexed.len());
331                for file in indexed {
332                    check_cancelled(cancellation)?;
333                    let source_token_count = source_token_counts
334                        .remove(&file.path)
335                        .expect("prepared file has a source token count");
336                    staged.stage_indexed(file, source_token_count);
337                }
338                if let Some(progress) = &progress {
339                    progress.staged(staged_files);
340                }
341                staged.flush()?;
342                Ok(())
343            },
344        )?;
345        source_bytes.enforce()?;
346        let staged = staged.finish()?;
347        let staging = staged.diagnostics();
348        check_cancelled(cancellation)?;
349
350        // Phase 2: Publication inside BEGIN IMMEDIATE performs only fast
351        // DELETE + INSERT operations via staged.apply.  The transaction
352        // rechecks the baseline generation and config_hash before applying.
353        let publication_started = Instant::now();
354        let publish = |writer: &mut ReconciliationWriter<'_, '_>| {
355            staged.apply(writer)?;
356            Ok(preparation)
357        };
358        let observe_publication =
359            |phase| observe_publication_phase(progress.as_ref(), cancellation, phase);
360        let (generation, preparation, mut publication_detail) = if profiling
361            == StorageProfiling::Collect
362        {
363            self.storage
364                .publish_reconciliation_profiled_at_with_progress(
365                    &baseline,
366                    &config_hash,
367                    rebuild,
368                    observe_publication,
369                    publish,
370                )?
371        } else {
372            let (generation, preparation) = self.storage.publish_reconciliation_at_with_progress(
373                &baseline,
374                &config_hash,
375                rebuild,
376                observe_publication,
377                publish,
378            )?;
379            (generation, preparation, PublicationDiagnostics::default())
380        };
381        after_publication();
382        if let Some(progress) = &mut progress {
383            progress.complete(generation);
384        }
385        publication_detail.stage_write_ms = staging.write_ms;
386        publication_detail.stage_write_bytes = staging.write_bytes;
387        publication_detail.stage_database_bytes = staging.database_bytes;
388        let publication_elapsed = publication_started.elapsed();
389
390        // Storage returns only after the generation transaction commits. A
391        // cancellation observed after this commit point cannot roll publication
392        // back and must not turn committed success into a `Cancelled` outcome.
393        let files_seen = unchanged + candidates.len();
394        let files_removed = removed_paths.len();
395        let files_skipped = skip_reasons.total();
396
397        let response = IndexResponse {
398            repository_generation: generation,
399            files_seen,
400            files_indexed,
401            files_unchanged: unchanged,
402            files_removed,
403            files_skipped,
404            warnings,
405        };
406        let report = IndexReport::with_skip_reasons(response, skip_reasons);
407        let process_write_after = (profiling == StorageProfiling::Collect)
408            .then(process_write_bytes)
409            .flatten();
410        let process_write_bytes = process_write_before
411            .zip(process_write_after)
412            .map(|(before, after)| after.saturating_sub(before));
413        let storage_phase_write_bytes = publication_detail.measured_write_bytes();
414        let unattributed_process_write_bytes = process_write_bytes
415            .zip(storage_phase_write_bytes)
416            .map(|(total, attributed)| total.saturating_sub(attributed));
417        let diagnostics = IndexingDiagnostics {
418            total_ms: duration_ms(total_started.elapsed()),
419            discovery_ms: duration_ms(discovery_elapsed),
420            hash_and_plan_ms: duration_ms(planning_elapsed),
421            preparation_ms: duration_ms(preparation.preparation),
422            preparation_detail: preparation.detail.report(),
423            preparation_by_language: preparation
424                .detail_by_language
425                .iter()
426                .map(|(language, detail)| (language.clone(), detail.report()))
427                .collect(),
428            insertion_ms: duration_ms(preparation.insertion),
429            publication_ms: duration_ms(publication_elapsed),
430            publication_detail,
431            process_write_bytes,
432            storage_phase_write_bytes,
433            unattributed_process_write_bytes,
434            generation_before: baseline.repository_generation,
435            generation_after: generation,
436            generation_published: generation != baseline.repository_generation,
437            preparation_batches: preparation.batches,
438            max_batch_files: preparation.max_batch_files,
439            max_batch_source_bytes: preparation.max_batch_source_bytes,
440            walk_entries: discovery_stats.walk_entries,
441            discovered_files: discovery_stats.files,
442            discovered_source_bytes: discovery_stats.total_source_bytes,
443        };
444        tracing::debug!(
445            total_ms = diagnostics.total_ms,
446            discovery_ms = diagnostics.discovery_ms,
447            hash_and_plan_ms = diagnostics.hash_and_plan_ms,
448            preparation_ms = diagnostics.preparation_ms,
449            insertion_ms = diagnostics.insertion_ms,
450            publication_ms = diagnostics.publication_ms,
451            preparation_batches = diagnostics.preparation_batches,
452            max_batch_files = diagnostics.max_batch_files,
453            max_batch_source_bytes = diagnostics.max_batch_source_bytes,
454            "repository reconciliation profile"
455        );
456        Ok(ProfiledIndexReport {
457            report,
458            diagnostics,
459        })
460    }

463pub(super) fn observe_publication_phase(
464    progress: Option<&IndexProgressAttempt>,
465    cancellation: &CancellationToken,
466    phase: ReconciliationPublicationPhase,
467) -> Result<()> {
468    if let Some(progress) = progress {
469        progress.phase(match phase {
470            ReconciliationPublicationPhase::ChunkWordFts => IndexProgressPhase::ChunkWordFts,
471            ReconciliationPublicationPhase::ChunkTrigramFts => IndexProgressPhase::ChunkTrigramFts,
472            ReconciliationPublicationPhase::SymbolFts => IndexProgressPhase::SymbolFts,
473            ReconciliationPublicationPhase::ReferenceFts => IndexProgressPhase::ReferenceFts,
474            ReconciliationPublicationPhase::CommitAndCheckpoint => {
475                IndexProgressPhase::CommitAndCheckpoint
476            }
477        });
478    }
479    check_cancelled(cancellation)
