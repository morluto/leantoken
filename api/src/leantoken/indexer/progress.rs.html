44impl IndexProgressRegistry {
45    pub(super) fn new(cache_namespace: String) -> Self {
46        Self {
47            shared: Arc::new(Mutex::new(IndexProgressRegistryState::default())),
48            cache_namespace: Arc::from(cache_namespace),
49            registry_id: NEXT_INDEX_PROGRESS_REGISTRY_ID
50                .fetch_add(1, std::sync::atomic::Ordering::Relaxed),
51        }
52    }
53
54    pub(super) fn start(
55        &self,
56        current_generation: u64,
57        cancellation: &CancellationToken,
58    ) -> IndexProgressAttempt {
59        let started = Instant::now();
60        let unix_duration = UNIX_EPOCH.elapsed().unwrap_or_default();
61        let started_unix_ms = saturating_duration_millis(unix_duration);
62        let mut state = self
63            .shared
64            .lock()
65            .unwrap_or_else(|poisoned| poisoned.into_inner());
66        state.next_attempt = state.next_attempt.saturating_add(1);
67        let internal_id = state.next_attempt;
68
69        let mut hasher = blake3::Hasher::new();
70        hasher.update(b"leantoken-index-attempt-v1\0");
71        hasher.update(&std::process::id().to_le_bytes());
72        hasher.update(&unix_duration.as_nanos().to_le_bytes());
73        hasher.update(&self.registry_id.to_le_bytes());
74        hasher.update(&internal_id.to_le_bytes());
75        let attempt_id = hasher.finalize().to_hex()[..32].to_owned();
76
77        state.current = Some(IndexProgressAttemptState {
78            internal_id,
79            attempt_id,
80            active: true,
81            current_generation,
82            phase: IndexProgressPhase::Discovery,
83            started,
84            started_unix_ms,
85            last_progress_unix_ms: started_unix_ms,
86            update_sequence: 1,
87            walk_entries: 0,
88            files_discovered: 0,
89            discovered_source_bytes: 0,
90            files_prepared: 0,
91            files_staged: 0,
92            preparation_batches: 0,
93        });
94        drop(state);
95
96        IndexProgressAttempt {
97            registry: self.clone(),
98            cancellation: cancellation.clone(),
99            internal_id,
100            finished: false,
101        }
102    }
103
104    pub(super) fn snapshot(&self) -> Option<IndexProgressSnapshot> {
105        let state = self
106            .shared
107            .lock()
108            .unwrap_or_else(|poisoned| poisoned.into_inner());
109        let current = state.current.as_ref()?;
110        Some(IndexProgressSnapshot {
111            cache_namespace: self.cache_namespace.to_string(),
112            detail_available: true,
113            active: current.active,
114            current_generation: current.current_generation,
115            attempt_id: Some(current.attempt_id.clone()),
116            phase: Some(current.phase),
117            started_unix_ms: Some(current.started_unix_ms),
118            elapsed_ms: Some(saturating_duration_millis(current.started.elapsed())),
119            last_progress_unix_ms: Some(current.last_progress_unix_ms),
120            update_sequence: Some(current.update_sequence),
121            walk_entries: Some(current.walk_entries),
122            files_discovered: Some(current.files_discovered),
123            discovered_source_bytes: Some(current.discovered_source_bytes),
124            files_prepared: Some(current.files_prepared),
125            files_staged: Some(current.files_staged),
126            preparation_batches: Some(current.preparation_batches),
127        })
128    }
129
130    pub(super) fn update(
131        &self,
132        internal_id: u64,
133        update: impl FnOnce(&mut IndexProgressAttemptState),
134    ) {
135        let mut state = self
136            .shared
137            .lock()
138            .unwrap_or_else(|poisoned| poisoned.into_inner());
139        let Some(current) = state
140            .current
141            .as_mut()
142            .filter(|current| current.internal_id == internal_id && current.active)
143        else {
144            return;
145        };
146        update(current);
147        current.update_sequence = current.update_sequence.saturating_add(1);
148        current.last_progress_unix_ms =
149            saturating_duration_millis(UNIX_EPOCH.elapsed().unwrap_or_default());
150    }
151}

153impl IndexProgressAttempt {
154    pub(super) fn phase(&self, phase: IndexProgressPhase) {
155        self.registry
156            .update(self.internal_id, |current| current.phase = phase);
157    }
158
159    pub(super) fn discovered(&self, walk_entries: u64, files: u64, source_bytes: u64) {
160        self.registry.update(self.internal_id, |current| {
161            current.walk_entries = walk_entries;
162            current.files_discovered = files;
163            current.discovered_source_bytes = source_bytes;
164        });
165    }
166
167    pub(super) fn prepared_batch(&self, files: usize) {
168        let files = u64::try_from(files).unwrap_or(u64::MAX);
169        self.registry.update(self.internal_id, |current| {
170            current.files_prepared = current.files_prepared.saturating_add(files);
171            current.preparation_batches = current.preparation_batches.saturating_add(1);
172        });
173    }
174
175    pub(super) fn staged(&self, files: usize) {
176        let files = u64::try_from(files).unwrap_or(u64::MAX);
177        self.registry.update(self.internal_id, |current| {
178            current.files_staged = current.files_staged.saturating_add(files);
179        });
180    }
181
182    pub(super) fn complete(&mut self, generation: u64) {
183        self.registry.update(self.internal_id, |current| {
184            current.current_generation = generation;
185            current.phase = IndexProgressPhase::Completed;
186            current.active = false;
187        });
188        self.finished = true;
189    }
190}
191
192impl Drop for IndexProgressAttempt {
193    fn drop(&mut self) {
194        if self.finished {
195            return;
196        }
197        let phase = if self.cancellation.is_cancelled() {
198            IndexProgressPhase::Cancelled
199        } else {
200            IndexProgressPhase::Failed
201        };
202        self.registry.update(self.internal_id, |current| {
203            current.phase = phase;
204            current.active = false;
205        });
206    }
207}

209pub(super) fn saturating_duration_millis(duration: Duration) -> u64 {
210    u64::try_from(duration.as_millis()).unwrap_or(u64::MAX)
211}

213pub(crate) fn index_progress_cache_namespace(config: &Config) -> String {
214    let mut hasher = blake3::Hasher::new();
215    hasher.update(b"leantoken-index-progress-cache-v1\0");
216    hasher.update(&INDEX_CONTENT_VERSION.to_le_bytes());
217    hasher.update(config.database_path.as_os_str().as_encoded_bytes());
218    hasher.finalize().to_hex()[..32].to_owned()
