377impl CacheCompatibility {
378    const ALL: [Self; 5] = [
379        Self::CompatibleCurrent,
380        Self::ObsoleteOlder,
381        Self::LegacyUnversioned,
382        Self::NewerUnsupported,
383        Self::Unknown,
384    ];
385
386    fn classify(entry: &CacheEntry) -> Self {
387        if matches!(entry.state, CacheState::Corrupt | CacheState::Unrecognized) {
388            return Self::Unknown;
389        }
390        match entry.index_content_version {
391            Some(version) if version == INDEX_CONTENT_VERSION => Self::CompatibleCurrent,
392            Some(version) if version < INDEX_CONTENT_VERSION => Self::ObsoleteOlder,
393            Some(_) => Self::NewerUnsupported,
394            None => Self::LegacyUnversioned,
395        }
396    }
397
398    fn label(self) -> &'static str {
399        match self {
400            Self::CompatibleCurrent => "compatible_current",
401            Self::ObsoleteOlder => "obsolete_older",
402            Self::LegacyUnversioned => "legacy_unversioned",
403            Self::NewerUnsupported => "newer_unsupported",
404            Self::Unknown => "unknown",
405        }
406    }
407
408    fn safely_incompatible(self) -> bool {
409        matches!(self, Self::ObsoleteOlder | Self::LegacyUnversioned)
410    }
411}

491    fn list_with(&self, request: &CacheListRequest) -> Result<CacheListReport> {
492        validate_list_request(request)?;
493        let repository_root = request
494            .repository_root
495            .as_deref()
496            .map(normalize_repository_root_filter);
497        let filter_hash = cache_list_filter_hash(request, repository_root.as_deref());
498        let after_id = request
499            .cursor
500            .as_deref()
501            .map(|cursor| decode_cache_list_cursor(cursor, &filter_hash))
502            .transpose()?;
503
504        let (entries, ignored_entries) = self.inspect_all()?;
505        let total_bytes = entries.iter().fold(0u64, |total, cache| {
506            total.saturating_add(cache.entry.size_bytes)
507        });
508        let matching = entries
509            .iter()
510            .filter(|cache| {
511                (request.states.is_empty() || request.states.contains(&cache.entry.state))
512                    && repository_root
513                        .as_ref()
514                        .is_none_or(|root| cache.entry.repository_root.as_ref() == Some(root))
515            })
516            .collect::<Vec<_>>();
517        let matched_bytes = matching.iter().fold(0u64, |total, cache| {
518            total.saturating_add(cache.entry.size_bytes)
519        });
520        let active_entries = matching.iter().filter(|cache| cache.entry.active).count();
521        let missing_root_entries = matching
522            .iter()
523            .filter(|cache| cache.entry.repository_available == Some(false))
524            .count();
525        let mut state_counts = CacheState::ALL
526            .into_iter()
527            .map(|state| (state.label().to_owned(), 0usize))
528            .collect::<BTreeMap<_, _>>();
529        for cache in &matching {
530            *state_counts
531                .get_mut(cache.entry.state.label())
532                .expect("every cache state has a summary bucket") += 1;
533        }
534
535        let start = after_id.as_deref().map_or(0, |after_id| {
536            matching.partition_point(|cache| cache.entry.id.as_str() <= after_id)
537        });
538        let end = if request.summary {
539            start
540        } else {
541            start.saturating_add(request.limit).min(matching.len())
542        };
543        let page = matching[start..end]
544            .iter()
545            .map(|cache| cache.entry.clone())
546            .collect::<Vec<_>>();
547        let next_cursor = if !request.summary && end < matching.len() {
548            page.last()
549                .map(|entry| encode_cache_list_cursor(&filter_hash, &entry.id))
550        } else {
551            None
552        };
553        Ok(CacheListReport {
554            cache_root: self.root.clone(),
555            total_entries: entries.len(),
556            matched_entries: matching.len(),
557            returned_entries: page.len(),
558            total_bytes,
559            matched_bytes,
560            active_entries,
561            missing_root_entries,
562            state_counts,
563            ignored_entries,
564            summary_only: request.summary,
565            next_cursor,
566            entries: page,
567        })
568    }

570    fn list_v2_with(&self, request: &CacheListV2Request) -> Result<CacheListV2Report> {
571        validate_list_request(&request.request)?;
572        if request.compatibilities.len() > MAX_CACHE_COMPATIBILITY_FILTERS {
573            return Err(Error::RequestLimitExceeded {
574                field: "cache compatibility filters",
575                requested: request.compatibilities.len(),
576                limit: MAX_CACHE_COMPATIBILITY_FILTERS,
577            });
578        }
579        if request.index_content_versions.len() > MAX_CACHE_CONTENT_VERSION_FILTERS {
580            return Err(Error::RequestLimitExceeded {
581                field: "cache content-version filters",
582                requested: request.index_content_versions.len(),
583                limit: MAX_CACHE_CONTENT_VERSION_FILTERS,
584            });
585        }
586        if request.index_content_versions.contains(&0) {
587            return Err(Error::InvalidInput {
588                field: "cache content-version filter",
589                reason: "must be positive",
590            });
591        }
592        let repository_root = request
593            .request
594            .repository_root
595            .as_deref()
596            .map(normalize_repository_root_filter);
597        let filter_hash = cache_list_v2_filter_hash(request, repository_root.as_deref());
598        let after_id = request
599            .request
600            .cursor
601            .as_deref()
602            .map(|cursor| {
603                decode_cache_list_cursor_with_prefix(
604                    cursor,
605                    CACHE_LIST_V2_CURSOR_PREFIX,
606                    &filter_hash,
607                )
608            })
609            .transpose()?;
610
611        let (entries, ignored_entries) = self.inspect_all()?;
612        let total_bytes = entries.iter().fold(0u64, |total, cache| {
613            total.saturating_add(cache.entry.size_bytes)
614        });
615        let matching = entries
616            .iter()
617            .filter(|cache| {
618                (request.request.states.is_empty()
619                    || request.request.states.contains(&cache.entry.state))
620                    && repository_root
621                        .as_ref()
622                        .is_none_or(|root| cache.entry.repository_root.as_ref() == Some(root))
623                    && (request.compatibilities.is_empty()
624                        || request.compatibilities.contains(&cache.compatibility))
625                    && (request.index_content_versions.is_empty()
626                        || cache.entry.index_content_version.is_some_and(|version| {
627                            request.index_content_versions.contains(&version)
628                        }))
629                    && (!request.incompatible_with_current
630                        || cache.compatibility.safely_incompatible())
631            })
632            .collect::<Vec<_>>();
633        let matched_bytes = matching.iter().fold(0u64, |total, cache| {
634            total.saturating_add(cache.entry.size_bytes)
635        });
636        let active_entries = matching.iter().filter(|cache| cache.entry.active).count();
637        let missing_root_entries = matching
638            .iter()
639            .filter(|cache| cache.entry.repository_available == Some(false))
640            .count();
641        let mut state_counts = CacheState::ALL
642            .into_iter()
643            .map(|state| (state.label().to_owned(), 0usize))
644            .collect::<BTreeMap<_, _>>();
645        let mut compatibility_counts = CacheCompatibility::ALL
646            .into_iter()
647            .map(|compatibility| {
648                (
649                    compatibility.label().to_owned(),
650                    CacheCompatibilitySummary::default(),
651                )
652            })
653            .collect::<BTreeMap<_, _>>();
654        let mut safely_reclaimable_incompatible_entries = 0usize;
655        let mut safely_reclaimable_incompatible_bytes = 0u64;
656        for cache in &matching {
657            *state_counts
658                .get_mut(cache.entry.state.label())
659                .expect("every cache state has a summary bucket") += 1;
660            let summary = compatibility_counts
661                .get_mut(cache.compatibility.label())
662                .expect("every compatibility has a summary bucket");
663            summary.entries = summary.entries.saturating_add(1);
664            summary.bytes = summary.bytes.saturating_add(cache.entry.size_bytes);
665            if cache.compatibility.safely_incompatible()
666                && cache.safe_to_prune
667                && !cache.entry.active
668            {
669                safely_reclaimable_incompatible_entries =
670                    safely_reclaimable_incompatible_entries.saturating_add(1);
671                safely_reclaimable_incompatible_bytes =
672                    safely_reclaimable_incompatible_bytes.saturating_add(cache.entry.size_bytes);
673            }
674        }
675
676        let start = after_id.as_deref().map_or(0, |after_id| {
677            matching.partition_point(|cache| cache.entry.id.as_str() <= after_id)
678        });
679        let end = if request.request.summary {
680            start
681        } else {
682            start
683                .saturating_add(request.request.limit)
684                .min(matching.len())
685        };
686        let page = matching[start..end]
687            .iter()
688            .map(|cache| CacheEntryV2 {
689                entry: cache.entry.clone(),
690                compatibility: cache.compatibility,
691            })
692            .collect::<Vec<_>>();
693        let next_cursor = if !request.request.summary && end < matching.len() {
694            page.last().map(|entry| {
695                encode_cache_list_cursor_with_prefix(
696                    CACHE_LIST_V2_CURSOR_PREFIX,
697                    &filter_hash,
698                    &entry.entry.id,
699                )
700            })
701        } else {
702            None
703        };
704        Ok(CacheListV2Report {
705            report_version: 2,
706            cache_root: self.root.clone(),
707            total_entries: entries.len(),
708            matched_entries: matching.len(),
709            returned_entries: page.len(),
710            total_bytes,
711            matched_bytes,
712            active_entries,
713            missing_root_entries,
714            state_counts,
715            compatibility_counts,
716            safely_reclaimable_incompatible_entries,
717            safely_reclaimable_incompatible_bytes,
718            ignored_entries,
719            summary_only: request.request.summary,
720            next_cursor,
721            entries: page,
722        })
723    }

733    fn prune_with_compatibility(
734        &self,
735        request: &CachePruneRequest,
736        incompatible_with_current: bool,
737    ) -> Result<CachePruneReport> {
738        validate_prune_request(request, incompatible_with_current)?;
739        let (entries, _) = self.inspect_all()?;
740        let total_bytes_before = entries.iter().fold(0u64, |total, cache| {
741            total.saturating_add(cache.entry.size_bytes)
742        });
743        let selected = select_prune_candidates(
744            &entries,
745            request,
746            total_bytes_before,
747            incompatible_with_current,
748        );
749        let mut reclaimed_bytes = 0u64;
750        let mut results = Vec::with_capacity(entries.len());
751
752        for cache in entries {
753            let Some(mut reasons) = selected.get(&cache.entry.id).cloned() else {
754                results.push(prune_result(
755                    &cache,
756                    CachePruneAction::Kept,
757                    Vec::new(),
758                    None,
759                ));
760                continue;
761            };
762            if cache.entry.active {
763                results.push(prune_result(
764                    &cache,
765                    CachePruneAction::SkippedActive,
766                    reasons,
767                    None,
768                ));
769                continue;
770            }
771            if !cache.safe_to_prune {
772                results.push(prune_result(
773                    &cache,
774                    CachePruneAction::SkippedUnsafe,
775                    reasons,
776                    None,
777                ));
778                continue;
779            }
780            if request.dry_run {
781                reclaimed_bytes = reclaimed_bytes.saturating_add(cache.entry.size_bytes);
782                results.push(prune_result(
783                    &cache,
784                    CachePruneAction::WouldDelete,
785                    reasons,
786                    None,
787                ));
788                continue;
789            }
790
791            let database = cache.entry.path.join(DATABASE_NAME);
792            let coordination = IndexCoordination::for_database(&database);
793            let _lease = match coordination.try_acquire_prune_lease() {
794                Ok(Some(lease)) => lease,
795                Ok(None) => {
796                    reasons.push("prune_lease_unavailable".into());
797                    results.push(prune_result(
798                        &cache,
799                        CachePruneAction::SkippedActive,
800                        reasons,
801                        None,
802                    ));
803                    continue;
804                }
805                Err(error) => {
806                    results.push(prune_result(
807                        &cache,
808                        CachePruneAction::Failed,
809                        reasons,
810                        Some(error.to_string()),
811                    ));
812                    continue;
813                }
814            };
815            let current = match self.inspect_cache(&cache.entry.id, false) {
816                Ok(current) => current,
817                Err(error) => {
818                    results.push(prune_result(
819                        &cache,
820                        CachePruneAction::Failed,
821                        reasons,
822                        Some(error.to_string()),
823                    ));
824                    continue;
825                }
826            };
827            let selected_for_compatibility = reasons
828                .iter()
829                .any(|reason| reason.starts_with("incompatible_with_current:"));
830            if selected_for_compatibility && !current.compatibility.safely_incompatible() {
831                reasons.retain(|reason| !reason.starts_with("incompatible_with_current:"));
832                if reasons.is_empty() {
833                    reasons.push(format!(
834                        "incompatible_with_current_revalidated:{}",
835                        current.compatibility.label()
836                    ));
837                    results.push(prune_result(
838                        &current,
839                        CachePruneAction::Kept,
840                        reasons,
841                        None,
842                    ));
843                    continue;
844                }
845            }
846            if !current.safe_to_prune {
847                results.push(prune_result(
848                    &current,
849                    CachePruneAction::SkippedUnsafe,
850                    reasons,
851                    None,
852                ));
853                continue;
854            }
855            if reasons.len() == 1
856                && reasons[0] == "missing_repository"
857                && current.entry.repository_available != Some(false)
858            {
859                results.push(prune_result(
860                    &current,
861                    CachePruneAction::Kept,
862                    reasons,
863                    None,
864                ));
865                continue;
866            }
867
868            let removal = remove_managed_artifacts(&current.entry.path);
869            reclaimed_bytes = reclaimed_bytes.saturating_add(removal.reclaimed_bytes);
870            match removal.error {
871                None => {
872                    results.push(prune_result(
873                        &current,
874                        CachePruneAction::Deleted,
875                        reasons,
876                        None,
877                    ));
878                }
879                Some(error) => results.push(prune_result(
880                    &current,
881                    CachePruneAction::Failed,
882                    reasons,
883                    Some(error),
884                )),
885            }
886        }
887
888        results.sort_by(|left, right| left.id.cmp(&right.id));
889        Ok(CachePruneReport {
890            cache_root: self.root.clone(),
891            dry_run: request.dry_run,
892            total_bytes_before,
893            total_bytes_after: total_bytes_before.saturating_sub(reclaimed_bytes),
894            reclaimed_bytes,
895            results,
896        })
897    }

899    fn inspect_all(&self) -> Result<(Vec<InspectedCache>, usize)> {
900        let read_dir = match fs::read_dir(&self.root) {
901            Ok(entries) => entries,
902            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
903                return Ok((Vec::new(), 0));
904            }
905            Err(error) => return Err(error.into()),
906        };
907        let mut entries = Vec::new();
908        let mut ignored = 0usize;
909        for entry in read_dir {
910            let entry = entry?;
911            let file_type = entry.file_type()?;
912            let Some(id) = entry.file_name().to_str().map(str::to_owned) else {
913                ignored += 1;
914                continue;
915            };
916            if !file_type.is_dir() || !is_cache_id(&id) {
917                ignored += 1;
918                continue;
919            }
920            let cache = self.inspect_cache(&id, true)?;
921            if cache.entry.size_bytes == 0 && cache.entry.state == CacheState::Incomplete {
922                continue;
923            }
924            entries.push(cache);
925        }
926        entries.sort_by(|left, right| left.entry.id.cmp(&right.entry.id));
927        Ok((entries, ignored))
928    }

930    fn inspect_cache(&self, id: &str, probe_active: bool) -> Result<InspectedCache> {
931        let path = self.root.join(id);
932        let database = path.join(DATABASE_NAME);
933        let identity = parse_managed_cache_id(id).expect("validated managed cache identity");
934        let index_content_version = match identity {
935            ManagedCacheIdentity::Legacy => None,
936            ManagedCacheIdentity::Versioned(version) => Some(version),
937        };
938        let initial_scan = scan_artifacts(&path)?;
939        let latest_access_mtime = initial_scan.latest_access_mtime;
940        let mut unexpected = initial_scan.unexpected;
941        let mut metadata_safe = true;
942
943        let active = if probe_active && path.join(LEASE_NAME).exists() {
944            IndexCoordination::for_database(&database)
945                .try_acquire_prune_lease()?
946                .is_none()
947        } else {
948            false
949        };
950        let mut entry = CacheEntry {
951            id: id.into(),
952            path,
953            index_content_version,
954            repository_root: None,
955            repository_available: None,
956            last_access_unix_seconds: latest_access_mtime,
957            access_time_source: latest_access_mtime.map(|_| AccessTimeSource::FileMtime),
958            age_seconds: latest_access_mtime.map(|accessed| self.now.saturating_sub(accessed)),
959            schema_version: None,
960            size_bytes: initial_scan.size_bytes,
961            active,
962            state: CacheState::Incomplete,
963            detail: None,
964        };
965
966        let database_is_regular =
967            fs::symlink_metadata(&database).is_ok_and(|metadata| metadata.file_type().is_file());
968        if initial_scan.has_artifacts && database_is_regular {
969            match inspect_database(&database) {
970                Ok(metadata) => {
971                    entry.schema_version = metadata.schema_version;
972                    entry.repository_root = metadata.repository_root;
973                    entry.repository_available =
974                        entry.repository_root.as_deref().and_then(root_available);
975                    if let Some(accessed) = metadata.last_access_unix_seconds {
976                        entry.last_access_unix_seconds = Some(accessed);
977                        entry.access_time_source = Some(AccessTimeSource::Database);
978                        entry.age_seconds = Some(self.now.saturating_sub(accessed));
979                    }
980                    entry.state = if metadata.future_schema {
981                        metadata_safe = false;
982                        entry.detail = Some("cache uses a newer unsupported schema".into());
983                        CacheState::Unsupported
984                    } else if metadata.current {
985                        CacheState::Current
986                    } else {
987                        CacheState::Legacy
988                    };
989                    if let Some(repository_root) = &entry.repository_root
990                        && !managed_cache_id_matches_root(id, repository_root)
991                    {
992                        metadata_safe = false;
993                        entry.state = CacheState::Unsupported;
994                        entry.detail =
995                            Some("cache identity does not match its recorded root".into());
996                    }
997                }
998                Err(error) => {
999                    metadata_safe = false;
1000                    entry.state = CacheState::Corrupt;
1001                    entry.detail = Some(error.to_string());
1002                }
1003            }
1004        }
1005        if index_content_version.is_some_and(|version| version > INDEX_CONTENT_VERSION) {
1006            metadata_safe = false;
1007            entry.state = CacheState::Unsupported;
1008            entry.detail = Some("cache uses a newer index-content version".into());
1009        }
1010        let final_scan = scan_artifacts(&entry.path)?;
1011        entry.size_bytes = final_scan.size_bytes;
1012        unexpected |= final_scan.unexpected;
1013        if unexpected {
1014            entry.state = CacheState::Unrecognized;
1015            entry.detail = Some("cache directory contains unexpected entries".into());
1016        }
1017        let compatibility = CacheCompatibility::classify(&entry);
1018
1019        Ok(InspectedCache {
1020            safe_to_prune: final_scan.has_artifacts && !unexpected && metadata_safe,
1021            entry,
1022            compatibility,
1023        })
1024    }
1025}

1027fn scan_artifacts(path: &Path) -> Result<ArtifactScan> {
1028    let mut scan = ArtifactScan {
1029        size_bytes: 0,
1030        latest_access_mtime: None,
1031        has_artifacts: false,
1032        unexpected: false,
1033    };
1034    for child in fs::read_dir(path)? {
1035        let child = child?;
1036        let metadata = fs::symlink_metadata(child.path())?;
1037        let known = child
1038            .file_name()
1039            .to_str()
1040            .is_some_and(|name| PRUNABLE_ARTIFACTS.contains(&name) || name == LEASE_NAME);
1041        if !known || !metadata.file_type().is_file() {
1042            scan.unexpected = true;
1043            continue;
1044        }
1045        if child.file_name() == OsStr::new(LEASE_NAME) {
1046            continue;
1047        }
1048        scan.has_artifacts = true;
1049        scan.size_bytes = scan.size_bytes.saturating_add(metadata.len());
1050        let name = child.file_name();
1051        // Read-only WAL inspection can refresh SHM and lock-file mtimes.
1052        if (name == OsStr::new(DATABASE_NAME) || name == OsStr::new(WAL_NAME))
1053            && let Ok(modified) = metadata.modified()
1054        {
1055            let modified = unix_seconds(modified);
1056            scan.latest_access_mtime = Some(
1057                scan.latest_access_mtime
1058                    .map_or(modified, |current| current.max(modified)),
1059            );
1060        }
1061    }
1062    Ok(scan)
1063}

1074fn inspect_database(path: &Path) -> Result<DatabaseMetadata> {
1075    let connection = Connection::open_with_flags(
1076        path,
1077        OpenFlags::SQLITE_OPEN_READ_ONLY | OpenFlags::SQLITE_OPEN_NO_MUTEX,
1078    )?;
1079    connection.busy_timeout(Duration::from_millis(100))?;
1080    let migration_version =
1081        connection.pragma_query_value(None, "user_version", |row| row.get::<_, i64>(0))?;
1082    if migration_version > CURRENT_MIGRATION_VERSION {
1083        return Ok(DatabaseMetadata {
1084            schema_version: None,
1085            repository_root: None,
1086            last_access_unix_seconds: None,
1087            current: false,
1088            future_schema: true,
1089        });
1090    }
1091    let mut statement = connection.prepare("PRAGMA table_info(meta)")?;
1092    let columns = statement
1093        .query_map([], |row| row.get::<_, String>(1))?
1094        .collect::<std::result::Result<BTreeSet<_>, _>>()?;
1095    if !columns.contains("schema_version") {
1096        return Err(Error::InvalidConfiguration(
1097            "cache metadata table has no schema version".into(),
1098        ));
1099    }
1100    let schema_version =
1101        connection.query_row("SELECT schema_version FROM meta WHERE id = 1", [], |row| {
1102            row.get::<_, i64>(0)
1103        })?;
1104    if schema_version > CURRENT_SCHEMA_VERSION {
1105        return Ok(DatabaseMetadata {
1106            schema_version: Some(schema_version),
1107            repository_root: None,
1108            last_access_unix_seconds: None,
1109            current: false,
1110            future_schema: true,
1111        });
1112    }
1113    let repository_root = if columns.contains("repository_root") {
1114        let root =
1115            connection.query_row("SELECT repository_root FROM meta WHERE id = 1", [], |row| {
1116                row.get::<_, String>(0)
1117            })?;
1118        (!root.is_empty()).then(|| PathBuf::from(root))
1119    } else {
1120        None
1121    };
1122    let last_access_unix_seconds = if columns.contains("last_access_unix_seconds") {
1123        let accessed = connection.query_row(
1124            "SELECT last_access_unix_seconds FROM meta WHERE id = 1",
1125            [],
1126            |row| row.get::<_, i64>(0),
1127        )?;
1128        u64::try_from(accessed).ok().filter(|value| *value > 0)
1129    } else {
1130        None
1131    };
1132    Ok(DatabaseMetadata {
1133        schema_version: Some(schema_version),
1134        repository_root,
1135        last_access_unix_seconds,
1136        current: schema_version == CURRENT_SCHEMA_VERSION
1137            && columns.contains("last_access_unix_seconds"),
1138        future_schema: false,
1139    })
1140}

1170fn cache_list_filter_hash(request: &CacheListRequest, repository_root: Option<&Path>) -> String {
1171    let mut hasher = blake3::Hasher::new();
1172    if request.states.is_empty() {
1173        hasher.update(b"all-states");
1174    } else {
1175        for state in CacheState::ALL {
1176            if request.states.contains(&state) {
1177                hasher.update(state.label().as_bytes());
1178                hasher.update(b"\0");
1179            }
1180        }
1181    }
1182    hasher.update(b"\xff");
1183    if let Some(root) = repository_root {
1184        hasher.update(root.as_os_str().as_encoded_bytes());
1185    }
1186    hasher.finalize().to_hex()[..CACHE_LIST_CURSOR_HASH_CHARS].to_owned()
1187}

1189fn cache_list_v2_filter_hash(
1190    request: &CacheListV2Request,
1191    repository_root: Option<&Path>,
1192) -> String {
1193    let mut hasher = blake3::Hasher::new();
1194    hasher.update(cache_list_filter_hash(&request.request, repository_root).as_bytes());
1195    hasher.update(b"\xffcompatibility\0");
1196    if request.compatibilities.is_empty() {
1197        hasher.update(b"all");
1198    } else {
1199        for compatibility in CacheCompatibility::ALL {
1200            if request.compatibilities.contains(&compatibility) {
1201                hasher.update(compatibility.label().as_bytes());
1202                hasher.update(b"\0");
1203            }
1204        }
1205    }
1206    hasher.update(b"\xffcontent-versions\0");
1207    let mut versions = request.index_content_versions.clone();
1208    versions.sort_unstable();
1209    versions.dedup();
1210    if versions.is_empty() {
1211        hasher.update(b"all");
1212    } else {
1213        for version in versions {
1214            hasher.update(&version.to_le_bytes());
1215        }
1216    }
1217    hasher.update(b"\xffincompatible\0");
1218    hasher.update(&[u8::from(request.incompatible_with_current)]);
1219    hasher.finalize().to_hex()[..CACHE_LIST_CURSOR_HASH_CHARS].to_owned()
1220}

1234fn decode_cache_list_cursor_with_prefix(
1235    cursor: &str,
1236    expected_prefix: &str,
1237    expected_filter_hash: &str,
1238) -> Result<String> {
1239    if cursor.len() > MAX_CACHE_LIST_CURSOR_BYTES {
1240        return Err(Error::InputTooLong {
1241            field: "cache list cursor",
1242            max_bytes: MAX_CACHE_LIST_CURSOR_BYTES,
1243        });
1244    }
1245    let mut parts = cursor.splitn(3, ':');
1246    let prefix = parts.next();
1247    let filter_hash = parts.next();
1248    let after_id = parts.next();
1249    if prefix != Some(expected_prefix)
1250        || filter_hash.is_none_or(|hash| {
1251            hash.len() != CACHE_LIST_CURSOR_HASH_CHARS
1252                || !hash.bytes().all(|byte| byte.is_ascii_hexdigit())
1253        })
1254        || after_id.is_none_or(|id| !is_cache_id(id))
1255    {
1256        return Err(Error::InvalidInput {
1257            field: "cache list cursor",
1258            reason: "must be an opaque cursor returned by cache list",
1259        });
1260    }
1261    if filter_hash != Some(expected_filter_hash) {
1262        return Err(Error::InvalidInput {
1263            field: "cache list cursor",
1264            reason: "does not match the active cache filters",
1265        });
1266    }
1267    Ok(after_id.expect("validated cache cursor id").to_owned())
1268}

1270fn validate_prune_request(
1271    request: &CachePruneRequest,
1272    incompatible_with_current: bool,
1273) -> Result<()> {
1274    if request.older_than_days.is_none()
1275        && request.max_total_bytes.is_none()
1276        && !request.remove_missing_roots
1277        && !incompatible_with_current
1278    {
1279        return Err(Error::InvalidRequest(
1280            "cache prune requires --older-than, --max-total-bytes, \
1281             --remove-missing-roots, or --incompatible-with-current"
1282                .into(),
1283        ));
1284    }
1285    if request.older_than_days == Some(0) {
1286        return Err(Error::InvalidRequest(
1287            "--older-than must be at least one day".into(),
1288        ));
1289    }
1290    if !request.dry_run && !request.yes {
1291        return Err(Error::InvalidRequest(
1292            "cache prune requires --yes unless --dry-run is used".into(),
1293        ));
1294    }
1295    Ok(())
1296}

1298fn select_prune_candidates(
1299    entries: &[InspectedCache],
1300    request: &CachePruneRequest,
1301    total_bytes: u64,
1302    incompatible_with_current: bool,
1303) -> BTreeMap<String, Vec<String>> {
1304    let mut selected = BTreeMap::<String, Vec<String>>::new();
1305    let minimum_age = request
1306        .older_than_days
1307        .map(|days| days.saturating_mul(SECONDS_PER_DAY));
1308    for cache in entries {
1309        if incompatible_with_current && cache.compatibility.safely_incompatible() {
1310            selected
1311                .entry(cache.entry.id.clone())
1312                .or_default()
1313                .push(format!(
1314                    "incompatible_with_current:{}",
1315                    cache.compatibility.label()
1316                ));
1317        }
1318        if minimum_age.is_some_and(|age| cache.entry.age_seconds.is_some_and(|value| value >= age))
1319        {
1320            selected
1321                .entry(cache.entry.id.clone())
1322                .or_default()
1323                .push("older_than".into());
1324        }
1325        if request.remove_missing_roots && cache.entry.repository_available == Some(false) {
1326            selected
1327                .entry(cache.entry.id.clone())
1328                .or_default()
1329                .push("missing_repository".into());
1330        }
1331    }
1332
1333    let Some(max_total_bytes) = request.max_total_bytes else {
1334        return selected;
1335    };
1336    let mut projected = total_bytes;
1337    for cache in entries {
1338        if selected.contains_key(&cache.entry.id) && cache.safe_to_prune && !cache.entry.active {
1339            projected = projected.saturating_sub(cache.entry.size_bytes);
1340        }
1341    }
1342    let mut lru = entries
1343        .iter()
1344        .filter(|cache| {
1345            !selected.contains_key(&cache.entry.id) && cache.safe_to_prune && !cache.entry.active
1346        })
1347        .collect::<Vec<_>>();
1348    lru.sort_by(|left, right| {
1349        left.entry
1350            .last_access_unix_seconds
1351            .unwrap_or(0)
1352            .cmp(&right.entry.last_access_unix_seconds.unwrap_or(0))
1353            .then_with(|| left.entry.id.cmp(&right.entry.id))
1354    });
1355    for cache in lru {
1356        if projected <= max_total_bytes {
1357            break;
1358        }
1359        selected
1360            .entry(cache.entry.id.clone())
1361            .or_default()
1362            .push("max_total_bytes".into());
1363        projected = projected.saturating_sub(cache.entry.size_bytes);
1364    }
1365    selected
1366}

1415        }
1416    }
1417    RemovalOutcome {

1419        error: None,
1420    }
