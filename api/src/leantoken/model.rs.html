553impl ContextCoverageReceipt {
554    fn is_empty(&self) -> bool {
555        self.unmatched_focus_paths.is_empty()
556            && self.unmatched_focus_symbols.is_empty()
557            && self.focus_path_coverage.is_empty()
558            && self.changed_path_coverage.is_none()
559            && self.strict_scope_satisfied.is_none()
560            && self.unmatched_include_paths.is_empty()
561            && self.covered_must_include_paths.is_empty()
562            && self.covered_must_include_symbols.is_empty()
563            && self.unmatched_must_include_paths.is_empty()
564            && self.unmatched_must_include_symbols.is_empty()
565            && self.uncovered_must_include_paths.is_empty()
566            && self.uncovered_must_include_symbols.is_empty()
567    }
568}

828/// Bounded aggregate counts for files skipped during index preparation.
829#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
830pub struct IndexSkipReasonCounts {
831    /// Files detected as binary during preparation.
832    pub binary: usize,
833    /// Files admitted by discovery that exceeded the byte limit before reading completed.
834    pub oversized_during_read: usize,
835    /// Files whose preparation failed before searchable content could be produced.
836    pub failed: usize,
837}
838
839impl IndexSkipReasonCounts {
840    /// Return the total number of preparation skips across every public reason.
841    #[must_use]
842    pub fn total(&self) -> usize {
843        self.binary
844            .saturating_add(self.oversized_during_read)
845            .saturating_add(self.failed)
846    }
847}
