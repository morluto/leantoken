11/// Commands for centrally managed repository caches.

13pub enum CacheCommand {

20/// Filters and response bounds for `cache list`.
21#[derive(Debug, Clone, Args)]
22pub struct CacheListArgs {
23    /// Return aggregate diagnostics without per-cache entries.
24    #[arg(long, conflicts_with = "cursor")]
25    pub summary: bool,
26    /// Keep caches in this metadata state (repeatable).
27    #[arg(long, value_enum, value_name = "STATE")]
28    pub state: Vec<CacheStateArg>,
29    /// Keep caches in this content-compatibility class (repeatable).
30    #[arg(long, value_enum, value_name = "COMPATIBILITY")]
31    pub compatibility: Vec<CacheCompatibilityArg>,
32    /// Keep caches with this exact index-content version (repeatable).
33    #[arg(long, value_name = "VERSION")]
34    pub index_content_version: Vec<u32>,
35    /// Keep only older or legacy-unversioned content.
36    #[arg(long)]
37    pub incompatible_with_current: bool,
38    /// Keep the exact recorded repository root.
39    #[arg(long, value_name = "PATH")]
40    pub repository_root: Option<PathBuf>,
41    /// Maximum entries returned by one page (1-100).
42    #[arg(
43        long,
44        default_value_t = DEFAULT_CACHE_LIST_LIMIT,
45        value_parser = parse_cache_list_limit
46    )]
47    pub limit: usize,
48    /// Continue from an opaque cursor returned by the same filters.
49    #[arg(long, value_name = "CURSOR")]
50    pub cursor: Option<String>,
51}

53/// Cache metadata state accepted by `cache list --state`.
54#[derive(Debug, Clone, Copy, ValueEnum)]
55pub enum CacheStateArg {
56    /// Current schema and access metadata.
57    Current,
58    /// Readable older schema without current access metadata.
59    Legacy,
60    /// Known artifacts without a readable database.
61    Incomplete,
62    /// SQLite metadata inspection failed.
63    Corrupt,
64    /// Newer or mismatched metadata unsafe for this binary.
65    Unsupported,
66    /// Unexpected directory content.
67    Unrecognized,
68}

70/// Cache content compatibility accepted by `cache list --compatibility`.
71#[derive(Debug, Clone, Copy, ValueEnum)]
72pub enum CacheCompatibilityArg {
73    /// Content produced by the current index-content version.
74    CompatibleCurrent,
75    /// Content produced by a known older version.
76    ObsoleteOlder,
77    /// Legacy content without a versioned cache identity.
78    LegacyUnversioned,
79    /// Content produced by a newer unsupported version.
80    NewerUnsupported,
81    /// Content whose compatibility cannot be trusted.
82    Unknown,
83}
84
85impl From<CacheStateArg> for CacheState {
86    fn from(value: CacheStateArg) -> Self {
87        match value {
88            CacheStateArg::Current => Self::Current,
89            CacheStateArg::Legacy => Self::Legacy,
90            CacheStateArg::Incomplete => Self::Incomplete,
91            CacheStateArg::Corrupt => Self::Corrupt,
92            CacheStateArg::Unsupported => Self::Unsupported,
93            CacheStateArg::Unrecognized => Self::Unrecognized,
94        }
95    }
96}
97
98impl From<CacheCompatibilityArg> for CacheCompatibility {
99    fn from(value: CacheCompatibilityArg) -> Self {
100        match value {
101            CacheCompatibilityArg::CompatibleCurrent => Self::CompatibleCurrent,
102            CacheCompatibilityArg::ObsoleteOlder => Self::ObsoleteOlder,
103            CacheCompatibilityArg::LegacyUnversioned => Self::LegacyUnversioned,
104            CacheCompatibilityArg::NewerUnsupported => Self::NewerUnsupported,
105            CacheCompatibilityArg::Unknown => Self::Unknown,
106        }
107    }

127/// Selection and consent for `cache prune`.
128#[derive(Debug, Clone, Args)]
129pub struct CachePruneArgs {
130    /// Remove caches not accessed for at least this many days.
131    #[arg(long, value_name = "DAYS")]
132    pub older_than: Option<NonZeroU64>,
133    /// Reduce managed cache storage to at most this many bytes using LRU order.
134    #[arg(long, value_name = "BYTES")]
135    pub max_total_bytes: Option<u64>,
136    /// Remove caches whose recorded repository roots are currently missing.
137    #[arg(long)]
138    pub remove_missing_roots: bool,
139    /// Select inactive older or legacy-unversioned caches.
140    ///
141    /// Without `--yes`, this criterion defaults to a dry-run.
142    #[arg(long)]
143    pub incompatible_with_current: bool,
144    /// Show the exact prune plan without deleting files.
145    #[arg(long)]
146    pub dry_run: bool,
147    /// Apply the prune plan without prompting.
148    #[arg(short = 'y', long)]
149    pub yes: bool,
150}
151
152impl From<CachePruneArgs> for CachePruneV2Request {
153    fn from(args: CachePruneArgs) -> Self {
154        Self {
155            request: CachePruneRequest {
156                older_than_days: args.older_than.map(NonZeroU64::get),
157                max_total_bytes: args.max_total_bytes,
158                remove_missing_roots: args.remove_missing_roots,
159                dry_run: args.dry_run || (args.incompatible_with_current && !args.yes),
160                yes: args.yes,
161            },
162            incompatible_with_current: args.incompatible_with_current,
163        }
164    }
