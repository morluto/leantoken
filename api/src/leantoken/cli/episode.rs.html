3/// Local, read-only episode-analysis commands.
4#[derive(Debug, Clone, Args)]
5pub struct EpisodeArgs {
6    /// Episode subcommand.
7    #[command(subcommand)]
8    pub command: EpisodeCommand,
9}
10
11/// Episode-analysis operation.
12#[derive(Debug, Clone, Subcommand)]
13pub enum EpisodeCommand {
14    /// Normalize one existing redacted analyzer report.
15    Audit(EpisodeAuditArgs),
16}
17
18/// Arguments for `episode audit`.
19#[derive(Debug, Clone, Args)]
20pub struct EpisodeAuditArgs {
21    /// Explicit input adapter and schema version.
22    #[arg(long, value_enum)]
23    pub adapter: EpisodeAdapterArg,
24    /// Existing redacted analyzer report.
25    #[arg(long, value_name = "PATH")]
26    pub input: PathBuf,
27}
28
29/// Versioned existing-report adapter accepted by the episode auditor.
30#[derive(Debug, Clone, Copy, PartialEq, Eq, ValueEnum)]
31pub enum EpisodeAdapterArg {
32    /// Aggregate produced by `codex_multi_agent_suite`.
33    MultiAgentSuiteV1,
34    /// Classification produced by `model_ab_trajectory`.
35    ModelAbTrajectoryV1,
36    /// Version-two report produced by `mcp_wire_analyze`.
37    McpWireReportV2,
38    /// Publishable receipt produced by `codex_host_receipt`.
39    CodexHostReceiptV1,
40    /// Classification produced by `context_utilization`.
41    ContextUtilizationV1,
42}
43
44impl From<EpisodeAuditArgs> for crate::episode::EpisodeAuditRequest {
45    fn from(args: EpisodeAuditArgs) -> Self {
46        Self {
47            adapter: match args.adapter {
48                EpisodeAdapterArg::MultiAgentSuiteV1 => {
49                    crate::episode::EpisodeAdapter::MultiAgentSuiteV1
50                }
51                EpisodeAdapterArg::ModelAbTrajectoryV1 => {
52                    crate::episode::EpisodeAdapter::ModelAbTrajectoryV1
53                }
54                EpisodeAdapterArg::McpWireReportV2 => {
55                    crate::episode::EpisodeAdapter::McpWireReportV2
56                }
57                EpisodeAdapterArg::CodexHostReceiptV1 => {
58                    crate::episode::EpisodeAdapter::CodexHostReceiptV1
59                }
60                EpisodeAdapterArg::ContextUtilizationV1 => {
61                    crate::episode::EpisodeAdapter::ContextUtilizationV1
62                }
63            },
64            input: args.input,
65        }
66    }
