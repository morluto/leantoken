10/// Detect the parser language from a file path based on its extension.
11pub fn language_by_path(path: impl AsRef<Path>) -> Option<String> {
12    let ext = path.as_ref().extension()?.to_str()?;
13    Some(match ext.to_lowercase().as_str() {
14        "c" => "c".to_string(),
15        "cs" => "csharp".to_string(),
16        "cc" | "cpp" | "cxx" | "h" | "hh" | "hpp" | "hxx" | "inl" | "ipp" | "tpp" => {
17            "cpp".to_string()
18        }
19        "java" => "java".to_string(),
20        "rs" => "rust".to_string(),
21        "py" | "pyi" => "python".to_string(),
22        "php" => "php".to_string(),
23        "rb" => "ruby".to_string(),
24        "js" | "jsx" | "mjs" | "cjs" => "javascript".to_string(),
25        "ts" | "mts" | "cts" => "typescript".to_string(),
26        "tsx" => "tsx".to_string(),
27        "go" => "go".to_string(),
28        "html" | "htm" => "html".to_string(),
29        "css" => "css".to_string(),
30        "md" | "markdown" => "markdown".to_string(),
31        "tex" | "ltx" => "latex".to_string(),
32        _ => return None,
33    })
34}

36/// Parse a source file given its repository path and full text.
37///
38/// Files whose language is not supported still return `Ok`, but with an empty
39/// parse and `language: None` so callers can fall back to plain text indexing.
40pub fn parse(path: impl AsRef<Path>, source: &str) -> Result<ParseOutput> {
41    parse_with_cancellation(path, source, &CancellationToken::new())
42}

44pub(crate) fn parse_with_cancellation(
45    path: impl AsRef<Path>,
46    source: &str,
47    cancellation: &CancellationToken,
48) -> Result<ParseOutput> {
49    match language_by_path(path) {
50        Some(lang) => {
51            parse_language_with_cancellation(&lang, source, || cancellation.is_cancelled())
52        }
53        None if cancellation.is_cancelled() => Err(Error::Cancelled),
54        None => Ok(empty_parse()),
55    }
56}

58/// Parse source text for a known language name.
59pub fn parse_language(language: &str, source: &str) -> Result<ParseOutput> {
60    parse_language_with_cancellation(language, source, || false)
61}

63// Per-thread cache of a configured tree-sitter `Parser` and compiled query
64// objects, keyed by language name. `Parser` and `Query` are not `Send`, so
65// they cannot be shared across the rayon worker pool. A thread-local avoids
66// recreating and recompiling them for every source file parsed on the same
67// thread.
68thread_local! {
69    static PARSER_CACHE: RefCell<Option<ParserCache>> = const { RefCell::new(None) };
70}
71
72struct ParserCache {
73    parser: Parser,
74    language: Option<String>,
75    queries: HashMap<String, CompiledQueries>,
76}
77
78struct CompiledQueries {
79    tags_query: Option<Query>,
80    import_query: Option<Query>,
81}
82
83fn parse_language_with_cancellation(
84    language: &str,
85    source: &str,
86    mut is_cancelled: impl FnMut() -> bool,
87) -> Result<ParseOutput> {
88    if language == "markdown" {
89        return parse_markdown(source, &mut is_cancelled);
90    }
91    if language == "latex" {
92        return parse_latex(source, &mut is_cancelled);
93    }
94    let lang = language_object(language)
95        .ok_or_else(|| Error::UnsupportedLanguage(language.to_string()))?;
96
97    PARSER_CACHE.with(|cell| {
98        let mut cache = cell.borrow_mut();
99        let cache = cache.get_or_insert_with(|| ParserCache {
100            parser: Parser::new(),
101            language: None,
102            queries: HashMap::new(),
103        });
104        if !cache.queries.contains_key(language) {
105            let tags_query = build_tags_query(language, &lang)?;
106            let import_query = build_import_query(language, &lang)?;
107            cache.queries.insert(
108                language.to_string(),
109                CompiledQueries {
110                    tags_query,
111                    import_query,
112                },
113            );
114        }
115        if cache.language.as_deref() != Some(language) {
116            cache
117                .parser
118                .set_language(&lang)
119                .map_err(Error::TreeSitterLanguage)?;
120            cache.language = Some(language.to_string());
121        }
122
123        let tree = parse_tree(&mut cache.parser, source, &mut is_cancelled)?;
124        let root = tree.root_node();
125        let structurally_complete = !root.has_error();
126
127        let queries = cache
128            .queries
129            .get(language)
130            .expect("queries were just initialized");
131        let mut symbols = Vec::new();
132        let mut references = Vec::new();
133        let mut imports = Vec::new();
134
135        if let Some(tags_query) = &queries.tags_query {
136            run_query(source, tags_query, root, &mut is_cancelled, |qm| {
137                process_tags_match(
138                    language,
139                    source,
140                    tags_query,
141                    qm,
142                    &mut symbols,
143                    &mut references,
144                );
145            })?;
146        }
147        if let Some(import_query) = &queries.import_query {
148            run_query(source, import_query, root, &mut is_cancelled, |qm| {
149                process_imports_match(source, import_query, qm, &mut imports);
150            })?;
151        }
152        if matches!(language, "javascript" | "typescript" | "tsx") {
153            append_javascript_bindings(source, root, &mut symbols, &mut is_cancelled)?;
154        }
155        match language {
156            "csharp" => {
157                append_csharp_structure(
158                    source,
159                    root,
160                    &mut symbols,
161                    &mut references,
162                    &mut imports,
163                    &mut is_cancelled,
164                )?;
165                imports.sort_by(|a, b| {
166                    a.line
167                        .cmp(&b.line)
168                        .then_with(|| a.raw_target.cmp(&b.raw_target))
169                });
170            }
171            "css" => append_css_structure(
172                source,
173                root,
174                &mut symbols,
175                &mut references,
176                &mut is_cancelled,
177            )?,
178            "html" => {
179                append_html_structure(
180                    source,
181                    root,
182                    &mut symbols,
183                    &mut references,
184                    &mut imports,
185                    &mut is_cancelled,
186                )?;
187                imports.sort_by(|a, b| {
188                    a.line
189                        .cmp(&b.line)
190                        .then_with(|| a.raw_target.cmp(&b.raw_target))
191                });
192            }
193            _ => {}
194        }
195
196        if is_cancelled() {
197            return Err(Error::Cancelled);
198        }
199
200        deduplicate_symbols(&mut symbols);
201        compute_symbol_parents(&mut symbols);
202        compute_reference_enclosing(&symbols, &mut references);
203
204        Ok(ParseOutput {
205            language: Some(language.to_string()),
206            structurally_complete,
207            symbols,
208            references,
209            imports,
210        })
211    })
