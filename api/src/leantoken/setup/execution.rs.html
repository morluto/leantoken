6/// Run global MCP setup or removal using the current user environment.
7pub fn run(
8    operation: SetupOperation,
9    request: SetupRequest,
10    json_output: bool,
11) -> Result<SetupReport> {
12    let home = home_directory()
13        .ok_or_else(|| Error::SetupFailure("could not determine the home directory".into()))?;
14    let launcher = McpLauncher::current()?;
15    let native_executable = std::env::current_exe()?.canonicalize()?;
16    if operation == SetupOperation::Setup
17        && launcher.uses_npx()
18        && !request.allow_outdated
19        && npx_resolved_from_local_project(&native_executable, &std::env::current_dir()?)
20    {
21        require_current_npx_setup(
22            launcher.version(),
23            crate::upgrade::latest_npm_version().as_deref(),
24        )?;
25    }
26    let runtime_root = setup_runtime_root(&home);
27    let environment = SetupEnvironment {
28        home,
29        runtime_root,
30        native_executable,
31        persistent_cli: !launcher.is_ephemeral(),
32        launcher,
33        interactive: !json_output
34            && std::io::stdin().is_terminal()
35            && std::io::stderr().is_terminal(),
36    };
37    run_with(operation, request, &environment, &InteractivePrompt)
38}

40pub(super) fn run_with(
41    operation: SetupOperation,
42    request: SetupRequest,
43    environment: &SetupEnvironment,
44    prompt: &dyn SetupPrompt,
45) -> Result<SetupReport> {
46    let recovery_path = transaction_path(&environment.runtime_root);
47    if request.dry_run && recovery_path.exists() {
48        return Err(Error::SetupFailure(format!(
49            "interrupted setup requires recovery before dry-run: {}",
50            recovery_path.display()
51        )));
52    }
53    let _setup_lock = (!request.dry_run)
54        .then(|| acquire_setup_lock(&environment.runtime_root))
55        .transpose()?;
56    if !request.dry_run {
57        recover_interrupted_transaction(&environment.runtime_root)?;
58    }
59    if request.refresh && operation != SetupOperation::Setup {
60        return Err(Error::InvalidRequest(
61            "--refresh is only valid with the setup command".into(),
62        ));
63    }
64    if request.refresh && (request.all || !request.clients.is_empty()) {
65        return Err(Error::InvalidRequest(
66            "--refresh cannot be combined with client flags or --all".into(),
67        ));
68    }
69    if request.private_runtime && operation != SetupOperation::Setup {
70        return Err(Error::InvalidRequest(
71            "--private-runtime is only valid with the setup command".into(),
72        ));
73    }
74    if request.allow_outdated && operation != SetupOperation::Setup {
75        return Err(Error::InvalidRequest(
76            "--allow-outdated is only valid with the setup command".into(),
77        ));
78    }
79
80    let inventory_required_for_selection = request.refresh
81        || (operation == SetupOperation::Remove
82            && environment.interactive
83            && !request.all
84            && request.clients.is_empty()
85            && !request.yes);
86    let selection_inventory = inventory_required_for_selection
87        .then(|| {
88            configured_registrations_with_snapshots(
89                &environment.home,
90                &environment.launcher,
91                &SetupClient::ALL,
92            )
93        })
94        .transpose()?;
95
96    let detected = SetupClient::ALL
97        .into_iter()
98        .filter(|client| client.is_detected(&environment.home))
99        .collect::<Vec<_>>();
100
101    let clients = if request.refresh {
102        managed_clients_from_registrations(
103            &selection_inventory
104                .as_ref()
105                .expect("refresh captured configuration inventory")
106                .0,
107        )
108    } else if request.all {
109        SetupClient::ALL.to_vec()
110    } else if !request.clients.is_empty() {
111        deduplicate(request.clients)
112    } else if request.yes {
113        return Err(Error::InvalidRequest(
114            "--yes requires explicit client flags or --all; detection is not consent".into(),
115        ));
116    } else {
117        if !environment.interactive {
118            return Err(Error::InvalidRequest(
119                "interactive setup requires a terminal; pass client flags or --all with --yes"
120                    .into(),
121            ));
122        }
123        let preferred = if operation == SetupOperation::Setup {
124            detected.clone()
125        } else {
126            managed_clients_from_registrations(
127                &selection_inventory
128                    .as_ref()
129                    .expect("interactive remove captured configuration inventory")
130                    .0,
131            )
132        };
133        let Some(selected) = prompt.select(operation, &detected, &preferred)? else {
134            return Ok(empty_report(operation, environment.persistent_cli));
135        };
136        if selected.is_empty() {
137            return Ok(empty_report(operation, environment.persistent_cli));
138        }
139        selected
140    };
141
142    if !environment.interactive && !request.dry_run && !request.yes {
143        return Err(Error::InvalidRequest(
144            "non-interactive setup requires explicit client flags, --all, or --refresh with --yes"
145                .into(),
146        ));
147    }
148    let runtime = (request.private_runtime && !clients.is_empty())
149        .then(|| runtime_install_plan(environment))
150        .transpose()?;
151    let private_launcher = runtime.as_ref().map(|runtime| {
152        McpLauncher::from_executable_with_version(
153            &runtime.destination,
154            environment.launcher.version(),
155        )
156    });
157    let launcher = private_launcher.as_ref().unwrap_or(&environment.launcher);
158    let selected_discovery_paths = clients
159        .iter()
160        .map(|client| client.discovery_path(&environment.home))
161        .collect::<std::collections::BTreeSet<_>>();
162    let (registrations, configuration_snapshots) = match selection_inventory {
163        Some((_registrations, snapshots)) => (
164            configured_registrations_from_snapshots(
165                &environment.home,
166                launcher,
167                &SetupClient::ALL,
168                &snapshots,
169            )?,
170            snapshots,
171        ),
172        None => {
173            configured_registrations_with_snapshots(&environment.home, launcher, &SetupClient::ALL)?
174        }
175    };
176    let unselected_registrations = registrations
177        .into_iter()
178        .filter(|registration| !clients.contains(&registration.client))
179        .collect::<Vec<_>>();
180    let (discovery_paths, discovery_cleanup_paths) = if operation == SetupOperation::Setup {
181        let mut required_paths = unselected_registrations
182            .into_iter()
183            .map(|registration| registration.client.discovery_path(&environment.home))
184            .collect::<std::collections::BTreeSet<_>>();
185        required_paths.extend(selected_discovery_paths.iter().cloned());
186        let cleanup_paths = if request.refresh && clients.is_empty() {
187            Vec::new()
188        } else {
189            [
190                environment.home.join(".agents/skills/leantoken/SKILL.md"),
191                environment.home.join(".claude/skills/leantoken/SKILL.md"),
192            ]
193            .into_iter()
194            .filter(|path| !required_paths.contains(path))
195            .collect()
196        };
197        (
198            selected_discovery_paths.into_iter().collect(),
199            cleanup_paths,
200        )
201    } else {
202        let remaining_paths = unselected_registrations
203            .into_iter()
204            .map(|registration| registration.client)
205            .map(|client| client.discovery_path(&environment.home))
206            .collect::<std::collections::BTreeSet<_>>();
207        if remaining_paths.is_empty() {
208            (
209                [
210                    environment.home.join(".agents/skills/leantoken/SKILL.md"),
211                    environment.home.join(".claude/skills/leantoken/SKILL.md"),
212                ]
213                .into_iter()
214                .collect(),
215                Vec::new(),
216            )
217        } else {
218            (
219                selected_discovery_paths
220                    .difference(&remaining_paths)
221                    .cloned()
222                    .collect(),
223                Vec::new(),
224            )
225        }
226    };
227
228    let plan = resolve_plan(
229        operation,
230        &clients,
231        PlanEnvironment {
232            detected: &detected,
233            home: &environment.home,
234            launcher,
235            persistent_cli: environment.persistent_cli,
236            runtime,
237            discovery_paths,
238            discovery_cleanup_paths,
239            configuration_snapshots,
240            force_unmanaged: request.force_unmanaged,
241            transaction_root: &environment.runtime_root,
242        },
243    )?;
244
245    if request.dry_run {
246        return Ok(report_from_plan(&plan, false, true, Vec::new(), None, None));
247    }
248
249    if !request.yes && !prompt.confirm(operation, &plan)? {
250        return Ok(report_from_plan(&plan, true, false, Vec::new(), None, None));
251    }
252
253    let outcome = apply_plan(&plan);
254    let verification = verify_applied_setup(&plan, &outcome.results, outcome.error.as_deref());
255    Ok(report_from_plan(
256        &plan,
257        false,
258        false,
259        outcome.results,
260        outcome.error,
261        verification,
262    ))
263}

265fn verify_applied_setup(
266    plan: &ResolvedSetupPlan,
267    results: &[ClientSetupResult],
268    apply_error: Option<&str>,
269) -> Option<SetupVerification> {
270    let launcher = plan.launcher.as_ref()?;
271    let client_argument = plan
272        .edits
273        .first()
274        .map(|edit| format!(" --client {}", edit.public.client.cli_name()))
275        .unwrap_or_default();
276    let repair_command = if let Some(package) = &launcher.package {
277        format!("npx --yes {package} doctor{client_argument} --json")
278    } else if plan.persistent_cli {
279        format!("leantoken doctor{client_argument} --json")
280    } else {
281        format!(
282            "npx --yes leantoken@{} doctor{client_argument} --json",
283            launcher.version
284        )
285    };
286    if apply_error.is_some() || results.iter().any(|result| result.error.is_some()) {
287        return Some(SetupVerification {
288            status: SetupVerificationStatus::Skipped,
289            stage: None,
290            message: Some("setup transaction did not complete".into()),
291            repair_command: Some(repair_command),
292        });
293    }
294
295    let result = (|| {
296        let repository = tempfile::tempdir()?;
297        fs::write(
298            repository.path().join("lib.rs"),
299            "pub fn ready() -> bool { true }\n",
300        )?;
301        let config = crate::Config::discover(
302            repository.path(),
303            Some(repository.path().join("index.sqlite")),
304        )?;
305        let report = crate::doctor::run_launcher(
306            &config,
307            &launcher.command,
308            &launcher.args,
309            SETUP_VERIFICATION_TIMEOUT,
310        )?;
311        revalidate_applied_client_edits(&plan.edits)?;
312        Ok(report)
313    })();
314
315    Some(match result {
316        Ok(_) => SetupVerification {
317            status: SetupVerificationStatus::Passed,
318            stage: None,
319            message: None,
320            repair_command: Some(repair_command),
321        },
322        Err(Error::DoctorFailure { stage, message }) => SetupVerification {
323            status: SetupVerificationStatus::Failed,
324            stage: Some(stage.into()),
325            message: Some(message),
326            repair_command: Some(repair_command),
327        },
328        Err(error) => SetupVerification {
329            status: SetupVerificationStatus::Failed,
330            stage: Some("launch".into()),
331            message: Some(error.to_string()),
332            repair_command: Some(repair_command),
333        },
334    })
335}

337pub(super) fn revalidate_applied_client_edits(edits: &[PlannedClientEdit]) -> Result<()> {
338    for edit in edits {
339        let expected = edit.updated.as_ref().or(edit.original.as_ref());
340        let current = read_optional(&edit.public.path).map_err(|error| Error::DoctorFailure {
341            stage: "registration",
342            message: format!(
343                "could not revalidate {} after launcher verification: {error}",
344                edit.public.path.display()
345            ),
346        })?;
347        if current.as_ref() != expected {
348            return Err(Error::DoctorFailure {
349                stage: "registration",
350                message: format!(
351                    "{} changed during launcher verification",
352                    edit.public.path.display()
353                ),
354            });
355        }
356    }
357    Ok(())
358}

360pub(super) fn report_from_plan(
361    plan: &ResolvedSetupPlan,
362    cancelled: bool,
363    dry_run: bool,
364    results: Vec<ClientSetupResult>,
365    apply_error: Option<String>,
366    verification: Option<SetupVerification>,
367) -> SetupReport {
368    let discovery_skill_tokens = plan.discovery_edits.first().and_then(|edit| {
369        edit.updated
370            .as_ref()
371            .or(edit.original.as_ref())
372            .map(|content| crate::tokens::Tokenizer::Cl100kBase.count(content))
373    });
374    SetupReport {
375        operation: plan.operation,
376        cancelled,
377        dry_run,
378        ownership_override: plan.ownership_override,
379        persistent_cli: plan.persistent_cli,
380        launcher: plan.launcher.clone(),
381        plan: plan.edits.iter().map(|edit| edit.public.clone()).collect(),
382        discovery_plan: plan
383            .discovery_edits
384            .iter()
385            .map(|edit| edit.public.clone())
386            .collect(),
387        discovery_skill_tokens,
388        results,
389        apply_error,
390        verification,
391    }
392}

394pub(super) fn empty_report(operation: SetupOperation, persistent_cli: bool) -> SetupReport {
395    SetupReport {
396        operation,

411pub(super) fn deduplicate(clients: Vec<SetupClient>) -> Vec<SetupClient> {
