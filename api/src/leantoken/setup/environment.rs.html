19pub(super) fn require_current_npx_setup(current: &str, latest: Option<&str>) -> Result<()> {
20    let Some(latest) = latest else {
21        return Err(Error::InvalidRequest(
22            "could not verify the locally resolved npx release; retry online or pass \
23             --allow-outdated for an intentional pinned setup"
24                .into(),
25        ));
26    };
27    match crate::upgrade::version_update_available(current, latest) {
28        Some(false) => Ok(()),
29        Some(true) => Err(Error::InvalidRequest(format!(
30            "npx resolved stale local LeanToken v{current}; latest is v{latest}. Run \
31             `npx --yes leantoken@latest setup`, or pass --allow-outdated for an intentional \
32             rollback"
33        ))),
34        None => Err(Error::InvalidRequest(format!(
35            "could not compare locally resolved LeanToken v{current} with npm v{latest}; pass \
36             --allow-outdated only for an intentional pinned setup"
37        ))),
38    }
39}

56pub(super) fn home_directory() -> Option<PathBuf> {
57    std::env::var_os("HOME")
58        .map(PathBuf::from)
59        .filter(|path| path.is_absolute())
60        .or({
61            #[cfg(windows)]
62            {
63                std::env::var_os("USERPROFILE")
64                    .map(PathBuf::from)
65                    .filter(|path| path.is_absolute())
66            }
67            #[cfg(not(windows))]
68            {
69                None
70            }
71        })
72        .or_else(|| BaseDirs::new().map(|directories| directories.home_dir().to_path_buf()))
73}

75pub(crate) fn diagnostic_state() -> SetupDiagnostic {
76    let Some(home) = home_directory() else {
77        return SetupDiagnostic {
78            registration_status: "unknown",
79            configured_clients: Vec::new(),
80            registrations: Vec::new(),
81            discovery_status: "unknown",
82            discovery_paths: Vec::new(),
83        };
84    };
85    let configured =
86        McpLauncher::current().and_then(|launcher| configured_registrations(&home, &launcher));
87    let (registration_status, registrations) = match configured {
88        Ok(registrations) if registrations.is_empty() => ("not_registered", registrations),
89        Ok(registrations) => ("registered", registrations),
90        Err(_) => ("unknown", Vec::new()),
91    };
92    let configured_clients = registrations
93        .iter()
94        .map(|registration| registration.client)
95        .collect();
96    let discovery_paths = [
97        home.join(".agents/skills/leantoken/SKILL.md"),
98        home.join(".claude/skills/leantoken/SKILL.md"),
99    ]
100    .into_iter()
101    .filter(|path| {
102        read_optional(path)
103            .ok()
104            .flatten()
105            .is_some_and(|content| content.contains(DISCOVERY_SKILL_MARKER))
106    })
107    .collect::<Vec<_>>();
108    SetupDiagnostic {
109        registration_status,
110        configured_clients,
111        registrations,
112        discovery_status: match discovery_paths.len() {
113            0 => "missing",
114            2 => "installed",
115            _ => "partial",
116        },
117        discovery_paths,
118    }
119}

131pub(super) fn read_configured_registration(
132    client: SetupClient,
133    home: &Path,
134    launcher: &McpLauncher,
135) -> Result<Option<ConfiguredRegistration>> {
136    let definition = client.definition(home);
137    let Some(source) = read_optional(&definition.path)? else {
138        return Ok(None);
139    };
140    let (command, args) = match definition.format {
141        ConfigFormat::Json { section, shape } => {
142            let root: Value = jsonc_parser::parse_to_serde_value(&source, &ParseOptions::default())
143                .map_err(|error| invalid_config(&definition.path, error))?;
144            let Some(entry) = root
145                .get(section)
146                .and_then(Value::as_object)
147                .and_then(|section| section.get(SERVER_NAME))
148            else {
149                return Ok(None);
150            };
151            json_registration_command(entry, shape, &definition.path)?
152        }
153        ConfigFormat::Toml => {
154            let document = source
155                .parse::<DocumentMut>()
156                .map_err(|error| invalid_config(&definition.path, error))?;
157            let Some(entry) = document
158                .get("mcp_servers")
159                .and_then(Item::as_table)
160                .and_then(|servers| servers.get(SERVER_NAME))
161            else {
162                return Ok(None);
163            };
164            toml_registration_command(entry, &definition.path)?
165        }
166    };
167    let expected_command = launcher.command()?.to_owned();
168    let matches_current = command == expected_command && args == launcher.args;
169    Ok(Some(ConfiguredRegistration {
170        client,
171        path: definition.path,
172        version: registered_version(&command, &args),
173        command,
174        args,
175        expected_version: launcher.version().to_owned(),
176        matches_current,
177    }))
178}

180pub(super) fn json_registration_command(
181    entry: &Value,
182    shape: JsonEntryShape,
183    path: &Path,
184) -> Result<(String, Vec<String>)> {
185    let object = entry
186        .as_object()
187        .ok_or_else(|| invalid_config(path, "LeanToken MCP entry must be an object"))?;
188    match shape {
189        JsonEntryShape::CommandAndArgs => {
190            let command = object
191                .get("command")
192                .and_then(Value::as_str)
193                .ok_or_else(|| invalid_config(path, "LeanToken MCP command must be a string"))?;
194            let args = json_string_array(object.get("args"), path, "args")?;
195            Ok((command.to_owned(), args))
196        }
197        JsonEntryShape::OpenCode => {
198            let command = object
199                .get("command")
200                .and_then(Value::as_array)
201                .ok_or_else(|| invalid_config(path, "OpenCode MCP command must be an array"))?;
202            let mut values = command.iter();
203            let executable = values
204                .next()
205                .and_then(Value::as_str)
206                .ok_or_else(|| invalid_config(path, "OpenCode MCP command is empty"))?;
207            let args = values
208                .map(|value| {
209                    value.as_str().map(str::to_owned).ok_or_else(|| {
210                        invalid_config(path, "OpenCode MCP command arguments must be strings")
211                    })
212                })
213                .collect::<Result<Vec<_>>>()?;
214            Ok((executable.to_owned(), args))
215        }
216    }
217}

219pub(super) fn json_string_array(
220    value: Option<&Value>,
221    path: &Path,
222    field: &str,
223) -> Result<Vec<String>> {
224    let Some(value) = value else {
225        return Ok(Vec::new());
226    };
227    value
228        .as_array()
229        .ok_or_else(|| invalid_config(path, format!("LeanToken MCP {field} must be an array")))?
230        .iter()
231        .map(|value| {
232            value.as_str().map(str::to_owned).ok_or_else(|| {
233                invalid_config(path, format!("LeanToken MCP {field} must contain strings"))
234            })
235        })
236        .collect()
237}

239pub(super) fn toml_registration_command(
240    entry: &Item,
241    path: &Path,
242) -> Result<(String, Vec<String>)> {
243    let table = entry
244        .as_table()
245        .ok_or_else(|| invalid_config(path, "LeanToken MCP entry must be a table"))?;
246    let command = table
247        .get("command")
248        .and_then(Item::as_str)
249        .ok_or_else(|| invalid_config(path, "LeanToken MCP command must be a string"))?;
250    let args = table
251        .get("args")
252        .and_then(Item::as_array)
253        .map(|args| {
254            args.iter()
255                .map(|value| {
256                    value.as_str().map(str::to_owned).ok_or_else(|| {
257                        invalid_config(path, "LeanToken MCP arguments must be strings")
258                    })
259                })
260                .collect::<Result<Vec<_>>>()
261        })
262        .transpose()?
263        .unwrap_or_default();
264    Ok((command.to_owned(), args))
265}

267pub(super) fn registered_version(command: &str, args: &[String]) -> Option<String> {
268    args.iter()
269        .find_map(|argument| argument.strip_prefix("--package=leantoken@"))
270        .map(str::to_owned)
271        .or_else(|| {
272            Path::new(command)
273                .components()
274                .collect::<Vec<_>>()
275                .windows(2)
276                .find(|components| components[0].as_os_str() == "runtimes")
277                .and_then(|components| components[1].as_os_str().to_str())
278                .map(str::to_owned)
279        })
280}

282pub(super) fn configured_clients(home: &Path, launcher: &McpLauncher) -> Result<Vec<SetupClient>> {
283    SetupClient::ALL
284        .into_iter()
285        .filter_map(|client| {
286            let resolved = resolve_client_edit(SetupOperation::Remove, client, &[], home, launcher);
287            match resolved {
288                Ok(edit) if matches!(edit.status, EditStatus::Removed) => Some(Ok(client)),
289                Ok(_) => None,
290                Err(error) => Some(Err(error)),
291            }
292        })
293        .collect()
