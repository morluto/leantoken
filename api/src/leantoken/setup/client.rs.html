31/// Coding clients supported by the global setup wizard.
32#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, clap::ValueEnum)]
33#[serde(rename_all = "snake_case")]
34pub enum SetupClient {
35    /// Claude Code.

37    /// Cursor.
38    Cursor,
39    /// OpenCode.

41    OpenCode,

43    Codex,
44    /// Gemini CLI.
45    Gemini,
46    /// Google Antigravity.
47    Antigravity,
48}

83    pub(super) fn definition(self, home: &Path) -> ClientDefinition {
84        let path = match self {
85            Self::OpenCode => {
86                let candidates = self.configuration_paths(home);
87                candidates
88                    .iter()
89                    .find(|candidate| candidate.exists())
90                    .cloned()
91                    .unwrap_or_else(|| candidates[0].clone())
92            }
93            _ => self.configuration_paths(home)[0].clone(),
94        };
95        self.definition_at(path)
96    }
97
98    pub(super) fn definition_at(self, path: PathBuf) -> ClientDefinition {
99        match self {
100            Self::Claude => {
101                ClientDefinition::json(path, "mcpServers", JsonEntryShape::CommandAndArgs)
102            }
103            Self::Cursor => {
104                ClientDefinition::json(path, "mcpServers", JsonEntryShape::CommandAndArgs)
105            }
106            Self::OpenCode => ClientDefinition::json(path, "mcp", JsonEntryShape::OpenCode),
107            Self::Codex => ClientDefinition {
108                path,
109                format: ConfigFormat::Toml,
110            },
111            Self::Gemini => {
112                ClientDefinition::json(path, "mcpServers", JsonEntryShape::CommandAndArgs)
113            }
114            Self::Antigravity => {
115                ClientDefinition::json(path, "mcpServers", JsonEntryShape::CommandAndArgs)
116            }
117        }
118    }
119
120    pub(super) fn configuration_paths(self, home: &Path) -> Vec<PathBuf> {
121        match self {
122            Self::Claude => vec![home.join(".claude.json")],
123            Self::Cursor => vec![home.join(".cursor/mcp.json")],
124            Self::OpenCode => {
125                let directory = home.join(".config/opencode");
126                vec![
127                    directory.join("opencode.json"),
128                    directory.join("opencode.jsonc"),
129                    directory.join(".opencode.json"),
130                    directory.join(".opencode.jsonc"),
131                ]
132            }
133            Self::Codex => vec![home.join(".codex/config.toml")],
134            Self::Gemini => vec![home.join(".gemini/settings.json")],
135            Self::Antigravity => vec![home.join(".gemini/config/mcp_config.json")],
136        }
137    }
138
139    pub(super) fn is_detected(self, home: &Path) -> bool {
140        match self {
141            Self::Claude => home.join(".claude").exists() || home.join(".claude.json").exists(),
142            Self::Cursor => home.join(".cursor").exists(),
143            Self::OpenCode => home.join(".config/opencode").exists(),
144            Self::Codex => home.join(".codex").exists(),
145            Self::Gemini => home.join(".gemini").exists(),
146            Self::Antigravity => {
147                home.join(".gemini/antigravity").exists() || home.join(".agent").exists()
148            }
149        }
150    }
151
152    pub(super) fn discovery_path(self, home: &Path) -> PathBuf {
153        match self {
154            Self::Claude => home.join(".claude/skills/leantoken/SKILL.md"),
155            Self::Cursor | Self::OpenCode | Self::Codex | Self::Gemini | Self::Antigravity => {
156                home.join(".agents/skills/leantoken/SKILL.md")
157            }
158        }
159    }
