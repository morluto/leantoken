26/// Coding clients supported by the global setup wizard.
27#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
28#[serde(rename_all = "snake_case")]
29pub enum SetupClient {
30    /// Claude Code.
31    Claude,
32    /// Cursor.
33    Cursor,
34    /// OpenCode.
35    OpenCode,
36    /// Codex CLI, desktop, and IDE integrations.
37    Codex,
38    /// Gemini CLI.
39    Gemini,
40    /// Google Antigravity.
41    Antigravity,
42}
43
44impl SetupClient {
45    /// Every supported client in display order.
46    pub const ALL: [Self; 6] = [
47        Self::Claude,
48        Self::Cursor,
49        Self::OpenCode,
50        Self::Codex,
51        Self::Gemini,
52        Self::Antigravity,
53    ];
54
55    pub(super) fn display_name(self) -> &'static str {
56        match self {
57            Self::Claude => "Claude Code",
58            Self::Cursor => "Cursor",
59            Self::OpenCode => "OpenCode",
60            Self::Codex => "Codex",
61            Self::Gemini => "Gemini CLI",
62            Self::Antigravity => "Antigravity",
63        }
64    }
65
66    pub(super) fn definition(self, home: &Path) -> ClientDefinition {
67        match self {
68            Self::Claude => ClientDefinition::json(
69                home.join(".claude.json"),
70                "mcpServers",
71                JsonEntryShape::CommandAndArgs,
72            ),
73            Self::Cursor => ClientDefinition::json(
74                home.join(".cursor/mcp.json"),
75                "mcpServers",
76                JsonEntryShape::CommandAndArgs,
77            ),
78            Self::OpenCode => {
79                let directory = home.join(".config/opencode");
80                let candidates = [
81                    directory.join("opencode.json"),
82                    directory.join("opencode.jsonc"),
83                    directory.join(".opencode.json"),
84                    directory.join(".opencode.jsonc"),
85                ];
86                let path = candidates
87                    .iter()
88                    .find(|candidate| candidate.exists())
89                    .cloned()
90                    .unwrap_or_else(|| candidates[0].clone());
91                ClientDefinition::json(path, "mcp", JsonEntryShape::OpenCode)
92            }
93            Self::Codex => ClientDefinition {
94                path: home.join(".codex/config.toml"),
95                format: ConfigFormat::Toml,
96            },
97            Self::Gemini => ClientDefinition::json(
98                home.join(".gemini/settings.json"),
99                "mcpServers",
100                JsonEntryShape::CommandAndArgs,
101            ),
102            Self::Antigravity => ClientDefinition::json(
103                home.join(".gemini/config/mcp_config.json"),
104                "mcpServers",
105                JsonEntryShape::CommandAndArgs,
106            ),
107        }
108    }
109
110    pub(super) fn is_detected(self, home: &Path) -> bool {
111        match self {
112            Self::Claude => home.join(".claude").exists() || home.join(".claude.json").exists(),
113            Self::Cursor => home.join(".cursor").exists(),
114            Self::OpenCode => home.join(".config/opencode").exists(),
115            Self::Codex => home.join(".codex").exists(),
116            Self::Gemini => home.join(".gemini").exists(),
117            Self::Antigravity => {
118                home.join(".gemini/antigravity").exists() || home.join(".agent").exists()
119            }
120        }
121    }
122}
