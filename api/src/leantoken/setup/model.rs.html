3/// Setup or removal operation.
4#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
5#[serde(rename_all = "snake_case")]
6pub enum SetupOperation {
7    /// Add or update global MCP entries.
8    Setup,
9    /// Remove global MCP entries.
10    Remove,
11}
12
13impl SetupOperation {
14    pub(super) fn action(self) -> &'static str {
15        match self {
16            Self::Setup => "set up",
17            Self::Remove => "remove",
18        }
19    }
20
21    pub(super) fn action_label(self) -> &'static str {
22        match self {
23            Self::Setup => "Set up",
24            Self::Remove => "Remove",
25        }
26    }
27
28    pub(super) fn selection_prompt(self) -> &'static str {
29        match self {
30            Self::Setup => "Which coding agents should use LeanToken?",
31            Self::Remove => "Remove LeanToken from which coding agents?",
32        }
33    }
34
35    pub(super) fn plan_label(self) -> &'static str {
36        match self {
37            Self::Setup => "setup",
38            Self::Remove => "removal",
39        }
40    }
41}

43/// Parsed request for the setup or removal workflow.
44#[derive(Debug, Clone, PartialEq, Eq)]
45pub struct SetupRequest {
46    /// Explicitly selected clients.
47    pub clients: Vec<SetupClient>,
48    /// Select every supported client.
49    pub all: bool,
50    /// Refresh every existing LeanToken entry without selecting new clients.
51    pub refresh: bool,
52    /// Install and register a direct application-owned native runtime.
53    pub private_runtime: bool,
54    /// Apply an explicitly scoped plan without interactive confirmation.
55    pub yes: bool,
56    /// Resolve and print the setup plan without changing configuration.
57    pub dry_run: bool,
58    /// Permit an intentionally selected older npx release.
59    pub allow_outdated: bool,
60}

62/// Planned action for one client configuration.
63#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
64#[serde(rename_all = "snake_case")]
65pub enum ClientPlanAction {
66    /// Create a new configuration file.
67    Create,
68    /// Update an existing configuration file.
69    Update,
70    /// The requested setup is already current.
71    AlreadyCurrent,
72    /// Remove an existing LeanToken entry.
73    Remove,
74    /// No LeanToken entry exists to remove.
75    NotConfigured,
76}
77
78impl fmt::Display for ClientPlanAction {
79    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
80        let label = match self {
81            Self::Create => "create",
82            Self::Update => "update",
83            Self::AlreadyCurrent => "already current",
84            Self::Remove => "remove",
85            Self::NotConfigured => "not configured",
86        };
87        formatter.write_str(label)
88    }
89}

149/// Aggregate setup or removal report.
150#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
151pub struct SetupReport {
152    /// Operation that produced the report.
153    pub operation: SetupOperation,
154    /// Whether an interactive user cancelled before mutation.
155    pub cancelled: bool,
156    /// Whether this report describes a dry-run without mutation.
157    pub dry_run: bool,
158    /// Whether setup ran from a persistent CLI installation.
159    pub persistent_cli: bool,
160    /// Exact launcher considered for setup, omitted for removal.
161    #[serde(skip_serializing_if = "Option::is_none")]
162    pub launcher: Option<LauncherPlan>,
163    /// Secret-free resolved plan used for confirmation and execution.
164    pub plan: Vec<ClientSetupPlan>,
165    /// Agent-visible discovery artifacts included in the same transaction.
166    pub discovery_plan: Vec<DiscoverySetupPlan>,
167    /// Exact cl100k token count of one managed discovery skill.
168    #[serde(skip_serializing_if = "Option::is_none")]
169    pub discovery_skill_tokens: Option<usize>,
170    /// Per-client outcomes.
171    pub results: Vec<ClientSetupResult>,
172}

174impl SetupReport {
175    /// Return true when at least one selected client failed.
