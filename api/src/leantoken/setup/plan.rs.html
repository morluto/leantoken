37pub(super) fn resolve_plan(
38    operation: SetupOperation,
39    clients: &[SetupClient],
40    environment: PlanEnvironment<'_>,
41) -> Result<ResolvedSetupPlan> {
42    let edits = clients
43        .iter()
44        .copied()
45        .map(|client| {
46            resolve_client_edit(
47                operation,
48                client,
49                environment.detected,
50                environment.home,
51                environment.launcher,
52                &environment.configuration_snapshots,
53            )
54        })
55        .collect::<Result<Vec<_>>>()?;
56    for edit in &edits {
57        validate_client_edit_ownership(
58            operation,
59            edit,
60            environment.home,
61            environment.launcher,
62            environment.force_unmanaged,
63        )?;
64        if let Some(updated) = &edit.updated {
65            validate_setup_content_size(&edit.public.path, updated)?;
66        }
67    }
68    let mut discovery_edits = resolve_discovery_edits(
69        operation,
70        &environment.discovery_paths,
71        Some(environment.launcher),
72    )?;
73    if operation == SetupOperation::Setup {
74        let cleanup = resolve_discovery_edits(
75            SetupOperation::Remove,
76            &environment.discovery_cleanup_paths,
77            None,
78        )?;
79        discovery_edits.extend(
80            cleanup
81                .into_iter()
82                .filter(|edit| edit.public.action == ClientPlanAction::Remove),
83        );
84    }
85    let launcher = (operation == SetupOperation::Setup)
86        .then(|| launcher_plan(environment.launcher, environment.runtime.as_ref()))
87        .transpose()?;
88    Ok(ResolvedSetupPlan {
89        operation,
90        persistent_cli: environment.persistent_cli,
91        launcher,
92        runtime: environment.runtime,
93        edits,
94        discovery_edits,
95        configuration_snapshots: environment.configuration_snapshots,
96        ownership_override: environment.force_unmanaged,
97        transaction_root: environment.transaction_root.to_path_buf(),
98    })
99}

101pub(super) fn validate_client_edit_ownership(
102    operation: SetupOperation,
103    edit: &PlannedClientEdit,
104    home: &Path,
105    launcher: &McpLauncher,
106    force_unmanaged: bool,
107) -> Result<()> {
108    let definition = edit.public.client.definition_at(edit.public.path.clone());
109    if let Some(registration) = configured_registration_from_definition(
110        edit.public.client,
111        home,
112        launcher,
113        &definition,
114        edit.original.as_deref(),
115    )? && !registration.managed
116        && !force_unmanaged
117    {
118        return Err(Error::SetupFailure(format!(
119            "refusing to {} unmanaged LeanToken entry in {}; review it, then preview the override with --force-unmanaged --dry-run",
120            operation.action(),
121            registration.path.display()
122        )));
123    }
124    Ok(())
125}

142pub(super) fn resolve_discovery_edits(
143    operation: SetupOperation,
144    paths: &[PathBuf],
145    launcher: Option<&McpLauncher>,
146) -> Result<Vec<PlannedDiscoveryEdit>> {
147    let content = launcher.map(discovery_skill).transpose()?;
148    paths
149        .iter()
150        .cloned()
151        .map(|path| {
152            let original = read_optional(&path)?;
153            let owned = original
154                .as_deref()
155                .is_some_and(|value| value.contains(DISCOVERY_SKILL_MARKER));
156            let (action, updated) = match operation {
157                SetupOperation::Setup => {
158                    if original.as_deref() == content.as_deref() {
159                        (ClientPlanAction::AlreadyCurrent, None)
160                    } else if original.is_none() || owned {
161                        (
162                            if original.is_none() {
163                                ClientPlanAction::Create
164                            } else {
165                                ClientPlanAction::Update
166                            },
167                            content.clone(),
168                        )
169                    } else {
170                        return Err(Error::SetupFailure(format!(
171                            "refusing to overwrite unowned discovery skill {}",
172                            path.display()
173                        )));
174                    }
175                }
176                SetupOperation::Remove if owned => (ClientPlanAction::Remove, Some(String::new())),
177                SetupOperation::Remove => (ClientPlanAction::NotConfigured, None),
178            };
179            Ok(PlannedDiscoveryEdit {
180                public: DiscoverySetupPlan { path, action },
181                original,
182                updated,
183            })
184        })
185        .collect()
186}

188pub(super) fn discovery_skill(launcher: &McpLauncher) -> Result<String> {
189    let doctor = launcher.doctor_command();
190    Ok(format!(
191        "---\nname: leantoken\ndescription: Use LeanToken for token-bounded repository exploration, audits, codebase investigations, architecture reviews, source archaeology, code search, symbol outlines, exact source reads, symbol history, and structural JSON queries.\n---\n\n{DISCOVERY_SKILL_MARKER}\n\nBefore retrieving repository source, including for audits and code archaeology, discover the deferred `leantoken` MCP server and choose the narrowest applicable lane:\n\n1. For autonomous broad triage, call `leantoken.context` once with `plan_only=false` and use the materialized evidence directly. Make at most one focused follow-up only when coverage identifies a concrete gap.\n2. For known scope, `leantoken.files` finds paths, `leantoken.outline` maps definitions and imports, and `leantoken.search` locates symbols, references, identifiers, text, or regex matches. `leantoken.read` returns the exact current symbol or narrow line range; `leantoken.history` reads, diffs, or traces symbols across immutable Git revisions; `leantoken.json` queries, summarizes, or compares exact JSON artifacts without whole-file output.\n3. For human review or control-plane inspection before expensive or high-risk materialization, call `leantoken.context` with `plan_only=true`, inspect the bounded metadata and coverage, then materialize after approval.\n\nPass `BASE..HEAD` as `base_revision` with `strict_changed_paths` for immutable range context. Use `leantoken.savings` for repository-local savings. Use native workspace tools for edits, commands, tests, runtime probes, unsupported files, or evidence that is not source retrieval. If the server or tools cannot be discovered, run `{doctor}` and report its structured registration, launch, handshake, and catalog status instead of silently claiming LeanToken was used.\n"
192    ))
193}

195pub(super) fn resolve_client_edit(
196    operation: SetupOperation,
197    client: SetupClient,
198    detected: &[SetupClient],
199    home: &Path,
200    launcher: &McpLauncher,
201    configuration_snapshots: &[PlannedConfigurationSnapshot],
202) -> Result<PlannedClientEdit> {
203    let definition = client_definition_from_snapshots(client, home, configuration_snapshots);
204    let original = match configuration_snapshots
205        .iter()
206        .find(|snapshot| snapshot.path == definition.path)
207    {
208        Some(snapshot) => snapshot.original.clone(),
209        None => read_optional(&definition.path)?,
210    };
211    let (status, original, updated) = match definition.format {
212        ConfigFormat::Json { section, shape } => resolve_json_edit_from_source(
213            operation,
214            &definition.path,
215            section,
216            shape,
217            launcher,
218            original,
219        )?,
220        ConfigFormat::Toml => {
221            resolve_toml_edit_from_source(operation, &definition.path, launcher, original)?
222        }
223    };
224    let action = match status {
225        EditStatus::Configured if original.is_none() => ClientPlanAction::Create,
226        EditStatus::Configured | EditStatus::Updated => ClientPlanAction::Update,
227        EditStatus::AlreadyConfigured => ClientPlanAction::AlreadyCurrent,
228        EditStatus::Removed => ClientPlanAction::Remove,
229        EditStatus::NotConfigured => ClientPlanAction::NotConfigured,
230    };
231    Ok(PlannedClientEdit {
232        public: ClientSetupPlan {
233            client,
234            path: definition.path,
235            action,
236            detected: detected.contains(&client),
237        },
238        status,
239        original,
240        updated,
241    })
