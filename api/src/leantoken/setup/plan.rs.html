28pub(super) fn resolve_plan(
29    operation: SetupOperation,
30    clients: &[SetupClient],
31    environment: PlanEnvironment<'_>,
32) -> Result<ResolvedSetupPlan> {
33    let edits = clients
34        .iter()
35        .copied()
36        .map(|client| {
37            resolve_client_edit(
38                operation,
39                client,
40                environment.detected,
41                environment.home,
42                environment.launcher,
43            )
44        })
45        .collect::<Result<Vec<_>>>()?;
46    let discovery_edits = if environment.manage_discovery {
47        resolve_discovery_edits(operation, environment.home, Some(environment.launcher))?
48    } else {
49        Vec::new()
50    };
51    let launcher = (operation == SetupOperation::Setup)
52        .then(|| launcher_plan(environment.launcher, environment.runtime.as_ref()))
53        .transpose()?;
54    Ok(ResolvedSetupPlan {
55        operation,
56        persistent_cli: environment.persistent_cli,
57        launcher,
58        runtime: environment.runtime,
59        edits,
60        discovery_edits,
61        transaction_root: environment.transaction_root.to_path_buf(),
62    })
63}

65pub(super) fn launcher_plan(
66    launcher: &McpLauncher,
67    runtime: Option<&RuntimeInstallPlan>,
68) -> Result<LauncherPlan> {
69    Ok(LauncherPlan {
70        command: launcher.command()?.to_string(),
71        args: launcher.args.clone(),
72        version: launcher.version().into(),
73        package: launcher.npm_package().map(str::to_owned),
74        may_contact_network: launcher.uses_npx(),
75        runtime_path: runtime.map(|runtime| runtime.destination.clone()),
76        runtime_digest: runtime.map(|runtime| runtime.digest.clone()),
77    })
78}

80pub(super) fn resolve_discovery_edits(
81    operation: SetupOperation,
82    home: &Path,
83    launcher: Option<&McpLauncher>,
84) -> Result<Vec<PlannedDiscoveryEdit>> {
85    let content = launcher.map(discovery_skill).transpose()?;
86    [
87        home.join(".agents/skills/leantoken/SKILL.md"),
88        home.join(".claude/skills/leantoken/SKILL.md"),
89    ]
90    .into_iter()
91    .map(|path| {
92        let original = read_optional(&path)?;
93        let owned = original
94            .as_deref()
95            .is_some_and(|value| value.contains(DISCOVERY_SKILL_MARKER));
96        let (action, updated) = match operation {
97            SetupOperation::Setup => {
98                if original.as_deref() == content.as_deref() {
99                    (ClientPlanAction::AlreadyCurrent, None)
100                } else if original.is_none() || owned {
101                    (
102                        if original.is_none() {
103                            ClientPlanAction::Create
104                        } else {
105                            ClientPlanAction::Update
106                        },
107                        content.clone(),
108                    )
109                } else {
110                    return Err(Error::SetupFailure(format!(
111                        "refusing to overwrite unowned discovery skill {}",
112                        path.display()
113                    )));
114                }
115            }
116            SetupOperation::Remove if owned => (ClientPlanAction::Remove, Some(String::new())),
117            SetupOperation::Remove => (ClientPlanAction::NotConfigured, None),
118        };
119        Ok(PlannedDiscoveryEdit {
120            public: DiscoverySetupPlan { path, action },
121            original,
122            updated,
123        })
124    })
125    .collect()
126}

128pub(super) fn discovery_skill(launcher: &McpLauncher) -> Result<String> {
129    let doctor = if launcher.uses_npx() {
130        format!(
131            "npx --yes {} doctor --json",
132            launcher.npm_package().unwrap_or("leantoken")
133        )
134    } else {
135        "leantoken doctor --json".into()
136    };
137    Ok(format!(
138        "---\nname: leantoken\ndescription: Use LeanToken for token-bounded repository exploration, audits, codebase investigations, architecture reviews, source archaeology, code search, symbol outlines, exact source reads, symbol history, and structural JSON queries.\n---\n\n{DISCOVERY_SKILL_MARKER}\n\nBefore retrieving repository source, including for audits and code archaeology, discover the deferred `leantoken` MCP server and choose the narrowest applicable lane:\n\n1. For autonomous broad triage, call `leantoken.context` once with `plan_only=false` and use the materialized evidence directly. Make at most one focused follow-up only when coverage identifies a concrete gap.\n2. For known scope, `leantoken.files` finds paths, `leantoken.outline` maps definitions and imports, and `leantoken.search` locates symbols, references, identifiers, text, or regex matches. `leantoken.read` returns the exact current symbol or narrow line range; `leantoken.history` reads, diffs, or traces symbols across immutable Git revisions; `leantoken.json` queries, summarizes, or compares exact JSON artifacts without whole-file output.\n3. For human review or control-plane inspection before expensive or high-risk materialization, call `leantoken.context` with `plan_only=true`, inspect the bounded metadata and coverage, then materialize after approval.\n\nPass `BASE..HEAD` as `base_revision` with `strict_changed_paths` for immutable range context. Use `leantoken.savings` for repository-local savings. Use native workspace tools for edits, commands, tests, runtime probes, unsupported files, or evidence that is not source retrieval. If the server or tools cannot be discovered, run `{doctor}` and report its structured registration, launch, handshake, and catalog status instead of silently claiming LeanToken was used.\n"
139    ))
140}

142pub(super) fn resolve_client_edit(
143    operation: SetupOperation,
144    client: SetupClient,
145    detected: &[SetupClient],
146    home: &Path,
147    launcher: &McpLauncher,
148) -> Result<PlannedClientEdit> {
149    let definition = client.definition(home);
150    let (status, original, updated) = match definition.format {
151        ConfigFormat::Json { section, shape } => {
152            resolve_json_edit(operation, &definition.path, section, shape, launcher)?
153        }
154        ConfigFormat::Toml => resolve_toml_edit(operation, &definition.path, launcher)?,
155    };
156    let action = match status {
157        EditStatus::Configured if original.is_none() => ClientPlanAction::Create,
158        EditStatus::Configured | EditStatus::Updated => ClientPlanAction::Update,
159        EditStatus::AlreadyConfigured => ClientPlanAction::AlreadyCurrent,
160        EditStatus::Removed => ClientPlanAction::Remove,
161        EditStatus::NotConfigured => ClientPlanAction::NotConfigured,
162    };
163    Ok(PlannedClientEdit {
164        public: ClientSetupPlan {
165            client,
166            path: definition.path,
167            action,
168            detected: detected.contains(&client),
169        },
170        status,
171        original,
172        updated,
173    })
174}
