3pub(super) fn resolve_json_edit_from_source(
4    operation: SetupOperation,
5    path: &Path,
6    section_name: &str,
7    shape: JsonEntryShape,
8    launcher: &McpLauncher,
9    original: Option<String>,
10) -> Result<ResolvedEdit> {
11    let source = original.clone().unwrap_or_else(|| "{}\n".into());
12    let root = CstRootNode::parse(&source, &ParseOptions::default())
13        .map_err(|error| invalid_config(path, error))?;
14    let object = root
15        .object_value_or_create()
16        .ok_or_else(|| invalid_config(path, "top-level value must be an object"))?;
17    let section = match object.object_value_or_create(section_name) {
18        Some(section) => section,
19        None => {
20            return Err(invalid_config(
21                path,
22                format!("{section_name} must be an object"),
23            ));
24        }
25    };
26
27    match operation {
28        SetupOperation::Setup => {
29            let expected = json_entry(shape, launcher)?;
30            match section.get(SERVER_NAME) {
31                Some(property) => {
32                    let current = property
33                        .value()
34                        .ok_or_else(|| invalid_config(path, "LeanToken entry has no value"))?;
35                    let current_value: Value = jsonc_parser::parse_to_serde_value(
36                        &current.to_string(),
37                        &ParseOptions::default(),
38                    )
39                    .map_err(|error| invalid_config(path, error))?;
40                    if current_value == expected {
41                        return Ok(ResolvedEdit::AlreadyConfigured { original });
42                    }
43                    property.set_value(to_cst_input(&expected));
44                    Ok(ResolvedEdit::Updated {
45                        original,
46                        updated: root.to_string(),
47                    })
48                }
49                None => {
50                    section.append(SERVER_NAME, to_cst_input(&expected));
51                    Ok(ResolvedEdit::Configured {
52                        original,
53                        updated: root.to_string(),
54                    })
55                }
56            }
57        }
58        SetupOperation::Remove => {
59            let Some(original) = original else {
60                return Ok(ResolvedEdit::NotConfigured { original: None });
61            };
62            let Some(property) = section.get(SERVER_NAME) else {
63                return Ok(ResolvedEdit::NotConfigured {
64                    original: Some(original),
65                });
66            };
67            property.remove();
68            if section.properties().is_empty() {
69                object
70                    .get(section_name)
71                    .expect("section property exists")
72                    .remove();
73            }
74            Ok(ResolvedEdit::Removed {
75                original,
76                updated: root.to_string(),
77            })
78        }
79    }
80}

82pub(super) fn json_entry(shape: JsonEntryShape, launcher: &McpLauncher) -> Result<Value> {
83    let command = launcher.command()?;
84    Ok(match shape {
85        JsonEntryShape::CommandAndArgs => json!({
86            "command": command,
87            "args": launcher.args
88        }),
89        JsonEntryShape::OpenCode => json!({
90            "type": "local",
91            "command": std::iter::once(command).chain(launcher.args.iter().map(String::as_str)).collect::<Vec<_>>(),
92            "cwd": ".",
93            "enabled": true
94        }),
95    })
96}

98pub(super) fn to_cst_input(value: &Value) -> CstInputValue {
99    match value {
100        Value::Null => CstInputValue::Null,
101        Value::Bool(value) => CstInputValue::Bool(*value),
102        Value::Number(value) => CstInputValue::Number(value.to_string()),
103        Value::String(value) => CstInputValue::String(value.clone()),
104        Value::Array(values) => CstInputValue::Array(values.iter().map(to_cst_input).collect()),
105        Value::Object(values) => CstInputValue::Object(
106            values
107                .iter()
108                .map(|(key, value)| (key.clone(), to_cst_input(value)))
109                .collect(),
110        ),
111    }
