34pub(super) fn resolve_toml_edit_from_source(
35    operation: SetupOperation,
36    path: &Path,
37    launcher: &McpLauncher,
38    original: Option<String>,
39) -> Result<ResolvedEdit> {
40    let source = original.clone().unwrap_or_default();
41    let mut document = if source.trim().is_empty() {
42        DocumentMut::new()
43    } else {
44        source
45            .parse::<DocumentMut>()
46            .map_err(|error| invalid_config(path, error))?
47    };
48
49    match operation {
50        SetupOperation::Setup => {
51            let command = launcher.command()?;
52            let servers = ensure_toml_table(&mut document, "mcp_servers", path)?;
53            if let Some(existing) = servers.get(SERVER_NAME)
54                && toml_entry_matches(existing, command, &launcher.args)
55            {
56                return Ok(ResolvedEdit::AlreadyConfigured { original });
57            }
58            let existed = servers.contains_key(SERVER_NAME);
59            let mut server = Table::new();
60            server["command"] = value(command);
61            let mut args = Array::new();
62            launcher
63                .args
64                .iter()
65                .for_each(|argument| args.push(argument));
66            server["args"] = value(args);
67            server["startup_timeout_sec"] = value(CODEX_STARTUP_TIMEOUT_SECONDS as i64);
68            servers.insert(SERVER_NAME, Item::Table(server));
69            let updated = document.to_string();
70            if existed {
71                Ok(ResolvedEdit::Updated { original, updated })
72            } else {
73                Ok(ResolvedEdit::Configured { original, updated })
74            }
75        }
76        SetupOperation::Remove => {
77            let Some(original) = original else {
78                return Ok(ResolvedEdit::NotConfigured { original: None });
79            };
80            let Some(servers_item) = document.get_mut("mcp_servers") else {
81                return Ok(ResolvedEdit::NotConfigured {
82                    original: Some(original),
83                });
84            };
85            let servers = servers_item
86                .as_table_mut()
87                .ok_or_else(|| invalid_config(path, "mcp_servers must be a table"))?;
88            if servers.remove(SERVER_NAME).is_none() {
89                return Ok(ResolvedEdit::NotConfigured {
90                    original: Some(original),
91                });
92            }
93            if servers.is_empty() {
94                document.remove("mcp_servers");
95            }
96            Ok(ResolvedEdit::Removed {
97                original,
98                updated: document.to_string(),
99            })
100        }
101    }
102}

104pub(super) fn ensure_toml_table<'a>(
105    document: &'a mut DocumentMut,
106    name: &str,
107    path: &Path,
108) -> Result<&'a mut Table> {
109    if !document.contains_key(name) {
110        document.insert(name, Item::Table(Table::new()));
111    }
112    document
113        .get_mut(name)
114        .and_then(Item::as_table_mut)
115        .ok_or_else(|| invalid_config(path, format!("{name} must be a table")))
116}

118pub(super) fn toml_entry_matches(item: &Item, command: &str, expected_args: &[String]) -> bool {
119    let Some(table) = item.as_table() else {
120        return false;
121    };
122    let command_matches = table
123        .get("command")
124        .and_then(Item::as_str)
125        .is_some_and(|value| value == command);
126    let args_match = table
127        .get("args")
128        .and_then(Item::as_array)
129        .is_some_and(|args| {
130            args.iter()
131                .filter_map(|value| value.as_str())
132                .eq(expected_args.iter().map(String::as_str))
133                && args.len() == expected_args.len()
134        });
135    let timeout_matches = table
136        .get("startup_timeout_sec")
137        .and_then(toml_positive_integer)
138        == Some(CODEX_STARTUP_TIMEOUT_SECONDS);
139    command_matches && args_match && timeout_matches
140}
