12impl fmt::Display for EditStatus {

25pub(super) fn read_optional(path: &Path) -> Result<Option<String>> {
26    match fs::read_to_string(path) {
27        Ok(contents) => Ok(Some(contents)),
28        Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(None),
29        Err(error) => Err(error.into()),
30    }
31}

33pub(super) fn write_if_changed(path: &Path, original: &str, updated: &str) -> Result<()> {
34    if original == updated {
35        return Ok(());
36    }
37    let parent = path.parent().ok_or_else(|| {
38        Error::SetupFailure(format!("config path has no parent: {}", path.display()))
39    })?;
40    fs::create_dir_all(parent)?;
41    let mut temporary = NamedTempFile::new_in(parent)?;
42    temporary.write_all(updated.as_bytes())?;
43    temporary.as_file_mut().sync_all()?;
44    if let Ok(metadata) = fs::metadata(path) {
45        temporary
46            .as_file()
47            .set_permissions(metadata.permissions())?;
48    }
49    temporary
50        .persist(path)
51        .map_err(|error| Error::Io(error.error))?;
52    sync_parent_directory(path)?;
53    Ok(())
54}

56#[cfg(unix)]
57pub(super) fn sync_parent_directory(path: &Path) -> Result<()> {
58    let parent = path
59        .parent()
60        .ok_or_else(|| Error::SetupFailure(format!("path has no parent: {}", path.display())))?;
61    fs::File::open(parent)?.sync_all()?;
62    Ok(())
63}
64
65#[cfg(not(unix))]
66pub(super) fn sync_parent_directory(_path: &Path) -> Result<()> {
67    // Windows does not expose the Unix directory-fsync contract through
68    // std::fs. File contents are still synced before atomic replacement.
69    Ok(())
70}
