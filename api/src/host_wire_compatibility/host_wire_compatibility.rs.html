270fn validate(matrix: &Matrix, repository_root: &Path) -> AnyResult<ValidationSummary> {
271    validate_matrix(matrix)?;
272    let evidence = validate_evidence(matrix, repository_root)?;
273    validate_codex_01441(matrix, &evidence)?;
274    validate_codex_01445(matrix, &evidence)?;
275
276    let host_families = matrix
277        .host_observations
278        .iter()
279        .map(|observation| observation.host.as_str())
280        .collect::<BTreeSet<_>>()
281        .len();
282    let model_consumption_proofs = matrix
283        .host_observations
284        .iter()
285        .flat_map(|observation| &observation.modes)
286        .filter(|mode| mode.model_consumption == "proven")
287        .count();
288    let unavailable_host_families = matrix
289        .host_observations
290        .iter()
291        .filter(|observation| observation.availability == "unavailable")
292        .map(|observation| observation.host.clone())
293        .collect();
294
295    Ok(ValidationSummary {
296        schema_version: matrix.schema_version,
297        experiment: matrix.experiment.clone(),
298        audit_date: matrix.audit_date.clone(),
299        global_default: matrix.policy.global_default.clone(),
300        host_families,
301        host_version_observations: matrix.host_observations.len(),
302        model_consumption_proofs,
303        unavailable_host_families,
304        evidence_artifacts_verified: evidence.len(),
305        decision: matrix.policy.decision.clone(),
306    })
307}

309fn validate_matrix(matrix: &Matrix) -> AnyResult<()> {
310    if matrix.schema_version != 1 || matrix.experiment != "host-wire-compatibility-v1" {
311        return Err(invalid_data("unsupported host compatibility matrix schema"));
312    }
313    if matrix.audit_date != "2026-07-20" {
314        return Err(invalid_data("unexpected host compatibility audit date"));
315    }
316    if matrix.policy.global_default != "dual"
317        || !matrix.policy.unknown_values_are_null
318        || !matrix.policy.eligible_smaller_mode_host_versions.is_empty()
319        || matrix.policy.decision.trim().is_empty()
320    {
321        return Err(invalid_data(
322            "matrix policy does not preserve the dual default",
323        ));
324    }
325    if matrix.limitations.len() < 5 || matrix.limitations.iter().any(|item| item.trim().is_empty())
326    {
327        return Err(invalid_data("matrix requires explicit limitations"));
328    }
329
330    let hosts = matrix
331        .host_observations
332        .iter()
333        .map(|observation| observation.host.as_str())
334        .collect::<BTreeSet<_>>();
335    if hosts != REQUIRED_HOSTS.into_iter().collect() {
336        return Err(invalid_data(
337            "matrix does not cover every required host family",
338        ));
339    }
340
341    let mut observation_keys = BTreeSet::new();
342    for observation in &matrix.host_observations {
343        if !observation_keys.insert((observation.host.as_str(), observation.version.as_deref())) {
344            return Err(invalid_data("duplicate host/version observation"));
345        }
346        if observation.availability_note.trim().is_empty() {
347            return Err(invalid_data(
348                "host observation requires an availability note",
349            ));
350        }
351        let modes = observation
352            .modes
353            .iter()
354            .map(|mode| mode.mode.as_str())
355            .collect::<BTreeSet<_>>();
356        if modes != MODES.into_iter().collect() || observation.modes.len() != MODES.len() {
357            return Err(invalid_data(
358                "host observation must contain each result mode once",
359            ));
360        }
361        for mode in &observation.modes {
362            validate_mode(mode)?;
363        }
364        if observation.availability == "unavailable" {
365            validate_unavailable(observation)?;
366        } else if observation.version.is_none() {
367            return Err(invalid_data("captured host observation requires a version"));
368        }
369    }
370
371    let codex_versions = matrix
372        .host_observations
373        .iter()
374        .filter(|observation| observation.host == "codex-cli")
375        .filter_map(|observation| observation.version.as_deref())
376        .collect::<BTreeSet<_>>();
377    if codex_versions != ["0.144.1", "0.144.5"].into_iter().collect() {
378        return Err(invalid_data("matrix requires both frozen Codex versions"));
379    }
380
381    let mut evidence_ids = BTreeSet::new();
382    for item in &matrix.evidence {
383        if !evidence_ids.insert(item.id.as_str()) || item.kind.trim().is_empty() {
384            return Err(invalid_data(
385                "evidence identifiers must be unique and typed",
386            ));
387        }
388        validate_relative_evidence_path(&item.path)?;
389        if item.blake3.len() != 64 || !item.blake3.bytes().all(|byte| byte.is_ascii_hexdigit()) {
390            return Err(invalid_data(
391                "evidence BLAKE3 must be 64 hexadecimal characters",
392            ));
393        }
394    }
395    for mode in matrix
396        .host_observations
397        .iter()
398        .flat_map(|observation| &observation.modes)
399    {
400        if mode
401            .evidence_ids
402            .iter()
403            .any(|id| !evidence_ids.contains(id.as_str()))
404        {
405            return Err(invalid_data("mode references unknown evidence"));
406        }
407    }
408    Ok(())
409}

411fn validate_mode(mode: &ModeObservation) -> AnyResult<()> {
412    if !["complete", "tool_results_only", "unknown"].contains(&mode.wire_transport.as_str())
413        || !["proven", "unknown"].contains(&mode.model_consumption.as_str())
414        || !["partial", "unknown"].contains(&mode.provider_usage.as_str())
415        || mode.provider_framing != "unknown"
416        || mode.conclusion.trim().is_empty()
417    {
418        return Err(invalid_data("mode contains an unsupported evidence state"));
419    }
420    match (mode.provider_usage.as_str(), &mode.provider_tokens) {
421        ("partial", Some(tokens))
422            if tokens.total_input.is_some()
423                && tokens.uncached_input.is_some()
424                && tokens.cache_creation_input.is_none()
425                && tokens.cache_read_input.is_some()
426                && tokens.output.is_some()
427                && tokens.reasoning.is_some() => {}
428        ("unknown", None) => {}
429        _ => {
430            return Err(invalid_data(
431                "provider token values do not match their evidence state",
432            ));
433        }
434    }
435    if mode.model_consumption == "proven" && mode.evidence_ids.is_empty() {
436        return Err(invalid_data("model-consumption proof requires evidence"));
437    }
438    Ok(())
439}

441fn validate_unavailable(observation: &HostObservation) -> AnyResult<()> {
442    if observation.version.is_some() {
443        return Err(invalid_data("unavailable host version must remain null"));
444    }
445    for mode in &observation.modes {
446        let measurements_absent = mode.complete_wire_tokens.is_none()
447            && mode.result_text_tokens.is_none()
448            && mode.structured_content_tokens.is_none()
449            && mode.duplicated_result_tokens.is_none()
450            && mode.result_json_bytes.is_none()
451            && mode.result_text_bytes.is_none()
452            && mode.structured_content_bytes.is_none()
453            && mode.emitted_source_tokens.is_none()
454            && mode.provider_tokens.is_none();
455        if mode.wire_transport != "unknown"
456            || mode.model_consumption != "unknown"
457            || mode.provider_usage != "unknown"
458            || !mode.evidence_ids.is_empty()
459            || !measurements_absent
460        {
461            return Err(invalid_data(
462                "unavailable host measurements must remain unknown and null",
463            ));
464        }
465    }
466    Ok(())
467}

469fn validate_evidence(
470    matrix: &Matrix,
471    repository_root: &Path,
472) -> AnyResult<BTreeMap<String, Vec<u8>>> {
473    let mut loaded = BTreeMap::new();
474    for evidence in &matrix.evidence {
475        let bytes = canonical_json_artifact(fs::read(repository_root.join(&evidence.path))?)?;
476        let actual = blake3::hash(&bytes).to_hex().to_string();
477        if actual != evidence.blake3 {
478            return Err(invalid_data(&format!(
479                "evidence hash mismatch for {}",
480                evidence.id
481            )));
482        }
483        loaded.insert(evidence.id.clone(), bytes);
484    }
485    Ok(loaded)
486}
487
488fn canonical_json_artifact(bytes: Vec<u8>) -> AnyResult<Vec<u8>> {
489    let text = String::from_utf8(bytes)?;
490    let normalized = text.replace("\r\n", "\n");
491    if normalized.contains('\r') {
492        return Err(invalid_data(
493            "evidence JSON contains a lone carriage return",
494        ));
495    }
496    Ok(normalized.into_bytes())
497}

499fn validate_codex_01441(matrix: &Matrix, evidence: &BTreeMap<String, Vec<u8>>) -> AnyResult<()> {
500    let wire: WireAnalysisV2 = evidence_json(evidence, "codex-0.144.1-dual-wire")?;
501    let receipt: HostReceipt = evidence_json(evidence, "codex-0.144.1-host-receipt")?;
502    let structured: StructuredReceipt = evidence_json(evidence, "codex-0.144.1-structured-owner")?;
503
504    if wire.schema_version != 2
505        || wire.host != "codex-cli"
506        || wire.host_version != "0.144.1"
507        || !wire.token_count_exact
508        || wire.total_serialized_json_tokens != 4483
509        || wire.provider_total_input_tokens.is_some()
510        || provider_has_value(&wire.provider_usage)
511        || !required_exchange_complete(&wire.required_exchange_parts)
512        || wire.tool_result_modes.get("dual") != Some(&3)
513        || wire.exact_text_structured_duplicates != 3
514        || wire.duplicated_result_tokens != 776
515        || event_cost(&wire.event_categories, "initialize_request")? != (1, 67)
516        || event_cost(&wire.event_categories, "initialize_response")? != (1, 118)
517        || event_cost(&wire.event_categories, "initialized_notification")? != (1, 14)
518        || event_cost(&wire.event_categories, "tool_call_request")? != (3, 949)
519        || event_cost(&wire.event_categories, "tool_call_response")? != (3, 1694)
520        || event_cost(&wire.event_categories, "tools_list_request")? != (1, 27)
521        || event_cost(&wire.event_categories, "tools_list_response")? != (1, 1614)
522        || component_cost(&wire.components, "catalog")? != (1, 1595)
523        || component_cost(&wire.components, "call_arguments")? != (3, 69)
524        || component_cost(&wire.components, "result_text")? != (3, 854)
525        || component_cost(&wire.components, "structured_content")? != (3, 776)
526    {
527        return Err(invalid_data("Codex 0.144.1 dual wire evidence changed"));
528    }
529
530    let privacy_absent = !receipt.privacy.raw_rollout_retained
531        && !receipt.privacy.raw_mcp_messages_retained
532        && !receipt.privacy.prompts_retained
533        && !receipt.privacy.tool_arguments_retained
534        && !receipt.privacy.tool_outputs_retained
535        && !receipt.privacy.credentials_retained
536        && !receipt.privacy.absolute_paths_retained
537        && receipt.privacy.session_and_call_ids_hashed_or_omitted;
538    if receipt.schema_version != 1
539        || receipt.host != "codex-cli"
540        || receipt.host_version != "0.144.1"
541        || !receipt.token_count_exact
542        || receipt.total_input_tokens != 70904
543        || receipt.tool_calls.len() != 3
544        || receipt
545            .tool_calls
546            .iter()
547            .any(|call| !call.followed_by_model_response || !call.followed_by_provider_usage)
548        || receipt.mcp_correlation.rollout_tool_calls != 3
549        || receipt.mcp_correlation.mcp_tool_calls != 3
550        || !receipt.mcp_correlation.tool_order_matches
551        || !receipt.mcp_correlation.protocol_order_valid
552        || receipt.mcp_correlation.semantic_output_matches != 3
553        || !receipt.mcp_correlation.all_semantic_outputs_match
554        || !privacy_absent
555    {
556        return Err(invalid_data("Codex 0.144.1 host receipt evidence changed"));
557    }
558    require_provider_tokens(
559        &receipt.provider_usage,
560        [None, Some(7672), None, Some(63232), Some(282), Some(87)],
561    )?;
562
563    let structured_privacy_absent = !structured.privacy.raw_rollouts_retained
564        && !structured.privacy.prompts_retained
565        && !structured.privacy.tool_arguments_retained
566        && !structured.privacy.tool_outputs_retained
567        && !structured.privacy.credentials_retained
568        && !structured.privacy.absolute_paths_retained
569        && structured
570            .privacy
571            .session_call_and_agent_ids_hashed_or_omitted;
572    let child = structured
573        .thread_receipts
574        .iter()
575        .find(|thread| thread.role == "subagent")
576        .ok_or_else(|| invalid_data("structured receipt has no child thread"))?;
577    let task = &structured.task_evaluation;
578    if structured.schema_version != 1
579        || structured.experiment_id != "multi-agent-context-pilot"
580        || structured.arm != "thin-leantoken-structured-owner"
581        || structured.host != "codex-cli"
582        || structured.thread_receipts.len() != 2
583        || child.host_version != "0.144.1"
584        || !task.answer_json_valid
585        || task.expected_evidence_count != 4
586        || task.reported_evidence_count != 4
587        || task.matched_path_count != 4
588        || task.matched_exact_evidence_count != 4
589        || task.unexpected_evidence_count != 0
590        || !task.task_success
591        || child.tool_calls.mcp != 6
592        || child.tool_calls.failed_mcp != 0
593        || child.tool_calls.mcp_result_json_bytes != 38127
594        || child.tool_calls.mcp_text_content_bytes != 0
595        || child.tool_calls.mcp_structured_content_bytes != 37821
596        || child.tool_calls.mcp_emitted_source_tokens != 5265
597        || !structured_privacy_absent
598    {
599        return Err(invalid_data(
600            "Codex 0.144.1 structured model-consumption evidence changed",
601        ));
602    }
603    require_provider_tokens(
604        &child.provider_usage,
605        [
606            Some(83923),
607            Some(17619),
608            None,
609            Some(66304),
610            Some(658),
611            Some(217),
612        ],
613    )?;
614
615    let observation = host_observation(matrix, "codex-cli", "0.144.1")?;
616    let dual = mode_observation(observation, "dual")?;
617    if dual.complete_wire_tokens != Some(wire.total_serialized_json_tokens)
618        || dual.result_text_tokens != Some(component_cost(&wire.components, "result_text")?.1)
619        || dual.structured_content_tokens
620            != Some(component_cost(&wire.components, "structured_content")?.1)
621        || dual.duplicated_result_tokens != Some(wire.duplicated_result_tokens)
622        || !provider_tokens_equal_with_total(
623            dual.provider_tokens.as_ref(),
624            receipt.total_input_tokens,
625            &receipt.provider_usage,
626        )
627    {
628        return Err(invalid_data("Codex 0.144.1 dual matrix values drifted"));
629    }
630    let structured_mode = mode_observation(observation, "structured")?;
631    if structured_mode.result_json_bytes != Some(child.tool_calls.mcp_result_json_bytes)
632        || structured_mode.result_text_bytes != Some(child.tool_calls.mcp_text_content_bytes)
633        || structured_mode.structured_content_bytes
634            != Some(child.tool_calls.mcp_structured_content_bytes)
635        || structured_mode.emitted_source_tokens != Some(child.tool_calls.mcp_emitted_source_tokens)
636        || !provider_tokens_equal(
637            structured_mode.provider_tokens.as_ref(),
638            &child.provider_usage,
639        )

641        return Err(invalid_data(

648fn validate_codex_01445(matrix: &Matrix, evidence: &BTreeMap<String, Vec<u8>>) -> AnyResult<()> {
649    let wire: WireAnalysisV1 = evidence_json(evidence, "codex-0.144.5-dual-wire")?;
650    if wire.schema_version != 1
651        || wire.host != "codex-cli"
652        || wire.host_version != "0.144.5"
653        || !wire.token_count_exact
654        || wire.event_count != 9
655        || wire.total_local_tokens != 2896
656        || wire.total_provider_input_tokens.is_some()
657        || !required_exchange_complete(&wire.required_exchange_parts)
658        || wire.tool_result_modes.get("dual") != Some(&2)
659        || legacy_event_cost(&wire.categories, "initialize_request")? != (1, 67)
660        || legacy_event_cost(&wire.categories, "initialize_response")? != (1, 97)
661        || legacy_event_cost(&wire.categories, "initialized_notification")? != (1, 14)
662        || legacy_event_cost(&wire.categories, "tool_call_request")? != (2, 530)
663        || legacy_event_cost(&wire.categories, "tool_call_response")? != (2, 607)
664        || legacy_event_cost(&wire.categories, "tools_list_request")? != (1, 27)
665        || legacy_event_cost(&wire.categories, "tools_list_response")? != (1, 1554)
666        || wire
667            .categories
668            .values()
669            .any(|category| category.provider_input_tokens.is_some())
670    {
671        return Err(invalid_data("Codex 0.144.5 wire evidence changed"));
672    }
673    let dual = mode_observation(host_observation(matrix, "codex-cli", "0.144.5")?, "dual")?;
674    if dual.complete_wire_tokens != Some(wire.total_local_tokens) {
675        return Err(invalid_data("Codex 0.144.5 matrix values drifted"));
676    }
677    Ok(())
678}

774fn host_observation<'a>(

786}

809fn validate_relative_evidence_path(path: &Path) -> AnyResult<()> {
810    let mut components = path.components();
811    if components.next() != Some(Component::Normal("benchmarks".as_ref()))
812        || components.next() != Some(Component::Normal("reports".as_ref()))
813        || components.any(|component| !matches!(component, Component::Normal(_)))
814    {
815        return Err(invalid_data(
816            "evidence path must be a safe benchmarks/reports relative path",
817        ));
818    }
819    Ok(())
820}
