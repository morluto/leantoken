562#[tokio::main]
563async fn main() -> Result<(), Box<dyn Error>> {
564    let args = Args::parse();
565    let manifest_json = fs::read_to_string(&args.manifest)?;
566    let manifest_blake3 = blake3::hash(manifest_json.as_bytes()).to_hex().to_string();
567    let manifest: Manifest = serde_json::from_str(&manifest_json)?;
568    if !matches!(manifest.schema_version, 1..=4) {
569        return Err(format!(
570            "unsupported benchmark manifest schema version {}",
571            manifest.schema_version
572        )
573        .into());
574    }
575    validate_manifest(&manifest)?;
576    let mut concept_labels = args
577        .concept_labels
578        .as_deref()
579        .map(|path| load_concept_labels(path, &args.manifest, &manifest, &manifest_blake3))
580        .transpose()?;
581    if args.consumed_diagnostic && manifest.dataset_kind != "blind_holdout" {
582        return Err("--consumed-diagnostic requires a blind_holdout manifest".into());
583    }
584    let source_root = Path::new(env!("LEANTOKEN_REPOSITORY_ROOT"));
585    let harness_revision = git_output(source_root, &["rev-parse", "HEAD"])?
586        .trim()
587        .to_owned();
588    let harness_worktree_dirty = !git_output(
589        source_root,
590        &["status", "--porcelain=v1", "--untracked-files=all"],
591    )?
592    .trim()
593    .is_empty();
594    let candidate_runtime_tree_verified = if args.consumed_diagnostic {
595        None
596    } else {
597        verify_candidate_runtime_tree(&manifest, source_root)?
598    };
599    let ripgrep_version = command_version("rg")?;
600    preflight(&manifest, &args.repos_root)?;
601    if args.preflight_only {
602        println!(
603            "{}",
604            serde_json::to_string_pretty(&serde_json::json!({
605                "manifest_blake3": manifest_blake3,
606                "dataset_kind": manifest.dataset_kind,
607                "candidate_revision": manifest.candidate_revision,
608                "harness_revision": harness_revision,
609                "harness_worktree_dirty": harness_worktree_dirty,
610                "candidate_runtime_tree_verified": candidate_runtime_tree_verified,
611                "diagnostic_only": args.consumed_diagnostic,
612                "workflow_evidence_enabled": args.workflow_evidence,
613                "history_lane_enabled": args.history_lane,
614                "ast_structural_lane_enabled": args.ast_structural_lane,
615                "ast_structural_lane_v2_enabled": args.ast_structural_lane_v2,
616                "orientation_capsule_enabled": args.orientation_capsule,
617                "concept_labels_blake3": concept_labels.as_ref().map(|labels| &labels.blake3),
618                "concept_count": concept_labels.as_ref().map(|labels| {
619                    labels.tasks.values().map(|task| task.concepts.len()).sum::<usize>()
620                }),
621                "corpus_count": manifest.corpora.len(),
622                "task_count": manifest.corpora.iter().map(|corpus| corpus.tasks.len()).sum::<usize>(),
623                "status": "ready"
624            }))?
625        );
626        return Ok(());
627    }
628
629    let scratch = tempfile::tempdir()?;
630    let mut corpora = Vec::new();
631    let mut aggregate = AggregateReport::default();
632    let mut task_families = BTreeMap::<String, AggregateReport>::new();
633    for corpus in manifest.corpora {
634        let root = args.repos_root.join(&corpus.directory);
635        verify_revision(&root, &corpus.base_revision)?;
636        let database_path = scratch.path().join(format!("{}.sqlite", corpus.name));
637        let config = Config::discover(&root, Some(database_path.clone()))?;
638        let services = Services::open(config)?;
639
640        let started = Instant::now();
641        let indexed = services.index(true).await?;
642        let cold_index_ms = elapsed_ms(started);
643        let mut tasks = Vec::new();
644        for task in corpus.tasks {
645            let labels = concept_labels
646                .as_mut()
647                .and_then(|loaded| loaded.tasks.remove(&task.id));
648            let report = run_task(
649                &root,
650                &services,
651                task,
652                RunTaskOptions {
653                    rg_max_lines_per_query: manifest.rg_max_lines_per_query,
654                    concept_labels: labels.as_ref(),
655                    workflow_evidence_enabled: args.workflow_evidence,
656                    history_lane_enabled: args.history_lane,

664            accumulate(&mut aggregate, &report);

666                task_families.entry(report.task_family.clone()).or_default(),

668            );
669            tasks.push(report);
670        }
671        let status = services.status().await?;
672        let database_bytes = database_footprint(&database_path)?;
673        aggregate.cold_index_ms += cold_index_ms;
674        aggregate.database_bytes = aggregate.database_bytes.saturating_add(database_bytes);
675        aggregate.process_rss_bytes = match (aggregate.process_rss_bytes, status.process_rss_bytes)
676        {
677            (Some(left), Some(right)) => Some(left.max(right)),

679            (None, None) => None,
680        };
681        corpora.push(CorpusReport {
682            name: corpus.name,
683            url: corpus.url,
684            base_revision: corpus.base_revision,
685            fix_commit: corpus.fix_commit,

691            dataset_license: corpus.dataset_license,
692            external_limitations: corpus.external_limitations,
693            indexed_files: status.file_count,
694            indexed_chunks: status.chunk_count,
695            index_warnings: indexed.warnings,
696            cold_index_ms,
697            database_bytes,
698            process_rss_bytes: status.process_rss_bytes,
699            tasks,
700        });
701    }
702    aggregate.corpus_count = corpora.len();
703    finalize_aggregate(&mut aggregate);
704    for family in task_families.values_mut() {
705        finalize_aggregate(family);
706    }
707    if let Some(labels) = &concept_labels
708        && !labels.tasks.is_empty()
709    {
710        return Err(format!(
711            "concept labels contain tasks absent from the source manifest: {}",
712            labels.tasks.keys().cloned().collect::<Vec<_>>().join(", ")
713        )
714        .into());
715    }
716    let concept_coverage = concept_labels
717        .as_ref()
718        .map(|labels| concept_coverage_decision(labels, &aggregate, &corpora));
719    let concept_thresholds_passed = concept_coverage
720        .as_ref()
721        .is_none_or(|coverage| coverage.passed);
722
723    let report = Report {
724        schema_version: manifest.schema_version,
725        dataset_kind: manifest.dataset_kind.clone(),
726        manifest_blake3,
727        frozen_at: manifest.frozen_at,
728        candidate_revision: manifest.candidate_revision,
729        evaluation_protocol: manifest.evaluation_protocol,
730        reclassification_rule: manifest.reclassification_rule,
731        manifest_description: manifest.description,
732        leantoken_version: env!("CARGO_PKG_VERSION"),
733        harness_revision,
734        harness_worktree_dirty,
735        candidate_runtime_tree_verified,
736        diagnostic_only: args.consumed_diagnostic,

744        rustc_version: command_version("rustc")?,
745        ripgrep_version,
746        generated_at_unix_seconds: SystemTime::now().duration_since(UNIX_EPOCH)?.as_secs(),
747        tokenizer: tokens::Tokenizer::default().name(),
748        token_count_exact: tokens::Tokenizer::default().is_exact(),
749        methodology: Methodology {
750            oracle_baseline: "Full contents of fix-labeled relevant files, as if an agent chose every file perfectly and paid no discovery cost.",
751            rg_discovery_baseline: "Bounded, path-sorted ripgrep --json output for fixed-string queries derived from each public bug task.",
752            scripted_baseline: "One JSON envelope containing the ripgrep discovery output and oracle full-file reads.",
753            source_tokens: "Tokens in source content only; excludes paths, scores, reasons, receipts, and JSON syntax.",
754            serialized_tokens: "Tokens in the complete serialized JSON payload, including metadata and syntax.",
755        },
756        concept_coverage,
757        aggregate,
758        task_families,
759        corpora,
760        limitations: benchmark_limitations(
761            &manifest.dataset_kind,
762            args.consumed_diagnostic,
763            args.concept_labels.is_some(),
764            args.history_lane,
765            args.ast_structural_lane,
766            args.ast_structural_lane_v2,
767            args.orientation_capsule,
768        ),
769    };
770    let json = serde_json::to_string_pretty(&report)?;
771    if let Some(parent) = args
772        .output
773        .parent()
774        .filter(|path| !path.as_os_str().is_empty())
775    {
776        fs::create_dir_all(parent)?;
777    }
778    fs::write(&args.output, &json)?;
779    println!("{json}");
780    if args.require_concept_thresholds && !concept_thresholds_passed {
781        return Err("frozen context concept-coverage thresholds failed".into());
782    }
783    Ok(())
784}

790fn validate_manifest(manifest: &Manifest) -> Result<(), Box<dyn Error>> {
791    if manifest.schema_version >= 5 {
792        for task in manifest.corpora.iter().flat_map(|corpus| &corpus.tasks) {
793            if task
794                .task_family
795                .as_deref()
796                .is_none_or(|family| family.trim().is_empty())
797            {
798                return Err(
799                    format!("manifest schema v5+ task {} requires task_family", task.id).into(),
800                );
801            }
802        }
803    }
804    if manifest.dataset_kind == "external_retrieval_corpus" {
805        if manifest.schema_version < 4 {
806            return Err("external retrieval corpora require manifest schema v4".into());
807        }
808        if manifest.frozen_at.as_deref().is_none_or(str::is_empty) {
809            return Err("external retrieval corpora require frozen_at".into());
810        }
811        for (field, value) in [
812            (
813                "evaluation_protocol",
814                manifest.evaluation_protocol.as_deref(),
815            ),
816            (
817                "reclassification_rule",
818                manifest.reclassification_rule.as_deref(),
819            ),
820        ] {
821            if value.is_none_or(str::is_empty) {
822                return Err(format!("external retrieval corpora require {field}").into());
823            }
824        }
825        for corpus in &manifest.corpora {
826            if corpus.fix_commit.is_some() {
827                return Err(
828                    format!("external corpus {} must not name a future fix", corpus.name).into(),
829                );
830            }
831            for (field, value) in [
832                ("dataset_url", corpus.dataset_url.as_deref()),
833                ("dataset_revision", corpus.dataset_revision.as_deref()),
834                ("dataset_license", corpus.dataset_license.as_deref()),
835                ("prompt_provenance", corpus.prompt_provenance.as_deref()),
836                ("label_provenance", corpus.label_provenance.as_deref()),
837            ] {
838                if value.is_none_or(str::is_empty) {
839                    return Err(format!("external corpus {} requires {field}", corpus.name).into());
840                }
841            }
842            let revision = corpus
843                .dataset_revision
844                .as_deref()
845                .expect("validated dataset revision");
846            if revision.len() != 40 || !revision.bytes().all(|byte| byte.is_ascii_hexdigit()) {
847                return Err(format!(
848                    "external corpus {} dataset revision is not a full Git object ID",
849                    corpus.name
850                )
851                .into());
852            }
853            if corpus.tasks.is_empty() {
854                return Err(format!("external corpus {} has no tasks", corpus.name).into());
855            }
856            if corpus.external_limitations.is_empty()
857                || corpus
858                    .external_limitations
859                    .iter()
860                    .any(|limitation| limitation.trim().is_empty())
861            {
862                return Err(format!(
863                    "external corpus {} requires explicit limitations",
864                    corpus.name
865                )
866                .into());
867            }
868            for task in &corpus.tasks {
869                if task.languages.is_empty()
870                    || task
871                        .languages
872                        .iter()
873                        .any(|language| language.trim().is_empty())
874                    || task.task_shapes.is_empty()
875                    || task.task_shapes.iter().any(|shape| shape.trim().is_empty())
876                {
877                    return Err(format!(
878                        "external task {} requires language and task-shape strata",
879                        task.id
880                    )
881                    .into());
882                }
883            }
884        }
885    }
886    if is_patch_free_dataset(&manifest.dataset_kind) {
887        if manifest.frozen_at.as_deref().is_none_or(str::is_empty) {
888            return Err(format!("{} set requires frozen_at", manifest.dataset_kind).into());
889        }
890        for corpus in &manifest.corpora {
891            if corpus.fix_commit.is_some() {
892                return Err(format!(
893                    "{} corpus {} must not name a future fix",
894                    manifest.dataset_kind, corpus.name
895                )
896                .into());
897            }
898            for (field, value) in [
899                ("issue_url", corpus.issue_url.as_deref()),
900                ("prompt_provenance", corpus.prompt_provenance.as_deref()),
901                ("label_provenance", corpus.label_provenance.as_deref()),
902            ] {
903                if value.is_none_or(str::is_empty) {
904                    return Err(format!(
905                        "{} corpus {} requires {field}",
906                        manifest.dataset_kind, corpus.name
907                    )
908                    .into());
909                }
910            }
911        }
912    }
913    if manifest.schema_version >= 3 && manifest.dataset_kind == "blind_holdout" {
914        for (field, value) in [
915            ("candidate_revision", manifest.candidate_revision.as_deref()),
916            (
917                "evaluation_protocol",
918                manifest.evaluation_protocol.as_deref(),
919            ),
920            (
921                "reclassification_rule",
922                manifest.reclassification_rule.as_deref(),
923            ),
924        ] {
925            if value.is_none_or(str::is_empty) {
926                return Err(format!("blind_holdout schema v3 requires {field}").into());
927            }
928        }
929
930        let tasks = manifest
931            .corpora
932            .iter()
933            .flat_map(|corpus| &corpus.tasks)
934            .collect::<Vec<_>>();
935        if tasks.len() < 8 {
936            return Err("blind_holdout schema v3 requires at least eight tasks".into());
937        }
938        let mut languages = HashSet::new();
939        let mut task_shapes = HashSet::new();
940        let allowed_shapes = HashSet::from([
941            "configuration",
942            "cross_file_behavior",
943            "definition_discovery",
944            "framework_behavior",
945            "regression_test_discovery",
946        ]);
947        for task in tasks {
948            if task.languages.is_empty() || task.task_shapes.is_empty() {
949                return Err(format!("{} requires language and task-shape tags", task.id).into());
950            }
951            for language in &task.languages {
952                if language.trim().is_empty() {
953                    return Err(format!("{} has an empty language tag", task.id).into());
954                }
955                languages.insert(language.as_str());
956            }
957            for shape in &task.task_shapes {
958                if !allowed_shapes.contains(shape.as_str()) {
959                    return Err(format!("{} has unsupported task shape {shape}", task.id).into());
960                }
961                task_shapes.insert(shape.as_str());
962            }
963        }
964        if languages.len() < 6 {
965            return Err("blind_holdout schema v3 requires at least six languages".into());
966        }
967        if task_shapes.len() < 4 {
968            return Err("blind_holdout schema v3 requires at least four task shapes".into());
969        }
970    }
971    Ok(())
972}

974fn load_concept_labels(
975    path: &Path,
976    source_manifest_path: &Path,
977    manifest: &Manifest,
978    manifest_blake3: &str,
979) -> Result<LoadedConceptLabels, Box<dyn Error>> {
980    let json = fs::read_to_string(path)?;
981    let blake3 = blake3::hash(json.as_bytes()).to_hex().to_string();
982    let labels: ConceptLabelManifest = serde_json::from_str(&json)?;
983    if labels.schema_version != 1 {
984        return Err(format!(
985            "unsupported concept-label schema version {}",
986            labels.schema_version
987        )
988        .into());
989    }
990    if labels.source_manifest.trim().is_empty()
991        || labels
992            .source_manifest
993            .rsplit('/')
994            .next()
995            .is_none_or(|name| {
996                Some(name) != source_manifest_path.file_name().and_then(|v| v.to_str())
997            })
998    {
999        return Err("concept labels name a different source manifest".into());
1000    }
1001    if labels.source_manifest_blake3 != manifest_blake3 {
1002        return Err("concept labels do not match the source manifest BLAKE3".into());
1003    }
1004    if labels.dataset_kind != manifest.dataset_kind {
1005        return Err("concept labels do not match the source dataset kind".into());
1006    }
1007    if labels.frozen_at.trim().is_empty() || labels.methodology.trim().is_empty() {
1008        return Err("concept labels require frozen_at and methodology".into());
1009    }
1010    for (name, value) in [
1011        (
1012            "minimum_candidate_concept_recall",
1013            labels.thresholds.minimum_candidate_concept_recall,
1014        ),
1015        (
1016            "minimum_selected_concept_recall",
1017            labels.thresholds.minimum_selected_concept_recall,
1018        ),
1019        (
1020            "minimum_selection_retention",
1021            labels.thresholds.minimum_selection_retention,
1022        ),
1023        (
1024            "minimum_task_selected_concept_recall",
1025            labels.thresholds.minimum_task_selected_concept_recall,
1026        ),
1027    ] {
1028        if !value.is_finite() || !(0.0..=1.0).contains(&value) {
1029            return Err(format!("concept threshold {name} must be between zero and one").into());
1030        }
1031    }
1032
1033    let mut source_tasks = BTreeMap::new();
1034    for task in manifest.corpora.iter().flat_map(|corpus| &corpus.tasks) {
1035        if source_tasks.insert(task.id.as_str(), task).is_some() {
1036            return Err(format!("source manifest repeats task id {}", task.id).into());
1037        }
1038    }
1039    let mut tasks = BTreeMap::new();
1040    for task_labels in labels.tasks {
1041        let Some(source_task) = source_tasks.get(task_labels.id.as_str()) else {
1042            return Err(format!("concept labels contain unknown task {}", task_labels.id).into());
1043        };
1044        if task_labels.concepts.is_empty() {
1045            return Err(format!("concept task {} has no concepts", task_labels.id).into());
1046        }
1047
1048        let source_anchors = source_task
1049            .relevant_files
1050            .iter()
1051            .flat_map(|file| {
1052                file.line_anchors
1053                    .iter()
1054                    .map(|line| (file.path.clone(), *line))
1055            })
1056            .collect::<BTreeSet<_>>();
1057        if source_anchors.is_empty() {
1058            return Err(format!(
1059                "concept task {} requires source line-anchor labels",
1060                task_labels.id
1061            )
1062            .into());
1063        }
1064        let mut concept_ids = BTreeSet::new();
1065        let mut labeled_anchors = BTreeSet::new();
1066        for concept in &task_labels.concepts {
1067            if concept.id.trim().is_empty()
1068                || concept.description.trim().is_empty()
1069                || concept.evidence.is_empty()
1070            {
1071                return Err(format!(
1072                    "concepts for task {} require an id, description, and evidence",
1073                    task_labels.id
1074                )
1075                .into());
1076            }
1077            if !concept_ids.insert(concept.id.as_str()) {
1078                return Err(
1079                    format!("task {} repeats concept id {}", task_labels.id, concept.id).into(),
1080                );
1081            }
1082            for evidence in &concept.evidence {
1083                validate_benchmark_path(&evidence.path)?;
1084                if evidence.line_anchors.is_empty() {
1085                    return Err(format!(
1086                        "concept {} in task {} has no line anchors",
1087                        concept.id, task_labels.id
1088                    )
1089                    .into());
1090                }
1091                for &line in &evidence.line_anchors {
1092                    let anchor = (evidence.path.clone(), line);
1093                    if line == 0 || !source_anchors.contains(&anchor) {
1094                        return Err(format!(
1095                            "concept {} in task {} uses an anchor absent from the source manifest",
1096                            concept.id, task_labels.id
1097                        )
1098                        .into());
1099                    }
1100                    if !labeled_anchors.insert(anchor) {
1101                        return Err(format!(
1102                            "task {} assigns one source anchor to multiple concepts",
1103                            task_labels.id
1104                        )
1105                        .into());
1106                    }
1107                }
1108            }
1109        }
1110        if labeled_anchors != source_anchors {
1111            return Err(format!(
1112                "task {} concept labels must partition every source-manifest anchor exactly once",
1113                task_labels.id
1114            )
1115            .into());
1116        }
1117        let task_id = task_labels.id.clone();
1118        if tasks.insert(task_id.clone(), task_labels).is_some() {
1119            return Err(format!("concept labels repeat task {task_id}").into());
1120        }
1121    }
1122    if tasks.len() != source_tasks.len()
1123        || source_tasks
1124            .keys()
1125            .any(|task_id| !tasks.contains_key(*task_id))
1126    {
1127        return Err("concept labels must cover every source-manifest task exactly once".into());
1128    }
1129
1130    Ok(LoadedConceptLabels {
1131        schema_version: labels.schema_version,
1132        blake3,
1133        source_manifest: labels.source_manifest,
1134        source_manifest_blake3: labels.source_manifest_blake3,
1135        dataset_kind: labels.dataset_kind,
1136        frozen_at: labels.frozen_at,
1137        methodology: labels.methodology,
1138        thresholds: labels.thresholds,
1139        tasks,
1140    })
1141}

1143fn evaluate_concept_coverage(
1144    labels: &ConceptTaskLabels,
1145    candidates: &[ContextCandidateEvaluation],
1146    selected: &[ContextFragment],
1147) -> Result<TaskConceptCoverage, Box<dyn Error>> {
1148    let mut evidence = Vec::with_capacity(labels.concepts.len());
1149    for concept in &labels.concepts {
1150        let anchors = concept
1151            .evidence
1152            .iter()
1153            .flat_map(|item| {
1154                item.line_anchors.iter().map(|line| MatchedAnchor {
1155                    path: item.path.clone(),
1156                    line: *line,
1157                })
1158            })
1159            .collect::<BTreeSet<_>>();
1160        let candidate_anchors_found = anchors
1161            .iter()
1162            .filter(|anchor| {
1163                candidates.iter().any(|candidate| {
1164                    candidate.path == anchor.path
1165                        && candidate.start_line <= anchor.line
1166                        && candidate.end_line >= anchor.line
1167                })
1168            })
1169            .cloned()
1170            .collect::<Vec<_>>();
1171        let selected_anchors_found = anchors
1172            .iter()
1173            .filter(|anchor| {
1174                selected.iter().any(|fragment| {
1175                    fragment.path == anchor.path
1176                        && fragment.start_line <= anchor.line
1177                        && fragment.end_line >= anchor.line
1178                })
1179            })
1180            .cloned()
1181            .collect::<Vec<_>>();
1182        let candidate_covered = !candidate_anchors_found.is_empty();
1183        let selected_covered = !selected_anchors_found.is_empty();
1184        if selected_covered && !candidate_covered {
1185            return Err(format!(
1186                "selected concept {} for task {} was absent from candidate diagnostics",
1187                concept.id, labels.id
1188            )
1189            .into());
1190        }
1191        evidence.push(ConceptEvidenceCoverage {
1192            id: concept.id.clone(),
1193            description: concept.description.clone(),
1194            candidate_covered,
1195            selected_covered,
1196            candidate_anchors_found,
1197            selected_anchors_found,
1198        });
1199    }
1200    evidence.sort_by(|left, right| left.id.cmp(&right.id));
1201    let candidate_concepts_found = evidence
1202        .iter()
1203        .filter(|concept| concept.candidate_covered)
1204        .count();
1205    let selected_concepts_found = evidence
1206        .iter()
1207        .filter(|concept| concept.selected_covered)
1208        .count();
1209    let concepts = evidence.len();
1210    Ok(TaskConceptCoverage {
1211        concepts,
1212        candidate_concepts_found,
1213        candidate_concept_recall: ratio(candidate_concepts_found, concepts),
1214        selected_concepts_found,
1215        selected_concept_recall: ratio(selected_concepts_found, concepts),
1216        selection_retention: optional_ratio(selected_concepts_found, candidate_concepts_found),
1217        evidence,
1218    })
1219}

1221fn concept_coverage_decision(
1222    labels: &LoadedConceptLabels,
1223    aggregate: &AggregateReport,
1224    corpora: &[CorpusReport],
1225) -> ConceptCoverageDecision {
1226    let mut failures = Vec::new();
1227    let candidate_recall = aggregate.candidate_concept_recall.unwrap_or(0.0);
1228    let selected_recall = aggregate.selected_concept_recall.unwrap_or(0.0);
1229    let selection_retention = aggregate.concept_selection_retention.unwrap_or(0.0);
1230    for (name, actual, minimum) in [
1231        (
1232            "candidate concept recall",
1233            candidate_recall,
1234            labels.thresholds.minimum_candidate_concept_recall,
1235        ),
1236        (
1237            "selected concept recall",
1238            selected_recall,
1239            labels.thresholds.minimum_selected_concept_recall,
1240        ),
1241        (
1242            "concept selection retention",
1243            selection_retention,
1244            labels.thresholds.minimum_selection_retention,
1245        ),
1246    ] {
1247        if actual < minimum {
1248            failures.push(format!("{name} {actual:.4} is below minimum {minimum:.4}"));
1249        }
1250    }
1251    for task in corpora.iter().flat_map(|corpus| &corpus.tasks) {
1252        let Some(coverage) = &task.concept_coverage else {
1253            failures.push(format!("task {} has no concept coverage", task.id));
1254            continue;
1255        };
1256        if coverage.selected_concept_recall < labels.thresholds.minimum_task_selected_concept_recall
1257        {
1258            failures.push(format!(
1259                "task {} selected concept recall {:.4} is below minimum {:.4}",
1260                task.id,
1261                coverage.selected_concept_recall,
1262                labels.thresholds.minimum_task_selected_concept_recall
1263            ));
1264        }
1265    }
1266    ConceptCoverageDecision {
1267        labels_schema_version: labels.schema_version,
1268        labels_blake3: labels.blake3.clone(),
1269        source_manifest: labels.source_manifest.clone(),
1270        source_manifest_blake3: labels.source_manifest_blake3.clone(),
1271        dataset_kind: labels.dataset_kind.clone(),
1272        frozen_at: labels.frozen_at.clone(),
1273        methodology: labels.methodology.clone(),
1274        thresholds: labels.thresholds,
1275        passed: failures.is_empty(),
1276        failures,
1277    }
1278}

1280fn benchmark_limitations(
1281    dataset_kind: &str,
1282    consumed_diagnostic: bool,
1283    concept_labels: bool,
1284    history_lane: bool,
1285    ast_structural_lane: bool,
1286    ast_structural_lane_v2: bool,

1379fn workflow_evidence_from_json_prompt(prompt: &str) -> Result<WorkflowEvidence, Box<dyn Error>> {
1380    let query: serde_json::Value = serde_json::from_str(prompt)?;
1381    let object = query
1382        .as_object()
1383        .ok_or("workflow-evidence prompt must be a JSON object")?;
1384    let failure_trace = object
1385        .get("failure_excerpt")
1386        .and_then(serde_json::Value::as_str)
1387        .ok_or("workflow-evidence prompt has no failure_excerpt")?;
1388    let failure_trace = utf8_tail(failure_trace, 8 * 1024);
1389    let mut test_intents = Vec::new();
1390    let mut seen_test_intents = HashSet::new();
1391    for line in failure_trace.lines() {
1392        let trimmed = line.trim();
1393        if let Some(failed) = trimmed.strip_prefix("FAILED ") {
1394            retain_observed_value(
1395                &mut test_intents,
1396                &mut seen_test_intents,
1397                failed.trim().to_owned(),
1398            );
1399        } else if let Some(test_name) = trimmed.strip_prefix("def test_") {
1400            let suffix = test_name
1401                .split(|character: char| !character.is_alphanumeric() && character != '_')
1402                .next()
1403                .unwrap_or_default();
1404            if !suffix.is_empty() {
1405                retain_observed_value(
1406                    &mut test_intents,
1407                    &mut seen_test_intents,
1408                    format!("test_{suffix}"),
1409                );
1410            }
1411        }
1412    }
1413    if let Some(command) = object.get("command").and_then(serde_json::Value::as_str)
1414        && !command.trim().is_empty()
1415    {
1416        retain_observed_value(
1417            &mut test_intents,
1418            &mut seen_test_intents,
1419            command.trim().to_owned(),
1420        );
1421    }
1422    Ok(WorkflowEvidence::new()
1423        .with_failure_traces([failure_trace.clone()])
1424        .with_symbols(observed_trace_symbols(&failure_trace))
1425        .with_paths(observed_trace_paths(&failure_trace))

1427}

1429fn utf8_tail(value: &str, max_bytes: usize) -> String {
1430    if value.len() <= max_bytes {
1431        return value.to_owned();
1432    }
1433    let mut start = value.len() - max_bytes;
1434    while !value.is_char_boundary(start) {
1435        start += 1;
1436    }
1437    value[start..].to_owned()
1438}

1440fn observed_trace_paths(trace: &str) -> Vec<String> {
1441    let mut paths = BTreeSet::new();
1442    for token in trace.split_whitespace() {
1443        let token = token.trim_matches(|character: char| {
1444            !character.is_alphanumeric() && !matches!(character, '/' | '.' | '_' | '-' | ':' | '\\')
1445        });
1446        let path = token.split_once("::").map_or(token, |(path, _)| path);
1447        let path = path
1448            .split_once(':')
1449            .filter(|(_, suffix)| {
1450                suffix
1451                    .chars()
1452                    .next()
1453                    .is_some_and(|character| character.is_ascii_digit())
1454            })
1455            .map_or(path, |(path, _)| path)
1456            .trim_start_matches("./");
1457        if path.contains('/') && path.contains('.') && validate_benchmark_path(path).is_ok() {
1458            paths.insert(path.to_owned());
1459        }
1460    }
1461    paths.into_iter().take(8).collect()
1462}

1464fn observed_trace_symbols(trace: &str) -> Vec<String> {
1465    let mut symbols = Vec::new();
1466    let mut seen = HashSet::new();
1467    for (index, value) in trace.split('`').enumerate() {
1468        if index % 2 == 1 {
1469            retain_trace_symbol(&mut symbols, &mut seen, value);
1470        }
1471    }
1472    for token in trace.split_whitespace() {
1473        let token = token.trim_matches(|character: char| {
1474            !character.is_alphanumeric() && !matches!(character, '_' | '.' | ':')
1475        });
1476        retain_trace_symbol(&mut symbols, &mut seen, token);
1477    }
1478    symbols.into_iter().take(8).collect()
1479}

1481fn retain_trace_symbol(symbols: &mut Vec<String>, seen: &mut HashSet<String>, value: &str) {
1482    if (3..=128).contains(&value.len())
1483        && !value.contains('/')
1484        && !value.chars().any(char::is_whitespace)
1485        && value.chars().any(char::is_alphabetic)
1486        && (value.contains('_')
1487            || value.contains('.')
1488            || value.contains("::")
1489            || value.chars().any(char::is_uppercase))
1490        && seen.insert(value.to_owned())
1491    {
1492        symbols.push(value.to_owned());
1493    }
1494}

1599async fn discover_ast_structural_lane(
1600    services: &Services,
1601    languages: &[String],
1602    failure_traces: &[String],
1603    orientation_capsule_enabled: bool,
1604    version: AstStructuralLaneVersion,
1605) -> Result<
1606    (
1607        AstStructuralLaneReport,
1608        OrientationCapsuleReport,
1609        Option<AstOwnerEvidence>,
1610    ),
1611    Box<dyn Error>,
1612> {
1613    let trace = utf8_tail(&failure_traces.join("\n"), AST_LANE_MAX_TRACE_BYTES);
1614    let mut report = AstStructuralLaneReport {
1615        enabled: true,
1616        version: version.number(),
1617        trace_bytes_examined: trace.len(),
1618        ..AstStructuralLaneReport::default()
1619    };
1620    if trace.is_empty() {
1621        let capsule = if orientation_capsule_enabled {
1622            OrientationCapsuleReport {
1623                enabled: true,
1624                unavailable_reason: Some("no_failure_trace"),
1625                ..OrientationCapsuleReport::default()
1626            }
1627        } else {
1628            OrientationCapsuleReport::default()
1629        };
1630        return Ok((report, capsule, None));
1631    }
1632
1633    let terms = collect_ast_structural_terms(languages, &trace, version, &mut report)?;
1634    report.terms = terms.structural.clone();
1635    report.owner_terms = terms.owners.clone();
1636    report.named_argument_terms = terms.named_arguments.clone();
1637
1638    let mut paths =
1639        search_ast_structural_definitions(services, &terms.structural, &mut report).await?;
1640    if version == AstStructuralLaneVersion::V2 {
1641        corroborate_ast_paths(
1642            services,
1643            &mut paths,
1644            &terms.owners,
1645            &terms.named_arguments,
1646            &mut report,
1647        )
1648        .await?;
1649    }
1650
1651    let ranked = rank_ast_structural_paths(paths, &report.languages_attempted, version);
1652    let capsule = build_orientation_capsule(&ranked, orientation_capsule_enabled)?;
1653    report.candidate_paths = ranked
1654        .iter()
1655        .map(|(path, _)| path)
1656        .take(AST_LANE_MAX_PATHS)
1657        .cloned()
1658        .collect();
1659    let owner_evidence = if version == AstStructuralLaneVersion::V2 {
1660        build_ast_owner_evidence(&ranked)?
1661    } else {
1662        None
1663    };
1664    report.owner_reservation = owner_evidence
1665        .as_ref()
1666        .map(|evidence| evidence.report.clone());
1667    Ok((report, capsule, owner_evidence))
1668}

1670fn collect_ast_structural_terms(
1671    languages: &[String],
1672    trace: &str,
1673    version: AstStructuralLaneVersion,
1674    report: &mut AstStructuralLaneReport,
1675) -> Result<AstQueryTerms, Box<dyn Error>> {
1676    let mut terms = Vec::new();
1677    let mut seen_terms = HashSet::new();
1678    let mut owner_terms = Vec::new();
1679    let mut seen_owner_terms = HashSet::new();
1680    let mut named_argument_terms = Vec::new();
1681    let mut seen_named_argument_terms = HashSet::new();
1682    for language in languages
1683        .iter()
1684        .map(|language| language.to_ascii_lowercase())
1685        .collect::<BTreeSet<_>>()
1686        .into_iter()
1687        .take(AST_LANE_MAX_LANGUAGES)
1688    {
1689        let parsed = parser::parse_language(&language, trace)?;
1690        report.languages_attempted.push(language.clone());
1691        report.structurally_complete_languages += usize::from(parsed.structurally_complete);
1692        for reference in parsed.references {
1693            if let Some(term) = normalize_ast_search_term(&reference.name) {
1694                retain_bounded_term(&mut terms, &mut seen_terms, term, AST_LANE_MAX_TERMS);
1695                if terms.len() == AST_LANE_MAX_TERMS {
1696                    break;
1697                }
1698            }
1699        }
1700        if version == AstStructuralLaneVersion::V2 {
1701            retain_ast_v2_trace_signals(
1702                extract_ast_v2_trace_signals(&language, trace),
1703                &mut terms,
1704                &mut seen_terms,
1705                &mut owner_terms,
1706                &mut seen_owner_terms,
1707                &mut named_argument_terms,
1708                &mut seen_named_argument_terms,
1709            );
1710        }
1711        terms.truncate(AST_LANE_MAX_TERMS);
1712        if version == AstStructuralLaneVersion::V1 && terms.len() == AST_LANE_MAX_TERMS {
1713            break;
1714        }
1715    }
1716    Ok(AstQueryTerms {
1717        structural: terms,
1718        owners: owner_terms,
1719        named_arguments: named_argument_terms,
1720    })
1721}

1780fn record_ast_structural_hit(
1781    paths: &mut BTreeMap<String, AstPathStats>,
1782    term: &str,
1783    hit: SearchHit,
1784) {
1785    let stats = paths.entry(hit.path).or_default();
1786    stats.terms.insert(term.to_owned());
1787    let symbol = hit.symbol.unwrap_or_else(|| term.to_owned());
1788    stats.definitions.insert(symbol.clone());
1789    let owner_hit = AstOwnerHit {
1790        start_line: hit.start_line,
1791        end_line: hit.end_line,
1792        excerpt: hit.excerpt,
1793        symbol,
1794        matched_term: term.to_owned(),
1795        normalized_score: hit.normalized_score,
1796        corroborating_owner_terms: BTreeSet::new(),
1797        corroborating_named_argument_terms: BTreeSet::new(),
1798    };
1799    if ast_owner_hit_exact(&owner_hit) {
1800        retain_best_ast_owner_hit(stats, owner_hit);
1801    }
1802    stats.structural_hits += 1;
1803    stats.best_score = stats.best_score.max(hit.normalized_score);
1804}

1820async fn corroborate_ast_paths(
1821    services: &Services,
1822    paths: &mut BTreeMap<String, AstPathStats>,
1823    owner_terms: &[String],
1824    named_argument_terms: &[String],
1825    report: &mut AstStructuralLaneReport,
1826) -> Result<(), Box<dyn Error>> {
1827    for (term, role) in owner_terms
1828        .iter()
1829        .map(|term| (term, AstAuxiliaryTermRole::Owner))
1830        .chain(
1831            named_argument_terms
1832                .iter()
1833                .map(|term| (term, AstAuxiliaryTermRole::NamedArgument)),
1834        )
1835    {
1836        let response = services
1837            .search(ast_search_request(
1838                term,
1839                SearchMode::Auto,
1840                paths.keys().cloned().collect(),
1841                true,
1842            ))
1843            .await?;
1844        report.searches += 1;
1845        report.auxiliary_searches += 1;
1846        for hit in response.hits {
1847            let Some(stats) = paths.get_mut(&hit.path) else {
1848                continue;
1849            };
1850            if record_ast_corroborating_hit(stats, term, role, &hit) {
1851                report.corroborating_hits += 1;
1852            }
1853        }
1854    }
1855    corroborate_ast_owner_excerpts(paths, named_argument_terms);
1856    Ok(())
1857}

1859fn record_ast_corroborating_hit(
1860    stats: &mut AstPathStats,
1861    term: &str,
1862    role: AstAuxiliaryTermRole,
1863    hit: &SearchHit,
1864) -> bool {
1865    if hit.symbol.is_none()
1866        && hit.enclosing_symbol.is_none()
1867        && hit.match_kind != "symbol"
1868        && !hit.match_kinds.iter().any(|kind| kind == "symbol")
1869    {
1870        return false;
1871    }
1872    let mut owner_local = false;
1873    if let Some(corroborating_symbol) = hit.enclosing_symbol.as_deref().or(hit.symbol.as_deref()) {
1874        for owner_hit in &mut stats.owner_hits {
1875            let range_cooccurs =
1876                owner_hit.start_line <= hit.start_line && owner_hit.end_line >= hit.end_line;
1877            if range_cooccurs || ast_symbols_cooccur(&owner_hit.symbol, corroborating_symbol) {
1878                record_ast_owner_corroboration(owner_hit, term, role);
1879                owner_local = true;
1880            }
1881        }
1882    }
1883    if !owner_local {
1884        return false;
1885    }
1886    match role {
1887        AstAuxiliaryTermRole::Owner => {
1888            stats.owner_terms.insert(term.to_owned());
1889        }
1890        AstAuxiliaryTermRole::NamedArgument => {
1891            stats.named_argument_terms.insert(term.to_owned());
1892        }
1893    }
1894    stats.corroborating_hits += 1;
1895    true
1896}

1898fn record_ast_owner_corroboration(
1899    owner_hit: &mut AstOwnerHit,
1900    term: &str,
1901    role: AstAuxiliaryTermRole,
1902) {
1903    match role {
1904        AstAuxiliaryTermRole::Owner => {
1905            owner_hit.corroborating_owner_terms.insert(term.to_owned());
1906        }
1907        AstAuxiliaryTermRole::NamedArgument => {
1908            owner_hit

1959fn rank_ast_structural_paths(
1960    paths: BTreeMap<String, AstPathStats>,
1961    languages: &[String],
1962    version: AstStructuralLaneVersion,
1963) -> Vec<(String, AstPathStats)> {

1966    ranked.sort_by(|(left_path, left), (right_path, right)| {
1967        let v2_order = || {
1968            ast_path_matches_source_extensions(right_path, &source_extensions)
1969                .cmp(&ast_path_matches_source_extensions(
1970                    left_path,
1971                    &source_extensions,
1972                ))
1973                .then_with(|| {
1974                    right
1975                        .named_argument_terms
1976                        .len()
1977                        .cmp(&left.named_argument_terms.len())
1978                })
1979                .then_with(|| right.owner_terms.len().cmp(&left.owner_terms.len()))

1984            std::cmp::Ordering::Equal

1989            .then_with(|| right.best_score.total_cmp(&left.best_score))

2208                    AST_LANE_V2_MAX_NAMED_ARGUMENT_TERMS,
2209                );
2210            }
2211            index = index.saturating_add(1);
2212            continue;
2213        }
2214        if !is_ascii_identifier_start(bytes[index]) {
2215            index += 1;
2216            continue;
2217        }
2218        let start = index;
2219        index += 1;

2221            index += 1;
2222        }

2235                && bytes

2240                let mut member_end = member_start + 1;

2242                    member_end += 1;
2243                }
2244                if let Some(member) = normalize_ast_v2_member_term(&line[member_start..member_end])
2245                {
2246                    if let Some(owner) = normalize_ast_auxiliary_term(identifier) {

2249                            seen_owners,

2251                            AST_LANE_V2_MAX_OWNER_TERMS,

2255                        &mut signals.member_terms,
2256                        seen_members,
2257                        member,
2258                        AST_LANE_MAX_TERMS,
2259                    );
2260                }
2261            }
2262        }
2263        let is_named_equals = supports_named_equals
2264            && previous.is_some_and(|byte| matches!(byte, b'(' | b','))
2265            && next.is_some_and(|(byte, offset)| {
2266                byte == b'=' && bytes.get(offset + 1).copied() != Some(b'=')
2267            });
2268        let is_named_field = supports_named_fields
2269            && previous.is_some_and(|byte| matches!(byte, b'{' | b','))
2270            && next.is_some_and(|(byte, offset)| {
2271                byte == b':' && bytes.get(offset + 1).copied() != Some(b':')

3200                return Err(format!("{} has an invalid token budget", task.id).into());

3206                return Err(format!("{} has no relevance labels", task.id).into());
3207            }
3208            let mut relevant_paths = HashSet::new();
3209            for file in &task.relevant_files {
3210                if !relevant_paths.insert(file.path.as_str()) {

3212                }

3224                    )

3227            }
3228        }
3229    }

3386#[derive(Debug, PartialEq, Eq)]
3387struct CompleteResponseAccounting {
3388    protocol_tokens: usize,
3389    path_and_metadata_tokens: usize,
3390    total_response_tokens: usize,
3391}
3392
3393fn calculate_complete_response_accounting(
3394    response: &ContextResponse,
3395) -> Result<CompleteResponseAccounting, Box<dyn Error>> {
3396    let tokenizer = tokens::Tokenizer::from_str(&response.meta.tokenizer, true)
3397        .map_err(|error| format!("invalid response tokenizer: {error}"))?;
3398    let payload = serde_json::to_string(response)?;
3399    let total_response_tokens = tokens::count_with(&payload, tokenizer);
3400    let mut protocol_skeleton = serde_json::to_value(response)?;
3401    strip_complete_response_values(&mut protocol_skeleton);
3402    let available_overhead = total_response_tokens.saturating_sub(response.meta.source_tokens);
3403    let protocol_tokens =
3404        tokens::count_with(&protocol_skeleton.to_string(), tokenizer).min(available_overhead);
3405    Ok(CompleteResponseAccounting {
3406        protocol_tokens,
3407        path_and_metadata_tokens: available_overhead.saturating_sub(protocol_tokens),
3408        total_response_tokens,
3409    })
3410}

3412fn strip_complete_response_values(value: &mut serde_json::Value) {
3413    match value {
3414        serde_json::Value::Null => {}
3415        serde_json::Value::Bool(value) => *value = false,
3416        serde_json::Value::Number(value) => *value = 0.into(),
3417        serde_json::Value::String(value) => value.clear(),
3418        serde_json::Value::Array(values) => values.clear(),
3419        serde_json::Value::Object(values) => {
3420            for value in values.values_mut() {
3421                strip_complete_response_values(value);
3422            }
3423        }
3424    }
3425}

3427fn finalize_complete_response_accounting(
3428    response: &mut ContextResponse,
3429) -> Result<(), Box<dyn Error>> {
3430    response.meta.protocol_tokens = 0;
3431    response.meta.path_and_metadata_tokens = 0;
3432    response.meta.total_response_tokens = 0;
3433    for _ in 0..32 {
3434        let expected = calculate_complete_response_accounting(response)?;
3435        let actual = CompleteResponseAccounting {
3436            protocol_tokens: response.meta.protocol_tokens,
3437            path_and_metadata_tokens: response.meta.path_and_metadata_tokens,
3438            total_response_tokens: response.meta.total_response_tokens,
3439        };
3440        if actual == expected {
3441            return Ok(());
3442        }
3443        response.meta.protocol_tokens = expected.protocol_tokens;
3444        response.meta.path_and_metadata_tokens = expected.path_and_metadata_tokens;
3445        response.meta.total_response_tokens = expected.total_response_tokens;
3446    }
3447    Err("complete response token accounting did not reach a fixed point".into())
3448}

3454        path_and_metadata_tokens: response.meta.path_and_metadata_tokens,

3458        return Err(format!(
3459            "complete response token accounting mismatch: declared={actual:?}, counted={expected:?}"

3462    }

3544fn repeated_range_token_estimate(
3545    start_line: usize,
3546    end_line: usize,
3547    token_count: usize,
3548    prior_ranges: &[(usize, usize)],
3549) -> usize {
3550    if end_line < start_line || token_count == 0 {
3551        return 0;
3552    }
3553    let line_count = end_line - start_line + 1;
3554    let mut repeated = vec![false; line_count];

3556        let overlap_start = start_line.max(prior_start);
3557        let overlap_end = end_line.min(prior_end);
3558        if overlap_start > overlap_end {

3560        }

3562            repeated[line - start_line] = true;
3563        }
3564    }
3565    let repeated_lines = repeated.into_iter().filter(|value| *value).count();
3566    token_count
3567        .saturating_mul(repeated_lines)
3568        .div_ceil(line_count)
3569}
