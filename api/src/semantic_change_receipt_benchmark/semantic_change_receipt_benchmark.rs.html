79#[tokio::main]
80async fn main() -> Result<(), Box<dyn Error>> {
81    let args = Args::parse();
82    if args.iterations == 0 {
83        return Err("--iterations must be greater than zero".into());
84    }
85
86    let repository = PathBuf::from(env!("LEANTOKEN_REPOSITORY_ROOT"));
87    let harness_revision = git_output(&repository, &["rev-parse", "HEAD"]);
88    let harness_tracked_worktree_dirty = git_output(
89        &repository,
90        &["status", "--porcelain=v1", "--untracked-files=no"],
91    )
92    .map(|status| !status.trim().is_empty());
93    let fixture = Fixture::new().await?;
94    let mut latencies = Vec::with_capacity(args.iterations);
95    let mut final_response = None;
96    for _ in 0..args.iterations {
97        let started = Instant::now();
98        let response = fixture.review().await?;
99        latencies.push(started.elapsed().as_secs_f64() * 1_000.0);
100        final_response = Some(response);
101    }
102    let response = final_response.expect("positive iteration count");
103    let semantic = response
104        .diff_scope
105        .as_ref()
106        .and_then(|scope| scope.evidence.as_ref())
107        .and_then(|evidence| evidence.semantic_change.as_ref())
108        .ok_or("semantic receipt missing")?;
109
110    let expected = expected_truth();
111    let returned = returned_truth(&semantic.symbol_changes, &semantic.configuration_changes);
112    let true_positives = expected.intersection(&returned).count();
113    let false_positives = returned.difference(&expected).count();
114    let false_negatives = expected.difference(&returned).count();
115    let precision = ratio(true_positives, true_positives + false_positives);
116    let recall = ratio(true_positives, true_positives + false_negatives);
117    let classification = Classification {
118        expected_items: expected.len(),
119        returned_items: returned.len(),
120        true_positives,
121        false_positives,
122        false_negatives,
123        precision,
124        recall,
125    };
126
127    let serialized_semantic = serde_json::to_string(semantic)?;
128    let no_configuration_values_exposed = !serialized_semantic.contains("base-secret-value")
129        && !serialized_semantic.contains("head-secret-value");
130    let owner_test_statuses_correct = semantic.owner_tests.iter().any(|coverage| {
131        coverage.changed_path == "src/lib.rs"
132            && coverage.status == DiffOwnerTestStatus::Found
133            && coverage.paths == ["tests/lib_test.rs"]
134    }) && semantic.owner_tests.iter().any(|coverage| {
135        coverage.changed_path == "package.json" && coverage.status == DiffOwnerTestStatus::Missing
136    });
137    let tokenizer = Tokenizer::default();
138    let response_tokens_with_receipt = serialized_tokens(&tokenizer, &response)?;
139    let mut without_receipt = response.clone();
140    if let Some(evidence) = without_receipt
141        .diff_scope
142        .as_mut()
143        .and_then(|scope| scope.evidence.as_mut())
144    {
145        evidence.semantic_change = None;
146    }
147    let response_tokens_without_receipt = serialized_tokens(&tokenizer, &without_receipt)?;
148    let receipt_overhead_tokens =
149        response_tokens_with_receipt.saturating_sub(response_tokens_without_receipt);
150    let payload = Payload {
151        response_tokens_with_receipt,
152        response_tokens_without_receipt,
153        receipt_overhead_tokens,
154        receipt_overhead_fraction: ratio(receipt_overhead_tokens, response_tokens_without_receipt),
155    };
156    latencies.sort_by(f64::total_cmp);
157    let latency = Latency {
158        iterations: latencies.len(),
159        min: latencies[0],
160        p50: percentile(&latencies, 0.50),
161        p95: percentile(&latencies, 0.95),
162        max: latencies[latencies.len() - 1],
163    };
164    let gates = Gates {
165        exact_truth_set: false_positives == 0 && false_negatives == 0,
166        no_configuration_values_exposed,
167        owner_test_statuses_correct,
168        bounded_payload_overhead: receipt_overhead_tokens <= 1_000,
169        bounded_p95_latency: latency.p95 <= 2_000.0,
170    };
171    let adopted = gates.exact_truth_set
172        && gates.no_configuration_values_exposed
173        && gates.owner_test_statuses_correct
174        && gates.bounded_payload_overhead
175        && gates.bounded_p95_latency;
176    let report = Report {
177        schema_version: 1,
178        leantoken_version: env!("CARGO_PKG_VERSION"),
179        tokenizer: tokenizer.name(),
180        harness_revision,
181        harness_tracked_worktree_dirty,
182        decision: if adopted { "adopt" } else { "reject" },
183        gates,
184        classification,
185        payload,
186        latency_ms: latency,
187        limitations: vec![
188            "This deterministic fixture validates receipt classification, not downstream model task success.",
189            "Rename classification requires a unique normalized body fingerprint and intentionally under-classifies ambiguous cases.",
190            "Configuration classification is limited to recognized JSON configuration filenames and emits key paths without values.",
191            "Owner-test discovery remains a filename heuristic and reports unknown when its bounded scan truncates.",
192        ],
193    };
194    let json = serde_json::to_string_pretty(&report)?;
195    if let Some(parent) = args
196        .output
197        .parent()
198        .filter(|parent| !parent.as_os_str().is_empty())
199    {
200        fs::create_dir_all(parent)?;
201    }
202    fs::write(&args.output, format!("{json}\n"))?;
203    println!("{json}");
204    if !adopted {
205        return Err("semantic change receipt adoption gates failed".into());
206    }
207    Ok(())
208}

324fn returned_truth(
325    symbols: &[DiffSymbolChange],
326    configuration: &[leantoken::DiffConfigurationChange],
327) -> BTreeSet<String> {
328    let mut returned = symbols.iter().map(symbol_truth).collect::<BTreeSet<_>>();
329    returned.extend(configuration.iter().map(|change| {
330        format!(
331            "config:{}:{}",
332            match change.kind {
333                DiffConfigurationChangeKind::Added => "added",
334                DiffConfigurationChangeKind::Removed => "removed",
335                DiffConfigurationChangeKind::Modified => "modified",
336            },
337            change.key_path
338        )
339    }));
340    returned
341}
342
343fn symbol_truth(change: &DiffSymbolChange) -> String {
344    let before = change
345        .before
346        .as_ref()
347        .map(|symbol| symbol.name.as_str())
348        .unwrap_or("");
349    let after = change
350        .after
351        .as_ref()
352        .map(|symbol| symbol.name.as_str())
353        .unwrap_or("");
354    match change.kind {
355        DiffSymbolChangeKind::Added => format!("symbol:added:{after}"),
356        DiffSymbolChangeKind::Removed => format!("symbol:removed:{before}"),
357        DiffSymbolChangeKind::Renamed => format!("symbol:renamed:{before}->{after}"),
358        DiffSymbolChangeKind::Modified => format!(
359            "symbol:modified:{after}:{}",
360            match change.modification {
361                Some(DiffSymbolModification::SignatureChanged) => "signature_changed",
362                Some(DiffSymbolModification::BodyOnly) => "body_only",
363                None => "unknown",
364            }
365        ),
366    }
367}

369fn serialized_tokens(
370    tokenizer: &Tokenizer,
371    response: &ContextResponse,

376fn ratio(numerator: usize, denominator: usize) -> f64 {
377    if denominator == 0 {
378        1.0
379    } else {
380        numerator as f64 / denominator as f64
381    }
382}
383
384fn percentile(sorted: &[f64], percentile: f64) -> f64 {
385    let rank = (percentile * sorted.len() as f64).ceil() as usize;
386    sorted[rank.saturating_sub(1).min(sorted.len() - 1)]
387}

389fn init_git(root: &Path) -> Result<(), Box<dyn Error>> {
390    run_git(root, &["init"])?;
391    run_git(root, &["config", "user.email", "benchmark@example.com"])?;
392    run_git(root, &["config", "user.name", "Benchmark"])?;
393    run_git(root, &["add", "."])?;
394    run_git(root, &["commit", "-m", "base"])?;
395    Ok(())
396}

398fn run_git(root: &Path, args: &[&str]) -> Result<(), Box<dyn Error>> {
399    let output = Command::new("git").args(args).current_dir(root).output()?;
400    if !output.status.success() {
401        return Err(format!(
402            "git {} failed: {}",
403            args.join(" "),
404            String::from_utf8_lossy(&output.stderr).trim()
405        )
406        .into());
407    }
408    Ok(())
409}

411fn required_git_output(root: &Path, args: &[&str]) -> Result<String, Box<dyn Error>> {
412    git_output(root, args).ok_or_else(|| format!("git {} failed", args.join(" ")).into())
413}

415fn git_output(root: &Path, args: &[&str]) -> Option<String> {
416    let output = Command::new("git")
417        .args(args)
418        .current_dir(root)
419        .output()
420        .ok()?;
421    output
422        .status
423        .success()
424        .then(|| String::from_utf8_lossy(&output.stdout).trim().to_owned())
