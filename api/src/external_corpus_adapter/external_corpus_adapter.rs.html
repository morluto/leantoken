275fn main() -> Result<(), Box<dyn Error>> {
276    let args = Args::parse();
277    let lock: CorpusLock = serde_json::from_slice(&fs::read(&args.lock)?)?;
278    validate_lock(&lock)?;
279
280    let (manifest, skipped) = match args.command {
281        AdapterCommand::Semble {
282            source,
283            output,
284            repository,
285            limit,
286            token_budget,
287        } => {
288            verify_source_revision(&source, &lock.semble.dataset_revision)?;
289            let manifest = convert_semble(
290                &source,
291                &lock,
292                &repository.into_iter().collect(),
293                limit,
294                token_budget,
295            )?;
296            write_manifest(&output, &manifest)?;
297            (manifest, 0)
298        }
299        AdapterCommand::Sverklo {
300            source,
301            output,
302            limit,
303            token_budget,
304        } => {
305            verify_source_revision(&source, &lock.sverklo.dataset_revision)?;
306            let (manifest, skipped) = convert_sverklo(&source, &lock, limit, token_budget)?;
307            write_manifest(&output, &manifest)?;
308            (manifest, skipped)
309        }
310        AdapterCommand::ArbTrace2code {
311            source,
312            output,
313            sample_id,
314            repository,
315            limit,
316            token_budget,
317        } => {
318            let arb = lock
319                .arb
320                .as_ref()
321                .ok_or("external corpus lock does not define ARB")?;
322            let (manifest, skipped) = convert_arb_trace2code(
323                &source,
324                &lock,
325                arb,
326                &sample_id.into_iter().collect(),
327                &repository.into_iter().collect(),
328                limit,
329                token_budget,
330            )?;
331            write_manifest(&output, &manifest)?;
332            (manifest, skipped)
333        }
334    };
335
336    eprintln!(
337        "converted {} corpora and {} tasks; skipped {skipped} unsupported tasks",
338        manifest.corpora.len(),
339        manifest
340            .corpora
341            .iter()
342            .map(|corpus| corpus.tasks.len())
343            .sum::<usize>()
344    );
345    Ok(())
346}

348fn validate_lock(lock: &CorpusLock) -> Result<(), Box<dyn Error>> {
349    if lock.schema_version != LOCK_SCHEMA_VERSION {
350        return Err(format!(
351            "unsupported external corpus lock schema {}",
352            lock.schema_version
353        )
354        .into());
355    }
356    if lock.frozen_at.trim().is_empty() {
357        return Err("external corpus lock requires frozen_at".into());
358    }
359    for (name, url, revision, license, prompt, labels) in [
360        (
361            "semble",
362            lock.semble.dataset_url.as_str(),
363            lock.semble.dataset_revision.as_str(),
364            lock.semble.dataset_license.as_str(),
365            lock.semble.prompt_provenance.as_str(),
366            lock.semble.label_provenance.as_str(),
367        ),
368        (
369            "sverklo",
370            lock.sverklo.dataset_url.as_str(),
371            lock.sverklo.dataset_revision.as_str(),
372            lock.sverklo.dataset_license.as_str(),
373            lock.sverklo.prompt_provenance.as_str(),
374            lock.sverklo.label_provenance.as_str(),
375        ),
376    ] {
377        if [url, license, prompt, labels]
378            .into_iter()
379            .any(str::is_empty)
380        {
381            return Err(format!("{name} corpus lock has an empty required field").into());
382        }
383        validate_revision(revision)?;
384    }
385    validate_repository(&lock.sverklo.repository)?;
386    if lock.sverklo.supported_categories.is_empty() {
387        return Err("sverklo corpus lock requires supported_categories".into());
388    }
389    for path in [
390        lock.semble.repositories_file.as_str(),
391        lock.semble.annotations_directory.as_str(),
392        lock.sverklo.tasks_file.as_str(),
393    ] {
394        validate_relative_path(path)?;
395    }
396    if let Some(arb) = &lock.arb {
397        if [
398            arb.dataset_url.as_str(),
399            arb.dataset_license.as_str(),
400            arb.release_id.as_str(),
401            arb.supported_task_type.as_str(),
402            arb.prompt_provenance.as_str(),
403            arb.label_provenance.as_str(),
404        ]
405        .into_iter()
406        .any(str::is_empty)
407        {
408            return Err("ARB corpus lock has an empty required field".into());
409        }
410        validate_revision(&arb.dataset_revision)?;
411        validate_hex_digest(&arb.release_sha256, 64, "ARB release SHA-256")?;
412        validate_hex_digest(&arb.samples_blake3, 64, "ARB samples BLAKE3")?;
413        validate_relative_path(&arb.samples_file)?;
414        if arb.repositories.is_empty() || arb.limitations.is_empty() {
415            return Err("ARB corpus lock requires repositories and limitations".into());
416        }
417        let mut repository_names = HashSet::new();
418        for repository in &arb.repositories {
419            if [
420                repository.name.as_str(),
421                repository.url.as_str(),
422                repository.language.as_str(),
423            ]
424            .into_iter()
425            .any(str::is_empty)
426            {
427                return Err("ARB repository lock has an empty required field".into());
428            }
429            if !repository_names.insert(repository.name.as_str()) {
430                return Err(format!("duplicate ARB repository {}", repository.name).into());
431            }
432        }
433    }
434    Ok(())
435}

437fn convert_semble(
438    source: &Path,
439    lock: &CorpusLock,
440    repositories: &HashSet<String>,
441    limit: Option<usize>,
442    token_budget: usize,
443) -> Result<Manifest, Box<dyn Error>> {
444    validate_budget(token_budget)?;
445    let repository_path = source.join(&lock.semble.repositories_file);
446    let repository_specs: Vec<SembleRepository> =
447        serde_json::from_slice(&fs::read(repository_path)?)?;
448    if !repositories.is_empty() {
449        let available = repository_specs
450            .iter()
451            .map(|repository| repository.name.as_str())
452            .collect::<HashSet<_>>();
453        let mut missing = repositories
454            .iter()
455            .filter(|name| !available.contains(name.as_str()))
456            .cloned()
457            .collect::<Vec<_>>();
458        missing.sort();
459        if !missing.is_empty() {
460            return Err(format!("unknown Semble repositories: {}", missing.join(", ")).into());
461        }
462    }
463    let annotations_root = source.join(&lock.semble.annotations_directory);
464    let mut corpora = Vec::new();
465    let mut remaining = limit.unwrap_or(usize::MAX);
466
467    for repository in repository_specs {
468        if !repositories.is_empty() && !repositories.contains(&repository.name) {
469            continue;
470        }
471        if remaining == 0 {
472            break;
473        }
474        validate_repository_fields(
475            &repository.name,
476            &repository.url,
477            &repository.name,
478            &repository.revision,
479        )?;
480        let annotation_path = annotations_root.join(format!("{}.json", repository.name));
481        let annotations: Vec<SembleAnnotation> =
482            serde_json::from_slice(&fs::read(&annotation_path)?)?;
483        let mut tasks = Vec::new();
484        for (index, annotation) in annotations.into_iter().enumerate() {
485            if remaining == 0 {
486                break;
487            }
488            let task = convert_semble_annotation(&repository, annotation, index, token_budget)?;
489            tasks.push(task);
490            remaining -= 1;
491        }
492        if tasks.is_empty() {
493            continue;
494        }
495        corpora.push(Corpus {
496            name: repository.name.clone(),
497            url: repository.url,
498            directory: repository.name,
499            base_revision: repository.revision,
500            fix_commit: None,
501            issue_url: None,
502            prompt_provenance: lock.semble.prompt_provenance.clone(),
503            label_provenance: lock.semble.label_provenance.clone(),
504            dataset_url: lock.semble.dataset_url.clone(),
505            dataset_revision: lock.semble.dataset_revision.clone(),
506            dataset_license: lock.semble.dataset_license.clone(),
507            external_limitations: lock.semble.limitations.clone(),
508            tasks,
509        });
510    }
511    if corpora.is_empty() {
512        return Err("Semble filters selected no annotated repositories".into());
513    }
514    if !repositories.is_empty() {
515        let found = corpora
516            .iter()
517            .map(|corpus| corpus.name.as_str())
518            .collect::<HashSet<_>>();
519        let mut missing = repositories
520            .iter()
521            .filter(|name| !found.contains(name.as_str()))
522            .cloned()
523            .collect::<Vec<_>>();
524        missing.sort();
525        if !missing.is_empty() {
526            return Err(format!(
527                "unknown or empty Semble repositories: {}",
528                missing.join(", ")
529            )
530            .into());
531        }
532    }
533    Ok(external_manifest(
534        &lock.frozen_at,
535        "semble",
536        "Pinned Semble public retrieval annotations converted without changing labels.",
537        corpora,
538    ))
539}

541fn convert_semble_annotation(
542    repository: &SembleRepository,
543    annotation: SembleAnnotation,
544    index: usize,
545    token_budget: usize,
546) -> Result<Task, Box<dyn Error>> {
547    if annotation.query.trim().is_empty() || annotation.category.trim().is_empty() {
548        return Err(format!("{} annotation {index} has an empty field", repository.name).into());
549    }
550    if annotation.relevant.is_empty() {
551        return Err(format!(
552            "{} annotation {index} has no primary labels",
553            repository.name
554        )
555        .into());
556    }
557    let _secondary_labels_are_intentionally_not_promoted = annotation.secondary.len();
558    let relevant_files = labels_to_relevant_files(annotation.relevant)?;
559    Ok(Task {
560        id: format!("semble:{}:{index:03}", repository.name),
561        prompt: annotation.query.clone(),
562        languages: vec![repository.language.clone()],
563        task_shapes: vec![format!("semble_{}", annotation.category)],
564        rg_queries: vec![annotation.query],
565        relevant_files,
566        token_budget,
567    })
568}

570fn labels_to_relevant_files(labels: Vec<SembleLabel>) -> Result<Vec<RelevantFile>, Box<dyn Error>> {
571    let mut by_path = BTreeMap::<String, Vec<usize>>::new();
572    for label in labels {
573        let (path, anchor) = match label {
574            SembleLabel::File(path) => (path, None),
575            SembleLabel::Region {
576                path,
577                start_line,
578                end_line,
579            } => {
580                if start_line == 0 || end_line < start_line {
581                    return Err(
582                        format!("invalid Semble region {path}:{start_line}-{end_line}").into(),
583                    );
584                }
585                (path, Some(start_line))
586            }
587        };
588        validate_relative_path(&path)?;
589        if let Some(anchor) = anchor {
590            by_path.entry(path).or_default().push(anchor);
591        } else {
592            by_path.entry(path).or_default();
593        }
594    }
595    Ok(by_path
596        .into_iter()
597        .map(|(path, mut line_anchors)| {
598            line_anchors.sort_unstable();
599            line_anchors.dedup();
600            RelevantFile { path, line_anchors }
601        })
602        .collect())
603}

605fn convert_sverklo(
606    source: &Path,
607    lock: &CorpusLock,
608    limit: Option<usize>,
609    token_budget: usize,
610) -> Result<(Manifest, usize), Box<dyn Error>> {
611    validate_budget(token_budget)?;
612    let tasks = read_jsonl::<SverkloTask>(&source.join(&lock.sverklo.tasks_file))?;
613    let supported = lock
614        .sverklo
615        .supported_categories
616        .iter()
617        .map(String::as_str)
618        .collect::<HashSet<_>>();
619    let mut converted = Vec::new();
620    let mut skipped = 0;
621    for task in tasks {
622        if !supported.contains(task.category.as_str()) {
623            skipped += 1;
624            continue;
625        }
626        if converted.len() == limit.unwrap_or(usize::MAX) {
627            break;
628        }
629        converted.push(convert_sverklo_task(
630            task,
631            &lock.sverklo.repository,
632            token_budget,
633        )?);
634    }
635    if converted.is_empty() {
636        return Err("Sverklo lock selected no coordinate-bearing tasks".into());
637    }
638    let repository = &lock.sverklo.repository;
639    let corpus = Corpus {
640        name: repository.name.clone(),
641        url: repository.url.clone(),
642        directory: repository.directory.clone(),
643        base_revision: repository.revision.clone(),
644        fix_commit: None,
645        issue_url: None,
646        prompt_provenance: lock.sverklo.prompt_provenance.clone(),
647        label_provenance: lock.sverklo.label_provenance.clone(),
648        dataset_url: lock.sverklo.dataset_url.clone(),
649        dataset_revision: lock.sverklo.dataset_revision.clone(),
650        dataset_license: lock.sverklo.dataset_license.clone(),
651        external_limitations: lock.sverklo.limitations.clone(),
652        tasks: converted,
653    };
654    Ok((
655        external_manifest(
656            &lock.frozen_at,
657            "sverklo",
658            "Pinned Sverklo coordinate-bearing primitive tasks; unsupported name-only tasks are excluded.",
659            vec![corpus],
660        ),
661        skipped,
662    ))
663}

665fn convert_sverklo_task(
666    task: SverkloTask,
667    repository: &LockedRepository,
668    token_budget: usize,
669) -> Result<Task, Box<dyn Error>> {
670    if task.id.trim().is_empty()
671        || task.query.trim().is_empty()
672        || task.category.trim().is_empty()
673        || task.dataset.trim().is_empty()
674    {
675        return Err("supported Sverklo task has an empty required field".into());
676    }
677    let relevant_files = match task.expected {
678        SverkloExpected::Locations { locations } => {
679            let mut by_path = BTreeMap::<String, Vec<usize>>::new();
680            for location in locations {
681                validate_relative_path(&location.file)?;
682                if location.line == 0 {
683                    return Err(format!("{} has a zero line label", task.id).into());
684                }
685                by_path
686                    .entry(location.file)
687                    .or_default()
688                    .push(location.line);
689            }
690            by_path
691                .into_iter()
692                .map(|(path, mut line_anchors)| {
693                    line_anchors.sort_unstable();
694                    line_anchors.dedup();
695                    RelevantFile { path, line_anchors }
696                })
697                .collect()
698        }
699        SverkloExpected::Deps { imports, importers } => labels_to_relevant_files(
700            imports
701                .into_iter()
702                .chain(importers)
703                .map(SembleLabel::File)
704                .collect(),
705        )?,
706        SverkloExpected::Names { names } => {
707            return Err(format!(
708                "{} unexpectedly selected a name-only task with {} labels",
709                task.id,
710                names.len()
711            )
712            .into());
713        }
714    };
715    if relevant_files.is_empty() {
716        return Err(format!("{} has no coordinate-bearing labels", task.id).into());
717    }
718    Ok(Task {
719        id: format!("sverklo:{}", task.id),
720        prompt: task.query.clone(),
721        languages: vec![repository.language.clone()],
722        task_shapes: vec![format!("sverklo_{}", task.category.to_ascii_lowercase())],
723        rg_queries: vec![task.query],
724        relevant_files,
725        token_budget,
726    })
727}

729fn convert_arb_trace2code(
730    source: &Path,
731    lock: &CorpusLock,
732    arb: &ArbLock,
733    sample_ids: &HashSet<String>,
734    repositories: &HashSet<String>,
735    limit: Option<usize>,
736    token_budget: usize,
737) -> Result<(Manifest, usize), Box<dyn Error>> {
738    validate_budget(token_budget)?;
739    if arb.supported_task_type != "trace2code" {
740        return Err(format!(
741            "unsupported locked ARB task type {}",
742            arb.supported_task_type
743        )
744        .into());
745    }
746    let samples_path = source.join(&arb.samples_file);
747    verify_blake3(&samples_path, &arb.samples_blake3)?;
748    let samples = read_jsonl::<ArbSample>(&samples_path)?;
749    let repository_locks = arb
750        .repositories
751        .iter()
752        .map(|repository| (repository.name.as_str(), repository))
753        .collect::<BTreeMap<_, _>>();
754
755    let mut seen_sample_ids = HashSet::<&str>::new();
756    let mut seen_repositories = HashSet::<&str>::new();
757    for sample in &samples {
758        if sample.task_type != arb.supported_task_type {
759            continue;
760        }
761        seen_sample_ids.insert(sample.id.as_str());
762        seen_repositories.insert(sample.repo.as_str());
763    }
764
765    if !sample_ids.is_empty() {
766        let mut missing = sample_ids
767            .iter()
768            .filter(|sample_id| !seen_sample_ids.contains(sample_id.as_str()))
769            .cloned()
770            .collect::<Vec<_>>();
771        missing.sort();
772        if !missing.is_empty() {
773            return Err(format!("unknown ARB sample IDs: {}", missing.join(", ")).into());
774        }
775    }
776
777    if !repositories.is_empty() {
778        let mut missing = repositories
779            .iter()
780            .filter(|repository| !seen_repositories.contains(repository.as_str()))
781            .cloned()
782            .collect::<Vec<_>>();
783        missing.sort();
784        if !missing.is_empty() {
785            return Err(
786                format!("unknown or empty ARB repositories: {}", missing.join(", ")).into(),
787            );
788        }
789    }
790
791    let mut corpora = BTreeMap::<(String, String), Corpus>::new();
792    let mut converted = 0usize;
793    let mut skipped = 0usize;
794
795    for sample in samples {
796        if sample.task_type != arb.supported_task_type {
797            skipped += 1;
798            continue;
799        }
800        if !sample_ids.is_empty() && !sample_ids.contains(&sample.id) {
801            continue;
802        }
803        if !repositories.is_empty() && !repositories.contains(&sample.repo) {
804            continue;
805        }
806        if limit.is_some_and(|limit| converted >= limit) {
807            break;
808        }
809        let repository = repository_locks.get(sample.repo.as_str()).ok_or_else(|| {
810            format!(
811                "ARB sample {} uses unlocked repository {}",
812                sample.id, sample.repo
813            )
814        })?;
815        validate_revision(&sample.base_commit)?;
816        let task = convert_arb_trace2code_sample(&sample, repository, token_budget)?;
817        let key = (sample.repo.clone(), sample.base_commit.clone());
818        let corpus = corpora.entry(key).or_insert_with(|| Corpus {
819            name: format!("arb:{}:{}", sample.repo, &sample.base_commit[..12]),
820            url: repository.url.clone(),
821            directory: format!(
822                "arb-{}-{}",
823                sample.repo.replace('/', "__"),
824                &sample.base_commit[..12]
825            ),
826            base_revision: sample.base_commit.clone(),
827            fix_commit: None,
828            issue_url: None,
829            prompt_provenance: arb.prompt_provenance.clone(),
830            label_provenance: arb.label_provenance.clone(),
831            dataset_url: arb.dataset_url.clone(),
832            dataset_revision: arb.dataset_revision.clone(),
833            dataset_license: arb.dataset_license.clone(),
834            external_limitations: arb.limitations.clone(),
835            tasks: Vec::new(),
836        });
837        corpus.tasks.push(task);
838        converted += 1;
839    }
840    if converted == 0 {
841        return Err("ARB filters selected no trace2code samples".into());
842    }
843    let manifest = external_manifest(
844        &lock.frozen_at,
845        "Agent Retrieval Bench trace2code smoke",
846        &format!(
847            "{} deterministically selected task(s) from pinned release {} (archive SHA-256 {}). This is a bounded smoke baseline, not a complete ARB leaderboard run.",
848            converted, arb.release_id, arb.release_sha256
849        ),
850        corpora.into_values().collect(),
851    );
852    Ok((manifest, skipped))
853}

855fn convert_arb_trace2code_sample(
856    sample: &ArbSample,
857    repository: &ArbRepository,
858    token_budget: usize,
859) -> Result<Task, Box<dyn Error>> {
860    if sample.gold.root_cause_files.is_empty() {
861        return Err(format!(
862            "ARB trace2code sample {} has no root-cause files",
863            sample.id
864        )
865        .into());
866    }
867    let root_cause_files = sample
868        .gold
869        .root_cause_files
870        .iter()
871        .cloned()
872        .collect::<HashSet<_>>();
873    let mut by_path = BTreeMap::<String, Vec<usize>>::new();
874    for path in &sample.gold.root_cause_files {
875        validate_relative_path(path)?;
876        by_path.entry(path.clone()).or_default();
877    }
878    for span in &sample.gold_spans {
879        if span.start_line == 0 || span.end_line < span.start_line {
880            return Err(format!(
881                "invalid ARB span {}:{}-{}",
882                span.path, span.start_line, span.end_line
883            )
884            .into());
885        }
886        validate_relative_path(&span.path)?;
887        if root_cause_files.contains(&span.path) {
888            by_path
889                .entry(span.path.clone())
890                .or_default()
891                .push(span.start_line);
892        }
893    }
894    let relevant_files = by_path
895        .into_iter()
896        .map(|(path, mut line_anchors)| {
897            line_anchors.sort_unstable();
898            line_anchors.dedup();
899            RelevantFile { path, line_anchors }
900        })
901        .collect();
902    let prompt = serde_json::to_string(&sample.query)?;
903    if prompt == "{}" || prompt == "null" {
904        return Err(format!("ARB sample {} has an empty query", sample.id).into());
905    }
906    Ok(Task {
907        id: format!("arb:{}", sample.id),
908        prompt,
909        languages: vec![repository.language.clone()],
910        task_shapes: vec!["failure_trace_root_cause".into()],
911        rg_queries: Vec::new(),
912        relevant_files,
913        token_budget,
914    })
915}

936fn verify_blake3(path: &Path, expected: &str) -> Result<(), Box<dyn Error>> {
937    let actual = blake3::hash(&fs::read(path)?).to_hex().to_string();
938    if actual != expected {
939        return Err(format!(
940            "BLAKE3 mismatch for {}: expected {expected}, got {actual}",
941            path.display()
942        )
943        .into());
944    }
945    Ok(())
946}

948fn validate_repository(repository: &LockedRepository) -> Result<(), Box<dyn Error>> {
949    if repository.language.trim().is_empty() {
950        return Err("locked repository requires language".into());
951    }
952    validate_repository_fields(
953        &repository.name,
954        &repository.url,
955        &repository.directory,
956        &repository.revision,
957    )
958}
959
960fn validate_repository_fields(
961    name: &str,
962    url: &str,
963    directory: &str,
964    revision: &str,
965) -> Result<(), Box<dyn Error>> {
966    if name.trim().is_empty() || url.trim().is_empty() || directory.trim().is_empty() {
967        return Err("repository has an empty required field".into());
968    }
969    validate_relative_path(directory)?;
970    validate_revision(revision)
971}
972
973fn validate_revision(revision: &str) -> Result<(), Box<dyn Error>> {
974    if revision.len() != 40 || !revision.bytes().all(|byte| byte.is_ascii_hexdigit()) {
975        return Err(
976            format!("revision must be a full 40-character Git object ID: {revision}").into(),
977        );
978    }
979    Ok(())
980}
981
982fn validate_hex_digest(value: &str, length: usize, name: &str) -> Result<(), Box<dyn Error>> {
983    if value.len() != length || !value.bytes().all(|byte| byte.is_ascii_hexdigit()) {
984        return Err(format!("{name} must be exactly {length} hexadecimal characters").into());
985    }
986    Ok(())
987}
988
989fn validate_budget(token_budget: usize) -> Result<(), Box<dyn Error>> {
990    if token_budget == 0 || token_budget > 1_000_000 {
991        return Err(format!("token budget must be within 1..=1000000: {token_budget}").into());
992    }
993    Ok(())
994}
995
996fn validate_relative_path(path: &str) -> Result<(), Box<dyn Error>> {
997    let parsed = Path::new(path);
998    if path.trim().is_empty()
999        || path.contains('\\')
1000        || parsed.is_absolute()
1001        || parsed
1002            .components()
1003            .any(|component| !matches!(component, Component::Normal(_)))
1004    {
1005        return Err(format!("path must be normalized and repository-relative: {path}").into());
1006    }
1007    Ok(())
1008}

1010fn verify_source_revision(source: &Path, expected: &str) -> Result<(), Box<dyn Error>> {
1011    let output = Command::new("git")
1012        .args(["rev-parse", "HEAD"])
1013        .current_dir(source)
1014        .output()?;
1015    if !output.status.success() {
1016        return Err(format!("cannot resolve dataset revision in {}", source.display()).into());
1017    }
1018    let actual = String::from_utf8(output.stdout)?.trim().to_owned();
1019    if actual != expected {
1020        return Err(format!(
1021            "dataset checkout {} is at {actual}, expected {expected}",
1022            source.display()
1023        )
1024        .into());
1025    }
1026    Ok(())
1027}
1028
1029fn read_jsonl<T: for<'de> Deserialize<'de>>(path: &Path) -> Result<Vec<T>, Box<dyn Error>> {
1030    let bytes = fs::read(path)?;
1031    bytes
1032        .split(|byte| *byte == b'\n')
1033        .filter(|line| !line.iter().all(u8::is_ascii_whitespace))
1034        .enumerate()
1035        .map(|(index, line)| {
1036            serde_json::from_slice(line)
1037                .map_err(|error| format!("{} line {}: {error}", path.display(), index + 1).into())
1038        })
1039        .collect()
1040}
1041
1042fn write_manifest(path: &Path, manifest: &Manifest) -> Result<(), Box<dyn Error>> {
1043    if let Some(parent) = path
1044        .parent()
1045        .filter(|parent| !parent.as_os_str().is_empty())
1046    {
1047        fs::create_dir_all(parent)?;
1048    }
1049    let mut json = serde_json::to_string_pretty(manifest)?;
1050    json.push('\n');
1051    fs::write(path, json)?;
1052    Ok(())
1053}
