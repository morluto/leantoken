49#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize)]
50struct UsageReceipt {
51    total_input_tokens: u64,
52    uncached_input_tokens: u64,
53    cache_read_input_tokens: u64,
54    output_tokens: u64,
55    reasoning_output_tokens: u64,
56    non_reasoning_output_tokens: u64,
57}
58
59impl UsageReceipt {
60    fn parse(value: &Value) -> Result<Self, DynError> {
61        let input = required_u64(value, "/input_tokens", "input tokens")?;
62        let cached = required_u64(value, "/cached_input_tokens", "cached input tokens")?;
63        let output = required_u64(value, "/output_tokens", "output tokens")?;
64        let reasoning = required_u64(value, "/reasoning_output_tokens", "reasoning output tokens")?;
65        let total = required_u64(value, "/total_tokens", "total tokens")?;
66        if cached > input || reasoning > output || input.checked_add(output) != Some(total) {
67            return Err("Codex provider usage fields are internally inconsistent".into());
68        }
69        Ok(Self {
70            total_input_tokens: input,
71            uncached_input_tokens: input - cached,
72            cache_read_input_tokens: cached,
73            output_tokens: output,
74            reasoning_output_tokens: reasoning,
75            non_reasoning_output_tokens: output - reasoning,
76        })
77    }
78
79    fn checked_add(self, other: Self) -> Result<Self, DynError> {
80        Ok(Self {
81            total_input_tokens: checked_add(self.total_input_tokens, other.total_input_tokens)?,
82            uncached_input_tokens: checked_add(
83                self.uncached_input_tokens,
84                other.uncached_input_tokens,
85            )?,
86            cache_read_input_tokens: checked_add(
87                self.cache_read_input_tokens,
88                other.cache_read_input_tokens,
89            )?,
90            output_tokens: checked_add(self.output_tokens, other.output_tokens)?,
91            reasoning_output_tokens: checked_add(
92                self.reasoning_output_tokens,
93                other.reasoning_output_tokens,
94            )?,
95            non_reasoning_output_tokens: checked_add(
96                self.non_reasoning_output_tokens,
97                other.non_reasoning_output_tokens,
98            )?,
99        })
100    }
101
102    fn is_monotonic_after(self, previous: Self) -> bool {
103        self.total_input_tokens >= previous.total_input_tokens
104            && self.uncached_input_tokens >= previous.uncached_input_tokens
105            && self.cache_read_input_tokens >= previous.cache_read_input_tokens
106            && self.output_tokens >= previous.output_tokens
107            && self.reasoning_output_tokens >= previous.reasoning_output_tokens
108            && self.non_reasoning_output_tokens >= previous.non_reasoning_output_tokens
109    }
110}

281fn main() -> Result<(), DynError> {
282    let args = Args::parse();
283    validate_label(&args.experiment_id, "experiment ID")?;
284    validate_label(&args.arm, "experiment arm")?;
285    let receipt = build_receipt(
286        &args.root_rollout,
287        &args.sessions_root,
288        args.experiment_id,
289        args.arm,
290        args.expected_children,
291        args.gold_manifest.as_deref(),
292    )?;
293    fs::write(&args.output, serde_json::to_vec_pretty(&receipt)?)?;
294    if let Some(path) = args.svg {
295        fs::write(path, render_svg(&receipt))?;
296    }
297    Ok(())
298}

300fn build_receipt(
301    root_rollout: &Path,
302    sessions_root: &Path,
303    experiment_id: String,
304    arm: String,
305    expected_children: Option<usize>,
306    gold_manifest: Option<&Path>,
307) -> Result<FamilyReceipt, DynError> {
308    let root_meta = read_session_locator(root_rollout)?
309        .ok_or("root rollout does not start with Codex session_meta")?;
310    if root_meta.parent_id.is_some() || root_meta.is_subagent {
311        return Err("root rollout identifies itself as a subagent".into());
312    }
313    let locators = discover_session_locators(sessions_root)?;
314    let family = resolve_family(&root_meta, locators)?;
315    if let Some(expected) = expected_children
316        && family.len().saturating_sub(1) != expected
317    {
318        return Err(format!(
319            "expected {expected} child threads, discovered {}",
320            family.len().saturating_sub(1)
321        )
322        .into());
323    }
324
325    let mut parsed = Vec::with_capacity(family.len());
326    for locator in &family {
327        parsed.push(parse_rollout(&locator.path)?);
328    }
329    let index_by_id: HashMap<&str, usize> = parsed
330        .iter()
331        .enumerate()
332        .map(|(index, rollout)| (rollout.id.as_str(), index))
333        .collect();
334    let depths = compute_depths(&parsed, &index_by_id)?;
335
336    let mut family_usage = UsageReceipt::default();
337    let mut first_request_usage = UsageReceipt::default();
338    let mut collaboration_calls = BTreeMap::<String, usize>::new();
339    let mut fork_turns = BTreeMap::<String, usize>::new();
340    let mut spawn_calls = 0usize;
341    let mut matched_spawn_calls = 0usize;
342    let mut thread_receipts = Vec::with_capacity(parsed.len());
343    let mut family_hash_entries = Vec::with_capacity(parsed.len());
344
345    for (index, rollout) in parsed.iter().enumerate() {
346        family_usage = family_usage.checked_add(rollout.provider_usage)?;
347        first_request_usage = first_request_usage.checked_add(rollout.first_request_usage)?;
348        family_hash_entries.push(format!("{index}:{}", rollout.source_hash));
349        for call in &rollout.collaboration_calls {
350            *collaboration_calls.entry(call.tool.clone()).or_default() += 1;
351            if call.tool == "spawn_agent" {
352                spawn_calls += 1;
353                let mode = call.fork_turns.as_deref().unwrap_or("unknown");
354                *fork_turns.entry(mode.to_owned()).or_default() += 1;
355                if rollout
356                    .started_children
357                    .get(&call.call_id)
358                    .is_some_and(|child| index_by_id.contains_key(child.as_str()))
359                {
360                    matched_spawn_calls += 1;
361                }
362            }
363        }
364        let parent_thread_index = rollout
365            .parent_id
366            .as_deref()
367            .map(|parent| {
368                index_by_id
369                    .get(parent)
370                    .copied()
371                    .ok_or("family thread references an undiscovered parent")
372            })
373            .transpose()?;
374        thread_receipts.push(ThreadReceipt {
375            thread_index: index,
376            parent_thread_index,
377            depth: depths[index],
378            role: if index == 0 { "root" } else { "subagent" },
379            source_rollout_blake3: rollout.source_hash.clone(),
380            host_version: rollout.host_version.clone(),
381            model: rollout.model.clone(),
382            provider: rollout.provider.clone(),
383            reasoning_effort: rollout.reasoning_effort.clone(),
384            multi_agent_version: rollout.multi_agent_version.clone(),
385            rollout_record_count: rollout.record_count,
386            inherited_history_record_count: rollout.inherited_history_record_count,
387            inherited_history_bytes: rollout.inherited_history_bytes,
388            turn_count: rollout.turn_count,
389            completed_turns: rollout.completed_turns,
390            aborted_turns: rollout.aborted_turns,
391            provider_request_count: rollout.provider_request_count,
392            compaction_count: rollout.compaction_count,
393            task_duration_ms: rollout.task_duration_ms,
394            first_request_usage: rollout.first_request_usage,
395            provider_usage: rollout.provider_usage,
396            tool_calls: ToolCallCounts {
397                collaboration: rollout.tool_calls.collaboration,
398                mcp: rollout.tool_calls.mcp,
399                failed_mcp: rollout.tool_calls.failed_mcp,
400                mcp_result_json_bytes: rollout.tool_calls.mcp_result_json_bytes,
401                mcp_text_content_bytes: rollout.tool_calls.mcp_text_content_bytes,
402                mcp_structured_content_bytes: rollout.tool_calls.mcp_structured_content_bytes,
403                mcp_emitted_source_tokens: rollout.tool_calls.mcp_emitted_source_tokens,
404                shell: rollout.tool_calls.shell,
405                other: rollout.tool_calls.other,
406            },
407        });
408    }
409    let source_family_blake3 = blake3::hash(family_hash_entries.join("\n").as_bytes())
410        .to_hex()
411        .to_string();
412    let task_evaluation = gold_manifest
413        .map(|path| evaluate_task(path, &experiment_id, parsed[0].final_answer.as_deref()))
414        .transpose()?;
415    let mut limitations = vec![
416        "Provider usage is session-native accounting; provider request framing is unavailable.",
417        "Cached input is reported, but cache-creation input is unavailable.",
418        "Tool-call counts identify categories but do not retain arguments, outputs, or source evidence.",
419    ];
420    if task_evaluation.is_none() {
421        limitations.push(
422            "No gold manifest was supplied, so the receipt does not establish task correctness.",
423        );
424    }
425
426    Ok(FamilyReceipt {
427        schema_version: RECEIPT_SCHEMA_V1,
428        experiment_id,
429        arm,
430        source_root_rollout_blake3: parsed[0].source_hash.clone(),
431        source_family_blake3,
432        receipt_binary_blake3: hash_file(&std::env::current_exe()?)?,
433        host: "codex-cli",
434        topology: TopologyReceipt {
435            thread_count: parsed.len(),
436            child_thread_count: parsed.len().saturating_sub(1),
437            maximum_depth: depths.iter().copied().max().unwrap_or(0),
438            spawn_calls,
439            matched_spawn_calls,
440            collaboration_calls,
441            fork_turns,
442        },
443        provider_usage: family_usage,
444        first_request_usage,
445        task_evaluation,
446        thread_receipts,
447        privacy: PrivacyBoundary {
448            raw_rollouts_retained: false,
449            prompts_retained: false,
450            tool_arguments_retained: false,
451            tool_outputs_retained: false,
452            credentials_retained: false,
453            absolute_paths_retained: false,
454            session_call_and_agent_ids_hashed_or_omitted: true,
455        },
456        limitations,
457    })
458}

460fn evaluate_task(
461    manifest_path: &Path,
462    expected_experiment_id: &str,
463    final_answer: Option<&str>,
464) -> Result<TaskEvaluation, DynError> {
465    let bytes = read_bounded(manifest_path)?;
466    let manifest: GoldManifest = serde_json::from_slice(&bytes)?;
467    if manifest.schema_version != 1 {
468        return Err("unsupported multi-agent gold manifest schema".into());
469    }
470    if manifest.experiment_id != expected_experiment_id {
471        return Err("gold manifest experiment ID does not match the receipt".into());
472    }
473    validate_label(&manifest.task_id, "task ID")?;
474    validate_lower_hex(&manifest.repository_revision, 40, "repository revision")?;
475    if manifest.expected_evidence.is_empty() || manifest.expected_evidence.len() > 256 {
476        return Err("gold manifest must contain 1-256 evidence entries".into());
477    }
478    let expected_paths = manifest
479        .expected_evidence
480        .iter()
481        .map(|evidence| {
482            if evidence.path.trim().is_empty() {
483                return Err("gold evidence path must be non-empty".into());
484            }
485            Ok(evidence.path.clone())
486        })
487        .collect::<Result<HashSet<_>, DynError>>()?;
488    if matches!(manifest.evaluation_mode, EvaluationMode::Path)
489        && expected_paths.len() != manifest.expected_evidence.len()
490    {
491        return Err("gold manifest contains duplicate evidence paths".into());
492    }
493    let expected_exact = manifest
494        .expected_evidence
495        .iter()
496        .map(|evidence| {
497            let symbol = evidence.symbol.as_deref().unwrap_or_default();
498            if matches!(manifest.evaluation_mode, EvaluationMode::Exact) && symbol.trim().is_empty()
499            {
500                return Err("exact gold evidence symbols must be non-empty".into());
501            }
502            Ok((evidence.path.clone(), symbol.to_owned()))
503        })
504        .collect::<Result<HashSet<_>, DynError>>()?;
505    if expected_exact.len() != manifest.expected_evidence.len() {
506        return Err("gold manifest contains duplicate evidence".into());
507    }
508    let answer = final_answer.and_then(|answer| serde_json::from_str::<AgentAnswer>(answer).ok());
509    let answer_json_valid = answer.is_some();
510    let reported_evidence_count = answer.as_ref().map_or(0, |answer| answer.evidence.len());
511    let reported = answer
512        .as_ref()
513        .map(|answer| {
514            answer
515                .evidence
516                .iter()
517                .map(|evidence| {
518                    (
519                        evidence.path.clone(),
520                        evidence.symbol.clone().unwrap_or_default(),
521                    )
522                })
523                .collect::<HashSet<_>>()
524        })
525        .unwrap_or_default();
526    let reported_paths = reported
527        .iter()
528        .map(|(path, _)| path.clone())
529        .collect::<HashSet<_>>();
530    let matched_path_count = expected_paths.intersection(&reported_paths).count();
531    let matched_exact_evidence_count = expected_exact.intersection(&reported).count();
532    let unexpected_evidence_count = match manifest.evaluation_mode {
533        EvaluationMode::Exact => {
534            reported_evidence_count.saturating_sub(matched_exact_evidence_count)
535        }
536        EvaluationMode::Path => reported_paths.difference(&expected_paths).count(),
537    };
538    let task_success = answer_json_valid
539        && match manifest.evaluation_mode {
540            EvaluationMode::Exact => {
541                matched_exact_evidence_count == manifest.expected_evidence.len()
542                    && reported_evidence_count == manifest.expected_evidence.len()
543            }
544            EvaluationMode::Path => reported_paths == expected_paths,
545        };
546    Ok(TaskEvaluation {
547        task_id: manifest.task_id,
548        repository_revision: manifest.repository_revision,
549        gold_manifest_blake3: blake3::hash(&bytes).to_hex().to_string(),
550        evaluation_mode: manifest.evaluation_mode,
551        answer_json_valid,
552        expected_evidence_count: manifest.expected_evidence.len(),
553        reported_evidence_count,
554        reported_path_count: reported_paths.len(),
555        matched_path_count,
556        matched_exact_evidence_count,
557        unexpected_evidence_count,
558        task_success,
559    })
560}

562fn discover_session_locators(root: &Path) -> Result<Vec<SessionLocator>, DynError> {
563    let mut paths = Vec::new();
564    collect_jsonl_paths(root, &mut paths)?;
565    if paths.len() > MAX_DISCOVERED_ROLLOUTS {
566        return Err("sessions directory contains too many rollout files".into());
567    }
568    paths.sort();
569    let mut locators = Vec::new();
570    let mut ids = HashSet::new();
571    for path in paths {
572        let Some(locator) = read_session_locator(&path)? else {
573            continue;
574        };
575        if !ids.insert(locator.id.clone()) {
576            return Err("sessions directory contains a duplicate Codex session ID".into());
577        }
578        locators.push(locator);
579    }
580    Ok(locators)
581}

583fn collect_jsonl_paths(root: &Path, output: &mut Vec<PathBuf>) -> Result<(), DynError> {
584    for entry in fs::read_dir(root)? {
585        let entry = entry?;
586        let file_type = entry.file_type()?;
587        let path = entry.path();
588        if file_type.is_dir() {
589            collect_jsonl_paths(&path, output)?;
590        } else if file_type.is_file() && path.extension().is_some_and(|value| value == "jsonl") {
591            output.push(path);
592        }
593    }
594    Ok(())
595}

597fn read_session_locator(path: &Path) -> Result<Option<SessionLocator>, DynError> {
598    let file = File::open(path)?;
599    let mut line = String::new();
600    BufReader::new(file)
601        .take(MAX_SESSION_RECORD_BYTES)
602        .read_line(&mut line)?;
603    if line.trim().is_empty() {
604        return Ok(None);
605    }
606    let record: Value = match serde_json::from_str(&line) {
607        Ok(record) => record,
608        Err(_) => return Ok(None),
609    };
610    if record.pointer("/type").and_then(Value::as_str) != Some("session_meta") {
611        return Ok(None);
612    }
613    let payload = record
614        .pointer("/payload")
615        .ok_or("session_meta has no payload")?;
616    let id = required_str(payload, "/id", "session ID")?.to_owned();
617    let parent_id = payload
618        .pointer("/parent_thread_id")
619        .and_then(Value::as_str)
620        .map(str::to_owned);
621    let is_subagent = payload.pointer("/thread_source").and_then(Value::as_str) == Some("subagent")
622        || payload.pointer("/source/subagent").is_some();
623    Ok(Some(SessionLocator {
624        id,
625        parent_id,
626        is_subagent,
627        path: path.to_owned(),
628    }))
629}

632    root: &SessionLocator,
633    locators: Vec<SessionLocator>,
634) -> Result<Vec<SessionLocator>, DynError> {
635    let mut by_parent = HashMap::<String, Vec<SessionLocator>>::new();
636    for locator in locators {
637        if locator.id == root.id {
638            continue;
639        }
640        if locator.is_subagent
641            && let Some(parent) = locator.parent_id.clone()
642        {
643            by_parent.entry(parent).or_default().push(locator);
644        }
645    }
646    for children in by_parent.values_mut() {
647        children.sort_by(|left, right| left.path.cmp(&right.path));
648    }
649    let mut family = vec![SessionLocator {
650        id: root.id.clone(),
651        parent_id: None,
652        is_subagent: false,
653        path: root.path.clone(),
654    }];
655    let mut queue = VecDeque::from([root.id.clone()]);
656    let mut seen = HashSet::from([root.id.clone()]);
657    while let Some(parent) = queue.pop_front() {
658        for child in by_parent.remove(&parent).unwrap_or_default() {
659            if !seen.insert(child.id.clone()) {
660                return Err("Codex session family contains a cycle or duplicate".into());
661            }
662            queue.push_back(child.id.clone());
663            family.push(child);
664        }
665    }
666    Ok(family)
667}

669fn parse_rollout(path: &Path) -> Result<ParsedRollout, DynError> {
670    let bytes = read_bounded(path)?;
671    let text = std::str::from_utf8(&bytes)?;
672    let records = text
673        .lines()
674        .enumerate()
675        .filter(|(_, line)| !line.trim().is_empty())
676        .map(|(record_index, line)| {
677            serde_json::from_str::<Value>(line)
678                .map(|record| (record_index, line.len() + 1, record))
679                .map_err(|error| {
680                    format!("invalid Codex rollout JSONL record {record_index}: {error}")
681                })
682        })
683        .collect::<Result<Vec<_>, _>>()?;
684    let (_, _, first) = records.first().ok_or("Codex rollout is empty")?;
685    if first.pointer("/type").and_then(Value::as_str) != Some("session_meta") {
686        return Err("Codex rollout does not start with session_meta".into());
687    }
688    let session = first
689        .pointer("/payload")
690        .ok_or("session_meta has no payload")?;
691    let id = required_str(session, "/id", "session ID")?.to_owned();
692    let parent_id = session
693        .pointer("/parent_thread_id")
694        .and_then(Value::as_str)
695        .map(str::to_owned);
696    let host_version = required_str(session, "/cli_version", "Codex CLI version")?.to_owned();
697    let provider = required_str(session, "/model_provider", "Codex model provider")?.to_owned();
698    let multi_agent_version = session
699        .pointer("/multi_agent_version")
700        .and_then(Value::as_str)
701        .map(str::to_owned);
702    let live_start = live_record_start(&records, parent_id.is_some())?;
703    let inherited_history_record_count = live_start.saturating_sub(1);
704    let inherited_history_bytes = records[1..live_start]
705        .iter()
706        .try_fold(0u64, |total, (_, bytes, _)| {
707            checked_add(total, u64::try_from(*bytes)?)
708        })?;
709
710    let mut model = None;
711    let mut reasoning_effort = None;
712    let mut active_turn = false;
713    let mut turn_count = 0usize;
714    let mut completed_turns = 0usize;
715    let mut aborted_turns = 0usize;
716    let mut provider_request_count = 0usize;
717    let mut compaction_count = 0usize;
718    let mut task_duration_ms = 0u64;
719    let mut first_request_usage = None;
720    let mut provider_usage = None;
721    let mut final_answer = None;
722    let mut tool_calls = ToolCallCounts::default();
723    let mut collaboration_calls = Vec::new();
724    let mut started_children = HashMap::new();
725    for (_, _, record) in records.iter().skip(live_start) {
726        let record_type = required_str(record, "/type", "rollout record type")?;
727        let payload = record.pointer("/payload").unwrap_or(&Value::Null);
728        match record_type {
729            "session_meta" => {
730                return Err("Codex live rollout contains an unexpected session_meta".into());
731            }
732            "turn_context" => {
733                let current_model = required_str(payload, "/model", "Codex model")?;
734                if model.as_deref().is_some_and(|value| value != current_model) {
735                    return Err("Codex rollout changes model between turns".into());
736                }
737                model = Some(current_model.to_owned());
738                let current_effort = payload
739                    .pointer("/effort")
740                    .and_then(Value::as_str)
741                    .map(str::to_owned);
742                if reasoning_effort.is_some() && reasoning_effort != current_effort {
743                    return Err("Codex rollout changes reasoning effort between turns".into());
744                }
745                reasoning_effort = current_effort;
746            }
747            "event_msg" => ingest_event(
748                payload,
749                &mut RolloutEventAccumulator {
750                    active_turn: &mut active_turn,
751                    turn_count: &mut turn_count,
752                    completed_turns: &mut completed_turns,
753                    aborted_turns: &mut aborted_turns,
754                    provider_request_count: &mut provider_request_count,
755                    task_duration_ms: &mut task_duration_ms,
756                    first_request_usage: &mut first_request_usage,
757                    provider_usage: &mut provider_usage,
758                    final_answer: &mut final_answer,
759                    tool_calls: &mut tool_calls,
760                    started_children: &mut started_children,
761                },
762            )?,
763            "response_item" => {
764                ingest_response_item(payload, &mut tool_calls, &mut collaboration_calls)?
765            }
766            "compacted" => compaction_count += 1,
767            _ => {}
768        }
769    }
770    if active_turn || turn_count == 0 || completed_turns + aborted_turns != turn_count {
771        return Err("Codex rollout turn lifecycle is incomplete".into());
772    }
773    let provider_usage = provider_usage.ok_or("Codex rollout has no provider usage snapshot")?;
774    Ok(ParsedRollout {
775        id,
776        parent_id,
777        host_version,
778        provider,
779        model: model.ok_or("Codex rollout has no turn context")?,
780        reasoning_effort,
781        multi_agent_version,
782        record_count: records.len(),
783        inherited_history_record_count,
784        inherited_history_bytes,
785        turn_count,
786        completed_turns,
787        aborted_turns,
788        provider_request_count,
789        compaction_count,
790        task_duration_ms,
791        first_request_usage: first_request_usage
792            .ok_or("Codex rollout has no first-request usage")?,
793        provider_usage,
794        source_hash: blake3::hash(&bytes).to_hex().to_string(),
795        final_answer,
796        tool_calls,
797        collaboration_calls,
798        started_children,
799    })
800}

802fn live_record_start(
803    records: &[(usize, usize, Value)],
804    is_subagent: bool,
805) -> Result<usize, DynError> {
806    if !is_subagent {
807        return Ok(1);
808    }
809    let trigger_index = records
810        .iter()
811        .position(|(_, _, record)| {
812            record.pointer("/type").and_then(Value::as_str)
813                == Some("inter_agent_communication_metadata")
814                && record
815                    .pointer("/payload/trigger_turn")
816                    .and_then(Value::as_bool)
817                    == Some(true)
818        })
819        .ok_or("subagent rollout has no trigger-turn marker")?;
820    (1..=trigger_index)
821        .rev()
822        .find(|index| {
823            records[*index].2.pointer("/type").and_then(Value::as_str) == Some("event_msg")
824                && records[*index]
825                    .2
826                    .pointer("/payload/type")
827                    .and_then(Value::as_str)
828                    == Some("task_started")
829        })
830        .ok_or_else(|| "subagent rollout has no live task start".into())
831}

833struct RolloutEventAccumulator<'a> {
834    active_turn: &'a mut bool,
835    turn_count: &'a mut usize,
836    completed_turns: &'a mut usize,
837    aborted_turns: &'a mut usize,
838    provider_request_count: &'a mut usize,
839    task_duration_ms: &'a mut u64,
840    first_request_usage: &'a mut Option<UsageReceipt>,
841    provider_usage: &'a mut Option<UsageReceipt>,
842    final_answer: &'a mut Option<String>,
843    tool_calls: &'a mut ToolCallCounts,
844    started_children: &'a mut HashMap<String, String>,
845}
846
847fn ingest_event(
848    payload: &Value,
849    accumulator: &mut RolloutEventAccumulator<'_>,
850) -> Result<(), DynError> {
851    let active_turn = &mut *accumulator.active_turn;
852    let turn_count = &mut *accumulator.turn_count;
853    let completed_turns = &mut *accumulator.completed_turns;
854    let aborted_turns = &mut *accumulator.aborted_turns;
855    let provider_request_count = &mut *accumulator.provider_request_count;
856    let task_duration_ms = &mut *accumulator.task_duration_ms;
857    let first_request_usage = &mut *accumulator.first_request_usage;
858    let provider_usage = &mut *accumulator.provider_usage;
859    let final_answer = &mut *accumulator.final_answer;
860    let tool_calls = &mut *accumulator.tool_calls;
861    let started_children = &mut *accumulator.started_children;
862    match payload.pointer("/type").and_then(Value::as_str) {
863        Some("task_started") => {
864            if *active_turn {
865                return Err("Codex rollout starts a turn before closing the previous turn".into());
866            }
867            *active_turn = true;
868            *turn_count += 1;
869        }
870        Some("task_complete") => {
871            if !*active_turn {
872                return Err("Codex rollout completes a turn that was not started".into());
873            }
874            *active_turn = false;
875            *completed_turns += 1;
876            if let Some(answer) = payload
877                .pointer("/last_agent_message")
878                .and_then(Value::as_str)
879            {
880                *final_answer = Some(answer.to_owned());
881            }
882            *task_duration_ms = checked_add(
883                *task_duration_ms,
884                payload
885                    .pointer("/duration_ms")
886                    .and_then(Value::as_u64)
887                    .unwrap_or(0),
888            )?;
889        }
890        Some("turn_aborted") => {
891            if !*active_turn {
892                return Err("Codex rollout aborts a turn that was not started".into());
893            }
894            *active_turn = false;
895            *aborted_turns += 1;
896        }
897        Some("token_count") => {
898            if !*active_turn {
899                return Err("Codex provider usage occurs outside an active turn".into());
900            }
901            if let Some(total) = payload.pointer("/info/total_token_usage") {
902                let current = UsageReceipt::parse(total)?;
903                if provider_usage.is_some_and(|previous| !current.is_monotonic_after(previous)) {
904                    return Err("Codex cumulative provider usage regressed".into());
905                }
906                let last = payload
907                    .pointer("/info/last_token_usage")
908                    .map(UsageReceipt::parse)
909                    .transpose()?
910                    .unwrap_or(current);
911                if first_request_usage.is_none() {
912                    *first_request_usage = Some(last);
913                }
914                *provider_usage = Some(current);
915                *provider_request_count += 1;
916            }
917        }
918        Some("sub_agent_activity")
919            if payload.pointer("/kind").and_then(Value::as_str) == Some("started") =>
920        {
921            let call_id = required_str(payload, "/event_id", "subagent spawn event ID")?;
922            let child = required_str(payload, "/agent_thread_id", "subagent thread ID")?;
923            if started_children
924                .insert(call_id.to_owned(), child.to_owned())
925                .is_some()
926            {
927                return Err("Codex rollout repeats a subagent spawn event ID".into());
928            }
929        }
930        Some("mcp_tool_call_end") => ingest_mcp_result(payload, tool_calls)?,
931        _ => {}
932    }
933    Ok(())
934}

936fn ingest_mcp_result(payload: &Value, tool_calls: &mut ToolCallCounts) -> Result<(), DynError> {
937    tool_calls.mcp += 1;
938    let result = payload
939        .pointer("/result")
940        .ok_or("Codex MCP lifecycle event has no result")?;
941    let Some(ok) = result.pointer("/Ok") else {
942        tool_calls.failed_mcp += 1;
943        return Ok(());
944    };
945    if ok.pointer("/isError").and_then(Value::as_bool) == Some(true) {
946        tool_calls.failed_mcp += 1;
947    }
948    tool_calls.mcp_result_json_bytes = checked_add(
949        tool_calls.mcp_result_json_bytes,
950        u64::try_from(serde_json::to_vec(ok)?.len())?,
951    )?;
952    if let Some(content) = ok.pointer("/content").and_then(Value::as_array) {
953        let bytes = content.iter().try_fold(0u64, |total, item| {
954            let text_bytes = item
955                .pointer("/text")
956                .and_then(Value::as_str)
957                .map_or(0, str::len);
958            checked_add(total, u64::try_from(text_bytes)?)
959        })?;
960        tool_calls.mcp_text_content_bytes = checked_add(tool_calls.mcp_text_content_bytes, bytes)?;
961    }
962    if let Some(structured) = ok.pointer("/structuredContent") {
963        tool_calls.mcp_structured_content_bytes = checked_add(
964            tool_calls.mcp_structured_content_bytes,
965            u64::try_from(serde_json::to_vec(structured)?.len())?,
966        )?;
967        let emitted = structured
968            .pointer("/meta/source_tokens")
969            .and_then(Value::as_u64)
970            .unwrap_or(0);
971        tool_calls.mcp_emitted_source_tokens =
972            checked_add(tool_calls.mcp_emitted_source_tokens, emitted)?;
973    }
974    Ok(())
975}

977fn ingest_response_item(
978    payload: &Value,
979    tool_calls: &mut ToolCallCounts,
980    collaboration_calls: &mut Vec<CollaborationCall>,
981) -> Result<(), DynError> {
982    let payload_type = payload.pointer("/type").and_then(Value::as_str);
983    if payload_type != Some("function_call") && payload_type != Some("custom_tool_call") {
984        return Ok(());
985    }
986    let name = required_str(payload, "/name", "Codex tool name")?;
987    if is_collaboration_tool(name) {
988        tool_calls.collaboration += 1;
989        let call_id = required_str(payload, "/call_id", "Codex collaboration call ID")?;
990        let fork_turns = if name == "spawn_agent" {
991            payload
992                .pointer("/arguments")
993                .and_then(Value::as_str)
994                .map(serde_json::from_str::<Value>)
995                .transpose()?
996                .as_ref()
997                .and_then(|arguments| arguments.pointer("/fork_turns"))
998                .map(normalize_fork_turns)
999        } else {
1000            None
1001        };
1002        collaboration_calls.push(CollaborationCall {
1003            call_id: call_id.to_owned(),
1004            tool: name.to_owned(),
1005            fork_turns,
1006        });
1007    } else if payload.pointer("/namespace").is_some() || name.starts_with("mcp__") {
1008        // The matching mcp_tool_call_end lifecycle event is authoritative and
1009        // carries result-size and outcome metadata.
1010    } else if name == "exec" {
1011        let input = payload
1012            .pointer("/input")
1013            .and_then(Value::as_str)
1014            .unwrap_or("");
1015        if input.contains("tools.exec_command") || input.contains("tools.write_stdin") {
1016            tool_calls.shell += 1;
1017        } else if !input.contains("tools.mcp__") {
1018            tool_calls.other += 1;
1019        }
1020    } else if matches!(name, "exec_command" | "write_stdin" | "functions.exec") {
1021        tool_calls.shell += 1;
1022    } else {
1023        tool_calls.other += 1;
1024    }
1025    Ok(())
1026}

1028fn normalize_fork_turns(value: &Value) -> String {
1029    match value {
1030        Value::String(mode) if mode == "none" || mode == "all" => mode.clone(),
1031        Value::String(turns) if turns.parse::<u64>().is_ok() => "bounded".to_owned(),
1032        Value::Number(_) => "bounded".to_owned(),
1033        _ => "invalid".to_owned(),
1034    }
1035}

1049fn compute_depths(
1050    parsed: &[ParsedRollout],
1051    index_by_id: &HashMap<&str, usize>,
1052) -> Result<Vec<usize>, DynError> {
1053    let mut depths = vec![0usize; parsed.len()];
1054    for index in 1..parsed.len() {
1055        let parent = parsed[index]
1056            .parent_id
1057            .as_deref()
1058            .ok_or("subagent rollout has no parent thread")?;
1059        let parent_index = index_by_id
1060            .get(parent)
1061            .copied()
1062            .ok_or("subagent parent is outside the discovered family")?;
1063        if parent_index >= index {
1064            return Err("Codex session family is not in parent-before-child order".into());
1065        }
1066        depths[index] = depths[parent_index]
1067            .checked_add(1)
1068            .ok_or("thread depth overflow")?;
1069    }
1070    Ok(depths)
1071}
