5pub(super) fn run(workspace: &Path) -> Result<(), String> {
6    // Build executable fixtures under the workspace target: /tmp can be noexec.
7    let scratch = workspace.join("target/coverage-cache-fixtures");
8    fs::create_dir_all(&scratch).map_err(|error| error.to_string())?;
9    let fixture = tempfile::tempdir_in(&scratch).map_err(|error| error.to_string())?;
10    let root = fixture.path().join("workspace");
11    let output = std::process::Command::new(std::env::current_exe().unwrap())
12        .args(["coverage", "check-cache-cleanup-worker"])
13        .env("LEANTOKEN_COVERAGE_CACHE_FIXTURE", &root)
14        .env("CARGO_TARGET_DIR", root.join("target"))
15        .env_remove("RUSTFLAGS")
16        .env_remove("CARGO_ENCODED_RUSTFLAGS")
17        .env_remove("LLVM_PROFILE_FILE")
18        .env_remove("CARGO_LLVM_COV")
19        .env_remove("CARGO_LLVM_COV_TARGET_DIR")
20        .env_remove("CARGO_LLVM_COV_BUILD_DIR")
21        .env_remove("CARGO_BUILD_BUILD_DIR")
22        .output()
23        .map_err(|error| error.to_string())?;
24    if !output.status.success() {
25        return Err(format!(
26            "coverage cache regression failed\nstdout: {}\nstderr: {}",
27            String::from_utf8_lossy(&output.stdout),
28            String::from_utf8_lossy(&output.stderr)
29        ));
30    }
31    println!(
32        "coverage cache cleanup: obsolete maps excluded, current coverage retained, dependency artifacts unchanged"
33    );
34    Ok(())
35}

37pub(super) fn worker() -> Result<(), String> {
38    let root = std::env::var_os("LEANTOKEN_COVERAGE_CACHE_FIXTURE")
39        .ok_or("coverage cache worker requires an isolated fixture root")?;
40    let root = Path::new(&root);
41    let dependency = root.parent().unwrap().join("dependency");
42    fs::create_dir_all(root.join("src")).unwrap();
43    fs::create_dir_all(root.join("target/coverage")).unwrap();
44    fs::create_dir_all(dependency.join("src")).unwrap();
45    fs::write(
46        root.join("Cargo.toml"),
47        r#"[package]
48name = "leantoken_coverage_cache_fixture"
49version = "0.1.0"
50edition = "2024"
51[features]
52previous = []
53[dependencies]
54fixture_dependency = { path = "../dependency" }
55[workspace]
56"#,
57    )
58    .unwrap();
59    fs::write(
60        dependency.join("Cargo.toml"),
61        "[package]\nname = \"fixture_dependency\"\nversion = \"0.1.0\"\nedition = \"2024\"\n[workspace]\n",
62    )
63    .unwrap();
64    fs::write(
65        dependency.join("src/lib.rs"),
66        "pub fn constant() -> u64 { 4 }\n",
67    )
68    .unwrap();
69    fs::write(
70        root.join("src/lib.rs"),
71        "mod retired;\npub fn old_entry() -> u64 { retired::old_owner() + fixture_dependency::constant() }\n#[test] fn old_test() { assert_eq!(old_entry(), 11); }\n",
72    )
73    .unwrap();
74    fs::write(
75        root.join("src/retired.rs"),
76        "pub fn old_owner() -> u64 { 7 }\n",
77    )
78    .unwrap();
79    phase(root, "lock", &["cargo", "generate-lockfile", "--offline"]);
80    phase(
81        root,
82        "old-tests",
83        &[
84            "cargo",
85            "llvm-cov",
86            "test",
87            "--offline",
88            "--locked",
89            "--features",
90            "previous",
91            "--no-report",
92        ],
93    );
94    let before = report(root, "before");
95    assert!(has_retired_owner(&before));
96    assert!(has_executed_function(&before, "old_entry"));
97    let dependencies = dependency_objects(root);
98    assert!(!dependencies.is_empty(), "fixture must build a dependency");
99
100    fs::write(
101        root.join("src/lib.rs"),
102        "pub fn current_entry() -> u64 { fixture_dependency::constant() + 9 }\n#[test] fn current_test() { assert_eq!(current_entry(), 13); }\n",
103    )
104    .unwrap();
105    fs::remove_file(root.join("src/retired.rs")).unwrap();
106    // Exercise the production cleanup boundary, not a duplicate command.
107    clean_workspace(root).expect("clean cached workspace artifacts");
108    assert_eq!(
109        dependency_objects(root),
110        dependencies,
111        "workspace cleanup must retain dependency artifacts unchanged"
112    );
113    phase(
114        root,
115        "current-tests",
116        &[
117            "cargo",
118            "llvm-cov",
119            "test",
120            "--offline",
121            "--locked",
122            "--no-report",
123        ],
124    );
125    let current = report(root, "current");
126    assert!(
127        !has_retired_owner(&current),
128        "obsolete source owner in report"
129    );
130    assert!(
131        !current["data"][0]["functions"]
132            .as_array()
133            .unwrap()
134            .iter()
135            .any(|function| { function["name"].as_str().unwrap().contains("old_entry") }),
136        "obsolete function in report"
137    );
138    assert!(has_executed_function(&current, "current_entry"));
139    Ok(())
140}

142fn phase(root: &Path, name: &str, command: &[&str]) {
143    let command = command
144        .iter()
145        .map(|arg| (*arg).to_owned())
146        .collect::<Vec<_>>();
147    logged(root, &command, name).unwrap_or_else(|error| {
148        let output = root.join("target/coverage");
149        let stdout = fs::read_to_string(output.join(format!("{name}.stdout.log"))).unwrap();
150        let stderr = fs::read_to_string(output.join(format!("{name}.stderr.log"))).unwrap();
151        panic!("{error}\nstdout: {stdout}\nstderr: {stderr}");
152    });
153}

155fn report(root: &Path, name: &str) -> serde_json::Value {
156    let path = root.join(format!("{name}.json"));
157    phase(
158        root,
159        name,
160        &[
161            "cargo",
162            "llvm-cov",
163            "report",
164            "--json",
165            "--output-path",
166            path.to_str().unwrap(),
167        ],
168    );
169    serde_json::from_slice(&fs::read(path).unwrap()).unwrap()
170}

172fn has_retired_owner(report: &serde_json::Value) -> bool {
173    report["data"][0]["files"]
174        .as_array()
175        .unwrap()
176        .iter()
177        .any(|file| {
178            file["filename"]
179                .as_str()
180                .unwrap()
181                .replace('\\', "/")
182                .ends_with("/src/retired.rs")
183        })
184}
185
186fn has_executed_function(report: &serde_json::Value, name: &str) -> bool {
187    report["data"][0]["functions"]
188        .as_array()
189        .unwrap()
190        .iter()
191        .any(|function| {
192            function["name"].as_str().unwrap().contains(name)
193                && function["count"].as_u64().unwrap() > 0
194        })
195}

197fn dependency_objects(root: &Path) -> BTreeMap<String, String> {
198    fs::read_dir(root.join("target/llvm-cov-target/debug/deps"))
199        .unwrap()
200        .map(|entry| entry.unwrap().path())
201        .filter(|path| {
202            let name = path.file_name().unwrap().to_str().unwrap();
203            name.starts_with("libfixture_dependency-") && name.ends_with(".rlib")
204        })
205        .map(|path| {
206            let digest = blake3::hash(&fs::read(&path).unwrap()).to_hex().to_string();
207            (path.to_str().unwrap().to_owned(), digest)
208        })
209        .collect()
210}
