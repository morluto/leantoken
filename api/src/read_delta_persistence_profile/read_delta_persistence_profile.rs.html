1//! Profile cross-restart read-delta reuse against a disposable repository.
2//!
3//! Run this source with identical arguments at a base and candidate revision.
4//! Indexing is excluded from timed phases. Every measured request opens a fresh
5//! `Services` instance so process-local read-delta state cannot carry over.
6

21#[derive(Debug, Parser)]
22#[command(about = "Profile persistent read-delta base overhead and reuse")]
23struct Args {
24    /// Samples in each restart-reuse phase.
25    #[arg(long, default_value_t = 100)]
26    iterations: usize,
27    /// Generated source lines in the complete read target.
28    #[arg(long, default_value_t = 1_200)]
29    lines: usize,
30    /// Persistent database path. Omit to use and remove a temporary repository.

32    database: Option<PathBuf>,
33}

79#[tokio::main]
80async fn main() -> AnyResult<()> {
81    let args = Args::parse();
82    if args.iterations == 0 || args.lines < 20 {
83        return Err("iterations must be positive and lines must be at least 20".into());
84    }
85    let temporary = tempfile::tempdir()?;
86    let root = temporary.path();
87    let database = args
88        .database
89        .clone()
90        .unwrap_or_else(|| root.join("index.sqlite"));
91    let source_path = root.join("profile.rs");
92    let source = generated_source(args.lines);
93    fs::write(&source_path, &source)?;
94    let services = open_services(root, &database)?;
95    services.index(false).await?;
96    drop(services);
97    let database_bytes_after_index = file_bytes(&database);
98    let wal_bytes_after_index = file_bytes(&wal_path(&database));
99
100    let (seed, base_hash) = profile_seed(root, &database).await?;
101    let unchanged_restart_reuse = profile_restarts(root, &database, args.iterations, None).await?;
102    let changed = source.replacen("compute_value(10)", "compute_updated_value(10)", 1);
103    fs::write(&source_path, changed)?;
104    let edited_restart_reuse =
105        profile_restarts(root, &database, args.iterations, Some(&base_hash)).await?;
106
107    let report = Report {
108        schema_version: 1,
109        iterations: args.iterations,
110        lines: args.lines,
111        source_bytes: source.len(),
112        database_bytes_after_index,
113        wal_bytes_after_index,
114        seed,
115        unchanged_restart_reuse,
116        edited_restart_reuse,
117        final_database_bytes: file_bytes(&database),
118        final_wal_bytes: file_bytes(&wal_path(&database)),
119        peak_rss_kib: linux_status_kib("VmHWM:"),
120        limitations: vec![
121            "Fresh Services instances model restart isolation inside one benchmark process.",
122            "Phase timing includes complete Services::read work, not only delta-base storage.",
123            "Linux process write_bytes includes every write by this benchmark process.",
124            "SQLite database and WAL deltas are physical artifact sizes, not logical base bytes.",
125            "Provider task success is invariant here because candidate responses are checked by status and hash.",
126        ],
127    };
128    println!("{}", serde_json::to_string_pretty(&report)?);
129    Ok(())
130}

132async fn profile_seed(root: &Path, database: &Path) -> AnyResult<(PhaseReport, String)> {
133    let before_database = file_bytes(database);
134    let before_wal = file_bytes(&wal_path(database));
135    let before_write = linux_process_write_bytes();
136    let services = open_services(root, database)?;
137    let started = Instant::now();
138    let response = services.read(request(None)).await?;
139    let elapsed = started.elapsed().as_micros();
140    if response.truncated || response.status != ReadStatus::Content {
141        return Err("seed must return complete full content".into());
142    }
143    let hash = response.content_hash.clone();
144    let mut totals = OutcomeTotals::default();
145    observe(&response, &mut totals);
146    black_box(response);
147    Ok((
148        phase_report(
149            vec![elapsed],
150            totals,
151            artifact_delta(before_database, file_bytes(database)),
152            artifact_delta(before_wal, file_bytes(&wal_path(database))),
153            write_delta(before_write, linux_process_write_bytes()),
154        ),
155        hash,
156    ))
157}

159async fn profile_restarts(
160    root: &Path,
161    database: &Path,
162    iterations: usize,
163    expected_hash: Option<&str>,
164) -> AnyResult<PhaseReport> {
165    let before_database = file_bytes(database);
166    let before_wal = file_bytes(&wal_path(database));
167    let before_write = linux_process_write_bytes();
168    let mut samples = Vec::with_capacity(iterations);
169    let mut totals = OutcomeTotals::default();
170    for _ in 0..iterations {
171        let services = open_services(root, database)?;
172        let started = Instant::now();
173        let response = services
174            .read(request(expected_hash.map(str::to_owned)))
175            .await?;
176        samples.push(started.elapsed().as_micros());
177        observe(&response, &mut totals);
178        black_box(response);
179    }
180    Ok(phase_report(
181        samples,
182        totals,
183        artifact_delta(before_database, file_bytes(database)),
184        artifact_delta(before_wal, file_bytes(&wal_path(database))),
185        write_delta(before_write, linux_process_write_bytes()),
186    ))
187}

189fn open_services(root: &Path, database: &Path) -> AnyResult<Services> {
190    Ok(Services::open(Config::discover(
191        root,
192        Some(database.to_owned()),
193    )?)?)
194}

196fn request(expected_hash: Option<String>) -> ReadRequest {
197    ReadRequest {
198        path: "profile.rs".into(),
199        start_line: None,
200        end_line: None,
201        symbol: None,
202        heading: None,
203        heading_occurrence: None,
204        continuation_cursor: None,
205        max_tokens: Some(32_000),
206        expected_hash,
207        delta: true,
208        receipt_id: None,
209        policy: leantoken::model::ReadPolicy::Full,
210    }
211}

213fn generated_source(lines: usize) -> String {
214    (1..=lines)
215        .map(|line| format!("let value_{line} = compute_value({line});\n"))
216        .collect()
217}

219fn observe(response: &ReadResponse, totals: &mut OutcomeTotals) {
220    match response
221        .delta_receipt
222        .as_ref()
223        .map(|receipt| &receipt.outcome)
224    {
225        Some(ReadDeltaOutcome::Delta) => totals.delta = totals.delta.saturating_add(1),
226        Some(ReadDeltaOutcome::NotModified) => {
227            totals.not_modified = totals.not_modified.saturating_add(1);
228        }
229        _ => totals.full = totals.full.saturating_add(1),
230    }
231    totals.emitted_source_tokens = totals
232        .emitted_source_tokens
233        .saturating_add(response.meta.source_tokens);
234    totals.total_response_tokens = totals
235        .total_response_tokens
236        .saturating_add(response.meta.total_response_tokens);
237    totals.receipt_avoided_tokens = totals.receipt_avoided_tokens.saturating_add(
238        response
239            .delta_receipt
240            .as_ref()
241            .map_or(0, |receipt| receipt.avoided_tokens),
242    );
243}

245fn phase_report(
246    mut samples: Vec<u128>,
247    totals: OutcomeTotals,
248    database_bytes_delta: i128,
249    wal_bytes_delta: i128,
250    process_write_bytes_delta: Option<u64>,
251) -> PhaseReport {
252    samples.sort_unstable();
253    let total = samples.iter().copied().sum::<u128>();
254    PhaseReport {
255        samples: samples.len(),
256        p50_micros: percentile(&samples, 50),
257        p95_micros: percentile(&samples, 95),
258        mean_micros: total / samples.len().max(1) as u128,
259        full_responses: totals.full,
260        delta_responses: totals.delta,
261        not_modified_responses: totals.not_modified,
262        emitted_source_tokens: totals.emitted_source_tokens,
263        total_response_tokens: totals.total_response_tokens,
264        receipt_avoided_tokens: totals.receipt_avoided_tokens,
265        database_bytes_delta,
266        wal_bytes_delta,
267        process_write_bytes_delta,
268    }
269}

271fn percentile(sorted: &[u128], percentile: usize) -> u128 {
272    let index = sorted
273        .len()
274        .saturating_sub(1)
275        .saturating_mul(percentile)
276        .saturating_add(99)
277        / 100;
278    sorted.get(index).copied().unwrap_or_default()
279}

281fn wal_path(database: &Path) -> PathBuf {
282    let mut path = database.as_os_str().to_os_string();
283    path.push("-wal");
284    PathBuf::from(path)

287fn file_bytes(path: &Path) -> u64 {
288    fs::metadata(path).map_or(0, |metadata| metadata.len())
289}
290
291fn artifact_delta(before: u64, after: u64) -> i128 {
292    i128::from(after) - i128::from(before)
293}
294
295fn write_delta(before: Option<u64>, after: Option<u64>) -> Option<u64> {
296    Some(after?.saturating_sub(before?))
297}

299fn linux_process_write_bytes() -> Option<u64> {
300    fs::read_to_string("/proc/self/io")
301        .ok()?
302        .lines()
303        .find_map(|line| {
304            line.strip_prefix("write_bytes:")
305                .and_then(|value| value.trim().parse().ok())
306        })
307}
308
309fn linux_status_kib(field: &str) -> Option<u64> {
310    fs::read_to_string("/proc/self/status")
311        .ok()?
312        .lines()
313        .find_map(|line| {
314            line.strip_prefix(field)
315                .and_then(|value| value.split_whitespace().next())
316                .and_then(|value| value.parse().ok())
317        })
