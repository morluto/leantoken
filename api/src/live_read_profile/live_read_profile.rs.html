145async fn main() -> AnyResult<()> {
146    let args = Args::parse();
147    validate_args(&args)?;
148    let report = run_profile(&args).await?;
149    let json = serde_json::to_string_pretty(&report)?;
150    if let Some(parent) = args.output.parent() {
151        fs::create_dir_all(parent)?;
152    }
153    fs::write(&args.output, format!("{json}\n"))?;
154    println!("{json}");
155    Ok(())
156}

158fn validate_args(args: &Args) -> AnyResult<()> {
159    if !args.repository.is_dir() {
160        return Err(invalid_input("--repository must name a Git checkout"));
161    }
162    if args.repository_label.trim().is_empty() {
163        return Err(invalid_input("--repository-label must not be empty"));
164    }
165    if args.iterations == 0 || args.hot_set == 0 || args.max_tokens == 0 {
166        return Err(invalid_input(
167            "--iterations, --hot-set, and --max-tokens must be positive",
168        ));
169    }
170    if args.pressure_bytes > 1024 * 1024 * 1024 {
171        return Err(invalid_input("--pressure-bytes must not exceed 1 GiB"));
172    }
173    Ok(())
174}

176async fn run_profile(args: &Args) -> AnyResult<Report> {
177    let snapshot = snapshot_repository(&args.repository)?;
178    let config = Config::discover(&snapshot.root, Some(snapshot.database.clone()))?;
179    let services = Services::open(config)?;
180    let index_start = Instant::now();
181    services.index(leantoken::IndexingMode::Rebuild).await?;
182    let initial_index_ms = index_start.elapsed().as_secs_f64() * 1_000.0;
183
184    let targets = read_targets(&snapshot.root)?;
185    if targets.is_empty() {
186        return Err(invalid_data("repository contains no readable UTF-8 files"));
187    }
188    let hot_targets = evenly_spaced_targets(&targets, args.hot_set.min(targets.len()));
189    let hot_contents = hot_targets
190        .iter()
191        .map(|target| fs::read(&target.absolute_path))
192        .collect::<Result<Vec<_>, _>>()?;
193    let hot_payload_bytes = hot_contents.iter().try_fold(0u64, |total, content| {
194        total
195            .checked_add(content.len() as u64)
196            .ok_or_else(|| io::Error::other("hot payload byte count overflow"))
197    })?;
198
199    let post_index_spread_direct = measure_direct_reads(&targets, args.iterations)?;
200    prewarm_direct(&hot_targets)?;
201    prewarm_service(&services, &hot_targets, args.max_tokens).await?;
202    let warm_hot_direct = measure_direct_reads(&hot_targets, args.iterations)?;
203    let warm_hot_memory_copy = measure_memory_copies(&hot_contents, args.iterations);
204    let warm_hot_service =
205        measure_service_reads(&services, &hot_targets, args.iterations, args.max_tokens).await?;
206    let warm_spread_service =
207        measure_service_reads(&services, &targets, args.iterations, args.max_tokens).await?;
208
209    let pressure_buffer = touched_pressure(args.pressure_bytes);
210    let pressure = PressureMeasurement {
211        retained_touched_bytes: pressure_buffer.len(),
212        direct_hot: measure_direct_reads(&hot_targets, args.iterations)?,
213        service_hot: measure_service_reads(
214            &services,
215            &hot_targets,
216            args.iterations,
217            args.max_tokens,
218        )
219        .await?,
220    };
221    black_box(&pressure_buffer);
222
223    let live_target = targets
224        .iter()
225        .find(|target| !target.relative_path.contains(".gitignore"))
226        .unwrap_or(&targets[0]);
227    let live_change = verify_live_change(&services, live_target, args.max_tokens).await?;
228    let (leantoken_git_revision, leantoken_worktree_dirty) = leantoken_source_identity();
229
230    Ok(Report {
231        schema_version: 1,
232        leantoken_version: env!("LEANTOKEN_PRODUCT_VERSION"),
233        leantoken_git_revision,
234        leantoken_worktree_dirty,
235        host_os: std::env::consts::OS,
236        host_arch: std::env::consts::ARCH,
237        release_build: !cfg!(debug_assertions),
238        corpus: CorpusReport {
239            source_repository: args.repository_label.clone(),
240            revision: snapshot.revision,
241            files: snapshot.files,
242            total_bytes: snapshot.total_bytes,
243            text_read_targets: targets.len(),
244            iterations: args.iterations,
245            hot_set_files: hot_targets.len(),
246            max_tokens: args.max_tokens,
247            pressure_bytes: args.pressure_bytes,
248        },
249        initial_index_ms,
250        post_index_spread_direct,
251        warm_hot_direct,
252        warm_hot_memory_copy,
253        warm_hot_service,
254        warm_spread_service,
255        pressure,
256        live_change,
257        hot_payload_bytes,
258        limitations: vec![
259            "The checkout copy and initial index touch corpus files before measurement; post_index_spread_direct is a first profile pass, not proof of a cold operating-system page cache.".into(),
260            "The pressure condition retains and touches process memory but cannot force or verify cross-platform page-cache eviction.".into(),
261            "The direct-read p95 is treated as a deliberately generous upper bound in the adoption decision; a real cache would still pay lookup, synchronization, cloning, eviction, and invalidation costs.".into(),
262            "Service timing includes SQLite snapshot lookup, full live-file read and hash, UTF-8 validation, range extraction, token truncation, and response construction. Serialization is measured separately and in the complete request total.".into(),
263            "No model or provider call runs in this local profile. Comparing live reads only with the local service-plus-serialization request makes their end-to-end share an upper bound for agent workflows.".into(),
264            "GitHub-hosted local filesystems do not represent remote, encrypted, antivirus-heavy, or contended deployments; those environments require an in-situ frozen profile before a scoped cache decision.".into(),
265        ],
266    })
267}

269fn snapshot_repository(repository: &Path) -> AnyResult<CorpusSnapshot> {
270    let repository = repository.canonicalize()?;
271    let revision = git_output(&repository, &["rev-parse", "HEAD"])?;
272    if !git_output(
273        &repository,
274        &["status", "--porcelain", "--untracked-files=normal"],
275    )?
276    .is_empty()
277    {
278        return Err(invalid_data("profile repository must be clean"));
279    }
280    let limits = DiscoveryLimits::default();
281    let discovered = discover_files(&repository, limits.max_file_bytes)?;
282    let directory = tempfile::tempdir()?;
283    let root = directory.path().join("repository");
284    fs::create_dir_all(&root)?;
285    let mut total_bytes = 0u64;
286    for file in &discovered {
287        let destination = root.join(&file.relative_path);
288        if let Some(parent) = destination.parent() {
289            fs::create_dir_all(parent)?;
290        }
291        fs::copy(&file.absolute_path, destination)?;
292        total_bytes = total_bytes
293            .checked_add(file.size_bytes)
294            .ok_or_else(|| io::Error::other("corpus byte count overflow"))?;
295    }
296    let database = directory.path().join("index.sqlite");
297    Ok(CorpusSnapshot {
298        _directory: directory,
299        root,
300        database,
301        revision,
302        files: discovered.len(),
303        total_bytes,
304    })
305}

307fn read_targets(root: &Path) -> AnyResult<Vec<ReadTarget>> {
308    let limits = DiscoveryLimits::default();
309    let discovered = discover_files(root, limits.max_file_bytes)?;
310    let mut targets = Vec::new();
311    for DiscoveredFile {
312        absolute_path,
313        relative_path,
314        size_bytes,
315        ..
316    } in discovered
317    {
318        let bytes = fs::read(&absolute_path)?;
319        if !bytes.is_empty() && std::str::from_utf8(&bytes).is_ok() {
320            targets.push(ReadTarget {
321                absolute_path,
322                relative_path,
323                size_bytes,
324            });
325        }
326    }
327    targets.sort_by(|left, right| left.relative_path.cmp(&right.relative_path));
328    Ok(targets)
329}

355fn measure_direct_reads(targets: &[ReadTarget], samples: usize) -> AnyResult<ReadMeasurement> {
356    let mut durations = Vec::with_capacity(samples);
357    let mut total_bytes = 0u64;
358    for sample in 0..samples {
359        let target = &targets[sample % targets.len()];
360        let start = Instant::now();
361        let bytes = fs::read(&target.absolute_path)?;
362        durations.push(start.elapsed());
363        total_bytes = total_bytes
364            .checked_add(bytes.len() as u64)
365            .ok_or_else(|| io::Error::other("read byte count overflow"))?;
366        black_box(bytes);
367    }
368    Ok(ReadMeasurement::new(
369        durations,
370        total_bytes,
371        samples.min(targets.len()),
372    ))
373}

375fn measure_memory_copies(contents: &[Vec<u8>], samples: usize) -> ReadMeasurement {
376    let mut durations = Vec::with_capacity(samples);
377    let mut total_bytes = 0u64;
378    for sample in 0..samples {
379        let start = Instant::now();
380        let copy = contents[sample % contents.len()].clone();
381        durations.push(start.elapsed());
382        total_bytes += copy.len() as u64;
383        black_box(copy);
384    }
385    ReadMeasurement::new(durations, total_bytes, samples.min(contents.len()))
386}

388async fn measure_service_reads(
389    services: &Services,
390    targets: &[ReadTarget],
391    samples: usize,
392    max_tokens: usize,
393) -> AnyResult<ServiceMeasurement> {
394    let mut request_durations = Vec::with_capacity(samples);
395    let mut serialization_durations = Vec::with_capacity(samples);
396    let mut complete_durations = Vec::with_capacity(samples);
397    let mut total_source_bytes = 0u64;
398    let mut total_wire_bytes = 0u64;
399    for sample in 0..samples {
400        let target = &targets[sample % targets.len()];
401        let request_start = Instant::now();
402        let response = services.read(read_request(target, max_tokens)).await?;
403        let request_duration = request_start.elapsed();
404        let serialization_start = Instant::now();
405        let wire = serde_json::to_vec(&response)?;
406        let serialization_duration = serialization_start.elapsed();
407        request_durations.push(request_duration);
408        serialization_durations.push(serialization_duration);
409        complete_durations.push(request_duration + serialization_duration);
410        total_source_bytes = total_source_bytes
411            .checked_add(target.size_bytes)
412            .ok_or_else(|| io::Error::other("service source byte count overflow"))?;
413        total_wire_bytes = total_wire_bytes
414            .checked_add(wire.len() as u64)
415            .ok_or_else(|| io::Error::other("service wire byte count overflow"))?;
416        black_box((response, wire));
417    }
418    Ok(ServiceMeasurement {
419        request: TimingStats::from_durations(request_durations),
420        serialization: TimingStats::from_durations(serialization_durations),
421        complete: TimingStats::from_durations(complete_durations),
422        total_source_bytes,
423        total_wire_bytes,
424        mean_source_bytes: total_source_bytes as f64 / samples as f64,
425        mean_wire_bytes: total_wire_bytes as f64 / samples as f64,
426        unique_files: samples.min(targets.len()),
427    })
428}

448fn touched_pressure(bytes: usize) -> Vec<u8> {
449    let mut pressure = vec![0u8; bytes];
450    for (page, offset) in (0..bytes).step_by(4096).enumerate() {
451        pressure[offset] = (page % 251) as u8;
452    }
453    pressure
454}

456async fn verify_live_change(
457    services: &Services,
458    target: &ReadTarget,
459    max_tokens: usize,
460) -> AnyResult<LiveChangeCheck> {
461    let before = services.read(read_request(target, max_tokens)).await?;
462    OpenOptions::new()
463        .append(true)
464        .open(&target.absolute_path)?
465        .write_all(b"\n")?;
466    let stale = services.read(read_request(target, max_tokens)).await?;
467    if !stale.index_stale || stale.meta.repository_generation != before.meta.repository_generation {
468        return Err(invalid_data(
469            "live read did not preserve generation while reporting stale content",
470        ));
471    }
472    services
473        .index_paths(vec![target.relative_path.clone()])
474        .await?;
475    let current = services.read(read_request(target, max_tokens)).await?;
476    if current.index_stale
477        || current.meta.repository_generation <= before.meta.repository_generation
478        || current.indexed_hash == stale.indexed_hash
479    {
480        return Err(invalid_data(
481            "reconciliation did not publish a current newer generation",
482        ));
483    }
484    Ok(LiveChangeCheck {
485        path: target.relative_path.clone(),
486        generation_before: before.meta.repository_generation,
487        stale_generation: stale.meta.repository_generation,
488        generation_after: current.meta.repository_generation,
489        stale_before_reconciliation: stale.index_stale,
490        current_after_reconciliation: !current.index_stale,
491    })
492}

494impl ReadMeasurement {
495    fn new(durations: Vec<Duration>, total_bytes: u64, unique_files: usize) -> Self {
496        let samples = durations.len();
497        Self {
498            timing: TimingStats::from_durations(durations),
499            total_bytes,
500            mean_bytes: total_bytes as f64 / samples as f64,
501            unique_files,
502        }
503    }
504}
505
506impl TimingStats {
507    fn from_durations(durations: Vec<Duration>) -> Self {
508        let mut micros = durations
509            .iter()
510            .map(|duration| duration.as_secs_f64() * 1_000_000.0)
511            .collect::<Vec<_>>();
512        micros.sort_by(f64::total_cmp);
513        let total_us = micros.iter().sum::<f64>();
514        Self {
515            samples: micros.len(),
516            total_ms: total_us / 1_000.0,
517            mean_us: total_us / micros.len() as f64,
518            min_us: micros[0],
519            p50_us: nearest_rank(&micros, 50),
520            p95_us: nearest_rank(&micros, 95),
521            max_us: micros[micros.len() - 1],
522        }
523    }
524}
525
526fn nearest_rank(values: &[f64], percentile: usize) -> f64 {
527    let rank = (percentile * values.len()).div_ceil(100);
528    values[rank.saturating_sub(1).min(values.len() - 1)]
529}

531fn git_output(repository: &Path, args: &[&str]) -> AnyResult<String> {
532    let output = Command::new("git")
533        .arg("-C")
534        .arg(repository)
535        .args(args)
536        .output()?;
537    if !output.status.success() {
538        return Err(invalid_data("unable to resolve clean Git corpus identity"));
539    }
540    Ok(String::from_utf8(output.stdout)?.trim().to_string())
541}

543fn leantoken_source_identity() -> (Option<String>, Option<bool>) {
544    let root = Path::new(env!("LEANTOKEN_REPOSITORY_ROOT"));
545    let revision = git_output(root, &["rev-parse", "HEAD"]).ok();
546    let dirty = git_output(root, &["status", "--porcelain", "--untracked-files=normal"])
547        .ok()
548        .map(|status| !status.is_empty());
549    (revision, dirty)
550}
