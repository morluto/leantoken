270#[derive(Debug, Clone, Copy, PartialEq, Eq)]
271enum Arm {
272    LexicalSyntax,
273    ImportNeighbor,
274    ReverseDependency,
275    HighConfidenceCaller,
276}
277
278impl Arm {
279    const ALL: [Self; 4] = [
280        Self::LexicalSyntax,
281        Self::ImportNeighbor,
282        Self::ReverseDependency,
283        Self::HighConfidenceCaller,
284    ];
285
286    const fn name(self) -> &'static str {
287        match self {
288            Self::LexicalSyntax => "lexical_syntax",
289            Self::ImportNeighbor => "import_neighbor",
290            Self::ReverseDependency => "reverse_dependency",
291            Self::HighConfidenceCaller => "high_confidence_caller",
292        }
293    }
294
295    const fn policy(self) -> ContextSignalPolicy {
296        match self {
297            Self::LexicalSyntax => ContextSignalPolicy::LexicalSyntax,
298            Self::ImportNeighbor => ContextSignalPolicy::ImportNeighbor,
299            Self::ReverseDependency => ContextSignalPolicy::ReverseDependency,
300            Self::HighConfidenceCaller => ContextSignalPolicy::HighConfidenceCaller,
301        }
302    }
303}

305#[tokio::main]
306async fn main() -> Result<(), DynError> {
307    let args = Args::parse();
308    let manifest_bytes = fs::read(&args.manifest)?;
309    let manifest_blake3 = blake3::hash(&manifest_bytes).to_hex().to_string();
310    let manifest: AblationManifest = serde_json::from_slice(&manifest_bytes)?;
311    let source_bytes = fs::read(&manifest.source_manifest)?;
312    let observed_source_hash = blake3::hash(&source_bytes).to_hex().to_string();
313    if observed_source_hash != manifest.source_manifest_blake3 {
314        return Err(format!(
315            "source manifest hash mismatch: expected {}, observed {observed_source_hash}",
316            manifest.source_manifest_blake3
317        )
318        .into());
319    }
320    let source: SourceManifest = serde_json::from_slice(&source_bytes)?;
321    validate_manifest(&manifest, &source)?;
322    preflight_repositories(&source, &args.repos_root)?;
323
324    let source_root = Path::new(env!("LEANTOKEN_REPOSITORY_ROOT"));
325    let harness_revision = git_output(source_root, &["rev-parse", "HEAD"])?;
326    let harness_dirty = !git_output(
327        source_root,
328        &["status", "--porcelain=v1", "--untracked-files=all"],
329    )?
330    .is_empty();
331    let harness_source_blake3 = blake3::hash(include_bytes!("graph_signal_ablation.rs"))
332        .to_hex()
333        .to_string();
334    if args.preflight_only {
335        println!(
336            "{}",
337            serde_json::to_string_pretty(&serde_json::json!({
338                "experiment_id": manifest.experiment_id,
339                "manifest_blake3": manifest_blake3,
340                "source_manifest_blake3": observed_source_hash,
341                "harness_revision": harness_revision,
342                "harness_worktree_dirty": harness_dirty,
343                "corpus_count": source.corpora.len(),
344                "task_count": source.corpora.iter().map(|corpus| corpus.tasks.len()).sum::<usize>(),
345                "status": "ready"
346            }))?
347        );
348        return Ok(());
349    }
350    if harness_dirty {
351        return Err("formal ablation requires a clean harness worktree".into());
352    }
353
354    let scratch = tempfile::tempdir()?;
355    let labels = manifest
356        .task_labels
357        .iter()
358        .map(|label| (label.task_id.as_str(), label))
359        .collect::<BTreeMap<_, _>>();
360    let mut graph_index = GraphIndexAggregate::default();
361    let mut runs = Vec::new();
362    for corpus in source.corpora {
363        let root = args.repos_root.join(&corpus.directory);
364        let database = scratch.path().join(format!("{}.sqlite", corpus.name));
365        let services = Services::open(Config::discover(&root, Some(database.clone()))?)?;
366        let started = Instant::now();
367        let indexed = services.index(true).await?;
368        let cold_index_ms = elapsed_ms(started);
369        if !indexed.warnings.is_empty() {
370            return Err(format!(
371                "{} indexing emitted warnings: {:?}",
372                corpus.name, indexed.warnings
373            )
374            .into());
375        }
376        let status = services.status().await?;
377        let mut noop_reconcile_ms = Vec::new();
378        for _ in 0..manifest.repetitions {
379            let started = Instant::now();
380            let reconciled = services.index(false).await?;
381            if reconciled.files_indexed != 0 || reconciled.files_removed != 0 {
382                return Err(
383                    format!("{} no-op reconciliation changed the index", corpus.name).into(),
384                );
385            }
386            noop_reconcile_ms.push(elapsed_ms(started));
387        }
388        let imports = import_stats(&database)?;
389        let parsed_reference_edges = scalar_usize(&database, "SELECT COUNT(*) FROM symbol_refs")?;
390        let database_bytes = logical_database_bytes(&database)?;
391        accumulate_graph_index(
392            &mut graph_index,
393            database_bytes,
394            imports,
395            parsed_reference_edges,
396        );
397        graph_index.corpora.push(CorpusIndexReport {
398            name: corpus.name.clone(),
399            revision: corpus.base_revision.clone(),
400            indexed_files: status.file_count,
401            database_bytes,
402            cold_index_ms,
403            noop_reconcile_median_ms: median(&noop_reconcile_ms),
404            noop_reconcile_ms,
405            imports,
406            parsed_reference_edges,
407        });
408
409        for repetition in 1..=manifest.repetitions {
410            for task in &corpus.tasks {
411                let request = context_request(task);
412                let baseline = services
413                    .context_signal_evaluation(request.clone(), Arm::LexicalSyntax.policy())
414                    .await?;
415                let baseline_keys = candidate_keys(&baseline);
416                for arm in Arm::ALL {
417                    let evaluation = if arm == Arm::LexicalSyntax {
418                        baseline.clone()
419                    } else {
420                        services
421                            .context_signal_evaluation(request.clone(), arm.policy())
422                            .await?
423                    };
424                    runs.push(run_report(
425                        &corpus.name,
426                        task,
427                        repetition,
428                        arm,
429                        labels[task.id.as_str()],
430                        &baseline_keys,
431                        evaluation,
432                    )?);
433                }
434            }
435        }
436    }
437    finish_graph_index(&mut graph_index);
438    let arms = aggregate_arms(&runs, manifest.repetitions);
439    let decision = decide(&arms, manifest.thresholds);
440    let report = Report {
441        schema_version: REPORT_SCHEMA_VERSION,
442        experiment_id: manifest.experiment_id,
443        manifest_blake3,
444        source_manifest_blake3: observed_source_hash,
445        manifest_description: manifest.description,
446        source_manifest_description: source.description,
447        source_manifest_schema_version: source.schema_version,
448        harness_revision,
449        harness_source_blake3,
450        harness_worktree_dirty: harness_dirty,
451        leantoken_version: env!("CARGO_PKG_VERSION"),
452        host_os: std::env::consts::OS,
453        host_arch: std::env::consts::ARCH,
454        rustc_version: command_version("rustc")?,
455        tokenizer: tokens::Tokenizer::default().name(),
456        token_count_exact: tokens::Tokenizer::default().is_exact(),
457        methodology: Methodology {
458            baseline: "Symbol and full-text candidates with dependency expansion and parsed-reference candidates disabled.",
459            arms: "Each candidate arm starts from the identical baseline and enables exactly one of forward import expansion, reverse-import ranking boost, or parsed reference candidates.",
460            additive_invariant: "Every baseline candidate identity (path, inclusive range, representation) must remain present in every signal arm.",
461            edge_precision: "Resolved import precision counts resolved_path values that join an indexed file; unresolved rate counts imports with no resolved_path. This validates resolver integrity, not semantic usefulness.",
462            signal_precision: "Unique signal-bearing candidate or selected paths are relevant only when they match the frozen relevant-file labels.",
463            dead_end_source: "Selected source tokens in files outside the frozen relevant-file labels.",
464            complete_response_tokens: "Exact tokenizer count of the complete serialized ContextResponse; no evaluation diagnostics are included in that payload.",
465            index_size: "Logical SQLite page_count times page_size after indexing and no-op reconciliation; WAL and SHM sidecars are excluded.",
466            timing: "Cold index and three no-op reconciliations run in release mode on one host. Arms share the same graph-enabled index, so timing is a cost envelope, not a causal graph-disabled comparison.",
467        },
468        thresholds: manifest.thresholds,
469        graph_index,
470        arms,
471        runs,
472        decision,
473        limitations: vec![
474            "The retrospective prompts and labels use public future fixes and are development evidence, not blind generalization evidence.",
475            "Eight tasks and fifteen relevant files are not a statistically powered product claim.",
476            "Parsed import resolution precision does not establish that an edge is useful to a model.",
477            "The benchmark executes no model, edit, or tests, so it cannot justify exposing graph metadata to agents.",
478            "Index and reconciliation timings include all current indexing work; this harness does not implement a graph-disabled indexer.",
479            "Package aliases, re-exports, dynamic imports, same-package relationships, and cross-language calls can remain unresolved.",
480        ],
481    };
482    let json = serde_json::to_string_pretty(&report)?;
483    if let Some(parent) = args
484        .output
485        .parent()
486        .filter(|path| !path.as_os_str().is_empty())
487    {
488        fs::create_dir_all(parent)?;
489    }
490    fs::write(&args.output, &json)?;
491    println!("{json}");
492    Ok(())
493}

495fn validate_manifest(manifest: &AblationManifest, source: &SourceManifest) -> Result<(), DynError> {
496    if manifest.schema_version != 1 {
497        return Err(format!("unsupported manifest schema {}", manifest.schema_version).into());
498    }
499    if manifest.repetitions < 3 {
500        return Err("at least three repetitions are required".into());
501    }
502    if manifest.arms.iter().map(String::as_str).collect::<Vec<_>>() != EXPECTED_ARMS {
503        return Err("manifest arms must match the frozen four-arm order".into());
504    }
505    let task_ids = source
506        .corpora
507        .iter()
508        .flat_map(|corpus| corpus.tasks.iter().map(|task| task.id.as_str()))
509        .collect::<BTreeSet<_>>();
510    let label_ids = manifest
511        .task_labels
512        .iter()
513        .map(|label| label.task_id.as_str())
514        .collect::<BTreeSet<_>>();
515    if task_ids != label_ids || label_ids.len() != manifest.task_labels.len() {
516        return Err("task labels must match source tasks exactly and be unique".into());
517    }
518    for label in &manifest.task_labels {
519        if label.rationale.trim().is_empty() || label.applicable_signals.is_empty() {
520            return Err(format!("{} requires signal labels and rationale", label.task_id).into());
521        }
522        for signal in &label.applicable_signals {
523            if !EXPECTED_ARMS[1..].contains(&signal.as_str()) {
524                return Err(format!("{} names unknown signal {signal}", label.task_id).into());
525            }
526        }
527    }
528    let thresholds = manifest.thresholds;
529    if !(0.0..=1.0).contains(&thresholds.minimum_signal_candidate_precision) {
530        return Err("signal precision threshold must be between zero and one".into());
531    }
532    Ok(())
533}

535fn preflight_repositories(source: &SourceManifest, repos_root: &Path) -> Result<(), DynError> {
536    for corpus in &source.corpora {
537        let root = repos_root.join(&corpus.directory);
538        let revision = git_output(&root, &["rev-parse", "HEAD"])?;
539        if revision != corpus.base_revision {
540            return Err(format!(
541                "{} revision mismatch: expected {}, observed {revision}",
542                corpus.name, corpus.base_revision
543            )
544            .into());
545        }
546        if !git_output(
547            &root,
548            &["status", "--porcelain=v1", "--untracked-files=all"],
549        )?
550        .is_empty()
551        {
552            return Err(format!("{} benchmark worktree is dirty", corpus.name).into());
553        }
554    }
555    Ok(())
556}

598fn run_report(
599    corpus: &str,
600    task: &TaskSpec,
601    repetition: usize,
602    arm: Arm,
603    label: &TaskLabel,
604    baseline_keys: &BTreeSet<(String, usize, usize, String)>,
605    evaluation: ContextEvaluation,
606) -> Result<RunReport, DynError> {
607    let relevant = task
608        .relevant_files
609        .iter()
610        .map(|file| file.path.as_str())
611        .collect::<BTreeSet<_>>();
612    let candidate_keys = candidate_keys(&evaluation);
613    let missing_baseline_candidates = baseline_keys.difference(&candidate_keys).count();
614    let returned_files = evaluation
615        .response
616        .fragments
617        .iter()
618        .map(|fragment| fragment.path.clone())
619        .collect::<BTreeSet<_>>();
620    let signal_candidate_files = evaluation
621        .generated_candidates
622        .iter()
623        .filter(|candidate| candidate_has_signal(candidate, arm))
624        .map(|candidate| candidate.path.clone())
625        .collect::<BTreeSet<_>>();
626    let signal_selected_files = evaluation
627        .response
628        .fragments
629        .iter()
630        .filter(|fragment| selected_has_signal(&fragment.reason, arm))
631        .map(|fragment| fragment.path.clone())
632        .collect::<BTreeSet<_>>();
633    let relevant_files_found = returned_files
634        .iter()
635        .filter(|path| relevant.contains(path.as_str()))
636        .count();
637    let relevant_signal_candidate_files = signal_candidate_files
638        .iter()
639        .filter(|path| relevant.contains(path.as_str()))
640        .count();
641    let relevant_signal_selected_files = signal_selected_files
642        .iter()
643        .filter(|path| relevant.contains(path.as_str()))
644        .count();
645    let false_positive_signal_candidate_files = signal_candidate_files
646        .len()
647        .saturating_sub(relevant_signal_candidate_files);
648    let line_anchors = task
649        .relevant_files
650        .iter()
651        .map(|file| file.line_anchors.len())
652        .sum();
653    let line_anchors_found = count_line_anchors(&evaluation.response, &task.relevant_files);
654    let source_tokens = evaluation
655        .response
656        .fragments
657        .iter()
658        .map(|fragment| fragment.token_count)
659        .sum();
660    let dead_end_fragments = evaluation
661        .response
662        .fragments
663        .iter()
664        .filter(|fragment| !relevant.contains(fragment.path.as_str()))
665        .count();
666    let dead_end_source_tokens = evaluation
667        .response
668        .fragments
669        .iter()
670        .filter(|fragment| !relevant.contains(fragment.path.as_str()))
671        .map(|fragment| fragment.token_count)
672        .sum();
673    let complete_response_tokens = tokens::count(&serde_json::to_string(&evaluation.response)?);
674    let signal_applicable = label
675        .applicable_signals
676        .iter()
677        .any(|name| name == arm.name());
678    let applicable_signal_unresolved = signal_applicable && relevant_signal_candidate_files == 0;
679    Ok(RunReport {
680        corpus: corpus.to_owned(),
681        task_id: task.id.clone(),
682        repetition,
683        arm: arm.name().to_owned(),
684        signal_applicable,
685        applicable_signal_rationale: signal_applicable.then(|| label.rationale.clone()),
686        additive_baseline_preserved: missing_baseline_candidates == 0,
687        missing_baseline_candidates,
688        relevant_files: relevant.len(),
689        relevant_files_found,
690        returned_files: returned_files.into_iter().collect(),
691        line_anchors,
692        line_anchors_found,
693        source_tokens,
694        dead_end_fragments,
695        dead_end_source_tokens,
696        complete_response_tokens,
697        signal_candidate_files: signal_candidate_files.into_iter().collect(),
698        relevant_signal_candidate_files,
699        false_positive_signal_candidate_files,
700        signal_selected_files: signal_selected_files.into_iter().collect(),
701        relevant_signal_selected_files,
702        applicable_signal_unresolved,
703    })
704}

746fn aggregate_arms(runs: &[RunReport], repetitions: usize) -> Vec<ArmAggregate> {
747    Arm::ALL
748        .into_iter()
749        .map(|arm| {
750            let mut per_repetition = Vec::new();
751            for repetition in 1..=repetitions {
752                let mut totals = RetrievalTotals::default();
753                for run in runs
754                    .iter()
755                    .filter(|run| run.arm == arm.name() && run.repetition == repetition)
756                {
757                    totals.relevant_files += run.relevant_files;
758                    totals.relevant_files_found += run.relevant_files_found;
759                    totals.line_anchors += run.line_anchors;
760                    totals.line_anchors_found += run.line_anchors_found;
761                    totals.returned_files += run.returned_files.len();
762                    totals.source_tokens += run.source_tokens;
763                    totals.dead_end_fragments += run.dead_end_fragments;
764                    totals.dead_end_source_tokens += run.dead_end_source_tokens;
765                    totals.complete_response_tokens += run.complete_response_tokens;
766                    totals.signal_candidate_files += run.signal_candidate_files.len();
767                    totals.relevant_signal_candidate_files += run.relevant_signal_candidate_files;
768                    totals.false_positive_signal_candidate_files +=
769                        run.false_positive_signal_candidate_files;
770                    totals.signal_selected_files += run.signal_selected_files.len();
771                    totals.relevant_signal_selected_files += run.relevant_signal_selected_files;
772                    totals.applicable_signal_tasks += usize::from(run.signal_applicable);
773                    totals.applicable_signal_tasks_without_relevant_candidate +=
774                        usize::from(run.applicable_signal_unresolved);
775                    totals.additive_violations += run.missing_baseline_candidates;
776                }
777                per_repetition.push(totals);
778            }
779            let first = per_repetition.first().cloned().unwrap_or_default();
780            let deterministic_metrics_repeat = per_repetition.iter().all(|totals| totals == &first);
781            let reference_runs = normalized_task_runs(runs, arm, 1);
782            let deterministic_task_results_repeat = (2..=repetitions)
783                .all(|repetition| normalized_task_runs(runs, arm, repetition) == reference_runs);
784            ArmAggregate {
785                arm: arm.name().to_owned(),
786                repetitions,
787                relevant_file_recall: ratio(first.relevant_files_found, first.relevant_files),
788                line_anchor_recall: ratio(first.line_anchors_found, first.line_anchors),
789                signal_candidate_precision: optional_ratio(
790                    first.relevant_signal_candidate_files,
791                    first.signal_candidate_files,
792                ),
793                signal_candidate_false_positive_rate: optional_ratio(
794                    first.false_positive_signal_candidate_files,
795                    first.signal_candidate_files,
796                ),
797                signal_selected_precision: optional_ratio(
798                    first.relevant_signal_selected_files,
799                    first.signal_selected_files,
800                ),
801                applicable_signal_unresolved_rate: optional_ratio(
802                    first.applicable_signal_tasks_without_relevant_candidate,
803                    first.applicable_signal_tasks,
804                ),
805                mean_dead_end_source_tokens: mean(
806                    &per_repetition
807                        .iter()
808                        .map(|totals| totals.dead_end_source_tokens)
809                        .collect::<Vec<_>>(),
810                ),
811                mean_complete_response_tokens: mean(
812                    &per_repetition
813                        .iter()
814                        .map(|totals| totals.complete_response_tokens)
815                        .collect::<Vec<_>>(),
816                ),
817                per_repetition,
818                deterministic_metrics_repeat,
819                deterministic_task_results_repeat,
820            }
821        })
822        .collect()
823}

825fn normalized_task_runs(runs: &[RunReport], arm: Arm, repetition: usize) -> Vec<RunReport> {
826    runs.iter()
827        .filter(|run| run.arm == arm.name() && run.repetition == repetition)
828        .cloned()
829        .map(|mut run| {
830            run.repetition = 0;
831            run
832        })
833        .collect()
834}

836fn decide(arms: &[ArmAggregate], thresholds: Thresholds) -> Decision {
837    let baseline = &arms[0];
838    let mut ranking_signal_decisions = Vec::new();
839    let mut retained_ranking_signals = Vec::new();
840    for arm in &arms[1..] {
841        let additive = arm
842            .per_repetition
843            .iter()
844            .all(|totals| totals.additive_violations == 0);
845        let recall_gate_passed_every_repetition = arm
846            .per_repetition
847            .iter()
848            .zip(&baseline.per_repetition)
849            .all(|(candidate, base)| {
850                candidate.relevant_files_found
851                    >= base.relevant_files_found + thresholds.minimum_relevant_file_gain
852                    || candidate.line_anchors_found
853                        >= base.line_anchors_found + thresholds.minimum_line_anchor_gain
854            });
855        let dead_end_gate_passed_every_repetition = arm
856            .per_repetition
857            .iter()
858            .zip(&baseline.per_repetition)
859            .all(|(candidate, base)| {
860                candidate.dead_end_source_tokens
861                    <= base.dead_end_source_tokens
862                        + thresholds.maximum_dead_end_source_token_increase
863            });
864        let response_cost_gate_passed_every_repetition = arm
865            .per_repetition
866            .iter()
867            .zip(&baseline.per_repetition)
868            .all(|(candidate, base)| {
869                candidate.complete_response_tokens
870                    <= base.complete_response_tokens
871                        + thresholds.maximum_complete_response_token_increase
872            });
873        let precision_gate_passed = arm
874            .signal_candidate_precision
875            .is_some_and(|precision| precision >= thresholds.minimum_signal_candidate_precision);
876        let repeatable = arm.deterministic_metrics_repeat && arm.deterministic_task_results_repeat;
877        let retain = repeatable
878            && additive
879            && recall_gate_passed_every_repetition
880            && dead_end_gate_passed_every_repetition
881            && response_cost_gate_passed_every_repetition
882            && precision_gate_passed;
883        if retain {
884            retained_ranking_signals.push(arm.arm.clone());
885        }
886        ranking_signal_decisions.push(SignalDecision {
887            arm: arm.arm.clone(),
888            repeatable,
889            additive,
890            recall_gate_passed_every_repetition,
891            dead_end_gate_passed_every_repetition,
892            response_cost_gate_passed_every_repetition,
893            precision_gate_passed,
894            retain_ranking_signal: retain,
895            reason: if retain {
896                "Passed every preregistered repeatability, additivity, recall, dead-end, response-cost, and precision gate.".into()
897            } else {
898                "Rejected because one or more preregistered repeatability, additivity, recall, dead-end, response-cost, or precision gates failed.".into()
899            },
900        });
901    }
902    Decision {
903        issue_outcome: if retained_ranking_signals.is_empty() {
904            "no_go"
905        } else {
906            "ranking_only_go"
907        },
908        ranking_signal_decisions,
909        retained_ranking_signals,
910        expose_graph_metadata: false,
911        metadata_decision: "Do not expose graph metadata: this retrieval-only ablation provides no model-visible value measurement, while complete response cost is already measured without new fields.",
912    }
913}

915fn import_stats(database: &Path) -> Result<ImportStats, DynError> {
916    let conn = Connection::open(database)?;
917    let (total, resolved, resolved_existing): (i64, i64, i64) = conn.query_row(
918        "SELECT COUNT(*),
919                COALESCE(SUM(CASE WHEN imports.resolved_path IS NOT NULL THEN 1 ELSE 0 END), 0),
920                COALESCE(SUM(CASE WHEN files.id IS NOT NULL THEN 1 ELSE 0 END), 0)
921         FROM imports
922         LEFT JOIN files ON files.path = imports.resolved_path",
923        [],
924        |row| Ok((row.get(0)?, row.get(1)?, row.get(2)?)),
925    )?;
926    let total = usize::try_from(total)?;
927    let resolved = usize::try_from(resolved)?;
928    let resolved_existing = usize::try_from(resolved_existing)?;
929    let unresolved = total.saturating_sub(resolved);
930    let false_resolved = resolved.saturating_sub(resolved_existing);
931    Ok(ImportStats {
932        total,
933        resolved,
934        resolved_existing,
935        unresolved,
936        false_resolved,
937        resolution_precision: optional_ratio(resolved_existing, resolved),
938        unresolved_rate: optional_ratio(unresolved, total),
939    })
940}
