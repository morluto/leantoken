394pub(super) fn run(args: &ColdMatrixArgs) -> AnyResult<()> {
395    let policy = validate_args(args)?;
396    let corpus = prepare_corpus(args)?;
397    let (revision, dirty) = leantoken_source_identity();
398    let revision =
399        revision.ok_or_else(|| invalid_input("LeanToken Git revision is unavailable"))?;
400    let dirty = dirty.ok_or_else(|| invalid_input("LeanToken worktree state is unavailable"))?;
401    if dirty && !args.allow_dirty {
402        return Err(invalid_input(
403            "cold-matrix requires a clean LeanToken worktree; commit the harness first",
404        ));
405    }
406
407    let run_root = tempfile::tempdir()?;
408    let mut runs = Vec::with_capacity(policy.worker_order.len());
409    let mut parity: Option<ParityReport> = None;
410    for (sequence, workers) in policy.worker_order.iter().copied().enumerate() {
411        let report = measure_cold_run(sequence, workers, &corpus, run_root.path(), &policy)?;
412        require_parity(&mut parity, &report)?;
413        runs.push(report);
414    }
415    let parity = parity.ok_or_else(|| invalid_input("worker matrix produced no runs"))?;
416    let cancellation_probes =
417        measure_cancellation_probes(&corpus, run_root.path(), &policy, &parity)?;
418    let worker_summaries = summarize_workers(&runs);
419    let decision = decide(&runs, &worker_summaries, &policy);
420    let host = host_report();
421    let report = ColdMatrixReport {
422        schema_version: SCHEMA_VERSION,
423        generated_unix_ms: unix_millis(),
424        leantoken_version: env!("LEANTOKEN_PRODUCT_VERSION"),
425        leantoken_git_revision: revision,
426        leantoken_worktree_dirty: dirty,
427        release_build: !cfg!(debug_assertions),
428        host,
429        corpus: corpus_report(&corpus),
430        measurement_policy: policy,
431        runs,
432        worker_summaries,
433        cancellation_probes,
434        parity,
435        decision,
436        limitations: vec![
437            "The report characterizes one pinned corpus on one host; worker or storage conclusions do not transfer automatically to other repositories or platforms.".into(),
438            "Phase CPU attribution is sampled process CPU assigned to the latest observed process-local phase; exact IndexingDiagnostics wall times remain the phase-owner source of truth.".into(),
439            "RSS and SQLite main/WAL/SHM values are sampled high-water observations and can miss peaks shorter than the configured interval.".into(),
440            "The per-index deadline requests cooperative cancellation. An in-process worker cannot safely kill its indexing thread and joins it after a grace violation; the supervising parent imposes a separate hard subprocess bound and kills a worker that fails to exit. cancellation_grace_exceeded records the in-process contract violation.".into(),
441            "Cancellation probes request cooperative cancellation after the target phase is observed. A single SQLite FTS statement or commit may finish before cancellation is checked; the report preserves the resulting generation instead of assuming rollback.".into(),
442            "Fresh subprocesses and SQLite paths isolate process initialization and database state, but the profiler does not evict the corpus from the operating-system page cache. The mirrored worker order counterbalances order effects; this is not a cold-disk measurement.".into(),
443            "The snapshot copies ignore-visible regular files and does not vendor the external corpus or preserve Git metadata.".into(),
444        ],
445    };
446    let json = serde_json::to_string_pretty(&report)?;
447    if let Some(parent) = args.output.parent() {
448        fs::create_dir_all(parent)?;
449    }
450    fs::write(&args.output, format!("{json}\n"))?;
451    println!("{json}");
452    Ok(())
453}

455fn validate_args(args: &ColdMatrixArgs) -> AnyResult<MeasurementPolicy> {
456    if !cfg!(target_os = "linux") {
457        return Err(invalid_input(
458            "cold-matrix currently requires Linux /proc resource accounting",
459        ));
460    }
461    if cfg!(debug_assertions) && !args.allow_debug {
462        return Err(invalid_input(
463            "cold-matrix decisions require a release build; use --allow-debug only for smoke tests",
464        ));
465    }
466    if !args.repository.is_dir() {
467        return Err(invalid_input("--repository must name a Git checkout"));
468    }
469    if args.repository_label.trim().is_empty() || args.repository_label.len() > 512 {
470        return Err(invalid_input(
471            "--repository-label must contain 1..=512 bytes",
472        ));
473    }
474    if args.expected_revision.trim().is_empty() || args.expected_revision.len() > 64 {
475        return Err(invalid_input(
476            "--expected-revision must contain 1..=64 bytes",
477        ));
478    }
479    if !(1..=MAX_SAMPLE_INTERVAL_MS).contains(&args.sample_interval_ms) {
480        return Err(invalid_input("--sample-interval-ms must be in 1..=1000"));
481    }
482    if !(1..=MAX_TIMEOUT_SECONDS).contains(&args.timeout_seconds) {
483        return Err(invalid_input("--timeout-seconds must be in 1..=14400"));
484    }
485    if args.parity_queries.is_empty() || args.parity_queries.len() > MAX_PARITY_QUERIES {
486        return Err(invalid_input(
487            "--parity-query requires 1..=16 bounded queries",
488        ));
489    }
490    for query in &args.parity_queries {
491        if query.is_empty() || query.len() > MAX_PARITY_QUERY_BYTES {
492            return Err(invalid_input(
493                "each --parity-query must contain 1..=256 bytes",
494            ));
495        }
496    }
497
498    let cancellation_phases = parse_cancellation_phases(&args.cancellation_phases)?;
499    if !args.allow_incomplete_matrix
500        && (cancellation_phases.len() != REQUIRED_CANCELLATION_PHASES.len()
501            || REQUIRED_CANCELLATION_PHASES
502                .iter()
503                .any(|phase| !cancellation_phases.contains(phase)))
504    {
505        return Err(invalid_input(
506            "decision matrices require every cancellation phase; use --allow-incomplete-matrix only for mechanical smoke tests",
507        ));
508    }
509    let worker_order = parse_worker_order(
510        args.worker_order
511            .as_deref()
512            .unwrap_or_else(|| args.matrix_kind.default_worker_order()),
513    )?;
514    if !args.allow_incomplete_matrix {
515        match args.matrix_kind {
516            ColdMatrixKind::Screening => validate_screening_order(&worker_order)?,
517            ColdMatrixKind::TwoWorkerFollowUp => {
518                validate_two_worker_follow_up_order(&worker_order)?;
519            }
520        }
521    }
522    Ok(MeasurementPolicy {
523        matrix_kind: args.matrix_kind,
524        worker_order,
525        minimum_samples_per_worker: args.matrix_kind.minimum_samples_per_worker(),
526        sample_interval_ms: args.sample_interval_ms,
527        timeout_seconds: args.timeout_seconds,
528        cancellation_grace_seconds: CANCELLATION_GRACE_SECONDS,
529        cancellation_phases,
530        parity_queries: args.parity_queries.clone(),
531        minimum_wall_reduction: 0.20,
532        minimum_wall_p95_reduction: (args.matrix_kind == ColdMatrixKind::TwoWorkerFollowUp)
533            .then_some(0.20),
534        maximum_cpu_increase: 0.25,
535        maximum_peak_rss_increase: 0.25,
536        maximum_write_increase: 0.05,
537        maximum_footprint_increase: 0.05,
538        preparation_owner_threshold: 0.35,
539        require_cancellation_observation: !args.allow_incomplete_matrix,
540    })
541}

543fn validate_screening_order(worker_order: &[usize]) -> AnyResult<()> {
544    let reverse = worker_order.iter().rev().copied().collect::<Vec<_>>();
545    let workers = worker_order.iter().copied().collect::<BTreeSet<_>>();
546    let balanced = [1, 2, 4].into_iter().all(|worker| {
547        worker_order
548            .iter()
549            .filter(|actual| **actual == worker)
550            .count()
551            >= 2
552    });
553    if worker_order != reverse || !balanced || workers != BTreeSet::from([1, 2, 4]) {
554        return Err(invalid_input(
555            "screening --worker-order must be a mirrored counterbalance containing 1,2,4 at least twice",
556        ));
557    }
558    Ok(())
559}

561fn validate_two_worker_follow_up_order(worker_order: &[usize]) -> AnyResult<()> {
562    let workers = worker_order.iter().copied().collect::<BTreeSet<_>>();
563    let one_samples = worker_order.iter().filter(|workers| **workers == 1).count();
564    let two_samples = worker_order.iter().filter(|workers| **workers == 2).count();
565    let blocks_are_counterbalanced =
566        worker_order
567            .chunks_exact(4)
568            .enumerate()
569            .all(|(index, block)| {
570                let expected = if index % 2 == 0 {
571                    [1, 2, 2, 1]
572                } else {
573                    [2, 1, 1, 2]
574                };
575                block == expected
576            });
577    if worker_order.len() < 8
578        || !worker_order.len().is_multiple_of(4)
579        || workers != BTreeSet::from([1, 2])
580        || one_samples != two_samples
581        || one_samples < ColdMatrixKind::TwoWorkerFollowUp.minimum_samples_per_worker()
582        || !blocks_are_counterbalanced
583    {
584        return Err(invalid_input(
585            "two-worker-follow-up --worker-order must contain alternating 1,2,2,1 / 2,1,1,2 blocks with at least four samples per worker",
586        ));
587    }
588    Ok(())
589}

591fn parse_worker_order(value: &str) -> AnyResult<Vec<usize>> {
592    let workers = value
593        .split(',')
594        .map(|part| {
595            part.trim()
596                .parse::<usize>()
597                .map_err(|_| invalid_input("--worker-order must be comma-separated integers"))
598        })
599        .collect::<AnyResult<Vec<_>>>()?;
600    if workers.is_empty() || workers.len() > MAX_MATRIX_RUNS {
601        return Err(invalid_input("--worker-order must contain 1..=16 runs"));
602    }
603    if workers
604        .iter()
605        .any(|workers| !(1..=MAX_WORKERS).contains(workers))
606    {
607        return Err(invalid_input("each --worker-order value must be in 1..=64"));
608    }
609    Ok(workers)
610}

612fn parse_cancellation_phases(value: &str) -> AnyResult<Vec<IndexProgressPhase>> {
613    if value.trim() == "none" {
614        return Ok(Vec::new());
615    }
616    let mut phases = Vec::new();
617    for part in value.split(',') {
618        let phase = match part.trim() {
619            "preparation" => IndexProgressPhase::Preparation,
620            "relational_write" => IndexProgressPhase::RelationalWrite,
621            "chunk_word_fts" => IndexProgressPhase::ChunkWordFts,
622            "chunk_trigram_fts" => IndexProgressPhase::ChunkTrigramFts,
623            "symbol_fts" => IndexProgressPhase::SymbolFts,
624            "reference_fts" => IndexProgressPhase::ReferenceFts,
625            "commit_and_checkpoint" => IndexProgressPhase::CommitAndCheckpoint,
626            _ => {
627                return Err(invalid_input(
628                    "--cancellation-phases contains an unsupported phase",
629                ));
630            }
631        };
632        if !phases.contains(&phase) {
633            phases.push(phase);
634        }
635    }
636    Ok(phases)
637}

794fn spawn_cold_worker(args: &ColdWorkerArgs, maximum_wall: Duration) -> AnyResult<ColdWorkerOutput> {
795    let executable = std::env::current_exe()?;
796    let stderr_path = args.output.with_extension("stderr.log");
797    let stderr = fs::File::create(&stderr_path)?;
798    let mut command = Command::new(executable);
799    command
800        .arg("cold-worker")
801        .arg("--repository")
802        .arg(&args.repository)
803        .arg("--database")
804        .arg(&args.database)
805        .arg("--sequence")
806        .arg(args.sequence.to_string())
807        .arg("--workers")
808        .arg(args.workers.to_string())
809        .arg("--sample-interval-ms")
810        .arg(args.sample_interval_ms.to_string())
811        .arg("--timeout-seconds")
812        .arg(args.timeout_seconds.to_string())
813        .arg("--output")
814        .arg(&args.output)
815        .stdout(Stdio::null())
816        .stderr(Stdio::from(stderr));
817    for query in &args.parity_queries {
818        command.arg("--parity-query").arg(query);
819    }
820    if let Some(target_phase) = &args.target_phase {
821        command.arg("--target-phase").arg(target_phase);
822    }
823    if let Some(baseline) = &args.baseline {
824        command.arg("--baseline").arg(baseline);
825    }
826    if args.allow_missed_phase {
827        command.arg("--allow-missed-phase");
828    }
829
830    let mut child = command.spawn()?;
831    let started = Instant::now();
832    let status = loop {
833        if let Some(status) = child.try_wait()? {
834            break status;
835        }
836        if started.elapsed() >= maximum_wall {
837            let _ = child.kill();
838            let status = child.wait()?;
839            let stderr = bounded_stderr(&stderr_path);
840            return Err(invalid_input(&format!(
841                "cold worker exceeded its hard subprocess bound ({status}): {stderr}"
842            )));
843        }
844        std::thread::sleep(Duration::from_millis(25));
845    };
846    if !status.success() {
847        let stderr = bounded_stderr(&stderr_path);
848        return Err(invalid_input(&format!(
849            "cold worker failed ({status}): {stderr}"
850        )));
851    }
852    let output = serde_json::from_slice(&read_bounded_file(
853        &args.output,
854        MAX_WORKER_REPORT_BYTES,
855        "cold worker report",
856    )?)?;
857    Ok(output)
858}

860fn bounded_stderr(path: &Path) -> String {
861    let Ok(mut file) = fs::File::open(path) else {
862        return String::new();
863    };
864    let length = file.metadata().map(|metadata| metadata.len()).unwrap_or(0);
865    let start = length.saturating_sub(8 * 1024);
866    if file.seek(SeekFrom::Start(start)).is_err() {
867        return String::new();
868    }
869    let mut bytes = Vec::with_capacity(usize::try_from(length - start).unwrap_or(0));
870    if file.take(8 * 1024).read_to_end(&mut bytes).is_err() {
871        return String::new();
872    }
873    String::from_utf8_lossy(&bytes).trim().to_owned()
874}

876fn read_bounded_file(path: &Path, maximum_bytes: u64, label: &str) -> AnyResult<Vec<u8>> {
877    let file = fs::File::open(path)?;
878    let length = file.metadata()?.len();
879    if length > maximum_bytes {
880        return Err(invalid_input(&format!(
881            "{label} exceeds its {maximum_bytes}-byte bound"
882        )));
883    }
884    let mut bytes = Vec::with_capacity(usize::try_from(length)?);
885    file.take(maximum_bytes.saturating_add(1))
886        .read_to_end(&mut bytes)?;
887    if u64::try_from(bytes.len()).unwrap_or(u64::MAX) > maximum_bytes {
888        return Err(invalid_input(&format!(
889            "{label} exceeds its {maximum_bytes}-byte bound"
890        )));
891    }
892    Ok(bytes)
893}

895pub(super) fn run_worker(args: &ColdWorkerArgs) -> AnyResult<()> {
896    if !args.repository.is_dir() {
897        return Err(invalid_input("cold worker repository does not exist"));
898    }
899    if !(1..=MAX_WORKERS).contains(&args.workers) {
900        return Err(invalid_input("cold worker count is out of bounds"));
901    }
902    if !(1..=MAX_SAMPLE_INTERVAL_MS).contains(&args.sample_interval_ms)
903        || !(1..=MAX_TIMEOUT_SECONDS).contains(&args.timeout_seconds)
904    {
905        return Err(invalid_input("cold worker timing bounds are invalid"));
906    }
907    if args.parity_queries.is_empty() || args.parity_queries.len() > MAX_PARITY_QUERIES {
908        return Err(invalid_input("cold worker parity query count is invalid"));
909    }
910    if args
911        .parity_queries
912        .iter()
913        .any(|query| query.is_empty() || query.len() > MAX_PARITY_QUERY_BYTES)
914    {
915        return Err(invalid_input("cold worker parity query size is invalid"));
916    }
917    if args.sequence >= MAX_MATRIX_RUNS {
918        return Err(invalid_input("cold worker sequence is out of bounds"));
919    }
920    if storage_footprint(&args.database)?.total_bytes != 0 {
921        return Err(invalid_input("cold worker database must be fresh"));
922    }
923    let output = if let Some(target) = &args.target_phase {
924        let phases = parse_cancellation_phases(target)?;
925        let phase = phases
926            .first()
927            .copied()
928            .ok_or_else(|| invalid_input("cold worker target phase is missing"))?;
929        if phases.len() != 1 {
930            return Err(invalid_input(
931                "cold worker accepts exactly one cancellation phase",
932            ));
933        }
934        let baseline_path = args
935            .baseline
936            .as_deref()
937            .ok_or_else(|| invalid_input("cancellation worker requires --baseline"))?;
938        let baseline: ParityReport = serde_json::from_slice(&read_bounded_file(
939            baseline_path,
940            MAX_BASELINE_BYTES,
941            "cold worker baseline",
942        )?)?;
943        ColdWorkerOutput::Cancellation(Box::new(measure_cancellation_probe_in_process(
944            phase,
945            CancellationProbeInput {
946                workers: args.workers,
947                repository: &args.repository,
948                database: &args.database,
949                parity_queries: &args.parity_queries,
950                sample_interval_ms: args.sample_interval_ms,
951                timeout_seconds: args.timeout_seconds,
952                require_observation: !args.allow_missed_phase,
953                parity: &baseline,
954            },
955        )?))
956    } else {
957        if args.baseline.is_some() {
958            return Err(invalid_input(
959                "matrix cold worker does not accept --baseline",
960            ));
961        }
962        ColdWorkerOutput::Matrix(Box::new(measure_cold_run_in_process(
963            args.sequence,
964            args.workers,
965            &args.repository,
966            &args.database,
967            &args.parity_queries,
968            args.sample_interval_ms,
969            args.timeout_seconds,
970        )?))
971    };
972    let json = serde_json::to_string_pretty(&output)?;
973    if u64::try_from(json.len()).unwrap_or(u64::MAX) > MAX_WORKER_REPORT_BYTES {
974        return Err(invalid_input("cold worker report exceeds its byte bound"));
975    }
976    if let Some(parent) = args.output.parent() {
977        fs::create_dir_all(parent)?;
978    }
979    fs::write(&args.output, format!("{json}\n"))?;
980    Ok(())
981}

983fn run_profiled_monitored(
984    indexer: &Indexer,
985    database: &Path,
986    cancellation: &CancellationToken,
987    settings: MonitorSettings,
988) -> AnyResult<MonitoredProfile> {
989    let MonitorSettings {
990        started,
991        baseline,
992        target,
993        sample_interval_ms,
994        timeout_seconds,
995    } = settings;
996    let deadline = started + Duration::from_secs(timeout_seconds);
997    let grace_deadline = deadline + Duration::from_secs(CANCELLATION_GRACE_SECONDS);
998    let mut sampler = ResourceSampler::new(database, sample_interval_ms, baseline)?;
999    sampler.sample(
1000        indexer
1001            .progress_snapshot()
1002            .and_then(|snapshot| snapshot.phase),
1003    )?;
1004
1005    let (sender, receiver) = mpsc::sync_channel(1);
1006    let worker = indexer.clone();
1007    let worker_cancellation = cancellation.clone();
1008    let handle = std::thread::spawn(move || {
1009        let result = worker.reconcile_cancellable_profiled(false, &worker_cancellation);
1010        let _ = sender.send(result);
1011    });
1012    let interval = Duration::from_millis(sample_interval_ms);
1013    let mut timeout_triggered = false;
1014    let mut cancellation_grace_exceeded = false;
1015    let mut target_observed = false;
1016    let mut cancellation_requested_at = None;
1017    let result = loop {
1018        let phase = indexer
1019            .progress_snapshot()
1020            .and_then(|snapshot| snapshot.phase);
1021        sampler.sample(phase)?;
1022        if let Some(MonitoredTarget::Phase(target_phase)) = target
1023            && phase == Some(target_phase)
1024            && cancellation_requested_at.is_none()
1025        {
1026            target_observed = true;
1027            cancellation_requested_at = Some(Instant::now());
1028            cancellation.cancel();
1029        }
1030        if Instant::now() >= deadline && !timeout_triggered {
1031            timeout_triggered = true;
1032            cancellation.cancel();
1033        }
1034        if Instant::now() >= grace_deadline && !handle.is_finished() {
1035            cancellation_grace_exceeded = true;
1036        }
1037        match receiver.recv_timeout(interval) {
1038            Ok(result) => break result,
1039            Err(mpsc::RecvTimeoutError::Timeout) => {}
1040            Err(mpsc::RecvTimeoutError::Disconnected) => {
1041                return Err(invalid_input("index worker stopped without a result"));
1042            }
1043        }
1044    };
1045    handle
1046        .join()
1047        .map_err(|_| invalid_input("index worker panicked"))?;
1048    sampler.sample(
1049        indexer
1050            .progress_snapshot()
1051            .and_then(|snapshot| snapshot.phase),
1052    )?;
1053    let resources = sampler.finish(timeout_triggered, cancellation_grace_exceeded)?;
1054    Ok(MonitoredProfile {
1055        result,
1056        wall: started.elapsed(),
1057        resources,
1058        target_observed,
1059        cancellation_requested_at,
1060    })
1061}

1063impl ResourceSampler {
1064    fn new(
1065        database: &Path,
1066        sample_interval_ms: u64,
1067        baseline: ProcessResourceBaseline,
1068    ) -> AnyResult<Self> {
1069        let database = database.to_path_buf();
1070        Ok(Self {
1071            peak_storage: storage_footprint(&database)?,
1072            database,
1073            sample_interval_ms,
1074            clock_ticks_per_second: clock_ticks_per_second(),
1075            initial_cpu_ticks: baseline.cpu_ticks,
1076            previous_cpu_ticks: baseline.cpu_ticks,
1077            initial_write_bytes: baseline.write_bytes,
1078            previous_write_bytes: baseline.write_bytes,
1079            samples: 0,
1080            peak_rss_bytes: max_option(baseline.rss_bytes, process_rss_bytes()),
1081            phases: BTreeMap::new(),
1082        })
1083    }
1084
1085    fn sample(&mut self, phase: Option<IndexProgressPhase>) -> AnyResult<()> {
1086        let key = phase_name(phase).to_owned();
1087        let cpu_ticks = process_cpu_ticks();
1088        let write_bytes = process_write_bytes();
1089        let rss = process_rss_bytes();
1090        let footprint = storage_footprint(&self.database)?;
1091        self.samples = self.samples.saturating_add(1);
1092        self.peak_rss_bytes = max_option(self.peak_rss_bytes, rss);
1093        merge_footprint(&mut self.peak_storage, &footprint);
1094
1095        let current = self.phases.entry(key).or_default();
1096        current.samples = current.samples.saturating_add(1);
1097        if let (Some(previous), Some(current_ticks)) = (self.previous_cpu_ticks, cpu_ticks) {
1098            current.cpu_ticks = Some(
1099                current
1100                    .cpu_ticks
1101                    .unwrap_or(0)
1102                    .saturating_add(current_ticks.total().saturating_sub(previous.total())),
1103            );
1104        }
1105        if let (Some(previous), Some(current_bytes)) = (self.previous_write_bytes, write_bytes) {
1106            current.write_bytes = Some(
1107                current
1108                    .write_bytes
1109                    .unwrap_or(0)
1110                    .saturating_add(current_bytes.saturating_sub(previous)),
1111            );
1112        }
1113        current.peak_rss_bytes = max_option(current.peak_rss_bytes, rss);
1114        current.peak_database_bytes = current.peak_database_bytes.max(footprint.database_bytes);
1115        current.peak_wal_bytes = current.peak_wal_bytes.max(footprint.wal_bytes);
1116        current.peak_shm_bytes = current.peak_shm_bytes.max(footprint.shm_bytes);
1117        current.peak_total_storage_bytes =
1118            current.peak_total_storage_bytes.max(footprint.total_bytes);
1119        self.previous_cpu_ticks = cpu_ticks;
1120        self.previous_write_bytes = write_bytes;
1121        Ok(())
1122    }
1123
1124    fn finish(
1125        self,
1126        timeout_triggered: bool,
1127        cancellation_grace_exceeded: bool,
1128    ) -> AnyResult<ResourceReport> {
1129        let user_cpu_ticks = self
1130            .previous_cpu_ticks
1131            .zip(self.initial_cpu_ticks)
1132            .map(|(after, before)| after.user.saturating_sub(before.user));
1133        let system_cpu_ticks = self
1134            .previous_cpu_ticks
1135            .zip(self.initial_cpu_ticks)
1136            .map(|(after, before)| after.system.saturating_sub(before.system));
1137        let cpu_ticks = user_cpu_ticks
1138            .zip(system_cpu_ticks)
1139            .map(|(user, system)| user.saturating_add(system));
1140        let process_write_bytes = self
1141            .previous_write_bytes
1142            .zip(self.initial_write_bytes)
1143            .map(|(after, before)| after.saturating_sub(before));
1144        let by_phase = self
1145            .phases
1146            .into_iter()
1147            .map(|(phase, value)| {
1148                (
1149                    phase,
1150                    PhaseResourceReport {
1151                        samples: value.samples,
1152                        approximate_cpu_ms: ticks_to_milliseconds(
1153                            value.cpu_ticks,
1154                            self.clock_ticks_per_second,
1155                        ),
1156                        approximate_process_write_bytes: value.write_bytes,
1157                        peak_rss_bytes: value.peak_rss_bytes,
1158                        peak_database_bytes: value.peak_database_bytes,
1159                        peak_wal_bytes: value.peak_wal_bytes,
1160                        peak_shm_bytes: value.peak_shm_bytes,
1161                        peak_total_storage_bytes: value.peak_total_storage_bytes,
1162                    },
1163                )
1164            })
1165            .collect();
1166        Ok(ResourceReport {
1167            sample_interval_ms: self.sample_interval_ms,
1168            samples: self.samples,
1169            cpu_ms: ticks_to_milliseconds(cpu_ticks, self.clock_ticks_per_second),
1170            process_user_cpu_ms: ticks_to_milliseconds(user_cpu_ticks, self.clock_ticks_per_second),
1171            process_system_cpu_ms: ticks_to_milliseconds(
1172                system_cpu_ticks,
1173                self.clock_ticks_per_second,
1174            ),
1175            process_write_bytes,
1176            peak_rss_bytes: self.peak_rss_bytes,
1177            peak_storage_footprint: self.peak_storage,
1178            by_phase,
1179            timeout_triggered,
1180            cancellation_grace_exceeded,
1181        })
1182    }
1183}

1237fn logical_index_digest(database: &Path) -> AnyResult<String> {
1238    let connection = open_read_only(database)?;
1239    let mut hasher = blake3::Hasher::new();
1240    for (label, sql) in [
1241        (
1242            "files",
1243            "SELECT path, language, structurally_complete, size_bytes, modified_ns, content_hash, source_token_count, source_tokenizer FROM files ORDER BY path",
1244        ),
1245        (
1246            "chunks",
1247            "SELECT files.path, chunks.content, chunks.start_line, chunks.end_line, chunks.start_byte, chunks.end_byte, chunks.token_count FROM chunks JOIN files ON files.id = chunks.file_id ORDER BY files.path, chunks.start_byte, chunks.end_byte, chunks.id",
1248        ),
1249        (
1250            "symbols",
1251            "SELECT files.path, symbols.name, symbols.kind, symbols.parent, symbols.signature, symbols.start_line, symbols.end_line, symbols.start_byte, symbols.end_byte FROM symbols JOIN files ON files.id = symbols.file_id ORDER BY files.path, symbols.start_byte, symbols.end_byte, symbols.name, symbols.kind",
1252        ),
1253        (
1254            "references",
1255            "SELECT files.path, symbol_refs.name, symbol_refs.kind, symbol_refs.role, symbol_refs.enclosing_symbol, symbol_refs.start_line, symbol_refs.end_line, symbol_refs.start_byte, symbol_refs.end_byte FROM symbol_refs JOIN files ON files.id = symbol_refs.file_id ORDER BY files.path, symbol_refs.start_byte, symbol_refs.end_byte, symbol_refs.name, symbol_refs.role",
1256        ),
1257        (
1258            "imports",
1259            "SELECT files.path, imports.raw_target, imports.resolved_path, imports.line FROM imports JOIN files ON files.id = imports.file_id ORDER BY files.path, imports.line, imports.raw_target, imports.resolved_path",
1260        ),
1261        (
1262            "import_candidates",
1263            "SELECT files.path, imports.raw_target, imports.line, import_candidates.candidate_path FROM import_candidates JOIN imports ON imports.id = import_candidates.import_id JOIN files ON files.id = imports.file_id ORDER BY files.path, imports.line, imports.raw_target, import_candidates.candidate_path",
1264        ),
1265        (
1266            "path_entries",
1267            "SELECT path_entries.path, path_entries.depth, files.path FROM path_entries LEFT JOIN files ON files.id = path_entries.file_id ORDER BY path_entries.path",
1268        ),

1271    }

1275fn update_query_digest(
1276    connection: &Connection,
1277    hasher: &mut blake3::Hasher,
1278    label: &str,
1279    sql: &str,
1280) -> AnyResult<()> {
1281    update_bytes(hasher, label.as_bytes());
1282    let mut statement = connection.prepare(sql)?;
1283    let column_count = statement.column_count();
1284    let mut rows = statement.query([])?;
1285    let mut row_count = 0u64;
1286    while let Some(row) = rows.next()? {
1287        row_count = row_count.saturating_add(1);
1288        hasher.update(b"row\0");
1289        for column in 0..column_count {
1290            match row.get_ref(column)? {
1291                ValueRef::Null => {
1292                    hasher.update(b"null\0");
1293                }
1294                ValueRef::Integer(value) => {
1295                    hasher.update(b"integer\0");
1296                    hasher.update(&value.to_le_bytes());
1297                }
1298                ValueRef::Real(value) => {
1299                    hasher.update(b"real\0");
1300                    hasher.update(&value.to_bits().to_le_bytes());
1301                }
1302                ValueRef::Text(value) => {
1303                    hasher.update(b"text\0");
1304                    update_bytes(hasher, value);
1305                }
1306                ValueRef::Blob(value) => {
1307                    hasher.update(b"blob\0");
1308                    update_bytes(hasher, value);
1309                }
1310            }
1311        }
1312    }
1313    hasher.update(&row_count.to_le_bytes());
1314    Ok(())
1315}

1317fn retrieval_digest(storage: &Storage, queries: &[String]) -> AnyResult<String> {
1318    let mut hasher = blake3::Hasher::new();
1319    for query in queries {
1320        update_bytes(&mut hasher, query.as_bytes());
1321        for hit in storage.search_word(query, 100)? {
1322            hasher.update(b"word\0");
1323            update_chunk_hit(&mut hasher, &hit);
1324        }
1325        for hit in storage.search_trigram(query, 100)? {
1326            hasher.update(b"trigram\0");
1327            update_chunk_hit(&mut hasher, &hit);
1328        }
1329        for hit in storage.search_symbols(query, false, 100)? {
1330            hasher.update(b"symbol\0");
1331            update_bytes(&mut hasher, hit.path.as_bytes());
1332            update_bytes(&mut hasher, hit.content_hash.as_bytes());
1333            update_bytes(&mut hasher, hit.symbol.name.as_bytes());
1334            update_bytes(&mut hasher, hit.symbol.kind.as_bytes());
1335            update_optional_string(&mut hasher, hit.symbol.parent.as_deref());
1336            update_optional_string(&mut hasher, hit.symbol.signature.as_deref());
1337            update_usize(&mut hasher, hit.symbol.start_line);
1338            update_usize(&mut hasher, hit.symbol.end_line);
1339            update_usize(&mut hasher, hit.symbol.start_byte);
1340            update_usize(&mut hasher, hit.symbol.end_byte);
1341        }
1342        for hit in storage.search_references(query, false, 100)? {
1343            hasher.update(b"reference\0");
1344            update_bytes(&mut hasher, hit.path.as_bytes());
1345            update_bytes(&mut hasher, hit.content_hash.as_bytes());
1346            update_bytes(&mut hasher, hit.reference.name.as_bytes());
1347            update_bytes(&mut hasher, hit.reference.kind.as_bytes());
1348            update_bytes(
1349                &mut hasher,
1350                serde_json::to_string(&hit.reference.role)?.as_bytes(),
1351            );
1352            update_optional_string(&mut hasher, hit.reference.enclosing_symbol.as_deref());
1353            update_usize(&mut hasher, hit.reference.start_line);
1354            update_usize(&mut hasher, hit.reference.end_line);
1355            update_usize(&mut hasher, hit.reference.start_byte);
1356            update_usize(&mut hasher, hit.reference.end_byte);
1357        }
1358    }
1359    Ok(hasher.finalize().to_hex().to_string())
1360}

1362fn update_chunk_hit(hasher: &mut blake3::Hasher, hit: &leantoken::storage::ChunkHit) {
1363    update_bytes(hasher, hit.path.as_bytes());
1364    update_bytes(hasher, hit.content.as_bytes());
1365    update_usize(hasher, hit.start_line);
1366    update_usize(hasher, hit.end_line);
1367    update_usize(hasher, hit.start_byte);
1368    update_usize(hasher, hit.end_byte);
1369    update_usize(hasher, hit.token_count);
1370    hasher.update(&hit.score.to_bits().to_le_bytes());
1371}

1373fn update_optional_string(hasher: &mut blake3::Hasher, value: Option<&str>) {
1374    if let Some(value) = value {
1375        hasher.update(b"some\0");
1376        update_bytes(hasher, value.as_bytes());
1377    } else {
1378        hasher.update(b"none\0");
1379    }
1380}

1382fn update_usize(hasher: &mut blake3::Hasher, value: usize) {

1388    hasher.update(value);

1398fn require_parity(parity: &mut Option<ParityReport>, run: &ColdRunReport) -> AnyResult<()> {
1399    if run.response.repository_generation == 0 {
1400        return Err(invalid_input(
1401            "cold run did not publish a complete generation",
1402        ));
1403    }
1404    let observed = ParityReport {
1405        logical_index_blake3: run.logical_index_blake3.clone(),
1406        retrieval_blake3: run.retrieval_blake3.clone(),
1407        shape: run.shape.clone(),
1408        complete: true,
1409    };
1410    if let Some(expected) = parity {
1411        if expected.logical_index_blake3 != observed.logical_index_blake3
1412            || expected.retrieval_blake3 != observed.retrieval_blake3
1413            || expected.shape != observed.shape
1414        {
1415            return Err(invalid_input(
1416                "worker matrix changed logical index or retrieval output",
1417            ));
1418        }
1419    } else {
1420        *parity = Some(observed);
1421    }
1422    Ok(())
1423}

1425fn measure_cancellation_probes(
1426    corpus: &PreparedColdCorpus,
1427    run_root: &Path,
1428    policy: &MeasurementPolicy,
1429    parity: &ParityReport,
1430) -> AnyResult<Vec<CancellationProbeReport>> {
1431    let workers = policy.worker_order.iter().copied().max().unwrap_or(1);
1432    let baseline = run_root.join("parity-baseline.json");
1433    fs::write(&baseline, serde_json::to_vec_pretty(parity)?)?;
1434    let mut reports = Vec::with_capacity(policy.cancellation_phases.len());
1435    for (sequence, phase) in policy.cancellation_phases.iter().copied().enumerate() {
1436        let database = run_root.join(format!(
1437            "cancel-{sequence:02}-{}.sqlite",
1438            phase_name(Some(phase))
1439        ));
1440        let output = run_root.join(format!(
1441            "cancel-{sequence:02}-{}.json",
1442            phase_name(Some(phase))
1443        ));
1444        let worker = ColdWorkerArgs {
1445            repository: corpus.root.clone(),
1446            database,
1447            sequence,
1448            workers,
1449            parity_queries: policy.parity_queries.clone(),
1450            sample_interval_ms: policy.sample_interval_ms,
1451            timeout_seconds: policy.timeout_seconds,
1452            target_phase: Some(phase_name(Some(phase)).into()),
1453            baseline: Some(baseline.clone()),
1454            allow_missed_phase: !policy.require_cancellation_observation,
1455            output,
1456        };
1457        let one_attempt = policy
1458            .timeout_seconds
1459            .saturating_add(policy.cancellation_grace_seconds);
1460        let maximum_wall = Duration::from_secs(one_attempt.saturating_mul(2).saturating_add(30));

1462            ColdWorkerOutput::Cancellation(report) => reports.push(*report),

1464                return Err(invalid_input(

1466                ));
1467            }
1468        }
1469    }
1470    Ok(reports)
1471}

1484fn measure_cancellation_probe_in_process(
1485    phase: IndexProgressPhase,
1486    input: CancellationProbeInput<'_>,
1487) -> AnyResult<CancellationProbeReport> {
1488    let CancellationProbeInput {
1489        workers,
1490        repository,
1491        database,
1492        parity_queries,
1493        sample_interval_ms,
1494        timeout_seconds,
1495        require_observation,
1496        parity,
1497    } = input;
1498    let started = Instant::now();
1499    let baseline = ProcessResourceBaseline::capture();
1500    let mut config = Config::discover(repository, Some(database.to_path_buf()))?;
1501    config.max_index_workers = workers;
1502    let storage = Storage::open(database)?;
1503    let indexer = Indexer::new(std::sync::Arc::new(config), storage.clone())?;
1504    let cancellation = CancellationToken::new();
1505    let monitored = run_profiled_monitored(
1506        &indexer,
1507        database,
1508        &cancellation,
1509        MonitorSettings {
1510            started,
1511            baseline,
1512            target: Some(MonitoredTarget::Phase(phase)),
1513            sample_interval_ms,
1514            timeout_seconds,
1515        },
1516    )?;
1517    if monitored.resources.timeout_triggered {
1518        return Err(invalid_input(
1519            "cancellation probe exceeded --timeout-seconds",
1520        ));
1521    }
1522    if monitored.resources.cancellation_grace_exceeded {
1523        return Err(invalid_input(
1524            "cancellation probe exceeded the cancellation grace period",
1525        ));
1526    }
1527    if require_observation && !monitored.target_observed {
1528        return Err(invalid_input(
1529            "requested cancellation phase was too short to observe; reduce --sample-interval-ms or report the incomplete probe explicitly",
1530        ));
1531    }
1532
1533    let cancellation_to_return_ms = monitored
1534        .cancellation_requested_at
1535        .map(|requested| requested.elapsed().as_secs_f64() * 1_000.0);
1536    let result = match &monitored.result {
1537        Ok(_) if monitored.cancellation_requested_at.is_some() => "completed_after_cancellation",
1538        Ok(_) => "completed_without_observation",
1539        Err(leantoken::Error::Cancelled) => "cancelled",
1540        Err(error) => {
1541            return Err(invalid_input(&format!(
1542                "cancellation probe failed unexpectedly: {error}"
1543            )));
1544        }
1545    }
1546    .to_owned();
1547    let generation_after_attempt = storage.repository_generation()?;
1548    let footprint_after_attempt = storage_footprint(database)?;
1549
1550    let restart_started = Instant::now();
1551    let restart_baseline = ProcessResourceBaseline::capture();
1552    let restart_cancellation = CancellationToken::new();
1553    let restart = run_profiled_monitored(
1554        &indexer,
1555        database,
1556        &restart_cancellation,
1557        MonitorSettings {
1558            started: restart_started,
1559            baseline: restart_baseline,
1560            target: None,
1561            sample_interval_ms,
1562            timeout_seconds,
1563        },
1564    )?;
1565    if restart.resources.timeout_triggered {
1566        return Err(invalid_input(
1567            "restart after cancellation exceeded --timeout-seconds",
1568        ));
1569    }
1570    if restart.resources.cancellation_grace_exceeded {
1571        return Err(invalid_input(
1572            "restart after cancellation exceeded the cancellation grace period",
1573        ));
1574    }
1575    restart.result?;
1576    let restart_generation = storage.repository_generation()?;
1577    let restart_logical_index_blake3 = logical_index_digest(database)?;
1578    let restart_retrieval_blake3 = retrieval_digest(&storage, parity_queries)?;
1579    let restart_matches_baseline = restart_generation > 0
1580        && restart_logical_index_blake3 == parity.logical_index_blake3
1581        && restart_retrieval_blake3 == parity.retrieval_blake3
1582        && index_shape(&storage, database)? == parity.shape;
1583    if !restart_matches_baseline {
1584        return Err(invalid_input(
1585            "restart after cancellation changed logical index or retrieval output",
1586        ));
1587    }
1588
1589    Ok(CancellationProbeReport {
1590        target_phase: phase,
1591        workers,
1592        phase_observed: monitored.target_observed,
1593        cancellation_requested: monitored.cancellation_requested_at.is_some(),
1594        cancellation_to_return_ms,
1595        result,
1596        attempt_wall_ms: monitored.wall.as_secs_f64() * 1_000.0,
1597        resources: monitored.resources,
1598        generation_after_attempt,
1599        footprint_after_attempt,
1600        restart_wall_ms: restart.wall.as_secs_f64() * 1_000.0,
1601        restart_generation,
1602        restart_logical_index_blake3,
1603        restart_retrieval_blake3,
1604        restart_matches_baseline,
1605    })
1606}

1608fn summarize_workers(runs: &[ColdRunReport]) -> Vec<WorkerSummary> {
1609    let workers = runs.iter().map(|run| run.workers).collect::<BTreeSet<_>>();
1610    workers
1611        .into_iter()
1612        .map(|workers| {
1613            let matching = runs
1614                .iter()
1615                .filter(|run| run.workers == workers)
1616                .collect::<Vec<_>>();
1617            let mut wall = matching.iter().map(|run| run.wall_ms).collect::<Vec<_>>();
1618            wall.sort_by(f64::total_cmp);
1619            WorkerSummary {
1620                workers,
1621                samples: matching.len(),
1622                wall_p50_ms: percentile(&wall, 0.50),
1623                wall_p95_ms: percentile(&wall, 0.95),
1624                mean_cpu_ms: mean_option(matching.iter().map(|run| run.resources.cpu_ms)),
1625                peak_rss_bytes: matching
1626                    .iter()
1627                    .filter_map(|run| run.resources.peak_rss_bytes)
1628                    .max(),
1629                mean_process_write_bytes: mean_option(
1630                    matching
1631                        .iter()
1632                        .map(|run| run.resources.process_write_bytes.map(|value| value as f64)),
1633                ),
1634                max_final_storage_bytes: matching
1635                    .iter()
1636                    .map(|run| run.final_storage_footprint.total_bytes)
1637                    .max()
1638                    .unwrap_or(0),
1639            }
1640        })

1644fn decide(
1645    runs: &[ColdRunReport],
1646    summaries: &[WorkerSummary],
1647    policy: &MeasurementPolicy,
1648) -> DecisionReport {
1649    let baseline_workers = 1;
1650    let baseline_runs = runs
1651        .iter()
1652        .filter(|run| run.workers == baseline_workers)
1653        .collect::<Vec<_>>();
1654    let phase_totals = diagnostic_phase_totals(&baseline_runs);
1655    let total_ms = phase_totals.values().copied().sum::<f64>();
1656    let (dominant_phase, dominant_ms) = phase_totals
1657        .iter()
1658        .max_by(|left, right| left.1.total_cmp(right.1))
1659        .map(|(phase, milliseconds)| (phase.clone(), *milliseconds))
1660        .unwrap_or_else(|| ("unavailable".into(), 0.0));
1661    let dominant_phase_share = if total_ms > 0.0 {
1662        dominant_ms / total_ms
1663    } else {
1664        0.0
1665    };
1666    let baseline = summaries
1667        .iter()
1668        .find(|summary| summary.workers == baseline_workers);
1669    let mut comparisons = Vec::new();
1670    if let Some(baseline) = baseline {
1671        for candidate in summaries
1672            .iter()
1673            .filter(|candidate| candidate.workers != baseline_workers)
1674        {
1675            let wall_reduction = relative_reduction(baseline.wall_p50_ms, candidate.wall_p50_ms);
1676            let wall_p95_reduction =
1677                relative_reduction(baseline.wall_p95_ms, candidate.wall_p95_ms);
1678            let cpu_increase =
1679                relative_increase_option(baseline.mean_cpu_ms, candidate.mean_cpu_ms);
1680            let peak_rss_increase = relative_increase_option(
1681                baseline.peak_rss_bytes.map(|value| value as f64),
1682                candidate.peak_rss_bytes.map(|value| value as f64),
1683            );
1684            let write_increase = relative_increase_option(
1685                baseline.mean_process_write_bytes,
1686                candidate.mean_process_write_bytes,
1687            );
1688            let footprint_increase = relative_increase(
1689                baseline.max_final_storage_bytes as f64,
1690                candidate.max_final_storage_bytes as f64,
1691            );
1692            let passes = wall_reduction >= policy.minimum_wall_reduction
1693                && policy
1694                    .minimum_wall_p95_reduction
1695                    .is_none_or(|minimum| wall_p95_reduction >= minimum)
1696                && cpu_increase.is_some_and(|value| value <= policy.maximum_cpu_increase)
1697                && peak_rss_increase.is_some_and(|value| value <= policy.maximum_peak_rss_increase)
1698                && write_increase.is_some_and(|value| value <= policy.maximum_write_increase)
1699                && footprint_increase <= policy.maximum_footprint_increase;
1700            comparisons.push(WorkerComparison {
1701                workers: candidate.workers,
1702                wall_reduction,
1703                wall_p95_reduction,
1704                cpu_increase,
1705                peak_rss_increase,
1706                write_increase,
1707                footprint_increase,
1708                passes,
1709            });
1710        }
1711    }
1712    let preparation_owns_cost = dominant_phase == "preparation"
1713        && dominant_phase_share >= policy.preparation_owner_threshold;
1714    let sample_counts_complete = summaries
1715        .iter()
1716        .all(|summary| summary.samples >= policy.minimum_samples_per_worker);
1717    let candidate_workers = (preparation_owns_cost && sample_counts_complete)
1718        .then(|| {
1719            comparisons
1720                .iter()
1721                .filter(|comparison| comparison.passes)
1722                .max_by(|left, right| left.wall_reduction.total_cmp(&right.wall_reduction))
1723                .map(|comparison| comparison.workers)
1724        })
1725        .flatten();
1726    let (outcome, rationale) = if baseline.is_none() {
1727        (
1728            "insufficient_evidence",
1729            "The matrix did not contain a one-worker baseline.".into(),
1730        )
1731    } else if !sample_counts_complete {
1732        (
1733            "insufficient_evidence",
1734            format!(
1735                "Every worker arm requires at least {} samples before this matrix can make a decision.",
1736                policy.minimum_samples_per_worker
1737            ),
1738        )
1739    } else if !preparation_owns_cost {
1740        (
1741            "optimize_measured_owner",
1742            format!(
1743                "{dominant_phase} owns {:.1}% of measured leaf-phase time, so changing preparation workers is not justified.",
1744                dominant_phase_share * 100.0
1745            ),
1746        )
1747    } else if let Some(workers) = candidate_workers {
1748        match policy.matrix_kind {
1749            ColdMatrixKind::Screening => (
1750                "candidate_for_follow_up",
1751                format!(
1752                    "{workers} workers passed the preregistered wall, CPU, RSS, write, and footprint thresholds; this measurement PR does not change production defaults."
1753                ),
1754            ),
1755            ColdMatrixKind::TwoWorkerFollowUp => (
1756                "candidate_for_mcp_contention_measurement",
1757                format!(
1758                    "{workers} workers passed the preregistered p50, p95, CPU, RSS, write, and footprint thresholds; production defaults remain unchanged until MCP contention and multi-process evidence also passes."
1759                ),
1760            ),
1761        }
1762    } else {
1763        (
1764            "keep_current_worker_default",
1765            "No candidate passed every preregistered resource and wall-time threshold.".into(),
1766        )
1767    };
1768    DecisionReport {
1769        baseline_workers,
1770        dominant_phase,
1771        dominant_phase_share,
1772        candidate_workers,
1773        outcome,
1774        rationale,
1775        comparisons,
1776    }
1777}

1779fn diagnostic_phase_totals(runs: &[&ColdRunReport]) -> BTreeMap<String, f64> {
1780    let mut totals = BTreeMap::new();
1781    for run in runs {
1782        let publication = &run.diagnostics.publication_detail;
1783        for (phase, milliseconds) in [
1784            ("discovery", run.diagnostics.discovery_ms),
1785            ("hash_and_plan", run.diagnostics.hash_and_plan_ms),
1786            ("preparation", run.diagnostics.preparation_ms),
1787            ("relational_write", publication.relational_write_ms),

1863fn current_executable_blake3() -> Option<String> {
1864    let executable = std::env::current_exe().ok()?;
1865    let mut file = fs::File::open(executable).ok()?;
1866    let mut hasher = blake3::Hasher::new();
1867    let mut buffer = [0u8; 64 * 1024];
1868    loop {
1869        let read = file.read(&mut buffer).ok()?;
1870        if read == 0 {
1871            break;
1872        }
1873        hasher.update(&buffer[..read]);
1874    }
1875    Some(hasher.finalize().to_hex().to_string())
1876}

1902fn process_cpu_ticks() -> Option<ProcessCpuTicks> {
1903    let stat = fs::read_to_string("/proc/self/stat").ok()?;
1904    let after_command = stat.rsplit_once(')')?.1;
1905    let fields = after_command.split_ascii_whitespace().collect::<Vec<_>>();
1906    let user = fields.get(11)?.parse::<u64>().ok()?;
1907    let system = fields.get(12)?.parse::<u64>().ok()?;
1908    Some(ProcessCpuTicks { user, system })
1909}

1917        .split_ascii_whitespace()
1918        .next()?
1919        .parse::<u64>()
1920        .ok()?;
1921    kibibytes.checked_mul(1024)
1922}
