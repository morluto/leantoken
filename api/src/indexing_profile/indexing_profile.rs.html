192fn main() -> AnyResult<()> {
193    let args = Args::parse();
194    match &args.command {
195        Some(ProfileCommand::ColdMatrix(cold_args)) => return cold::run(cold_args),
196        Some(ProfileCommand::ColdWorker(worker_args)) => {
197            return cold::run_worker(worker_args);
198        }
199        None => {}
200    }
201    validate_args(&args)?;
202    let report = run_profile(&args)?;
203    let json = serde_json::to_string_pretty(&report)?;
204    if let Some(parent) = args.output.parent() {
205        fs::create_dir_all(parent)?;
206    }
207    fs::write(&args.output, format!("{json}\n"))?;
208    println!("{json}");
209    Ok(())
210}

212fn validate_args(args: &Args) -> AnyResult<()> {
213    if args.repository.is_none() && args.files < 2 {
214        return Err(invalid_input("--files must be at least 2"));
215    }
216    if args.repository.is_none() && args.file_bytes < 128 {
217        return Err(invalid_input("--file-bytes must be at least 128"));
218    }
219    if args
220        .repository
221        .as_ref()
222        .is_some_and(|repository| !repository.is_dir())
223    {
224        return Err(invalid_input("--repository must name a Git checkout"));
225    }
226    if args.iterations == 0 {
227        return Err(invalid_input("--iterations must be positive"));
228    }
229    if args.read_samples == 0 {
230        return Err(invalid_input("--read-samples must be positive"));
231    }
232    if args.hot_set == 0 {
233        return Err(invalid_input("--hot-set must be positive"));
234    }
235    if args.watcher_debounce_ms == 0 {
236        return Err(invalid_input("--watcher-debounce-ms must be positive"));
237    }
238    Ok(())
239}

245fn run_profile(args: &Args) -> AnyResult<Report> {
246    let corpus = prepare_corpus(args)?;
247    let ignore_path = corpus.root.join(".gitignore");
248    if ignore_path.exists() {
249        if !fs::symlink_metadata(&ignore_path)?.file_type().is_file() {
250            return Err(invalid_input("profile .gitignore must be a regular file"));
251        }
252    } else {
253        fs::write(&ignore_path, "# LeanToken indexing profile\n")?;
254    }
255    let database = tempfile::tempdir()?;
256    let config = Arc::new(Config::discover(
257        &corpus.root,
258        Some(database.path().join("index.sqlite")),
259    )?);
260    let discovered = discover_files(&corpus.root, config.max_file_bytes)?;
261    if discovered.len() < 2 {
262        return Err(invalid_input(
263            "profile corpus must contain at least two ignore-visible files within max_file_bytes",
264        ));
265    }
266    let paths = discovered
267        .iter()
268        .map(|file| file.absolute_path.clone())
269        .collect::<Vec<_>>();
270    let mutation_files = discovered
271        .iter()
272        .filter(|file| {
273            Path::new(&file.relative_path)
274                .file_name()
275                .is_some_and(|name| name != ".gitignore" && name != ".ignore")
276                && fs::read_to_string(&file.absolute_path).is_ok()
277        })
278        .take(2)
279        .collect::<Vec<_>>();
280    if mutation_files.len() < 2 {
281        return Err(invalid_input(
282            "profile corpus must contain at least two UTF-8 files for mutation measurements",
283        ));
284    }
285    let storage = Storage::open(&config.database_path)?;
286    let indexer = Indexer::new(Arc::clone(&config), storage.clone())?;
287
288    let start = Instant::now();
289    let initial_profile = indexer.reconcile_profiled_report(false)?;
290    let initial_index = IndexSample {
291        elapsed_ms: milliseconds(start.elapsed()),
292        response: initial_profile.report,
293        diagnostics: initial_profile.diagnostics,
294        storage_footprint: storage_footprint(&config.database_path)?,
295    };
296
297    let mut full_noop_durations = Vec::with_capacity(args.iterations);
298    let mut full_noop_diagnostics = Vec::with_capacity(args.iterations);
299    for _ in 0..args.iterations {
300        let start = Instant::now();
301        let profiled = indexer.reconcile_profiled(false)?;
302        full_noop_durations.push(start.elapsed());
303        require_index_counts(&profiled.response, discovered.len(), 0, "full no-op")?;
304        full_noop_diagnostics.push(profiled.diagnostics);
305    }
306    let full_noop_phases = IndexingPhaseMeasurement::from_diagnostics(&full_noop_diagnostics);
307
308    let full_changed = measure_changed_indexing(
309        args.iterations,
310        &mutation_files[0].absolute_path,
311        || indexer.reconcile(false),
312        discovered.len(),
313        "full changed",
314    )?;
315    let targeted_changed = measure_changed_indexing(
316        args.iterations,
317        &mutation_files[1].absolute_path,
318        || indexer.reconcile_paths(&[mutation_files[1].relative_path.clone()]),
319        1,
320        "targeted changed",
321    )?;
322    let create_path = corpus.root.join("leantoken_profile_created.rs");
323    let create_relative = "leantoken_profile_created.rs".to_string();
324    if create_path.exists() {
325        return Err(invalid_input(
326            "profile corpus already contains leantoken_profile_created.rs",
327        ));
328    }
329    let create_delta = measure_lifecycle_indexing(
330        args.iterations,
331        |iteration| fs::write(&create_path, synthetic_source(iteration, 256)),
332        || indexer.reconcile_paths(std::slice::from_ref(&create_relative)),
333        || {
334            fs::remove_file(&create_path)?;
335            indexer.reconcile_paths(std::slice::from_ref(&create_relative))?;
336            Ok(())
337        },
338        MinimumIndexCounts {
339            seen: 1,
340            indexed: 1,
341            removed: 0,
342        },
343        "create delta",
344    )?;
345
346    let delete_path = mutation_files[0].absolute_path.clone();
347    let delete_relative = mutation_files[0].relative_path.clone();
348    let delete_content = fs::read(&delete_path)?;
349    let delete_targeted = measure_lifecycle_indexing(
350        args.iterations,
351        |_| fs::remove_file(&delete_path),
352        || indexer.reconcile_paths(std::slice::from_ref(&delete_relative)),
353        || {
354            fs::write(&delete_path, &delete_content)?;
355            indexer.reconcile(false)?;
356            Ok(())
357        },
358        MinimumIndexCounts {
359            seen: 1,
360            indexed: 0,
361            removed: 1,
362        },
363        "targeted delete",
364    )?;
365
366    let rename_path = mutation_files[1].absolute_path.clone();
367    let rename_relative = mutation_files[1].relative_path.clone();
368    let rename_destination = rename_path.with_file_name("leantoken_profile_renamed.rs");
369    if rename_destination.exists() {
370        return Err(invalid_input(
371            "profile corpus already contains leantoken_profile_renamed.rs",
372        ));
373    }
374    let rename_destination_relative = Path::new(&rename_relative)
375        .with_file_name("leantoken_profile_renamed.rs")
376        .to_string_lossy()
377        .replace('\\', "/");
378    let rename_paths = vec![rename_relative.clone(), rename_destination_relative.clone()];
379    let rename_delta = measure_lifecycle_indexing(
380        args.iterations,
381        |_| fs::rename(&rename_path, &rename_destination),
382        || indexer.reconcile_paths(&rename_paths),
383        || {
384            fs::rename(&rename_destination, &rename_path)?;
385            indexer.reconcile(false)?;
386            Ok(())
387        },
388        MinimumIndexCounts {
389            seen: 2,
390            indexed: 1,
391            removed: 1,
392        },
393        "rename delta",
394    )?;
395
396    let directory_rename_delta = select_directory_mutation(&corpus.root, &discovered)
397        .map(|mutation| measure_directory_rename(args.iterations, &indexer, mutation))
398        .transpose()?;
399
400    let ignore_relative = ".gitignore".to_string();
401    let ignore_change_delta = measure_lifecycle_indexing(
402        args.iterations,
403        |iteration| {
404            let mut file = OpenOptions::new().append(true).open(&ignore_path)?;
405            writeln!(file, "# profile ignore mutation {iteration}")
406        },
407        || indexer.reconcile_paths(std::slice::from_ref(&ignore_relative)),
408        || Ok(()),
409        MinimumIndexCounts {
410            seen: 1,
411            indexed: 1,
412            removed: 0,
413        },
414        "ignore-change delta",
415    )?;
416
417    let ignore_visibility_delta = measure_ignore_visibility(
418        args.iterations,
419        &indexer,
420        &corpus.root,
421        &mutation_files[0].relative_path,
422    )?;
423    let watcher_modify_delivery = measure_watcher_delivery(
424        args.iterations,
425        &indexer,
426        &config,
427        &mutation_files[1].absolute_path,
428        &mutation_files[1].relative_path,
429        Duration::from_millis(args.watcher_debounce_ms),
430    )?;
431
432    let hot_set = args.hot_set.min(paths.len());
433    let hot_paths = &paths[..hot_set];
434    let hot_contents = hot_paths
435        .iter()
436        .map(fs::read)
437        .collect::<Result<Vec<_>, _>>()?;
438    for path in hot_paths {
439        black_box(fs::read(path)?);
440    }
441
442    let warm_hot_file_reads = measure_file_reads(hot_paths, args.read_samples)?;
443    let warm_spread_file_reads = measure_file_reads(&paths, args.read_samples)?;
444    let memory_hot_file_copies = measure_memory_copies(&hot_contents, args.read_samples);
445    let mut unpooled_session_durations = Vec::with_capacity(args.read_samples);
446    for _ in 0..args.read_samples {
447        let start = Instant::now();
448        let connection = rusqlite::Connection::open_with_flags(
449            &config.database_path,
450            rusqlite::OpenFlags::SQLITE_OPEN_READ_ONLY | rusqlite::OpenFlags::SQLITE_OPEN_NO_MUTEX,
451        )?;
452        connection.busy_timeout(Duration::from_secs(5))?;
453        connection.execute_batch("BEGIN DEFERRED")?;
454        black_box(connection.query_row(
455            "SELECT repository_generation FROM meta WHERE id = 1",
456            [],
457            |row| row.get::<_, i64>(0),
458        )?);
459        connection.execute_batch("ROLLBACK")?;
460        unpooled_session_durations.push(start.elapsed());
461    }
462    let mut pooled_session_durations = Vec::with_capacity(args.read_samples);
463    for _ in 0..args.read_samples {
464        let start = Instant::now();
465        let session = storage.begin_read()?;
466        black_box(session.repository_generation()?);
467        pooled_session_durations.push(start.elapsed());
468    }
469    let session = storage.begin_read()?;
470    let mut pinned_query_durations = Vec::with_capacity(args.read_samples);
471    for _ in 0..args.read_samples {
472        let start = Instant::now();
473        black_box(session.repository_generation()?);
474        pinned_query_durations.push(start.elapsed());
475    }
476
477    let (leantoken_git_revision, leantoken_worktree_dirty) = leantoken_source_identity();
478    Ok(Report {
479        schema_version: 8,
480        leantoken_version: env!("CARGO_PKG_VERSION"),
481        leantoken_git_revision,
482        leantoken_worktree_dirty,
483        host_os: std::env::consts::OS,
484        host_arch: std::env::consts::ARCH,
485        release_build: !cfg!(debug_assertions),
486        corpus: corpus_report(args, &corpus, &discovered, hot_set),
487        initial_index,
488        full_noop: IndexMeasurement {
489            timing: TimingStats::from_durations(full_noop_durations),
490            files_seen_per_sample: discovered.len(),
491            files_indexed_per_sample: 0,
492            files_removed_per_sample: 0,
493        },
494        full_noop_phases,
495        full_changed,
496        targeted_changed,
497        create_delta,
498        delete_targeted,
499        rename_delta,
500        directory_rename_delta,
501        ignore_change_delta,
502        ignore_visibility_delta,
503        watcher_modify_delivery,
504        warm_hot_file_reads,
505        warm_spread_file_reads,
506        memory_hot_file_copies,
507        unpooled_read_session_open_and_generation: TimingStats::from_durations(
508            unpooled_session_durations,
509        ),
510        pooled_read_session_checkout_and_generation: TimingStats::from_durations(
511            pooled_session_durations,
512        ),
513        pinned_generation_query: TimingStats::from_durations(pinned_query_durations),
514        final_storage_footprint: storage_footprint(&config.database_path)?,
515        limitations: vec![
516            corpus.limitation.to_string(),
517            "File-read measurements use the operating system's warm page cache; they do not represent cold, remote, encrypted, or heavily contended filesystems.".into(),
518            "The in-memory comparison copies bytes but excludes cache lookup, eviction, synchronization, invalidation, and memory-pressure costs.".into(),
519            "Read-session measurements compare opening a read-only SQLite connection, checking out an established per-service pool connection, and a generation query on one already pinned session; they do not simulate concurrent process contention.".into(),
520            "Lifecycle reconciliation measurements invoke changed paths directly. watcher_modify_delivery separately includes the configured debounce and the host notify backend, but excludes reconciliation work.".into(),
521            "Watcher-overflow and interrupted reconciliation use deterministic integration tests because operating-system overflow cannot be triggered portably on demand.".into(),
522            "Timing is machine-specific. Compare runs only on the same host and build profile, and use release builds for decisions.".into(),
523        ],
524    })
525}

527fn storage_footprint(database_path: &Path) -> io::Result<StorageFootprint> {
528    let database_bytes = logical_file_bytes(database_path)?;
529    let wal_bytes = logical_file_bytes(&path_with_suffix(database_path, "-wal"))?;
530    let shm_bytes = logical_file_bytes(&path_with_suffix(database_path, "-shm"))?;
531    Ok(StorageFootprint {
532        database_bytes,
533        wal_bytes,
534        shm_bytes,
535        total_bytes: database_bytes + wal_bytes + shm_bytes,
536    })
537}
538
539fn logical_file_bytes(path: &Path) -> io::Result<u64> {
540    match fs::metadata(path) {
541        Ok(metadata) => Ok(metadata.len()),
542        Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(0),
543        Err(error) => Err(error),
544    }
545}

553struct DirectoryMutation {
554    source: PathBuf,
555    source_relative: String,
556    destination: PathBuf,
557    destination_relative: String,
558    affected_files: usize,
559}

561fn select_directory_mutation(root: &Path, files: &[DiscoveredFile]) -> Option<DirectoryMutation> {
562    const TARGET_FILES: usize = 32;
563
564    let mut counts = BTreeMap::<String, usize>::new();
565    for file in files {
566        let mut parent = Path::new(&file.relative_path).parent();
567        while let Some(directory) = parent.filter(|directory| !directory.as_os_str().is_empty()) {
568            let relative = directory.to_string_lossy().replace('\\', "/");
569            *counts.entry(relative).or_default() += 1;
570            parent = directory.parent();
571        }
572    }
573    let (source_relative, affected_files) = counts
574        .into_iter()
575        .filter(|(_, count)| *count >= 2)
576        .min_by(|left, right| {
577            left.1
578                .abs_diff(TARGET_FILES)
579                .cmp(&right.1.abs_diff(TARGET_FILES))
580                .then_with(|| left.0.cmp(&right.0))
581        })?;
582    let source = root.join(Path::new(&source_relative));
583    let file_name = source.file_name()?.to_string_lossy();
584    let destination = source.with_file_name(format!("{file_name}_leantoken_profile_renamed"));
585    let destination_relative = destination
586        .strip_prefix(root)
587        .ok()?
588        .to_string_lossy()
589        .replace('\\', "/");
590    Some(DirectoryMutation {
591        source,
592        source_relative,
593        destination,
594        destination_relative,
595        affected_files,
596    })
597}

599fn measure_directory_rename(
600    iterations: usize,
601    indexer: &Indexer,
602    mutation: DirectoryMutation,
603) -> AnyResult<DirectoryRenameMeasurement> {
604    if mutation.destination.exists() {
605        return Err(invalid_input(
606            "profile corpus already contains the directory rename destination",
607        ));
608    }
609    let paths = vec![
610        mutation.source_relative.clone(),
611        mutation.destination_relative.clone(),
612    ];
613    let mut durations = Vec::with_capacity(iterations);
614    let mut expected = None;
615    for _ in 0..iterations {
616        fs::rename(&mutation.source, &mutation.destination)?;
617        let start = Instant::now();
618        let response = indexer.reconcile_paths(&paths)?;
619        durations.push(start.elapsed());
620        if response.files_indexed < mutation.affected_files
621            || response.files_removed < mutation.affected_files
622        {
623            return Err(Box::new(io::Error::other(format!(
624                "directory rename expected at least {} indexed and removed files; got indexed={}, removed={}",
625                mutation.affected_files, response.files_indexed, response.files_removed
626            ))));
627        }
628        require_stable_counts(&mut expected, &response, "directory rename")?;
629
630        fs::rename(&mutation.destination, &mutation.source)?;
631        indexer.reconcile_paths(&paths)?;
632    }
633    let expected = expected.expect("positive iteration count is validated");
634    Ok(DirectoryRenameMeasurement {

636        indexing: IndexMeasurement {
637            timing: TimingStats::from_durations(durations),
638            files_seen_per_sample: expected.seen,
639            files_indexed_per_sample: expected.indexed,
640            files_removed_per_sample: expected.removed,
641        },
642    })

645fn measure_ignore_visibility(
646    iterations: usize,
647    indexer: &Indexer,
648    root: &Path,
649    target_relative: &str,
650) -> AnyResult<IndexMeasurement> {
651    let ignore_path = root.join(".leantokenignore");
652    let original = match fs::read(&ignore_path) {
653        Ok(contents) => Some(contents),
654        Err(error) if error.kind() == io::ErrorKind::NotFound => None,
655        Err(error) => return Err(error.into()),
656    };
657    let ignore_relative = ".leantokenignore".to_string();
658    let mut durations = Vec::with_capacity(iterations);
659    let mut expected = None;
660    for iteration in 0..iterations {
661        let mut contents = original.clone().unwrap_or_default();
662        if !contents.is_empty() && !contents.ends_with(b"\n") {
663            contents.push(b'\n');
664        }
665        contents.extend_from_slice(
666            format!("# profile visibility mutation {iteration}\n/{target_relative}\n").as_bytes(),
667        );
668        fs::write(&ignore_path, contents)?;
669
670        let start = Instant::now();
671        let response = indexer.reconcile_paths(std::slice::from_ref(&ignore_relative))?;
672        durations.push(start.elapsed());
673        if response.files_removed == 0 {
674            return Err(Box::new(io::Error::other(
675                "semantic ignore change did not remove its target",
676            )));
677        }
678        require_stable_counts(&mut expected, &response, "semantic ignore change")?;
679
680        match &original {
681            Some(contents) => fs::write(&ignore_path, contents)?,
682            None => fs::remove_file(&ignore_path)?,
683        }
684        indexer.reconcile_paths(std::slice::from_ref(&ignore_relative))?;
685    }
686    let expected = expected.expect("positive iteration count is validated");
687    Ok(IndexMeasurement {
688        timing: TimingStats::from_durations(durations),
689        files_seen_per_sample: expected.seen,
690        files_indexed_per_sample: expected.indexed,
691        files_removed_per_sample: expected.removed,
692    })
693}
694
695fn require_stable_counts(
696    expected: &mut Option<MinimumIndexCounts>,
697    response: &IndexResponse,
698    measurement: &str,
699) -> AnyResult<()> {
700    let observed = MinimumIndexCounts {
701        seen: response.files_seen,
702        indexed: response.files_indexed,
703        removed: response.files_removed,
704    };
705    if let Some(expected) = expected {
706        if expected.seen != observed.seen
707            || expected.indexed != observed.indexed
708            || expected.removed != observed.removed
709        {
710            return Err(Box::new(io::Error::other(format!(
711                "{measurement} counts changed across samples: expected {expected:?}, got {observed:?}"
712            ))));
713        }
714    } else {
715        *expected = Some(observed);
716    }
717    Ok(())
718}

720fn measure_watcher_delivery(
721    iterations: usize,
722    indexer: &Indexer,
723    config: &Config,
724    path: &Path,
725    relative_path: &str,
726    debounce: Duration,
727) -> AnyResult<WatcherDeliveryMeasurement> {
728    const EVENT_TIMEOUT: Duration = Duration::from_secs(10);
729
730    let runtime = tokio::runtime::Runtime::new()?;
731    runtime.block_on(async {
732        let cancellation = CancellationToken::new();
733        let (watcher, mut receiver) = RepositoryWatcher::start_with_policy(
734            &config.root,
735            256,
736            debounce,
737            config.discovery_policy(),
738            cancellation,
739        )
740        .await?;
741        let mut durations = Vec::with_capacity(iterations);
742        let mut changed_messages = 0usize;
743        let mut full_reconciliation_messages = 0usize;
744        for iteration in 0..iterations {
745            let start = Instant::now();
746            let mut file = OpenOptions::new().append(true).open(path)?;
747            writeln!(file, "// watcher delivery mutation {iteration}")?;
748            drop(file);
749
750            loop {
751                let message = timeout(EVENT_TIMEOUT, receiver.recv())
752                    .await
753                    .map_err(|_| io::Error::other("watcher delivery timed out"))?
754                    .ok_or_else(|| io::Error::other("watcher closed before delivery"))?;
755                match message {
756                    WatcherMessage::Changed { paths }
757                        if paths.iter().any(|candidate| candidate == relative_path) =>
758                    {
759                        changed_messages += 1;
760                        break;
761                    }
762                    WatcherMessage::ReconcileRequired => {
763                        full_reconciliation_messages += 1;
764                        break;
765                    }
766                    WatcherMessage::Changed { .. } => {}
767                }
768            }
769            durations.push(start.elapsed());
770            indexer.reconcile_paths(&[relative_path.to_string()])?;
771        }
772        watcher.shutdown().await?;
773        Ok(WatcherDeliveryMeasurement {
774            configured_debounce_ms: u64::try_from(debounce.as_millis()).unwrap_or(u64::MAX),
775            timing: TimingStats::from_durations(durations),
776            changed_messages,
777            full_reconciliation_messages,
778        })
779    })
780}

791fn prepare_corpus(args: &Args) -> AnyResult<PreparedCorpus> {
792    let temporary_root = tempfile::tempdir()?;
793    if let Some(repository) = &args.repository {
794        let repository = repository.canonicalize()?;
795        let revision = git_output(&repository, ["rev-parse", "HEAD"])?;
796        let status = git_output(
797            &repository,
798            ["status", "--porcelain", "--untracked-files=all"],
799        )?;
800        if !status.is_empty() {
801            return Err(invalid_input(
802                "--repository must be clean so the recorded revision identifies the corpus",
803            ));
804        }
805        let root = temporary_root.path().join("repository");
806        snapshot_repository(&repository, &root, temporary_root.path())?;
807        return Ok(PreparedCorpus {
808            _temporary_root: temporary_root,
809            root,
810            source_kind: "git_worktree_snapshot",
811            source_repository: args.repository_label.clone(),
812            revision: Some(revision),
813            limitation: "The repository profile uses an isolated ignore-aware snapshot of a clean checkout at the recorded commit.",
814        });
815    }
816
817    create_corpus(temporary_root.path(), args.files, args.file_bytes)?;
818    Ok(PreparedCorpus {
819        root: temporary_root.path().to_path_buf(),
820        _temporary_root: temporary_root,
821        source_kind: "synthetic_rust",
822        source_repository: None,
823        revision: None,
824        limitation: "The generated Rust corpus controls file count and size but does not model a real repository's language mix or directory shape.",
825    })
826}

828fn snapshot_repository(source: &Path, destination: &Path, temporary_root: &Path) -> AnyResult<()> {
829    let source_config = Config::discover(source, Some(temporary_root.join("source-probe.sqlite")))?;
830    let files = discover_files(source, source_config.max_file_bytes)?;
831    fs::create_dir_all(destination)?;
832    for file in files {
833        let target = destination.join(Path::new(&file.relative_path));
834        if let Some(parent) = target.parent() {
835            fs::create_dir_all(parent)?;
836        }
837        fs::copy(&file.absolute_path, target)?;
838    }
839    Ok(())
840}

842fn git_output<const N: usize>(repository: &Path, args: [&str; N]) -> AnyResult<String> {
843    let output = Command::new("git")
844        .arg("-C")
845        .arg(repository)
846        .args(args)

848        .output()?;

850        return Err(Box::new(io::Error::other(format!(
851            "git command failed with {}: {}",
852            output.status,

854        ))));
855    }

859fn leantoken_source_identity() -> (Option<String>, Option<bool>) {
860    let root = Path::new(env!("LEANTOKEN_REPOSITORY_ROOT"));
861    let revision = git_output(root, ["rev-parse", "HEAD"]).ok();
862    let dirty = revision.as_ref().and_then(|_| {
863        git_output(root, ["status", "--porcelain", "--untracked-files=all"])
864            .ok()
865            .map(|status| !status.is_empty())
866    });
867    (revision, dirty)

870fn corpus_report(
871    args: &Args,
872    corpus: &PreparedCorpus,
873    files: &[DiscoveredFile],
874    hot_set: usize,
875) -> CorpusReport {
876    let total_bytes = files.iter().map(|file| file.size_bytes).sum::<u64>();
877    let mut extensions = BTreeMap::new();
878    let mut max_directory_depth = 0usize;
879    for file in files {
880        let path = Path::new(&file.relative_path);
881        let extension = path
882            .extension()
883            .and_then(|extension| extension.to_str())
884            .filter(|extension| !extension.is_empty())
885            .unwrap_or("<none>")
886            .to_ascii_lowercase();
887        *extensions.entry(extension).or_insert(0) += 1;
888        max_directory_depth = max_directory_depth.max(path.components().count().saturating_sub(1));
889    }
890    CorpusReport {
891        source_kind: corpus.source_kind,
892        source_repository: corpus.source_repository.clone(),
893        revision: corpus.revision.clone(),
894        files: files.len(),
895        total_bytes,
896        mean_file_bytes: total_bytes as f64 / files.len() as f64,
897        max_directory_depth,
898        extensions,
899        synthetic_file_bytes: args.repository.is_none().then_some(args.file_bytes),
900        indexing_iterations: args.iterations,
901        read_samples: args.read_samples,
902        hot_set_files: hot_set,
903    }
904}

935fn measure_lifecycle_indexing<S, C, R>(
936    iterations: usize,
937    mut setup: S,
938    mut reconcile: C,
939    mut restore: R,
940    minimum: MinimumIndexCounts,
941    measurement: &str,
942) -> AnyResult<IndexMeasurement>
943where
944    S: FnMut(usize) -> io::Result<()>,
945    C: FnMut() -> leantoken::Result<IndexResponse>,
946    R: FnMut() -> AnyResult<()>,
947{
948    let mut durations = Vec::with_capacity(iterations);
949    let mut expected = None;
950    for iteration in 0..iterations {
951        setup(iteration)?;
952        let start = Instant::now();
953        let response = reconcile()?;
954        durations.push(start.elapsed());
955        if response.files_seen < minimum.seen
956            || response.files_indexed < minimum.indexed
957            || response.files_removed < minimum.removed
958        {
959            return Err(Box::new(io::Error::other(format!(
960                "{measurement} expected at least {minimum:?}; got seen={}, indexed={}, removed={}",
961                response.files_seen, response.files_indexed, response.files_removed
962            ))));
963        }
964        require_stable_counts(&mut expected, &response, measurement)?;
965        restore()?;
966    }
967    let expected = expected.expect("positive iteration count is validated");
968    Ok(IndexMeasurement {
969        timing: TimingStats::from_durations(durations),
970        files_seen_per_sample: expected.seen,
971        files_indexed_per_sample: expected.indexed,
972        files_removed_per_sample: expected.removed,
973    })
974}

984            "{measurement} expected files_seen={files_seen}, files_indexed={files_indexed}; got files_seen={}, files_indexed={}",
985            response.files_seen, response.files_indexed
986        ))));
987    }

1025fn create_corpus(root: &Path, files: usize, file_bytes: usize) -> AnyResult<Vec<PathBuf>> {
1026    let source = root.join("src");
1027    let rename_fixture = source.join("rename_fixture");
1028    fs::create_dir_all(&rename_fixture)?;
1029    let mut paths = Vec::with_capacity(files);
1030    for index in 0..files {
1031        let directory = if index < files.min(32) {
1032            &rename_fixture
1033        } else {
1034            &source
1035        };
1036        let path = directory.join(format!("file_{index:05}.rs"));
1037        fs::write(&path, synthetic_source(index, file_bytes))?;
1038        paths.push(path);
1039    }
1040    Ok(paths)
1041}

1043fn synthetic_source(index: usize, file_bytes: usize) -> String {
1044    let mut source = format!(
1045        "pub fn symbol_{index:05}() -> usize {{\n    {index}\n}}\n\n// deterministic profile padding: "
1046    );
1047    if source.len() < file_bytes {
1048        source.extend(std::iter::repeat_n('x', file_bytes - source.len()));
1049    }
1050    source.push('\n');
1051    source
1052}

1054impl TimingStats {
1055    fn from_durations(durations: Vec<Duration>) -> Self {
1056        Self::from_microseconds(
1057            durations
1058                .into_iter()
1059                .map(|duration| duration.as_secs_f64() * 1_000_000.0)
1060                .collect(),
1061        )
1062    }
1063
1064    fn from_milliseconds(milliseconds: Vec<f64>) -> Self {
1065        Self::from_microseconds(
1066            milliseconds
1067                .into_iter()
1068                .map(|milliseconds| milliseconds * 1_000.0)
1069                .collect(),
1070        )
1071    }
1072
1073    fn from_microseconds(mut micros: Vec<f64>) -> Self {
1074        assert!(!micros.is_empty());
1075        micros.sort_by(f64::total_cmp);
1076        let total_us = micros.iter().sum::<f64>();
1077        Self {
1078            samples: micros.len(),
1079            total_ms: total_us / 1_000.0,
1080            mean_us: total_us / micros.len() as f64,
1081            min_us: micros[0],
1082            p50_us: percentile(&micros, 0.50),
1083            p95_us: percentile(&micros, 0.95),
1084            max_us: micros[micros.len() - 1],
1085        }
1086    }
1087}

1107    sorted[rank.saturating_sub(1).min(sorted.len() - 1)]
1108}
