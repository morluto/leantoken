258fn seal(args: &SealArgs) -> Result<(), DynError> {
259    ensure_output_absent(&args.output)?;
260    ensure_private_file(&args.labels)?;
261
262    let policy = load_json::<Policy>(&args.policy)?;
263    validate_policy(&policy.value)?;
264    let tasks = load_jsonl::<PublicTask>(&args.tasks)?;
265    let summary = validate_tasks(&policy.value, &tasks.value)?;
266    let labels = load_jsonl::<PrivateLabel>(&args.labels)?;
267    validate_labels(&tasks.value, &labels.value)?;
268    let host = load_json::<HostProvenance>(&args.host)?;
269    validate_host(&host.value)?;
270    validate_revision(&args.candidate_revision, "candidate_revision")?;
271    if args.candidate_revision == policy.value.baseline_revision {
272        return Err("candidate_revision must differ from the frozen baseline_revision".into());
273    }
274    validate_revision(&args.harness_revision, "harness_revision")?;
275    validate_revision(&args.evaluator_revision, "evaluator_revision")?;
276    validate_nonempty_bounded(&args.toolchain, "toolchain", 256)?;
277
278    let receipt = SealReceipt {
279        schema_version: SCHEMA_VERSION,
280        dataset_kind: DATASET_KIND.into(),
281        state: SEALED_STATE.into(),
282        frozen_at: policy.value.frozen_at.clone(),
283        baseline_revision: policy.value.baseline_revision.clone(),
284        candidate_revision: args.candidate_revision.clone(),
285        harness_revision: args.harness_revision.clone(),
286        evaluator_revision: args.evaluator_revision.clone(),
287        toolchain: args.toolchain.clone(),
288        tokenizer: policy.value.tokenizer.clone(),
289        policy_blake3: hash(&policy.bytes),
290        tasks_blake3: hash(&tasks.bytes),
291        labels_blake3: hash(&labels.bytes),
292        host_blake3: hash(&host.bytes),
293        task_count: tasks.value.len(),
294        label_count: labels.value.len(),
295        task_counts_by_family: summary.task_counts_by_family,
296        task_counts_by_language: summary.task_counts_by_language,
297        repository_counts_by_family: summary
298            .repositories_by_family
299            .into_iter()
300            .map(|(family, repositories)| (family, repositories.len()))
301            .collect(),
302        task_shape_counts_by_family: summary
303            .task_shapes_by_family
304            .into_iter()
305            .map(|(family, shapes)| (family, shapes.len()))
306            .collect(),
307        limitations: vec![
308            "The receipt commits to private labels but contains no owner paths, line regions, prompts, or repository source.".into(),
309            "Sealing proves artifact integrity and coverage strata, not retrieval quality or agent task success.".into(),
310            "Blind status depends on evaluator access controls outside this repository; local file permissions are only a fail-closed hygiene check.".into(),
311            policy.value.reclassification_rule.clone(),
312        ],
313    };
314    write_new_json(&args.output, &receipt)
315}

328fn validate_policy(policy: &Policy) -> Result<(), DynError> {
329    require_schema(policy.schema_version, "policy")?;
330    if policy.dataset_kind != DATASET_KIND {
331        return Err(format!("policy dataset_kind must be {DATASET_KIND}").into());
332    }
333    validate_nonempty_bounded(&policy.frozen_at, "frozen_at", 128)?;
334    validate_revision(&policy.baseline_revision, "baseline_revision")?;
335    if policy.tokenizer != TOKENIZER {
336        return Err(format!("policy tokenizer must be {TOKENIZER}").into());
337    }
338    let expected = REQUIRED_FAMILIES
339        .into_iter()
340        .map(str::to_owned)
341        .collect::<Vec<_>>();
342    if policy.task_families != expected {
343        return Err("policy task_families must contain the frozen ordered family catalog".into());
344    }
345
346    let coverage = &policy.coverage;
347    if coverage.minimum_tasks < REQUIRED_FAMILIES.len()
348        || coverage.minimum_tasks > MAX_RECORDS
349        || coverage.minimum_tasks_per_family == 0
350        || coverage.minimum_repositories_per_family < 2
351        || coverage.minimum_languages_per_family < 2
352        || coverage.minimum_task_shapes_per_family < 2
353    {
354        return Err("policy coverage bounds are incomplete or unsafe".into());
355    }
356    if coverage.minimum_tasks_per_family * REQUIRED_FAMILIES.len() > coverage.minimum_tasks {
357        return Err("minimum_tasks cannot be smaller than the required per-family total".into());
358    }
359
360    let execution = &policy.execution;
361    for (name, value) in [
362        ("attempts_per_arm", execution.attempts_per_arm),
363        (
364            "maximum_tool_calls_per_attempt",
365            execution.maximum_tool_calls_per_attempt,
366        ),
367        (
368            "maximum_wall_time_seconds_per_attempt",
369            execution.maximum_wall_time_seconds_per_attempt,
370        ),
371        (
372            "maximum_source_tokens_per_retrieval",
373            execution.maximum_source_tokens_per_retrieval,
374        ),
375    ] {
376        if value == 0 {
377            return Err(format!("execution {name} must be positive").into());
378        }
379    }
380    if execution.maximum_source_tokens_per_retrieval > MAX_TOKEN_BUDGET {
381        return Err("execution source-token bound exceeds the harness maximum".into());
382    }
383    for (name, value) in [
384        ("policy_id", execution.policy_id.as_str()),
385        ("network_policy", execution.network_policy.as_str()),
386        ("workspace_policy", execution.workspace_policy.as_str()),
387        ("validation_policy", execution.validation_policy.as_str()),
388    ] {
389        validate_nonempty_bounded(value, name, 256)?;
390    }
391
392    validate_promotion(&policy.promotion)?;
393    let required = [
394        "task_source",
395        "repository_revision",
396        "label_method",
397        "evaluator_revision",
398        "harness_revision",
399        "toolchain",
400        "host",
401    ]
402    .into_iter()
403    .map(str::to_owned)
404    .collect::<BTreeSet<_>>();
405    if policy
406        .required_provenance
407        .iter()
408        .cloned()
409        .collect::<BTreeSet<_>>()
410        != required
411        || policy.required_provenance.len() != required.len()
412    {
413        return Err("policy required_provenance must match the frozen catalog".into());
414    }
415    validate_nonempty_bounded(
416        &policy.reclassification_rule,
417        "reclassification_rule",
418        2_048,
419    )
420}

422fn validate_promotion(promotion: &PromotionPolicy) -> Result<(), DynError> {
423    if !promotion.confidence_level.is_finite()
424        || !(0.5..1.0).contains(&promotion.confidence_level)
425        || promotion.bootstrap_resamples < 1_000
426    {
427        return Err("promotion statistical policy is incomplete".into());
428    }
429    for (name, value, minimum, maximum) in [
430        (
431            "minimum_task_success_delta",
432            promotion.minimum_task_success_delta,
433            -1.0,
434            1.0,
435        ),
436        (
437            "maximum_global_recall_regression",
438            promotion.maximum_global_recall_regression,
439            0.0,
440            1.0,
441        ),
442        (
443            "maximum_family_recall_regression",
444            promotion.maximum_family_recall_regression,
445            0.0,
446            1.0,
447        ),
448        (
449            "maximum_dead_end_source_regression",
450            promotion.maximum_dead_end_source_regression,
451            0.0,
452            1.0,
453        ),
454        (
455            "maximum_reread_source_regression",
456            promotion.maximum_reread_source_regression,
457            0.0,
458            1.0,
459        ),
460        (
461            "maximum_provider_input_regression",
462            promotion.maximum_provider_input_regression,
463            0.0,
464            1.0,
465        ),
466        (
467            "maximum_warm_p95_latency_regression",
468            promotion.maximum_warm_p95_latency_regression,
469            0.0,
470            1.0,
471        ),
472        (
473            "maximum_peak_rss_regression",
474            promotion.maximum_peak_rss_regression,
475            0.0,
476            1.0,
477        ),
478        (
479            "maximum_database_size_regression",
480            promotion.maximum_database_size_regression,
481            0.0,
482            1.0,
483        ),
484    ] {
485        if !value.is_finite() || value < minimum || value > maximum {
486            return Err(format!("promotion {name} is outside [{minimum}, {maximum}]").into());
487        }
488    }
489    Ok(())
490}

492fn validate_tasks(policy: &Policy, tasks: &[PublicTask]) -> Result<CoverageSummary, DynError> {
493    if tasks.len() < policy.coverage.minimum_tasks || tasks.len() > MAX_RECORDS {
494        return Err(format!(
495            "task count {} does not satisfy [{}, {MAX_RECORDS}]",
496            tasks.len(),
497            policy.coverage.minimum_tasks
498        )
499        .into());
500    }
501    let allowed_families = policy
502        .task_families
503        .iter()
504        .map(String::as_str)
505        .collect::<BTreeSet<_>>();
506    let mut previous_id: Option<&str> = None;
507    let mut summary = CoverageSummary::default();
508    for task in tasks {
509        require_schema(task.schema_version, "task")?;
510        validate_task_id(&task.id)?;
511        if previous_id.is_some_and(|previous| previous >= task.id.as_str()) {
512            return Err("public tasks must have unique IDs in ascending order".into());
513        }
514        previous_id = Some(&task.id);
515        validate_nonempty_bounded(&task.prompt, "task prompt", MAX_PROMPT_BYTES)?;
516        if !allowed_families.contains(task.task_family.as_str()) {
517            return Err(format!("task {} has unknown task_family", task.id).into());
518        }
519        validate_repository(&task.repository)?;
520        if task.languages.is_empty() || task.languages.len() > 4 {
521            return Err(format!("task {} requires one to four languages", task.id).into());
522        }
523        let mut language_set = BTreeSet::new();
524        for language in &task.languages {
525            validate_tag(language, "language")?;
526            if !language_set.insert(language.as_str()) {
527                return Err(format!("task {} repeats a language", task.id).into());
528            }
529            *summary
530                .task_counts_by_language
531                .entry(language.clone())
532                .or_default() += 1;
533            summary
534                .languages_by_family
535                .entry(task.task_family.clone())
536                .or_default()
537                .insert(language.clone());
538        }
539        validate_tag(&task.task_shape, "task_shape")?;
540        if task.token_budget == 0
541            || task.token_budget > policy.execution.maximum_source_tokens_per_retrieval
542        {
543            return Err(format!("task {} token_budget is outside policy", task.id).into());
544        }
545        if task.executor_policy_id != policy.execution.policy_id {
546            return Err(format!("task {} executor policy is not frozen policy", task.id).into());
547        }
548        validate_success_validator(&task.id, &task.success_validator)?;
549        validate_task_provenance(&task.id, &task.provenance)?;
550
551        *summary
552            .task_counts_by_family
553            .entry(task.task_family.clone())
554            .or_default() += 1;
555        summary
556            .repositories_by_family
557            .entry(task.task_family.clone())
558            .or_default()
559            .insert(task.repository.clone());
560        summary
561            .task_shapes_by_family
562            .entry(task.task_family.clone())
563            .or_default()
564            .insert(task.task_shape.clone());
565    }
566
567    for family in &policy.task_families {
568        let tasks = summary
569            .task_counts_by_family
570            .get(family)
571            .copied()
572            .unwrap_or_default();
573        let repositories = summary
574            .repositories_by_family
575            .get(family)
576            .map_or(0, BTreeSet::len);
577        let languages = summary
578            .languages_by_family
579            .get(family)
580            .map_or(0, BTreeSet::len);
581        let shapes = summary
582            .task_shapes_by_family
583            .get(family)
584            .map_or(0, BTreeSet::len);
585        if tasks < policy.coverage.minimum_tasks_per_family
586            || repositories < policy.coverage.minimum_repositories_per_family
587            || languages < policy.coverage.minimum_languages_per_family
588            || shapes < policy.coverage.minimum_task_shapes_per_family
589        {
590            return Err(format!(
591                "family {family} coverage is incomplete: tasks={tasks}, repositories={repositories}, languages={languages}, task_shapes={shapes}"
592            )
593            .into());
594        }
595    }
596    Ok(summary)
597}

599fn validate_labels(tasks: &[PublicTask], labels: &[PrivateLabel]) -> Result<(), DynError> {
600    if labels.len() != tasks.len() {
601        return Err("private labels must cover every public task exactly once".into());
602    }
603    for (task, label) in tasks.iter().zip(labels) {
604        require_schema(label.schema_version, "label")?;
605        if label.task_id != task.id {
606            return Err("private labels must use the same ascending task order".into());
607        }
608        require_hash(
609            &label.task_blake3,
610            &hash(&serde_json::to_vec(task)?),
611            "label task binding",
612        )?;
613        validate_nonempty_bounded(&label.label_method, "label_method", 256)?;
614        if label.relevant_files.is_empty() {
615            return Err(format!("label {} has no relevant files", label.task_id).into());
616        }
617        let mut files = BTreeSet::new();
618        for path in &label.relevant_files {
619            validate_relative_path(path)?;
620            if !files.insert(path.as_str()) {
621                return Err(format!("label {} repeats a relevant file", label.task_id).into());
622            }
623        }
624        if label.relevant_regions.is_empty() {
625            return Err(format!("label {} has no relevant regions", label.task_id).into());
626        }
627        let mut regions = BTreeSet::new();
628        for region in &label.relevant_regions {
629            validate_relative_path(&region.path)?;
630            if !files.contains(region.path.as_str()) {
631                return Err(
632                    format!("label {} region path is not a relevant file", label.task_id).into(),
633                );
634            }
635            if region.start_line == 0 || region.end_line < region.start_line {
636                return Err(format!("label {} has an invalid line region", label.task_id).into());
637            }
638            if !regions.insert((&region.path, region.start_line, region.end_line)) {
639                return Err(format!("label {} repeats a line region", label.task_id).into());
640            }
641        }
642    }
643    Ok(())
644}

648    policy: &Loaded<Policy>,
649    tasks: &Loaded<Vec<PublicTask>>,
650    host: &Loaded<HostProvenance>,
651    summary: &CoverageSummary,
652) -> Result<(), DynError> {
653    require_schema(receipt.schema_version, "receipt")?;
654    if receipt.dataset_kind != DATASET_KIND
655        || receipt.state != SEALED_STATE
656        || receipt.frozen_at != policy.value.frozen_at
657        || receipt.baseline_revision != policy.value.baseline_revision
658        || receipt.tokenizer != policy.value.tokenizer
659    {
660        return Err("receipt frozen identity does not match policy".into());
661    }
662    validate_revision(&receipt.candidate_revision, "receipt candidate_revision")?;
663    if receipt.candidate_revision == receipt.baseline_revision {
664        return Err("receipt candidate_revision must differ from baseline_revision".into());
665    }
666    validate_revision(&receipt.harness_revision, "receipt harness_revision")?;
667    validate_revision(&receipt.evaluator_revision, "receipt evaluator_revision")?;
668    validate_nonempty_bounded(&receipt.toolchain, "receipt toolchain", 256)?;
669    require_hash(&receipt.policy_blake3, &hash(&policy.bytes), "policy")?;
670    require_hash(&receipt.tasks_blake3, &hash(&tasks.bytes), "tasks")?;
671    require_hash(&receipt.host_blake3, &hash(&host.bytes), "host")?;
672    validate_blake3(&receipt.labels_blake3, "labels commitment")?;
673    if receipt.task_count != tasks.value.len()
674        || receipt.label_count != tasks.value.len()
675        || receipt.task_counts_by_family != summary.task_counts_by_family
676        || receipt.task_counts_by_language != summary.task_counts_by_language
677    {
678        return Err("receipt task aggregates do not match public tasks".into());
679    }
680    let repository_counts = summary
681        .repositories_by_family
682        .iter()
683        .map(|(family, repositories)| (family.clone(), repositories.len()))
684        .collect::<BTreeMap<_, _>>();
685    let shape_counts = summary
686        .task_shapes_by_family
687        .iter()
688        .map(|(family, shapes)| (family.clone(), shapes.len()))
689        .collect::<BTreeMap<_, _>>();
690    if receipt.repository_counts_by_family != repository_counts
691        || receipt.task_shape_counts_by_family != shape_counts
692    {
693        return Err("receipt family aggregates do not match public tasks".into());
694    }
695    if receipt.limitations.len() < 4
696        || receipt
697            .limitations
698            .iter()
699            .any(|limitation| limitation.trim().is_empty())
700        || !receipt
701            .limitations
702            .iter()
703            .any(|limitation| limitation == &policy.value.reclassification_rule)
704    {
705        return Err("receipt limitations omit the reclassification boundary".into());
706    }
707    Ok(())
708}

760fn validate_task_id(value: &str) -> Result<(), DynError> {

762        || value.len() > 128
763        || !value.bytes().all(|byte| {
764            byte.is_ascii_lowercase() || byte.is_ascii_digit() || b"-_.".contains(&byte)
765        })
766    {
767        return Err("task id must be bounded lowercase ASCII".into());

772fn validate_tag(value: &str, name: &str) -> Result<(), DynError> {
773    if value.is_empty()
774        || value.len() > 64
775        || !value
776            .bytes()
777            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'_')
778    {
779        return Err(format!("{name} must be bounded snake_case ASCII").into());
780    }
781    Ok(())
782}

784fn validate_relative_path(value: &str) -> Result<(), DynError> {
785    if value.is_empty() || value.len() > 4_096 || value.contains('\\') || value.contains('\0') {
786        return Err("label path must be a bounded slash-relative path".into());
787    }
788    let path = Path::new(value);
789    if path.is_absolute()
790        || path
791            .components()
792            .any(|component| !matches!(component, Component::Normal(_)))
793    {
794        return Err("label path must not contain absolute or traversal components".into());
795    }
796    Ok(())
797}

799fn require_schema(actual: u32, description: &str) -> Result<(), DynError> {
800    if actual != SCHEMA_VERSION {
801        return Err(format!("{description} schema_version must be {SCHEMA_VERSION}").into());
802    }
803    Ok(())
804}

806fn validate_revision(value: &str, description: &str) -> Result<(), DynError> {
807    if value.len() != 40
808        || !value
809            .bytes()
810            .all(|byte| byte.is_ascii_digit() || matches!(byte, b'a'..=b'f'))
811    {
812        return Err(format!("{description} must be a lowercase 40-character Git revision").into());
813    }
814    Ok(())
815}
816
817fn validate_blake3(value: &str, description: &str) -> Result<(), DynError> {
818    if value.len() != 64
819        || !value
820            .bytes()
821            .all(|byte| byte.is_ascii_digit() || matches!(byte, b'a'..=b'f'))
822    {
823        return Err(format!("{description} must be a lowercase BLAKE3 hash").into());
824    }
825    Ok(())
826}
827
828fn require_hash(actual: &str, expected: &str, description: &str) -> Result<(), DynError> {
829    validate_blake3(actual, description)?;
830    if actual != expected {
831        return Err(format!("{description} hash does not match").into());
832    }
833    Ok(())
834}

836fn validate_nonempty_bounded(
837    value: &str,
838    description: &str,
839    maximum_bytes: usize,
840) -> Result<(), DynError> {
841    if value.trim().is_empty() || value.len() > maximum_bytes {
842        return Err(
843            format!("{description} must be nonempty and at most {maximum_bytes} bytes").into(),
844        );
845    }
846    Ok(())
847}

849fn load_json<T: DeserializeOwned>(path: &Path) -> Result<Loaded<T>, DynError> {
850    let bytes = read_bounded(path)?;
851    let value = serde_json::from_slice(&bytes)
852        .map_err(|error| format!("invalid JSON {}: {error}", path.display()))?;
853    Ok(Loaded { bytes, value })
854}
855
856fn load_jsonl<T: DeserializeOwned>(path: &Path) -> Result<Loaded<Vec<T>>, DynError> {
857    let bytes = read_bounded(path)?;
858    let mut values = Vec::new();
859    for (index, line) in bytes.split(|byte| *byte == b'\n').enumerate() {
860        if line.is_empty() {
861            continue;
862        }
863        if values.len() == MAX_RECORDS {
864            return Err(format!("{} exceeds {MAX_RECORDS} records", path.display()).into());
865        }
866        values.push(serde_json::from_slice(line).map_err(|error| {
867            format!(
868                "invalid JSONL {} line {}: {error}",
869                path.display(),
870                index + 1
871            )
872        })?);
873    }
874    if values.is_empty() {
875        return Err(format!("{} contains no records", path.display()).into());
876    }
877    Ok(Loaded {
878        bytes,
879        value: values,
880    })
881}

883fn read_bounded(path: &Path) -> Result<Vec<u8>, DynError> {
884    let metadata = fs::metadata(path)?;
885    if !metadata.is_file() || metadata.len() > MAX_ARTIFACT_BYTES {
886        return Err(format!(
887            "{} must be a file no larger than {MAX_ARTIFACT_BYTES} bytes",
888            path.display()
889        )
890        .into());
891    }
892    Ok(fs::read(path)?)
893}

925fn write_new_json<T: Serialize>(path: &Path, value: &T) -> Result<(), DynError> {
926    if let Some(parent) = path.parent() {
927        fs::create_dir_all(parent)?;
928    }
929    let mut output = OpenOptions::new().write(true).create_new(true).open(path)?;
930    serde_json::to_writer_pretty(&mut output, value)?;
931    output.write_all(b"\n")?;
932    output.sync_all()?;
933    Ok(())
934}
