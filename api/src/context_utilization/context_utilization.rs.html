1//! Classify observable downstream signals for context evidence in one trajectory.
2//!
3//! This is an offline diagnostic. It deliberately reports separate signals
4//! instead of collapsing relevance, explicit reuse, rereads, and task outcome
5//! into a guessed utilization score.

149fn main() -> Result<(), DynError> {
150    let args = Args::parse();
151    if args.relevant_paths.len() > MAX_RELEVANT_PATHS {
152        return Err(format!(
153            "relevant paths exceed bound: {} > {MAX_RELEVANT_PATHS}",
154            args.relevant_paths.len()
155        )
156        .into());
157    }
158    let relevant_paths = args
159        .relevant_paths
160        .iter()
161        .map(|path| normalize_path(path))
162        .collect::<Result<BTreeSet<_>, _>>()?;
163    let trace_bytes = read_bounded(&args.tool_trace)?;
164    let trajectory_bytes = read_bounded(&args.trajectory)?;
165    let trace: ToolTrace = serde_json::from_slice(&trace_bytes)?;
166    let trajectory: Trajectory = serde_json::from_slice(&trajectory_bytes)?;
167    validate_artifacts(&trace, &trajectory)?;
168    let (summary, ranges) = classify(&trace, &trajectory, &relevant_paths)?;
169    let report = Report {
170        schema_version: REPORT_SCHEMA_V1,
171        report_kind: "context_utilization_trajectory",
172        diagnostic_only: true,
173        outcome: args.outcome,
174        source: SourceReceipt {
175            classifier_source_blake3: blake3::hash(include_bytes!("context_utilization.rs"))
176                .to_hex()
177                .to_string(),
178            tool_trace_blake3: blake3::hash(&trace_bytes).to_hex().to_string(),
179            trajectory_blake3: blake3::hash(&trajectory_bytes).to_hex().to_string(),
180            artifact_schema_version: ARTIFACT_SCHEMA_V1,
181            binding: trace.binding,
182        },
183        bounds: Bounds {
184            max_artifact_bytes: MAX_ARTIFACT_BYTES,
185            max_trace_calls: MAX_TRACE_CALLS,
186            max_trajectory_events: MAX_TRAJECTORY_EVENTS,
187            max_trace_ranges: MAX_TRACE_RANGES,
188            max_ranges_per_path: MAX_RANGES_PER_PATH,
189            max_context_calls: MAX_CONTEXT_CALLS,
190            max_context_ranges: MAX_CONTEXT_RANGES,
191            max_relevant_paths: MAX_RELEVANT_PATHS,
192            max_hash_inputs: MAX_HASH_INPUTS,
193            max_path_bytes: MAX_PATH_BYTES,
194        },
195        summary,
196        ranges,
197        limitations: vec![
198            "relevant_path_proxy is label- or final-patch-based and does not prove that the model used the returned source",
199            "explicit_hash_input_later proves that the caller retained an identity, not that the model reasoned from its content",
200            "exact and overlap rereads measure downstream retrieval pressure, not useful evidence reuse",
201            "no_observed_downstream_signal is not proof that evidence was unused",
202            "task outcome is associated with the trajectory but this report does not claim causality",
203        ],
204    };
205    if let Some(parent) = args.output.parent() {
206        fs::create_dir_all(parent)?;
207    }
208    fs::write(&args.output, serde_json::to_vec_pretty(&report)?)?;
209    println!("{}", serde_json::to_string_pretty(&report)?);
210    Ok(())
211}

213fn read_bounded(path: &Path) -> Result<Vec<u8>, DynError> {
214    let mut bytes = Vec::new();
215    File::open(path)?
216        .take(MAX_ARTIFACT_BYTES.saturating_add(1))
217        .read_to_end(&mut bytes)?;
218    if u64::try_from(bytes.len())? > MAX_ARTIFACT_BYTES {
219        return Err(format!("artifact exceeds byte bound: > {MAX_ARTIFACT_BYTES} bytes").into());
220    }
221    Ok(bytes)
222}

224fn validate_artifacts(trace: &ToolTrace, trajectory: &Trajectory) -> Result<(), DynError> {
225    if trace.schema_version != ARTIFACT_SCHEMA_V1 || trajectory.schema_version != ARTIFACT_SCHEMA_V1
226    {
227        return Err("unsupported trajectory artifact schema".into());
228    }
229    if trace.binding != trajectory.binding {
230        return Err("tool trace and trajectory bindings differ".into());
231    }
232    if trace.calls.len() > MAX_TRACE_CALLS {
233        return Err("tool trace exceeds call bound".into());
234    }
235    if trajectory.events.len() > MAX_TRAJECTORY_EVENTS {
236        return Err("trajectory exceeds event bound".into());
237    }
238    let mut prior_sequence = None;
239    let mut call_ids = BTreeSet::new();
240    let mut range_count = 0usize;
241    let mut ranges_by_path = BTreeMap::new();
242    for call in &trace.calls {
243        if prior_sequence.is_some_and(|prior| call.sequence <= prior) {
244            return Err("tool trace sequences are not strictly increasing".into());
245        }
246        prior_sequence = Some(call.sequence);
247        if !call_ids.insert(call.call_id.as_str()) {
248            return Err("tool trace contains a duplicate call ID".into());
249        }
250        for range in &call.ranges {
251            validate_range(range)?;
252            range_count = range_count.saturating_add(1);
253            if range_count > MAX_TRACE_RANGES {
254                return Err("tool trace exceeds the total range bound".into());
255            }
256            let path_count = ranges_by_path
257                .entry((range.repository_generation, normalize_path(&range.path)?))
258                .or_insert(0usize);
259            *path_count = path_count.saturating_add(1);
260            if *path_count > MAX_RANGES_PER_PATH {
261                return Err("tool trace exceeds the per-path range bound".into());
262            }
263        }
264    }
265    Ok(())
266}
267
268fn validate_range(range: &RangeIdentity) -> Result<(), DynError> {
269    if range.start_line == 0 || range.end_line < range.start_line {
270        return Err("tool trace contains an invalid range".into());
271    }
272    normalize_path(&range.path)?;
273    if range.content_hash.len() != 32
274        || !range
275            .content_hash
276            .bytes()
277            .all(|byte| byte.is_ascii_hexdigit())
278    {
279        return Err("tool trace contains an invalid content hash".into());
280    }
281    Ok(())
282}

284fn classify(
285    trace: &ToolTrace,
286    trajectory: &Trajectory,
287    relevant_paths: &BTreeSet<String>,
288) -> Result<(UtilizationSummary, Vec<RangeUtilization>), DynError> {
289    let trajectory_calls = trajectory_calls(trajectory, trace)?;
290    let trace_by_id = trace
291        .calls
292        .iter()
293        .map(|call| (call.call_id.as_str(), call))
294        .collect::<BTreeMap<_, _>>();
295    let mut context_calls = Vec::new();
296    for (trace_id, call) in &trajectory_calls {
297        if call.tool == "leantoken_context" {
298            let trace_call = trace_by_id
299                .get(trace_id.as_str())
300                .ok_or("context trajectory call is missing from tool trace")?;
301            context_calls.push((call, *trace_call));
302        }
303    }
304    if context_calls.len() > MAX_CONTEXT_CALLS {
305        return Err("trajectory exceeds the context-call bound".into());
306    }
307
308    let context_range_count = context_calls
309        .iter()
310        .map(|(_, call)| call.ranges.len())
311        .sum::<usize>();
312    if context_range_count > MAX_CONTEXT_RANGES {
313        return Err("context evidence exceeds range bound".into());
314    }
315
316    let retrieval_calls = trajectory_calls
317        .values()
318        .filter(|call| is_retrieval_tool(call.tool))
319        .collect::<Vec<_>>();
320    let mut retrieval_ranges_by_path = BTreeMap::new();
321    for call in &trace.calls {
322        if !trajectory_call_for_trace_id(&trajectory_calls, &call.call_id)
323            .is_some_and(|call| is_retrieval_tool(call.tool))
324        {
325            continue;
326        }
327        for range in &call.ranges {
328            retrieval_ranges_by_path
329                .entry((range.repository_generation, normalize_path(&range.path)?))
330                .or_insert_with(Vec::new)
331                .push((call.sequence, range.start_line, range.end_line));
332        }
333    }
334    let mut hash_input_sequences = BTreeMap::<&str, Vec<usize>>::new();
335    let mut hash_input_count = 0usize;
336    for call in &retrieval_calls {
337        for hash in argument_hashes(call.arguments) {
338            hash_input_count = hash_input_count.saturating_add(1);
339            if hash_input_count > MAX_HASH_INPUTS {
340                return Err("trajectory exceeds the hash-input bound".into());
341            }
342            hash_input_sequences
343                .entry(hash)
344                .or_default()
345                .push(call.sequence);
346        }
347    }
348    let mut receipt_input_sequences = BTreeMap::<&str, Vec<usize>>::new();
349    for call in &retrieval_calls {
350        if let Some(receipt_id) = call.arguments["receipt_id"].as_str() {
351            receipt_input_sequences
352                .entry(receipt_id)
353                .or_default()
354                .push(call.sequence);
355        }
356    }
357    let mut receipt_follow_up_ids = BTreeSet::new();
358    let first_context_sequence = context_calls.iter().map(|(_, call)| call.sequence).min();
359    let follow_up_retrieval_ids = retrieval_calls
360        .iter()
361        .filter(|call| first_context_sequence.is_some_and(|sequence| call.sequence > sequence))
362        .map(|call| call.sequence)
363        .collect::<BTreeSet<_>>();
364    let mut ranges = Vec::with_capacity(context_range_count);
365    for (trajectory_call, trace_call) in &context_calls {
366        let receipt_id = trajectory_call
367            .structured_result
368            .and_then(|value| value.pointer("/meta/receipt_id"))
369            .and_then(Value::as_str);
370        if let Some(sequences) = receipt_id.and_then(|id| receipt_input_sequences.get(id)) {
371            for sequence in sequences {
372                if *sequence > trace_call.sequence {
373                    receipt_follow_up_ids.insert(*sequence);
374                }
375            }
376        }
377        for range in &trace_call.ranges {
378            let normalized_path = normalize_path(&range.path)?;
379            let later_ranges = retrieval_ranges_by_path
380                .get(&(range.repository_generation, normalized_path.clone()))
381                .into_iter()
382                .flatten()
383                .filter(|(sequence, _, _)| *sequence > trace_call.sequence)
384                .collect::<Vec<_>>();
385            let explicit_hash_input_later = hash_input_sequences
386                .get(range.content_hash.as_str())
387                .is_some_and(|sequences| {
388                    sequences
389                        .iter()
390                        .any(|sequence| *sequence > trace_call.sequence)
391                });
392            let exact_reread_later = later_ranges.iter().any(|(_, start_line, end_line)| {
393                *start_line == range.start_line && *end_line == range.end_line
394            });
395            let overlap_reread_later = later_ranges.iter().any(|(_, start_line, end_line)| {
396                *start_line <= range.end_line
397                    && range.start_line <= *end_line
398                    && (*start_line != range.start_line || *end_line != range.end_line)
399            });
400            let first_follow_up_sequence = later_ranges
401                .iter()
402                .filter(|(_, start_line, end_line)| {
403                    *start_line <= range.end_line && range.start_line <= *end_line
404                })
405                .map(|(sequence, _, _)| *sequence)
406                .min();
407            let relevant_path_proxy = relevant_paths.contains(&normalized_path);
408            let no_observed_downstream_signal = !relevant_path_proxy
409                && !explicit_hash_input_later
410                && !exact_reread_later
411                && !overlap_reread_later;
412            ranges.push(RangeUtilization {
413                origin_sequence: trace_call.sequence,
414                repository_generation: range.repository_generation,
415                path: normalized_path,
416                start_line: range.start_line,
417                end_line: range.end_line,
418                content_hash: range.content_hash.clone(),
419                source_tokens: range.source_tokens,
420                relevant_path_proxy,
421                explicit_hash_input_later,
422                exact_reread_later,
423                overlap_reread_later,
424                no_observed_downstream_signal,
425                first_follow_up_sequence,
426            });
427        }
428    }
429    ranges.sort_by(|left, right| {
430        (
431            left.origin_sequence,
432            &left.path,
433            left.start_line,
434            left.end_line,
435            &left.content_hash,
436        )
437            .cmp(&(
438                right.origin_sequence,
439                &right.path,
440                right.start_line,
441                right.end_line,
442                &right.content_hash,
443            ))
444    });
445    let summary = UtilizationSummary {
446        context_calls: context_calls.len(),
447        successful_context_calls: context_calls
448            .iter()
449            .filter(|(_, call)| call.outcome == ToolOutcome::Success)
450            .count(),
451        failed_context_calls: context_calls
452            .iter()
453            .filter(|(_, call)| call.outcome != ToolOutcome::Success)
454            .count(),
455        context_ranges: signal_count(&ranges, |_| true)?,
456        relevant_path_proxy: signal_count(&ranges, |range| range.relevant_path_proxy)?,
457        explicit_hash_input_later: signal_count(&ranges, |range| range.explicit_hash_input_later)?,
458        exact_reread_later: signal_count(&ranges, |range| range.exact_reread_later)?,
459        overlap_reread_later: signal_count(&ranges, |range| range.overlap_reread_later)?,
460        no_observed_downstream_signal: signal_count(&ranges, |range| {
461            range.no_observed_downstream_signal
462        })?,
463        receipt_follow_up_calls: receipt_follow_up_ids.len(),
464        follow_up_retrieval_calls: follow_up_retrieval_ids.len(),
465    };
466    Ok((summary, ranges))
467}

469fn trajectory_calls<'a>(
470    trajectory: &'a Trajectory,
471    trace: &ToolTrace,
472) -> Result<BTreeMap<String, TrajectoryCall<'a>>, DynError> {
473    let sequences = trace
474        .calls
475        .iter()
476        .map(|call| (call.call_id.as_str(), call.sequence))
477        .collect::<BTreeMap<_, _>>();
478    let mut calls = BTreeMap::new();
479    for event in &trajectory.events {
480        if event["type"].as_str() != Some("item.completed")
481            || event.pointer("/item/type").and_then(Value::as_str) != Some("mcp_tool_call")
482        {
483            continue;
484        }
485        let item = &event["item"];
486        let id = item["id"]
487            .as_str()
488            .ok_or("completed MCP trajectory item has no ID")?;
489        let trace_id = resolve_trace_id(&sequences, id)
490            .ok_or("completed MCP trajectory item is missing from tool trace")?;
491        let sequence = sequences[trace_id];
492        let tool = item["tool"]
493            .as_str()
494            .ok_or("completed MCP trajectory item has no tool")?;
495        let structured_result = item
496            .pointer("/result/structured_content")
497            .or_else(|| item.pointer("/result/structuredContent"));
498        if calls
499            .insert(
500                trace_id.to_owned(),
501                TrajectoryCall {
502                    sequence,
503                    tool,
504                    arguments: &item["arguments"],
505                    structured_result,
506                },
507            )
508            .is_some()
509        {
510            return Err("trajectory contains a duplicate MCP call ID".into());
511        }
512    }
513    Ok(calls)
514}

516fn resolve_trace_id<'a>(
517    sequences: &BTreeMap<&'a str, usize>,
518    trajectory_id: &str,
519) -> Option<&'a str> {
520    if let Some((trace_id, _)) = sequences.get_key_value(trajectory_id) {
521        return Some(*trace_id);
522    }
523    if let Some(id) = trajectory_id.strip_prefix("prewalk:")
524        && let Some((trace_id, _)) = sequences.get_key_value(id)
525    {
526        return Some(*trace_id);
527    }
528    let prefixed = format!("prewalk:{trajectory_id}");
529    sequences
530        .get_key_value(prefixed.as_str())
531        .map(|(trace_id, _)| *trace_id)
532}

534fn trajectory_call_for_trace_id<'a>(
535    calls: &'a BTreeMap<String, TrajectoryCall<'a>>,
536    trace_id: &str,
537) -> Option<&'a TrajectoryCall<'a>> {
538    calls.get(trace_id).or_else(|| {
539        trace_id
540            .strip_prefix("prewalk:")
541            .and_then(|id| calls.get(id))
542    })
543}

545fn argument_hashes(arguments: &Value) -> BTreeSet<&str> {
546    let mut hashes = arguments["known_hashes"]

549        .flatten()

571fn signal_count(
572    ranges: &[RangeUtilization],
573    include: impl Fn(&RangeUtilization) -> bool,
574) -> Result<SignalCount, DynError> {
575    let selected = ranges
576        .iter()
577        .filter(|range| include(range))
578        .collect::<Vec<_>>();
579    let source_tokens = selected
580        .iter()
581        .filter_map(|range| range.source_tokens)
582        .try_fold(0u64, |total, tokens| -> Result<u64, DynError> {
583            let tokens = u64::try_from(tokens)
584                .map_err(|_| -> DynError { "range source tokens exceed u64".into() })?;
585            total
586                .checked_add(tokens)
587                .ok_or_else(|| -> DynError { "context-utilization source-token overflow".into() })
588        })?;
589    Ok(SignalCount {
590        ranges: selected.len(),
591        source_tokens,
592        source_tokens_complete: selected.iter().all(|range| range.source_tokens.is_some()),
593    })
594}

596fn normalize_path(path: &str) -> Result<String, DynError> {
597    if path.len() > MAX_PATH_BYTES {
598        return Err("path exceeds byte bound".into());
599    }
600    let path = Path::new(path);
601    if path.as_os_str().is_empty() || path.is_absolute() {
602        return Err("path must be non-empty and repository-relative".into());
603    }
604    let mut parts = Vec::new();
605    for component in path.components() {
606        match component {
607            Component::Normal(value) => parts.push(value.to_string_lossy().into_owned()),
608            Component::CurDir => {}
609            Component::ParentDir | Component::RootDir | Component::Prefix(_) => {
610                return Err("path escapes the repository".into());
611            }
612        }
613    }
614    if parts.is_empty() {
615        return Err("path must contain a repository-relative component".into());
616    }
617    Ok(parts.join("/"))
618}

662        let trace = ToolTrace {
663            schema_version: ARTIFACT_SCHEMA_V1,
664            binding: binding(),
665            calls: vec![
666                ToolCall {
667                    sequence: 1,
668                    tool_name: "leantoken".into(),
669                    call_id: "prewalk:context-1".into(),
670                    result_id: "result-1".into(),
671                    outcome: ToolOutcome::Success,
672                    result_source_tokens: 60,
673                    reread: false,
674                    ranges: vec![
675                        range("src/lib.rs", 10, 20, 'a'),
676                        range("src/dead.rs", 1, 5, 'b'),
677                        range("src/held.rs", 2, 8, 'c'),
678                    ],
679                },
680                ToolCall {
681                    sequence: 2,
682                    tool_name: "leantoken".into(),
683                    call_id: "read-1".into(),
684                    result_id: "result-2".into(),
685                    outcome: ToolOutcome::Success,
686                    result_source_tokens: 20,
687                    reread: true,
688                    ranges: vec![range("src/lib.rs", 8, 24, 'd')],
689                },
690            ],
691        };
692        let trajectory = Trajectory {
693            schema_version: ARTIFACT_SCHEMA_V1,
694            binding: binding(),
695            events: vec![
696                event(
697                    "context-1",
698                    "leantoken_context",
699                    json!({}),
700                    json!({"meta": {"receipt_id": "r1"}}),
701                ),
702                event(
703                    "read-1",
704                    "leantoken_read",
705                    json!({"receipt_id": "r1", "known_hashes": ["cccccccccccccccccccccccccccccccc"]}),
706                    json!({}),
707                ),
708            ],
709        };
710        validate_artifacts(&trace, &trajectory).expect("valid fixtures");
711        let (summary, ranges) =
712            classify(&trace, &trajectory, &BTreeSet::from(["src/lib.rs".into()]))
713                .expect("classification");
714
715        assert_eq!(summary.context_ranges.ranges, 3);
716        assert_eq!(summary.relevant_path_proxy.ranges, 1);
717        assert_eq!(summary.explicit_hash_input_later.ranges, 1);
718        assert_eq!(summary.overlap_reread_later.ranges, 1);
719        assert_eq!(summary.exact_reread_later.ranges, 0);
720        assert_eq!(summary.no_observed_downstream_signal.ranges, 1);
721        assert_eq!(summary.receipt_follow_up_calls, 1);
722        assert_eq!(summary.follow_up_retrieval_calls, 1);
723        let by_path = ranges
724            .iter()
725            .map(|range| (range.path.as_str(), range))
726            .collect::<BTreeMap<_, _>>();
727        assert!(by_path["src/lib.rs"].relevant_path_proxy);
728        assert!(by_path["src/lib.rs"].overlap_reread_later);
729        assert!(by_path["src/dead.rs"].no_observed_downstream_signal);
730        assert!(by_path["src/held.rs"].explicit_hash_input_later);
731    }
