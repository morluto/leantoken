1use std::fs;
2use std::path::{Path, PathBuf};
3use tempfile::Builder;
4
5#[derive(Debug)]
6pub struct Sandbox {
7    root: PathBuf,
8    repo: PathBuf,
9    rerun: String,
10}
11
12impl Sandbox {
13    /// Create an isolated test tree. `module` and `callsite` form the stable
14    /// diagnostic identity; the directory itself still receives a unique
15    /// suffix so concurrent tests cannot collide.
16    pub fn new(module: &str, callsite: &str) -> std::io::Result<Self> {
17        let id = stable_id(module, callsite);
18        let workspace_root = workspace_root();
19        let parent = workspace_root.join("target").join("test-sandboxes");
20        fs::create_dir_all(&parent)?;
21
22        let root = Builder::new()
23            .prefix(&format!("{id}-"))
24            .tempdir_in(&parent)?
25            .keep();
26        let sandbox = Self {
27            repo: root.join("repo"),
28            root,
29            rerun: rerun_command(module, callsite, std::thread::current().name()),
30        };
31        fs::create_dir_all(&sandbox.repo)?;
32        Ok(sandbox)
33    }
34
35    pub fn root(&self) -> &Path {
36        &self.root
37    }
38    pub fn repo(&self) -> &Path {
39        &self.repo
40    }
41
42    fn preserve(&self) -> bool {
43        std::env::var_os("LEANTOKEN_TEST_KEEP").is_some_and(|value| value == "1")
44            || std::thread::panicking()
45    }
46}
47
48impl Drop for Sandbox {
49    fn drop(&mut self) {
50        if !self.preserve() {
51            let _ = fs::remove_dir_all(&self.root);
52            return;
53        }
54
55        let workspace_root = workspace_root();
56        let failure_root = workspace_root.join("target").join("test-failures");
57        let _ = fs::create_dir_all(&failure_root);
58        if let Some(name) = self.root.file_name() {
59            let destination = failure_root.join(name);
60            let _ = fs::rename(&self.root, &destination);
61            eprintln!(
62                "LeanToken test sandbox preserved: {}",
63                destination.display()
64            );
65            eprintln!("LeanToken focused rerun: {}", self.rerun);
66        }
67    }
68}
69
70fn workspace_root() -> PathBuf {
71    PathBuf::from(env!("CARGO_MANIFEST_DIR"))
72        .parent()
73        .and_then(Path::parent)
74        .map(Path::to_path_buf)
75        .expect("test-support manifest is below the workspace root")
76}

78fn stable_id(module: &str, callsite: &str) -> String {
79    let mut hash = 0xcbf29ce484222325u64;
80    for byte in format!("{module}::{callsite}").bytes() {
81        hash ^= u64::from(byte);
82        hash = hash.wrapping_mul(0x100000001b3);
83    }
84    let safe_module = module
85        .chars()
86        .map(|character| {
87            if character.is_ascii_alphanumeric() || matches!(character, '-' | '_') {
88                character
89            } else {
90                '_'
91            }
92        })
93        .collect::<String>();
94    format!("{safe_module}-{hash:016x}")
95}

97fn rerun_command(module: &str, callsite: &str, test_name: Option<&str>) -> String {
98    let package = match module.split("::").next() {
99        Some("leantoken_test_support") => "leantoken-test-support",
100        Some("leantoken_test_suite") => "leantoken-test-suite",
101        _ => "leantoken",
102    };
103    let selector = test_name
104        .filter(|name| !name.is_empty())
105        .unwrap_or(callsite);
106    format!("cargo test --locked --package {package} --all-features --lib {selector}")
107}
