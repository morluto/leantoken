109#[tokio::main]
110async fn main() -> AnyResult<()> {
111    let manifest_path = option_value("--manifest")
112        .unwrap_or_else(|| PathBuf::from("benchmarks/compact_projection_tasks.json"));
113    let output_path = option_value("--output")
114        .unwrap_or_else(|| PathBuf::from("target/compact_projection_report.json"));
115    let repository_root = option_value("--repository-root").unwrap_or_else(|| PathBuf::from("."));
116    let source_revision =
117        option_string("--source-revision").unwrap_or_else(|| "working-tree".into());
118    let source_dirty = env::args().any(|argument| argument == "--source-dirty");
119
120    let manifest_bytes = fs::read(&manifest_path)?;
121    let manifest: Manifest = serde_json::from_slice(&manifest_bytes)?;
122    if manifest.schema_version != 1
123        || manifest.experiment != "compact-projections-v1"
124        || manifest.fixture.generated_reference_callers == 0
125    {
126        return Err("invalid compact projection manifest".into());
127    }
128
129    let fixture_source = repository_root.join(&manifest.fixture.root);
130    let fixture_hash = fixture_manifest_hash(&canonical_fixture_files(&fixture_source)?);
131    if fixture_hash != manifest.fixture.tree_blake3 {
132        return Err("fixture tree commitment mismatch".into());
133    }
134
135    let temp = tempfile::tempdir()?;
136    let root = temp.path().join("repo");
137    copy_tree(&fixture_source, &root)?;
138    let generated_path = root.join("src/compact_projection_fixture.rs");
139    let mut generated = String::from("pub fn compact_target() -> usize {\n    42\n}\n\n");
140    for index in 0..manifest.fixture.generated_reference_callers {
141        generated.push_str(&format!(
142            "pub fn compact_caller_{index:03}() -> usize {{\n    compact_target()\n}}\n\n"
143        ));
144    }
145    fs::write(generated_path, generated)?;
146
147    let config = Config::discover(&root, Some(temp.path().join("index.sqlite")))?;
148    if config.tokenizer.name() != manifest.fixture.tokenizer {
149        return Err("tokenizer commitment mismatch".into());
150    }
151    let tokenizer = config.tokenizer;
152    let services = Services::open(config)?;
153    services.index(true).await?;
154
155    let files_request = FilesRequest {
156        operation: FileOperation::Tree,
157        path: None,
158        query: None,
159        pattern: None,
160        max_results: Some(manifest.tasks.files.max_results),
161        cursor: None,
162        depth: Some(manifest.tasks.files.depth),
163    };
164    let files_full = services.files(files_request.clone()).await?;
165    let files_compact = services.files_paths(files_request).await?;
166    let files_parity = files_compact.paths
167        == files_full
168            .entries
169            .iter()
170            .map(|entry| entry.path.clone())
171            .collect::<Vec<_>>();
172    let files_verifiable = files_compact.paths.iter().all(|path| !path.is_empty());
173    let files = projection_result(
174        files_full.meta.total_response_tokens,
175        files_compact.meta.total_response_tokens,
176        files_full.meta.source_tokens,
177        files_compact.meta.source_tokens,
178        files_parity,
179        files_verifiable,
180    );
181
182    let outline_request = OutlineRequest {
183        paths: manifest.tasks.outline.paths.clone(),
184        symbol_name: None,
185        symbol_kind: None,
186        max_results: Some(manifest.tasks.outline.max_results),
187        max_tokens: Some(manifest.tasks.outline.max_tokens),
188        receipt_id: None,
189        cursor: None,
190    };
191    let outline_full = services.outline(outline_request.clone()).await?;
192    let outline_compact = services.outline_signatures(outline_request).await?;
193    let full_symbols = outline_full
194        .files
195        .iter()
196        .flat_map(|file| {
197            file.symbols.iter().map(|symbol| {
198                (
199                    file.path.as_str(),
200                    symbol.name.as_str(),
201                    symbol.start_line,
202                    symbol.end_line,
203                )
204            })
205        })
206        .collect::<Vec<_>>();
207    let compact_symbols = outline_compact
208        .files
209        .iter()
210        .flat_map(|file| {
211            file.signatures.iter().map(|symbol| {
212                (
213                    file.path.as_str(),
214                    symbol.name.as_str(),
215                    symbol.start_line,
216                    symbol.end_line,
217                )
218            })
219        })
220        .collect::<Vec<_>>();
221    let outline_parity = compact_symbols == full_symbols;
222    let outline_verifiable = outline_compact.files.iter().all(|file| {
223        file.content_hash.len() == 32
224            && file
225                .signatures
226                .iter()
227                .all(|symbol| symbol.start_line > 0 && symbol.end_line >= symbol.start_line)
228    });
229    let outline = projection_result(
230        outline_full.meta.total_response_tokens,
231        outline_compact.meta.total_response_tokens,
232        outline_full.meta.source_tokens,
233        outline_compact.meta.source_tokens,
234        outline_parity,
235        outline_verifiable,
236    );
237
238    let search_request = SearchRequest {
239        query: manifest.tasks.search.query.clone(),
240        mode: SearchMode::Auto,
241        include_paths: Vec::new(),
242        exclude_paths: Vec::new(),
243        focus_paths: Vec::new(),
244        max_results: Some(manifest.tasks.search.max_results),
245        max_tokens: Some(manifest.tasks.search.max_tokens),
246        context_lines: Some(manifest.tasks.search.context_lines),
247        case_sensitive: false,
248        all_occurrences: false,
249        prefer_structural: true,
250        receipt_id: None,
251        query_receipt: None,
252        cursor: None,
253    };
254    let search_full = services.search(search_request.clone()).await?;
255    let search_compact = services.search_grouped(search_request).await?;
256    let represented_hits = search_compact
257        .groups
258        .iter()
259        .map(|group| group.total_hits)
260        .sum::<usize>();
261    let full_references = search_full
262        .hits
263        .iter()
264        .filter(|hit| {
265            hit.role == Some(leantoken::ReferenceRole::Reference)
266                || hit.match_kinds.iter().any(|kind| kind == "reference")
267        })
268        .count();
269    let compact_references = search_compact
270        .groups
271        .iter()
272        .flat_map(|group| &group.references)
273        .map(|references| references.count)
274        .sum::<usize>();
275    let search_parity =
276        represented_hits == search_full.hits.len() && compact_references == full_references;
277    let search_verifiable = search_compact
278        .groups
279        .iter()
280        .any(|group| group.definition.is_some())
281        && search_compact.groups.iter().all(|group| {
282            group
283                .definition
284                .as_ref()
285                .or(group.representative.as_ref())
286                .is_some_and(|evidence| {
287                    !evidence.path.is_empty()
288                        && evidence.start_line > 0
289                        && evidence.end_line >= evidence.start_line
290                        && evidence.content_hash.len() == 32
291                })
292        });
293    let search = projection_result(
294        search_full.meta.total_response_tokens,
295        search_compact.meta.total_response_tokens,
296        search_full.meta.source_tokens,
297        search_compact.meta.source_tokens,
298        search_parity,
299        search_verifiable,
300    );
301
302    let aggregate_baseline_tokens = files
303        .baseline_total_response_tokens
304        .saturating_add(outline.baseline_total_response_tokens)
305        .saturating_add(search.baseline_total_response_tokens);
306    let aggregate_compact_tokens = files
307        .compact_total_response_tokens
308        .saturating_add(outline.compact_total_response_tokens)
309        .saturating_add(search.compact_total_response_tokens);
310    let retry_proxy_delta = files.compact_retry_proxy as isize
311        + outline.compact_retry_proxy as isize
312        + search.compact_retry_proxy as isize;
313    let each_delta_negative = [
314        files.complete_response_token_delta,
315        outline.complete_response_token_delta,
316        search.complete_response_token_delta,
317    ]
318    .into_iter()
319    .all(|delta| delta < 0);
320    let passed = (!manifest.acceptance.require_default_membership_parity
321        || (files.membership_or_concept_parity
322            && outline.membership_or_concept_parity
323            && search.membership_or_concept_parity))
324        && (!manifest.acceptance.require_verifiable_path_range_hash
325            || (files.verifiable && outline.verifiable && search.verifiable))
326        && (!manifest.acceptance.require_zero_retry_proxy_delta || retry_proxy_delta == 0)
327        && (!manifest
328            .acceptance
329            .require_each_complete_response_token_delta_negative
330            || each_delta_negative);
331
332    let report = Report {
333        schema_version: manifest.schema_version,
334        experiment: manifest.experiment,
335        source_revision,
336        source_dirty,
337        build_profile: if cfg!(debug_assertions) {
338            "debug"
339        } else {
340            "release"
341        },
342        operating_system: env::consts::OS,
343        architecture: env::consts::ARCH,
344        manifest_blake3: blake3::hash(&manifest_bytes).to_hex().to_string(),
345        fixture_tree_blake3: fixture_hash,
346        tokenizer: tokenizer.name().into(),
347        token_count_exact: tokenizer.is_exact(),
348        visible_entries: files_full.entries.len(),
349        generated_reference_callers: manifest.fixture.generated_reference_callers,
350        aggregate_baseline_tokens,
351        aggregate_compact_tokens,
352        aggregate_token_delta: aggregate_compact_tokens as isize
353            - aggregate_baseline_tokens as isize,
354        retry_proxy_delta,
355        files,
356        outline,
357        search,
358        acceptance: manifest.acceptance,
359        passed,
360        limitations: vec![
361            "Frozen multilingual fixture plus one generated broad-reference file; this is not a population estimate.",
362            "Retry proxy is zero only when labeled path/symbol coverage and verification coordinates remain available.",
363            "No model executes a maintenance task, so task success outside the declared proxy is unobserved.",
364            "The experiment measures exact service DTO tokens, not provider billing or host UI framing.",
365        ],
366    };
367    if let Some(parent) = output_path.parent() {
368        fs::create_dir_all(parent)?;
369    }
370    fs::write(&output_path, serde_json::to_vec_pretty(&report)?)?;
371    println!("{}", serde_json::to_string_pretty(&report)?);
372    if !report.passed {
373        return Err("compact projection acceptance gate failed".into());
374    }
375    Ok(())
376}

378fn projection_result(
379    baseline_total_response_tokens: usize,
380    compact_total_response_tokens: usize,
381    baseline_source_tokens: usize,
382    compact_source_tokens: usize,
383    membership_or_concept_parity: bool,
384    verifiable: bool,
385) -> ProjectionResult {
386    let compact_retry_proxy = usize::from(!(membership_or_concept_parity && verifiable));
387    ProjectionResult {
388        baseline_total_response_tokens,
389        compact_total_response_tokens,
390        complete_response_token_delta: compact_total_response_tokens as isize
391            - baseline_total_response_tokens as isize,
392        baseline_source_tokens,
393        compact_source_tokens,
394        membership_or_concept_parity,
395        verifiable,
396        baseline_retry_proxy: 0,
397        compact_retry_proxy,
398    }
399}

415fn canonical_fixture_files(root: &Path) -> AnyResult<Vec<(String, Vec<u8>)>> {
416    let mut files = Vec::new();
417    collect_files(root, root, &mut files)?;
418    files.sort_by(|left, right| left.0.cmp(&right.0));
419    Ok(files)
420}
421
422fn collect_files(
423    root: &Path,
424    directory: &Path,
425    files: &mut Vec<(String, Vec<u8>)>,
426) -> AnyResult<()> {
427    let mut entries = fs::read_dir(directory)?.collect::<Result<Vec<_>, _>>()?;
428    entries.sort_by_key(fs::DirEntry::file_name);
429    for entry in entries {
430        let path = entry.path();
431        let file_type = entry.file_type()?;
432        if file_type.is_dir() {
433            collect_files(root, &path, files)?;
434        } else if file_type.is_file() {
435            let relative = path
436                .strip_prefix(root)?
437                .to_string_lossy()
438                .replace('\\', "/");
439            let text = String::from_utf8(fs::read(path)?)?;
440            let normalized = text.replace("\r\n", "\n");
441            if normalized.contains('\r') {
442                return Err("fixture contains a lone carriage return".into());
443            }
444            files.push((relative, normalized.into_bytes()));
445        }
446    }
447    Ok(())
448}

450fn fixture_manifest_hash(files: &[(String, Vec<u8>)]) -> String {
451    let manifest = files
452        .iter()
453        .map(|(path, bytes)| format!("{}  {path}\n", blake3::hash(bytes).to_hex()))
454        .collect::<String>();
455    blake3::hash(manifest.as_bytes()).to_hex().to_string()
456}

458fn copy_tree(source: &Path, destination: &Path) -> AnyResult<()> {
459    fs::create_dir_all(destination)?;
460    for entry in fs::read_dir(source)? {
461        let entry = entry?;
462        let source_path = entry.path();
463        let destination_path = destination.join(entry.file_name());
464        if entry.file_type()?.is_dir() {
465            copy_tree(&source_path, &destination_path)?;
466        } else {
467            fs::copy(source_path, destination_path)?;
468        }
469    }
470    Ok(())
