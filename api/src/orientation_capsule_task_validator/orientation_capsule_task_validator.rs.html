1use std::error::Error;

9const RECEIPT_FILE: &str = "validation-receipt.json";
10
11#[derive(Debug, Parser)]
12#[command(about = "Validate one local orientation-capsule ARB task")]
13struct Args {
14    #[arg(long)]
15    task: String,
16    #[arg(long)]
17    cargo: PathBuf,
18    #[arg(long)]
19    cargo_blake3: String,
20    #[arg(long)]
21    python: PathBuf,
22    #[arg(long)]
23    python_blake3: String,
24}
25
26#[derive(Debug, Serialize)]
27struct ValidationReceipt {
28    schema_version: u32,
29    experiment_id: String,
30    manifest_blake3: String,
31    task_id: String,
32    repetition: usize,
33    arm: String,
34    task: String,
35    cargo_blake3: String,
36    python_blake3: String,
37    repository_revision: String,
38    patch_blake3: String,
39    validation_exit_code: Option<i32>,
40    resolved: bool,
41}
42
43fn main() -> Result<(), Box<dyn Error>> {
44    let args = Args::parse();
45    if !matches!(args.task.as_str(), "clap" | "click") {
46        return Err("task must be clap or click".into());
47    }
48    validate_hash(&args.cargo_blake3)?;
49    validate_hash(&args.python_blake3)?;
50    verify_file_hash(&args.cargo, &args.cargo_blake3)?;
51    verify_file_hash(&args.python, &args.python_blake3)?;
52
53    let artifacts = required_env_path("LEANTOKEN_MODEL_AB_ARTIFACTS_DIRECTORY")?;
54    let experiment_id = required_env("LEANTOKEN_MODEL_AB_EXPERIMENT_ID")?;
55    let manifest_blake3 = required_env("LEANTOKEN_MODEL_AB_MANIFEST_BLAKE3")?;
56    validate_hash(&manifest_blake3)?;
57    let task_id = required_env("LEANTOKEN_MODEL_AB_TASK_ID")?;
58    let repetition = required_env("LEANTOKEN_MODEL_AB_REPETITION")?.parse::<usize>()?;
59    let arm = required_env("LEANTOKEN_MODEL_AB_ARM")?;
60    let repository = std::env::current_dir()?.canonicalize()?;
61    let repository_revision = git_stdout(&repository, &["rev-parse", "HEAD"])?;
62    let patch = git_bytes(
63        &repository,
64        &["diff", "--binary", "--full-index", "HEAD", "--"],
65    )?;
66    let patch_blake3 = blake3::hash(&patch).to_hex().to_string();
67
68    let output = match args.task.as_str() {
69        "clap" => validate_clap(&repository, &args.cargo)?,
70        "click" => validate_click(&repository, &args.python)?,
71        _ => unreachable!(),
72    };
73    fs::write(artifacts.join("validation-stdout.log"), &output.stdout)?;
74    fs::write(artifacts.join("validation-stderr.log"), &output.stderr)?;
75    let resolved = output.status.success();
76    let receipt = ValidationReceipt {
77        schema_version: 1,
78        experiment_id,
79        manifest_blake3,
80        task_id,
81        repetition,
82        arm,
83        task: args.task,
84        cargo_blake3: args.cargo_blake3,
85        python_blake3: args.python_blake3,
86        repository_revision: repository_revision.trim().to_owned(),
87        patch_blake3,
88        validation_exit_code: output.status.code(),
89        resolved,
90    };
91    fs::write(
92        artifacts.join(RECEIPT_FILE),
93        serde_json::to_vec_pretty(&receipt)?,
94    )?;
95    if !resolved {
96        return Err(format!("orientation task validator exited with {}", output.status).into());
97    }
98    Ok(())
99}

101fn validate_clap(repository: &Path, cargo: &Path) -> Result<Output, Box<dyn Error>> {
102    let project = tempfile::tempdir()?;
103    fs::create_dir(project.path().join("src"))?;
104    let dependency_path = serde_json::to_string(repository.to_string_lossy().as_ref())?;
105    fs::write(
106        project.path().join("Cargo.toml"),
107        format!(
108            "[package]\nname = \"orientation-capsule-clap-validator\"\nversion = \"0.0.0\"\n\
109             edition = \"2021\"\n\n[dependencies]\nclap = {{ path = {dependency_path} }}\n"
110        ),
111    )?;
112    fs::write(
113        project.path().join("src/lib.rs"),
114        r#"#[cfg(test)]
115mod tests {
116    use clap::{Arg, ArgAction, Command};
117
118    fn command() -> Command {
119        Command::new("test")
120            .arg(
121                Arg::new("opt")
122                    .long("opt")
123                    .action(ArgAction::Set),
124            )
125            .arg(
126                Arg::new("args")
127                    .long("args")
128                    .num_args(2)
129                    .default_values_if("opt", "value", ["df1", "df2"]),
130            )
131    }
132
133    #[test]
134    fn conditional_multiple_defaults_are_injected_only_on_match() {
135        let matches = command()
136            .try_get_matches_from(["test", "--opt", "value"])
137            .expect("matching command");
138        let values = matches
139            .get_many::<String>("args")
140            .expect("conditional defaults")
141            .map(String::as_str)
142            .collect::<Vec<_>>();
143        assert_eq!(values, ["df1", "df2"]);
144
145        let matches = command()
146            .try_get_matches_from(["test", "--opt", "other"])
147            .expect("non-matching command");
148        assert!(matches.get_many::<String>("args").is_none());
149    }
150}
151"#,
152    )?;
153    Ok(Command::new(cargo)
154        .args(["test", "--offline", "--quiet"])
155        .current_dir(project.path())
156        .env("CARGO_NET_OFFLINE", "true")
157        .output()?)
158}

160fn validate_click(repository: &Path, python: &Path) -> Result<Output, Box<dyn Error>> {
161    const SCRIPT: &str = r#"
162import sys
163from pathlib import Path
164
165sys.path.insert(0, str(Path.cwd() / "src"))
166import click
167from click.testing import CliRunner
168
169CASES = [
170    ({"type": click.BOOL}, False, "False"),
171    ({"type": click.BOOL}, True, "True"),
172    ({"type": click.BOOL, "default": True}, False, "True"),
173    ({"type": click.BOOL, "default": True}, True, "False"),
174    ({"type": str}, False, ""),
175    ({"type": str}, True, "True"),
176]
177
178runner = CliRunner()
179for opts, pass_flag, expected in CASES:
180    @click.command()
181    @click.option("--foo", is_flag=True, **opts)
182    def cmd(foo):
183        click.echo(foo)
184
185    result = runner.invoke(cmd, ["--foo"] if pass_flag else [])
186    assert result.exception is None, result.output
187    assert result.output == f"{expected}\n", (opts, pass_flag, result.output)
188"#;
189    Ok(Command::new(python)
190        .args(["-c", SCRIPT])
191        .current_dir(repository)
192        .env("PYTHONDONTWRITEBYTECODE", "1")
193        .output()?)
194}

196fn required_env(name: &str) -> Result<String, Box<dyn Error>> {
197    let value = std::env::var(name)?;
198    if value.trim().is_empty() {
199        return Err(format!("{name} is empty").into());
200    }
201    Ok(value)
202}
203
204fn required_env_path(name: &str) -> Result<PathBuf, Box<dyn Error>> {
205    Ok(PathBuf::from(required_env(name)?).canonicalize()?)
206}
207
208fn validate_hash(value: &str) -> Result<(), Box<dyn Error>> {
209    if value.len() != 64 || !value.bytes().all(|byte| byte.is_ascii_hexdigit()) {
210        return Err("expected a 64-character hexadecimal hash".into());
211    }
212    Ok(())
213}
214
215fn verify_file_hash(path: &Path, expected: &str) -> Result<(), Box<dyn Error>> {
216    let actual = blake3::hash(&fs::read(path)?).to_hex().to_string();
217    if actual != expected {
218        return Err(format!("executable hash mismatch: {}", path.display()).into());
219    }
220    Ok(())
221}

223fn git_stdout(repository: &Path, args: &[&str]) -> Result<String, Box<dyn Error>> {
224    Ok(String::from_utf8(git_bytes(repository, args)?)?)
225}
226
227fn git_bytes(repository: &Path, args: &[&str]) -> Result<Vec<u8>, Box<dyn Error>> {
228    let output = Command::new("git")
229        .arg("-C")
230        .arg(repository)
231        .args(args)
232        .output()?;
233    if !output.status.success() {
234        return Err(format!("git command failed: {}", output.status).into());
235    }
236    Ok(output.stdout)
237}
