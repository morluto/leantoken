77#[tokio::main]
78async fn main() {
79    let source = Path::new(env!("LEANTOKEN_REPOSITORY_ROOT")).join("fixtures/sample_repo");
80    let temp = tempfile::tempdir().expect("temporary benchmark repository");
81    let root = temp.path().join("repo");
82    copy_tree(&source, &root);
83    let config = Config::discover(&root, Some(temp.path().join("index.sqlite"))).expect("config");
84    let services = Services::open(config).expect("services");
85
86    let cold_start = Instant::now();
87    services.index(true).await.expect("cold index");
88    let cold_index_ms = cold_start.elapsed().as_secs_f64() * 1_000.0;
89
90    let mut task_reports = Vec::new();
91    let mut baseline_total = 0usize;
92    let mut emitted_total = 0usize;
93    let mut baseline_json_total = 0usize;
94    let mut leantoken_json_total = 0usize;
95    let mut relevant_total = 0usize;
96    let mut found_total = 0usize;
97
98    for task in TASKS {
99        let request = ContextRequest {
100            task: task.prompt.into(),
101            token_budget: 500,
102            include_paths: Vec::new(),
103            must_include_paths: Vec::new(),
104            must_include_symbols: Vec::new(),
105            required_evidence: Vec::new(),
106            max_fragments: None,
107            plan_only: false,
108            focus_paths: Vec::new(),
109            strict_focus_paths: false,
110            minimum_fragments_per_focus_path: None,
111            focus_symbols: Vec::new(),
112            exclude_paths: Vec::new(),
113            known_hashes: Vec::new(),
114            receipt_id: None,
115            prior_repository_generation: None,
116            base_revision: None,
117            changed_paths: Vec::new(),
118            strict_changed_paths: false,
119            verbose_diagnostics: false,
120        };
121        let warm_start = Instant::now();
122        let response = services.context(request.clone()).await.expect("context");
123        let warm_latency_ms = warm_start.elapsed().as_secs_f64() * 1_000.0;
124        let returned = response
125            .fragments
126            .iter()
127            .map(|fragment| fragment.path.clone())
128            .collect::<HashSet<_>>();
129        let found = task
130            .relevant
131            .iter()
132            .filter(|path| returned.contains(**path))
133            .count();
134        let baseline_files = task
135            .relevant
136            .iter()
137            .map(|path| {
138                let content = fs::read_to_string(root.join(path)).expect("fixture file");
139                serde_json::json!({ "path": path, "content": content })
140            })
141            .collect::<Vec<_>>();
142        let baseline = baseline_files
143            .iter()
144            .map(|file| tokens::count(file["content"].as_str().expect("content string")))
145            .sum::<usize>();
146        let baseline_json = tokens::count(
147            &serde_json::to_string(&baseline_files).expect("baseline read responses"),
148        );
149        let emitted = response.meta.source_tokens;
150        let total_json = tokens::count(&serde_json::to_string(&response).expect("response JSON"));
151        let known = response
152            .fragments
153            .iter()
154            .map(|fragment| fragment.content_hash.clone())
155            .collect::<Vec<_>>();
156        let known_set = known.iter().cloned().collect::<HashSet<_>>();
157        let repeated = services
158            .context(ContextRequest {
159                known_hashes: known,
160                receipt_id: None,
161                prior_repository_generation: Some(response.meta.repository_generation),
162                ..request
163            })
164            .await
165            .expect("known-hash context");
166        let known_fragments_resent = repeated
167            .fragments
168            .iter()
169            .filter(|fragment| known_set.contains(&fragment.content_hash))
170            .count();
171        let estimated_repeated_range_source_tokens = repeated
172            .fragments
173            .iter()
174            .map(|fragment| {
175                let prior_ranges = response
176                    .fragments
177                    .iter()
178                    .filter(|prior| prior.path == fragment.path)
179                    .map(|prior| (prior.start_line, prior.end_line))
180                    .collect::<Vec<_>>();
181                repeated_range_token_estimate(
182                    fragment.start_line,
183                    fragment.end_line,
184                    fragment.token_count,
185                    &prior_ranges,
186                )
187            })
188            .sum();
189        assert!(
190            repeated.omission_summary.known_hash > 0,
191            "known-hash omission should be visible in the bounded omission summary"
192        );
193
194        baseline_total += baseline;
195        emitted_total += emitted;
196        baseline_json_total += baseline_json;
197        leantoken_json_total += total_json;
198        relevant_total += task.relevant.len();
199        found_total += found;
200        let mut returned_files = returned.into_iter().collect::<Vec<_>>();
201        returned_files.sort_unstable();
202        task_reports.push(TaskReport {
203            task: task.prompt,
204            relevant_files: task.relevant,
205            returned_files,
206            relevant_files_found: found,
207            recall: ratio(found, task.relevant.len()),
208            baseline_full_file_source_tokens: baseline,
209            baseline_minimal_read_json_tokens: baseline_json,
210            leantoken_source_tokens: emitted,
211            leantoken_total_json_tokens: total_json,
212            source_token_savings_fraction: savings(baseline, emitted),
213            total_json_savings_against_minimal_baseline_fraction: savings(
214                baseline_json,
215                total_json,
216            ),
217            warm_latency_ms,
218            second_response_source_tokens: repeated.meta.source_tokens,
219            estimated_repeated_range_source_tokens,
220            known_fragments_resent,
221        });
222    }
223
224    let aggregate_savings = savings(baseline_total, emitted_total);
225    let aggregate_json_savings = savings(baseline_json_total, leantoken_json_total);
226    let aggregate_recall = ratio(found_total, relevant_total);
227    let report = Report {
228        fixture: source.display().to_string(),
229        tokenizer: tokens::Tokenizer::default().name(),
230        token_count_exact: tokens::Tokenizer::default().is_exact(),
231        task_count: TASKS.len(),
232        cold_index_ms,
233        aggregate_baseline_source_tokens: baseline_total,
234        aggregate_leantoken_source_tokens: emitted_total,
235        aggregate_source_token_savings_fraction: aggregate_savings,
236        aggregate_baseline_minimal_read_json_tokens: baseline_json_total,
237        aggregate_leantoken_total_json_tokens: leantoken_json_total,
238        aggregate_total_json_savings_against_minimal_baseline_fraction: aggregate_json_savings,
239        aggregate_recall,
240        tasks: task_reports,
241        limitations: vec![
242            "Pinned synthetic fixture; not a substitute for benchmarks on maintained real repositories.",
243            "Baseline assumes an agent perfectly chooses the labeled relevant files, so it excludes grep output and wrong turns.",
244            "Baseline JSON is only a minimal path/content envelope, not a matched tool schema; its comparison with the full LeanToken response is a conservative diagnostic, not a product claim.",
245            "Source-token savings and total serialized tool-result tokens are reported separately.",
246            "Tiny labeled files can make direct oracle reads cheaper than ranked context; negative savings are retained rather than treated as a harness failure.",
247            "No model is executed; edit success, plan handoff, and prewalk quality require a separate agent evaluation.",
248        ],
249    };
250    let pretty = serde_json::to_string_pretty(&report).expect("serialize report");
251    let report_path =
252        Path::new(env!("LEANTOKEN_REPOSITORY_ROOT")).join("target/benchmark_contract_report.json");
253    fs::write(&report_path, &pretty).expect("write report");
254    println!("{pretty}");
255
256    assert!(aggregate_savings.is_finite());
257    assert!(
258        aggregate_recall >= 0.80,
259        "labeled-file recall below MVP target"
260    );
261    assert!(
262        report
263            .tasks
264            .iter()
265            .all(|task| task.known_fragments_resent == 0)
266    );
267}

269fn ratio(numerator: usize, denominator: usize) -> f64 {
270    if denominator == 0 {
271        0.0
272    } else {
273        numerator as f64 / denominator as f64
274    }
275}
276
277fn savings(baseline: usize, actual: usize) -> f64 {
278    if baseline == 0 {
279        0.0
280    } else {
281        1.0 - actual as f64 / baseline as f64
282    }
283}

285fn repeated_range_token_estimate(
286    start_line: usize,
287    end_line: usize,
288    token_count: usize,
289    prior_ranges: &[(usize, usize)],
290) -> usize {
291    if end_line < start_line || token_count == 0 {
292        return 0;
293    }
294    let line_count = end_line - start_line + 1;
295    let repeated_lines = (start_line..=end_line)
296        .filter(|line| {
297            prior_ranges
298                .iter()
299                .any(|(prior_start, prior_end)| prior_start <= line && line <= prior_end)
300        })
301        .count();
302    token_count
303        .saturating_mul(repeated_lines)
304        .div_ceil(line_count)
305}

307fn copy_tree(source: &Path, destination: &Path) {
308    fs::create_dir_all(destination).expect("create destination");
309    for entry in fs::read_dir(source).expect("read fixture directory") {
310        let entry = entry.expect("fixture entry");
311        let target: PathBuf = destination.join(entry.file_name());
312        if entry.file_type().expect("fixture type").is_dir() {
313            copy_tree(&entry.path(), &target);
314        } else {
315            fs::copy(entry.path(), target).expect("copy fixture file");
316        }
317    }
